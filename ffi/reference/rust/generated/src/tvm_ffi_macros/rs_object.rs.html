29pub(crate) fn rs_object(args: TokenStream, item: TokenStream) -> TokenStream {
30    let args = parse_macro_input!(args as RsObjectArgs);
31    let item = parse_macro_input!(item as DeriveInput);
32    expand(&args, &item)
33        .unwrap_or_else(|error| error.to_compile_error())
34        .into()
35}
36
37/// `#[rs_object(type_key = "type.key", final, ...)]`.
38struct RsObjectArgs {
39    type_key: LitStr,
40    type_final: bool,
41    immutable: bool,
42    s_eq_hash_kind: Option<LitStr>,
43    child_slots: Option<LitInt>,
44    child_slots_can_overflow: Option<LitBool>,
45    reflection: Option<Path>,
46    tvm_ffi_crate: Option<Path>,
47}
48
49impl Parse for RsObjectArgs {
50    fn parse(input: ParseStream) -> syn::Result<Self> {
51        if input.peek(LitStr) {
52            return Err(syn::Error::new(
53                input.span(),
54                "name the type key, as in `#[rs_object(type_key = \"my.Point\")]`",
55            ));
56        }
57        let mut type_key = None;
58        let mut args = RsObjectArgs {
59            type_key: LitStr::new("", input.span()),
60            type_final: false,
61            immutable: false,
62            s_eq_hash_kind: None,
63            child_slots: None,
64            child_slots_can_overflow: None,
65            reflection: None,
66            tvm_ffi_crate: None,
67        };
68        let mut seen = Vec::new();
69        while !input.is_empty() {
70            if !seen.is_empty() {
71                input.parse::<Token![,]>()?;
72                if input.is_empty() {
73                    break;
74                }
75            }
76            // `final` and `crate` are keywords.
77            let name = input.call(syn::Ident::parse_any)?;
78            let key = name.to_string();
79            if seen.contains(&key) {
80                return Err(syn::Error::new(name.span(), format!("duplicate `{key}`")));
81            }
82            if key != "final" && key != "immutable" {
83                input.parse::<Token![=]>()?;
84            }
85            match key.as_str() {
86                "type_key" => type_key = Some(input.parse()?),
87                "final" => args.type_final = true,
88                "immutable" => args.immutable = true,
89                "s_eq_hash_kind" => args.s_eq_hash_kind = Some(input.parse()?),
90                "child_slots" => args.child_slots = Some(input.parse()?),
91                "child_slots_can_overflow" => args.child_slots_can_overflow = Some(input.parse()?),
92                "reflection" => args.reflection = Some(input.parse()?),
93                "crate" => args.tvm_ffi_crate = Some(Path::parse_mod_style(input)?),
94                _ => {
95                    return Err(syn::Error::new(
96                        name.span(),
97                        "expected `type_key = \"<key>\"`, `final`, `immutable`, `s_eq_hash_kind = \"<kind>\"`, `child_slots = <n>`, \
98                         `child_slots_can_overflow = <bool>`, `reflection = <path>` or \
99                         `crate = <path>`",
100                    ));
101                }
102            }
103            seen.push(key);
104        }
105        args.type_key = type_key.ok_or_else(|| {
106            syn::Error::new(
107                input.span(),
108                "missing the type key, as in `#[rs_object(type_key = \"my.Point\")]`",
109            )
110        })?;
111        if args.type_final
112            && (args.child_slots.is_some() || args.child_slots_can_overflow.is_some())
113        {
114            return Err(syn::Error::new(
115                args.type_key.span(),
116                "a final type has no children to reserve slots for",
117            ));
118        }
119        Ok(args)
120    }
121}

123const EXPECTED_BASE: &str = "the first field holds the parent's object: \
124     `base_: Object` at the root, else `base_: ParentObj`";
125
126fn expand(args: &RsObjectArgs, item: &DeriveInput) -> syn::Result<TokenStream2> {
127    if !item.generics.params.is_empty() || item.generics.where_clause.is_some() {
128        return Err(syn::Error::new_spanned(
129            &item.generics,
130            "an `rs_object` cannot be generic",
131        ));
132    }
133    let Data::Struct(data) = &item.data else {
134        return Err(syn::Error::new(
135            item.ident.span(),
136            "`rs_object` expects a struct",
137        ));
138    };
139    let fields: Vec<&syn::Field> = match &data.fields {
140        Fields::Named(fields) => fields.named.iter().collect(),
141        Fields::Unit => Vec::new(),
142        Fields::Unnamed(fields) => {
143            return Err(syn::Error::new_spanned(
144                fields,
145                "`rs_object` expects named fields",
146            ));
147        }
148    };
149    let (base, fields) = match fields.split_first() {
150        Some((base, fields)) if base.ident.as_ref().is_some_and(|id| id.unraw() == "base_") => {
151            (*base, fields)
152        }
153        // The trailing `_` marks plumbing, so a reflected field can be named `base`.
154        Some((field, _)) if field.ident.as_ref().is_some_and(|id| id.unraw() == "base") => {
155            return Err(syn::Error::new_spanned(
156                field,
157                "the parent's object is named `base_`: rename `base` to `base_`",
158            ));
159        }
160        Some((field, _)) => return Err(syn::Error::new_spanned(field, EXPECTED_BASE)),
161        None => return Err(syn::Error::new(item.ident.span(), EXPECTED_BASE)),
162    };
163    if let Some(attr) = base.attrs.iter().find(|a| a.path.is_ident("reflect")) {
164        return Err(syn::Error::new_spanned(
165            attr,
166            "`base_` is the parent's object, whose fields the parent reflects",
167        ));
168    }
169    let base_id = &base.ident;
170    let base_ty = &base.ty;
171    // `new` takes a scalar as its own type, so that a literal infers as it, and any
172    // other field as `impl Into`. Tokens alone do not tell an object reference from a
173    // value, so this object-level allocator does not take `impl IntoArg` as a
174    // reference's constructor does.
175    let params = fields.iter().map(|field| {
176        let (name, ty) = (&field.ident, &field.ty);
177        if is_exact_param(ty) {
178            quote!(#name: #ty)
179        } else {
180            quote!(#name: impl ::core::convert::Into<#ty>)
181        }
182    });
183    let inits = fields.iter().map(|field| {
184        let name = &field.ident;
185        if is_exact_param(&field.ty) {
186            quote!(#name)
187        } else {
188            quote!(#name: ::core::convert::Into::into(#name))
189        }
190    });
191
192    let tvm_ffi = match &args.tvm_ffi_crate {
193        Some(path) => quote!(#path),
194        None => get_tvm_ffi_crate(),
195    };
196    let attrs = &item.attrs;
197    let vis = &item.vis;
198    let obj = &item.ident;
199    let key = &args.type_key;
200
201    // `derive(Object)` reads the type from these options; the struct written out
202    // carries neither them nor `#[reflect]`, which no derive on it would take.
203    let mut options = vec![quote!(type_key = #key), quote!(register)];
204    if args.type_final {
205        options.push(quote!(final));
206    }
207    if args.immutable {
208        options.push(quote!(immutable));
209    }
210    if let Some(kind) = &args.s_eq_hash_kind {
211        options.push(quote!(s_eq_hash_kind = #kind));
212    }
213    if let Some(slots) = &args.child_slots {
214        options.push(quote!(child_slots = #slots));
215    }
216    if let Some(overflow) = &args.child_slots_can_overflow {
217        options.push(quote!(child_slots_can_overflow = #overflow));
218    }
219    if let Some(reflection) = &args.reflection {
220        options.push(quote!(reflection = #reflection));
221    }
222    let reflected_fields = fields.iter().map(|field| {
223        if field.attrs.iter().any(|a| a.path.is_ident("reflect")) {
224            quote!(#field)
225        } else {
226            quote!(#[reflect] #field)
227        }
228    });
229    let object_input: DeriveInput = syn::parse2(quote! {
230        #[repr(C)]
231        #[object(#(#options),*)]
232        #vis struct #obj {
233            #base,
234            #(#reflected_fields,)*
235        }
236    })?;
237    let object_impl = expand_object(&object_input, &tvm_ffi, true);
238    let plain_fields = fields.iter().map(|field| {
239        let mut field = (*field).clone();
240        field.attrs.retain(|a| !a.path.is_ident("reflect"));
241        field
242    });
243
244    Ok(quote! {
245        #(#attrs)*
246        #[repr(C)]
247        #vis struct #obj {
248            #base,
249            #(#plain_fields,)*
250        }
251
252        #object_impl
253
254        impl ::core::ops::Deref for #obj {
255            type Target = #base_ty;
256            #[inline]
257            fn deref(&self) -> &#base_ty {
258                &self.#base_id
259            }
260        }
261
262        impl #obj {
263            /// The object from the parent's object, then the fields in order: a
264            /// scalar as its own type, any other from anything that converts into it.
265            #[allow(clippy::too_many_arguments)]
266            pub fn new(#base_id: #base_ty, #(#params),*) -> Self {
267                Self { #base_id, #(#inits,)* }
268            }
269        }
270    })
271}

273/// The scalars `new` takes as their own type, as the allocator of a type bound by
274/// tvm-ffi-stubgen does (`RUST_EXACT_PARAMS` there): a literal infers only from
275/// its exact type.
276const EXACT_PARAMS: &[&str] = &[
277    "bool", "i8", "i16", "i32", "i64", "isize", "u8", "u16", "u32", "u64", "usize", "f32", "f64",
278];
279
280fn is_exact_param(ty: &syn::Type) -> bool {
281    match ty {
282        syn::Type::Path(path) if path.qself.is_none() => path
283            .path
284            .get_ident()
285            .is_some_and(|ident| EXACT_PARAMS.iter().any(|name| ident == name)),
286        _ => false,
287    }
