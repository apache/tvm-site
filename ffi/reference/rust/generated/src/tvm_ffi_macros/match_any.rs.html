20use proc_macro2::{Ident, Span, TokenStream};
21use quote::quote;
22use syn::parse::{Parse, ParseStream};
23use syn::{braced, parenthesized, Expr, Pat, Path, Result, Token};
24
25use crate::utils::get_tvm_ffi_crate;
26
27struct MatchAnyInput {
28    scrutinee: Expr,
29    arms: Vec<TypedArm>,
30    fallback: Expr,
31}
32
33struct TypedArm {
34    matcher: Path,
35    binding: Pat,
36    guard: Option<Expr>,
37    body: Expr,
38}
39
40impl Parse for MatchAnyInput {
41    fn parse(input: ParseStream<'_>) -> Result<Self> {
42        let scrutinee = input.call(Expr::parse_without_eager_brace)?;
43        let content;
44        braced!(content in input);
45
46        let mut arms = Vec::new();
47        let mut fallback = None;
48        while !content.is_empty() {
49            if fallback.is_some() {
50                return Err(content.error("the `_` fallback must be the final arm"));
51            }
52
53            if content.peek(Token![_]) {
54                content.parse::<Token![_]>()?;
55                if content.peek(Token![if]) {
56                    return Err(content.error("the `_` fallback cannot have a guard"));
57                }
58                content.parse::<Token![=>]>()?;
59                fallback = Some(content.parse::<Expr>()?);
60            } else {
61                let matcher = content.parse::<Path>()?;
62                let binding_content;
63                parenthesized!(binding_content in content);
64                let binding = binding_content.parse::<Pat>()?;
65                if !binding_content.is_empty() {
66                    return Err(binding_content.error("expected one binding pattern"));
67                }
68                let guard = if content.peek(Token![if]) {
69                    content.parse::<Token![if]>()?;
70                    Some(content.parse::<Expr>()?)
71                } else {
72                    None
73                };
74                content.parse::<Token![=>]>()?;
75                let body = content.parse::<Expr>()?;
76                arms.push(TypedArm {
77                    matcher,
78                    binding,
79                    guard,
80                    body,
81                });
82            }
83
84            if content.peek(Token![,]) {
85                content.parse::<Token![,]>()?;
86            } else if !content.is_empty() {
87                return Err(content.error("expected `,` between match_any! arms"));
88            }
89        }
90
91        let fallback = fallback
92            .ok_or_else(|| content.error("match_any! requires a final `_` fallback arm"))?;
93        Ok(Self {
94            scrutinee,
95            arms,
96            fallback,
97        })
98    }
99}
100
101pub fn expand(input: proc_macro::TokenStream) -> proc_macro::TokenStream {
102    let input = syn::parse_macro_input!(input as MatchAnyInput);
103    expand_match_any(input).into()
104}
105
106fn expand_match_any(input: MatchAnyInput) -> TokenStream {
107    let tvm_ffi = get_tvm_ffi_crate();
108    let span = Span::mixed_site();
109    let source = Ident::new("__tvm_ffi_match_any_source", span);
110    let converted = Ident::new("__tvm_ffi_match_any_converted", span);
111    let view = Ident::new("__tvm_ffi_match_any_view", span);
112    let rejected = Ident::new("__tvm_ffi_match_any_rejected", span);
113    let scrutinee = input.scrutinee;
114    let fallback = input.fallback;
115    let dispatch_fallback = fallback.clone();
116    let arms = input.arms;
117    let dispatch = arms
118        .into_iter()
119        .rev()
120        .fold(quote!({ #dispatch_fallback }), |next, arm| {
121            let matcher = arm.matcher;
122            let binding = arm.binding;
123            let body = arm.body;
124            let matched = if let Some(guard) = arm.guard {
125                quote!(::core::result::Result::Ok(#binding) if #guard)
126            } else {
127                quote!(::core::result::Result::Ok(#binding))
128            };
129
130            quote! {
131                match ::core::convert::TryInto::<#matcher>::try_into(#view) {
132                    #matched => { #body },
133                    #rejected => {
134                        ::core::mem::drop(#rejected);
135                        #next
136                    },
137                }
138            }
139        });
140
141    quote! {
142        {
143            let #source = &(#scrutinee);
144            let #converted: ::core::result::Result<
145                #tvm_ffi::AnyView<'_>,
146                ::core::convert::Infallible,
147            > = ::core::convert::TryInto::<#tvm_ffi::AnyView<'_>>::try_into(#source);
148            let #view = match #converted {
149                ::core::result::Result::Ok(view) => view,
150                ::core::result::Result::Err(error) => match error {},
151            };
152            if #view.type_index()
153                >= #tvm_ffi::TypeIndex::kTVMFFIStaticObjectBegin as i32
154            {
155                #dispatch
156            } else {
157                #fallback
158            }
159        }
160    }
161}
