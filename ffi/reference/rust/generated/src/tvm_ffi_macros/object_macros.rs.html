28/// Derive Object trait for a struct to generate boilerplate code
29pub fn derive_object(input: proc_macro::TokenStream) -> TokenStream {
30    let derive_input = syn::parse_macro_input!(input as DeriveInput);
31    TokenStream::from(expand_object(&derive_input, &get_tvm_ffi_crate(), false))
32}

34/// The `ObjectCore` impl of an object struct, and its `ShallowCopy` when it
35/// registers reflection, naming the tvm-ffi crate `tvm_ffi_crate`.
36///
37/// `rs_object` registers the reflection of a type without `#[reflect]` fields
38/// or `s_eq_hash_kind` too, with the creator of its `UnsafeInit`, as
39/// `#[rs_object]` does.
40pub(crate) fn expand_object(
41    derive_input: &DeriveInput,
42    tvm_ffi_crate: &TokenStream2,
43    rs_object: bool,
44) -> TokenStream2 {
45    let struct_name = derive_input.ident.clone();
46    let options = object_options(derive_input);
47
48    let type_key = options
49        .type_key
50        .clone()
51        .expect("Expect #[object(type_key = \"<my_type_key>\")]");
52    let type_final = options.type_final;
53    let type_immutable = options.immutable;
54    // Whether an `rs_object` can derive from this type, which needs its
55    // `ShallowCopy` and `UnsafeInit`.
56    let type_rs_parent = options.rs_parent;
57    // Whether this type is defined here, so that it registers its key, as C++
58    // `TVM_FFI_DECLARE_OBJECT_INFO` does. Without it, the type binds a key
59    // that its defining library, usually C++, has registered.
60    let type_register = options.register;
61
62    // Reserved child slots and whether children may exceed them, as C++
63    // `_type_child_slots` and `_type_child_slots_can_overflow` declare them.
64    let child_slots = match &options.child_slots {
65        Some(n) => n
66            .base10_parse::<i32>()
67            .expect("Expect `child_slots = <non-negative integer>`"),
68        None => 0,
69    };
70    let child_slots_can_overflow = options
71        .child_slots_can_overflow
72        .as_ref()
73        .is_none_or(|b| b.value);
74    assert!(
75        child_slots >= 0,
76        "Expect `child_slots = <non-negative integer>`"
77    );
78    assert!(
79        !(type_final && child_slots > 0),
80        "a final object type cannot reserve child slots"
81    );
82    assert!(
83        type_register
84            || (options.child_slots.is_none() && options.child_slots_can_overflow.is_none()),
85        "`child_slots` and `child_slots_can_overflow` need `register`"
86    );
87    // Upcasts reinterpret a pointer to the object as one to its parent, which
88    // only `repr(C)` keeps at offset zero.
89    if !derive_input.attrs.iter().any(is_repr_c) {
90        abort!(struct_name, "#[derive(Object)] needs #[repr(C)]");
91    }
92    // we expect base always to be the first field
93    let base_ty = match &derive_input.data {
94        syn::Data::Struct(s) => s.fields.iter().next().map(|f| f.ty.clone()),
95        _ => None,
96    }
97    .expect("First field must be `<base_name>: <ObjectCoreType>`");
98
99    let reflection = Reflection::parse(derive_input, &options, rs_object);
100    let writable = reflection
101        .iter()
102        .flat_map(|reflection| &reflection.reflected)
103        .find(|field| field.flags.iter().any(|flag| flag == "writable"));
104    if let (true, Some(field)) = (type_immutable, writable) {
105        abort!(
106            field.id,
107            "an immutable type cannot have a #[reflect(writable)] field"
108        );
109    }
110    let has_writable_field = writable.is_some();
111    let immutable_impl = if type_immutable {
112        quote! {
113            unsafe impl #tvm_ffi_crate::object::ImmutableObject for #struct_name {}
114        }
115    } else {
116        quote! {}
117    };
118    assert!(
119        reflection.is_none() || (type_register && options.type_index.is_none()),
120        "#[reflect], `s_eq_hash_kind` and `reflection` need `register` and no `type_index`"
121    );
122    let register_reflection = match &reflection {
123        Some(reflection) => reflection.register_tokens(&struct_name, &type_key, tvm_ffi_crate),
124        None => quote! {},
125    };
126    let (base_id, field_ids) = struct_field_ids(derive_input);
127    let shallow_copy_impl = if reflection.is_some() || type_rs_parent {
128        shallow_copy_tokens(&struct_name, &base_ty, &base_id, &field_ids, tvm_ffi_crate)
129    } else {
130        quote! {}
131    };
132    let unsafe_init_impl = if rs_object || type_rs_parent {
133        if !derive_input.generics.params.is_empty() {
134            abort!(
135                struct_name,
136                "`rs_parent` needs a struct without generic parameters"
137            );
138        }
139        unsafe_init_tokens(&struct_name, &base_ty, &base_id, &field_ids, tvm_ffi_crate)
140    } else {
141        quote! {}
142    };
143
144    // A type with a static index has it; one defined here registers its key
145    // under its parent on its first `type_index()`, which a static init block
146    // calls at load, or takes the index the key already has,
147    // as C++ `TVM_FFI_DECLARE_OBJECT_INFO` does; any other type binds the
148    // index its defining library registered.
149    let type_index_tokens = match &options.type_index {
150        // A static index is trusted: the type must be the one the runtime
151        // gives that index.
152        Some(type_index_expr) => {
153            quote! {
154                #[inline]
155                fn type_index() -> i32 {
156                    #type_index_expr as i32
157                }
158            }
159        }
160        None if type_register => {
161            quote! {
162                #[inline]
163                fn type_index() -> i32 {
164                    static TYPE_INDEX: #tvm_ffi_crate::extra::cold_init_lazy_lock::ColdInitLazyLock<i32> = #tvm_ffi_crate::extra::cold_init_lazy_lock::ColdInitLazyLock::new(||
165                        unsafe {
166                            // The parent first, since registering it takes the
167                            // same lock.
168                            let parent =
169                                <#base_ty as #tvm_ffi_crate::object::ObjectCore>::type_index();
170                            let _registering =
171                                #tvm_ffi_crate::object::lock_type_registration();
172                            let type_key_arg =
173                                 #tvm_ffi_crate::tvm_ffi_sys::TVMFFIByteArray::from_str(#type_key);
174                            let tindex = #tvm_ffi_crate::tvm_ffi_sys::TVMFFITypeGetOrAllocIndex(
175                                &type_key_arg,
176                                -1,
177                                <#struct_name as #tvm_ffi_crate::object::ObjectCore>::TYPE_DEPTH,
178                                #child_slots,
179                                #child_slots_can_overflow as i32,
180                                parent,
181                            );
182                            if tindex < 0 {
183                                ::core::panic!(
184                                    "Failed to get or allocate type index for type key: {}",
185                                    #type_key
186                                );
187                            }
188                            #tvm_ffi_crate::object::assert_registered_type_matches::<#struct_name>(
189                                tindex, parent, false,
190                            );
191                            #register_reflection
192                            tindex
193                        }
194                    );
195                    *TYPE_INDEX
196                }
197
198                fn type_index_available() -> bool {
199                    <#base_ty as #tvm_ffi_crate::object::ObjectCore>::type_index_available()
200                }
201            }
202        }
203        None => {
204            quote! {
205                #[inline]
206                fn type_index() -> i32 {
207                    static TYPE_INDEX: #tvm_ffi_crate::extra::cold_init_lazy_lock::ColdInitLazyLock<i32> = #tvm_ffi_crate::extra::cold_init_lazy_lock::ColdInitLazyLock::new(||
208                        unsafe {
209                            let type_key_arg =
210                                 #tvm_ffi_crate::tvm_ffi_sys::TVMFFIByteArray::from_str(#type_key);
211                            let mut tindex = 0;
212                            let ret = #tvm_ffi_crate::tvm_ffi_sys::TVMFFITypeKeyToIndex(
213                                &type_key_arg, &mut tindex
214                            );
215                            if ret != 0 {
216                                ::core::panic!(
217                                    "Type key {} is not registered: load the library that defines \
218                                     it first, or add #[object(register)] if this crate defines \
219                                     it",
220                                    #type_key
221                                );
222                            }
223                            #tvm_ffi_crate::object::assert_registered_type_matches::<#struct_name>(
224                                tindex,
225                                <#base_ty as #tvm_ffi_crate::object::ObjectCore>::type_index(),
226                                true,
227                            );
228                            tindex
229                        }
230                    );
231                    *TYPE_INDEX
232                }
233
234                fn type_index_available() -> bool {
235                    let mut tindex = 0;
236                    // SAFETY: the key and the index outlive the call.
237                    let found = unsafe {
238                        let type_key_arg =
239                            #tvm_ffi_crate::tvm_ffi_sys::TVMFFIByteArray::from_str(#type_key);
240                        #tvm_ffi_crate::tvm_ffi_sys::TVMFFITypeKeyToIndex(&type_key_arg, &mut tindex)
241                            == 0
242                    };
243                    if !found {
244                        // Discard the error the failed lookup raised.
245                        ::core::mem::drop(#tvm_ffi_crate::Error::from_raised());
246                    }
247                    found
248                }
249            }
250        }
251    };
252    // A type defined here registers when its library loads, as a C++ type does
253    // through `refl::ObjectDef` in a static init block; a binding stays lazy,
254    // and so does a type under a binding whose library is not loaded yet.
255    let register_at_load = if type_register && options.type_index.is_none() {
256        if !derive_input.generics.params.is_empty() {
257            abort!(
258                struct_name,
259                "`register` needs a struct without generic parameters"
260            );
261        }
262        // The key is in the text of the block, which identifies the block, so
263        // two libraries that differ only in their keys both register.
264        quote! {
265            #tvm_ffi_crate::static_init_block! {
266                let _: &str = #type_key;
267                if <#struct_name as #tvm_ffi_crate::object::ObjectCore>::type_index_available() {
268                    let _ = <#struct_name as #tvm_ffi_crate::object::ObjectCore>::type_index();
269                }
270            }
271        }
272    } else {
273        quote! {}
274    };
275    // search for field name base and derive the base type
276    // we expect base always to be the first field
277    let final_parent_check = match &derive_input.data {
278        syn::Data::Struct(s) => s.fields.iter().next().map(|f| {
279            let base_ty = f.ty.clone();
280            quote! {
281                const _: () = {
282                    ::core::assert!(
283                        !<#base_ty as #tvm_ffi_crate::object::ObjectCore>::TYPE_FINAL,
284                        "an object type cannot derive from a final parent"
285                    );
286                    ::core::assert!(
287                        #type_immutable
288                            || !<#base_ty as #tvm_ffi_crate::object::ObjectCore>::TYPE_IMMUTABLE,
289                        "a subtype of an immutable object type must be immutable"
290                    );
291                    ::core::assert!(
292                        !#type_immutable
293                            || !<#base_ty as #tvm_ffi_crate::object::ObjectCore>::TYPE_HAS_WRITABLE_FIELD,
294                        "an immutable type cannot inherit a #[reflect(writable)] field"
295                    );
296                };
297            }
298        }),
299        _ => panic!("First field must be `<base_name>: <ObjectCoreType>`"),
300    };
301    let base_def_tokens = match &derive_input.data {
302        syn::Data::Struct(s) => s.fields.iter().next().and_then(|f| {
303            let (base_id, base_ty) = (f.ident.clone()?, f.ty.clone());
304            // The transitive case of subtyping
305            Some(quote! {
306                type Parent = #base_ty;
307
308                const TYPE_DEPTH: i32 =
309                    <#base_ty as #tvm_ffi_crate::object::ObjectCore>::TYPE_DEPTH + 1;
310
311                #[inline]
312                unsafe fn object_header_mut(
313                    this: &mut Self
314                ) -> &mut  #tvm_ffi_crate::tvm_ffi_sys::TVMFFIObject {
315                    const _: () = {
316                        fn assert_impl<T: #tvm_ffi_crate::object::ObjectCore>() {}
317                        let _ = assert_impl::<#base_ty>;
318                    };
319                    <#base_ty as #tvm_ffi_crate::object::ObjectCore>::object_header_mut(
320                        &mut this.#base_id
321                    )
322                }
323            })
324        }),
325        _ => panic!("First field must be `<base_name>: <ObjectCoreType>`"),
326    };
327
328    quote! {
329        #final_parent_check
330
331        const _: () = ::core::assert!(
332            ::core::mem::offset_of!(#struct_name, #base_id) == 0,
333            "the parent object must be the first field"
334        );
335
336        unsafe impl #tvm_ffi_crate::object::ObjectCore for #struct_name {
337            const TYPE_KEY: &'static str = #type_key;
338            const TYPE_FINAL: bool = #type_final;
339            const TYPE_IMMUTABLE: bool = #type_immutable;
340            const TYPE_HAS_WRITABLE_FIELD: bool =
341                <#base_ty as #tvm_ffi_crate::object::ObjectCore>::TYPE_HAS_WRITABLE_FIELD
342                    || #has_writable_field;
343
344            #type_index_tokens
345
346            #base_def_tokens
347        }
348
349        impl<'a> ::core::convert::From<&'a #struct_name> for #tvm_ffi_crate::AnyView<'a> {
350            #[inline(always)]
351            fn from(value: &'a #struct_name) -> Self {
352                Self::from_object(value)
353            }
354        }
355
356        #shallow_copy_impl
357
358        #unsafe_init_impl
359
360        #immutable_impl
361
362        #register_at_load
363    }
364}

366/// The reflection a `register` type declares: its `#[reflect]` fields, its
367/// `s_eq_hash_kind` and its `reflection` function.
368struct Reflection {
369    /// The structural kind; `None` inherits the parent's.
370    kind: Option<syn::Ident>,
371    /// The function `reflection = path` names, which registers methods.
372    methods: Option<syn::Path>,
373    /// The reflected fields.
374    reflected: Vec<ReflectedField>,
375    /// Whether the type registers the creator of its `UnsafeInit`, as an
376    /// `#[rs_object]` type does.
377    unsafe_init: bool,
378}
379
380/// A `#[reflect]` field: its identifier, type, flag names, and the name and
381/// doc string registered for it.
382struct ReflectedField {
383    id: syn::Ident,
384    ty: syn::Type,
385    flags: Vec<syn::Ident>,
386    name: String,
387    doc: String,
388}
389
390const EXPECTED_REFLECT: &str =
391    "Expect #[reflect] or #[reflect(<flag>, ..., name = \"<name>\", doc = \"<doc>\")]";
392
393impl ReflectedField {
394    /// The field's reflection from its `#[reflect]` attribute, whose `name`
395    /// and `doc` default to the field's name and its `///` doc comment.
396    fn parse(field: &syn::Field, attr: &syn::Attribute) -> Self {
397        let id = field
398            .ident
399            .clone()
400            .expect("#[derive(Object)] needs named fields");
401        let mut flags = Vec::new();
402        let mut name = None;
403        let mut doc = None;
404        let nested = match attr.parse_meta() {
405            Ok(syn::Meta::Path(_)) => Vec::new(),
406            Ok(syn::Meta::List(list)) => list.nested.into_iter().collect(),
407            _ => panic!("{EXPECTED_REFLECT}"),
408        };
409        for item in nested {
410            match item {
411                syn::NestedMeta::Meta(syn::Meta::Path(path)) => {
412                    flags.push(path.get_ident().cloned().expect(EXPECTED_REFLECT));
413                }
414                syn::NestedMeta::Meta(syn::Meta::NameValue(syn::MetaNameValue {
415                    path,
416                    lit: syn::Lit::Str(value),
417                    ..
418                })) if path.is_ident("name") || path.is_ident("doc") => {
419                    let slot = if path.is_ident("name") {
420                        &mut name
421                    } else {
422                        &mut doc
423                    };
424                    assert!(
425                        slot.is_none(),
426                        "duplicate `{}` in #[reflect] of field `{id}`",
427                        path.get_ident().unwrap()
428                    );
429                    *slot = Some(value.value());
430                }
431                _ => panic!("{EXPECTED_REFLECT}"),
432            }
433        }
434        Self {
435            name: name.unwrap_or_else(|| id.unraw().to_string()),
436            doc: doc.unwrap_or_else(|| doc_comment(&field.attrs)),
437            id,
438            ty: field.ty.clone(),
439            flags,
440        }
441    }
442}

444/// The `///` doc comment of an item, as rustdoc reads it: the lines of its
445/// `#[doc = "..."]` attributes without their leading space, joined by newlines.
446fn doc_comment(attrs: &[syn::Attribute]) -> String {
447    let lines: Vec<String> = attrs
448        .iter()
449        .filter(|attr| attr.path.is_ident("doc"))
450        .filter_map(|attr| match attr.parse_meta() {
451            Ok(syn::Meta::NameValue(syn::MetaNameValue {
452                lit: syn::Lit::Str(line),
453                ..
454            })) => Some(line.value()),
455            _ => None,
456        })
457        .map(|line| line.strip_prefix(' ').unwrap_or(&line).to_string())
458        .collect();
459    lines.join("\n").trim().to_string()
460}

462impl Reflection {
463    /// The type's reflection, or `None` when it declares none and is not an
464    /// `rs_object`.
465    fn parse(derive_input: &DeriveInput, options: &ObjectOptions, rs_object: bool) -> Option<Self> {
466        let kind = options
467            .s_eq_hash_kind
468            .as_ref()
469            .map(|kind| syn::Ident::new(&kind.value(), kind.span()));
470        let methods = options.reflection.clone();
471        let syn::Data::Struct(s) = &derive_input.data else {
472            return None;
473        };
474        let mut fields = s.fields.iter();
475        let base = fields.next()?;
476        assert!(
477            !base.attrs.iter().any(|a| a.path.is_ident("reflect")),
478            "the parent field cannot be #[reflect]; the parent registers its own fields"
479        );
480        let mut reflected: Vec<ReflectedField> = Vec::new();
481        for field in fields {
482            if let Some(attr) = field.attrs.iter().find(|a| a.path.is_ident("reflect")) {
483                let field = ReflectedField::parse(field, attr);
484                if let Some(other) = reflected.iter().find(|f| f.name == field.name) {
485                    panic!(
486                        "fields `{}` and `{}` of `{}` both reflect as `{}`",
487                        other.id, field.id, derive_input.ident, field.name
488                    );
489                }
490                reflected.push(field);
491            }
492        }
493        if kind.is_none() && methods.is_none() && reflected.is_empty() && !rs_object {
494            return None;
495        }
496        Some(Self {
497            kind,
498            methods,
499            reflected,
500            unsafe_init: rs_object,
501        })
502    }
503
504    /// Registers the reflection under `tindex`, inside the registration of
505    /// the type index, which holds the registration lock.
506    fn register_tokens(
507        &self,
508        struct_name: &syn::Ident,
509        type_key: &syn::LitStr,
510        tvm_ffi_crate: &proc_macro2::TokenStream,
511    ) -> proc_macro2::TokenStream {
512        let reflection = quote!(#tvm_ffi_crate::reflection);
513        let kind = match &self.kind {
514            Some(kind) => quote!(::core::option::Option::Some(#reflection::SEqHashKind::#kind)),
515            None => quote!(::core::option::Option::None),
516        };
517        let fields = self.reflected.iter().map(|field| {
518            let ReflectedField {
519                id,
520                ty,
521                flags,
522                name,
523                doc,
524            } = field;
525            // `#[reflect(seq_hash_ignore)]` names `FieldFlags::SEQ_HASH_IGNORE`.
526            let flags = flags.iter().map(|flag| {
527                format_ident!("{}", flag.to_string().to_uppercase(), span = flag.span())
528            });
529            quote! {
530                let def = def.field::<#ty>(
531                    #name,
532                    #doc,
533                    ::core::mem::offset_of!(#struct_name, #id),
534                    #reflection::FieldFlags::NONE #(| #reflection::FieldFlags::#flags)*,
535                )?;
536            }
537        });
538        // The `reflection` function registers the methods after the
539        // fields, as a C++ `ObjectDef` chains `def` after `def_ro`.
540        let methods = self.methods.as_ref().map(|path| {
541            // The registration is a `static` initializer, where `Self` does
542            // not resolve, so `Self::f` names the struct.
543            let mut path = path.clone();
544            if let Some(first) = path.segments.first_mut()
545                && first.ident == "Self"
546            {
547                first.ident = struct_name.clone();
548            }
549            quote!(#path)
550        });
551        let methods = methods.unwrap_or_else(|| quote!(|_| {}));
552        let creator = if self.unsafe_init {
553            quote!(::core::option::Option::Some(#reflection::unsafe_init_creator::<#struct_name>))
554        } else {
555            quote!(::core::option::Option::None)
556        };
557        quote! {
558            let registered = #reflection::register_object::<#struct_name>(
559                tindex,
560                #kind,
561                #creator,
562                |def| {
563                    #(#fields)*
564                    ::core::result::Result::Ok(def)
565                },
566                #methods,
567            );
568            if let ::core::result::Result::Err(error) = registered {
569                ::core::panic!("Failed to register the reflection of type key {}: {}", #type_key, error);
570            }
571        }
572    }
573}

575/// The parent's field name and every other field's name.
576fn struct_field_ids(derive_input: &DeriveInput) -> (syn::Ident, Vec<syn::Ident>) {
577    let syn::Data::Struct(s) = &derive_input.data else {
578        panic!("First field must be `<base_name>: <ObjectCoreType>`");
579    };
580    let mut ids = s.fields.iter().map(|field| {
581        field
582            .ident
583            .clone()
584            .expect("#[derive(Object)] needs named fields")
585    });
586    let base_id = ids
587        .next()
588        .expect("First field must be `<base_name>: <ObjectCoreType>`");
589    (base_id, ids.collect())
590}

592/// `ShallowCopy`: the parent's shallow copy and a clone of every other field.
593fn shallow_copy_tokens(
594    struct_name: &syn::Ident,
595    base_ty: &syn::Type,
596    base_id: &syn::Ident,
597    field_ids: &[syn::Ident],
598    tvm_ffi_crate: &TokenStream2,
599) -> TokenStream2 {
600    quote! {
601        impl #tvm_ffi_crate::reflection::ShallowCopy for #struct_name {
602            fn shallow_copy(&self) -> Self {
603                Self {
604                    #base_id: <#base_ty as #tvm_ffi_crate::reflection::ShallowCopy>
605                        ::shallow_copy(&self.#base_id),
606                    #(#field_ids: ::core::clone::Clone::clone(&self.#field_ids),)*
607                }
608            }
609        }
610    }
611}

613/// `UnsafeInit`: drops the parent's defined fields, then each other defined
614/// field in order, as the drop glue does; only through raw pointers, since
615/// some may be undefined.
616fn unsafe_init_tokens(
617    struct_name: &syn::Ident,
618    base_ty: &syn::Type,
619    base_id: &syn::Ident,
620    field_ids: &[syn::Ident],
621    tvm_ffi_crate: &TokenStream2,
622) -> TokenStream2 {
623    let field_drop = quote!(#tvm_ffi_crate::reflection::field_drop);
624    quote! {
625        unsafe impl #tvm_ffi_crate::reflection::UnsafeInit for #struct_name {
626            #[allow(unused_imports)]
627            unsafe fn drop_defined_fields(this: *mut Self) {
628                use #field_drop::{DropDefined as _, DropNothing as _};
629                unsafe {
630                    <#base_ty as #tvm_ffi_crate::reflection::UnsafeInit>::drop_defined_fields(
631                        ::core::ptr::addr_of_mut!((*this).#base_id),
632                    );
633                    #((&#field_drop::FieldPtr(::core::ptr::addr_of_mut!((*this).#field_ids)))
634                        .drop_field();)*
635                }
636            }
637        }
638    }
639}

642pub fn derive_object_ref(input: proc_macro::TokenStream) -> TokenStream {

645}

647/// The impls of a reference struct, as configured by `#[object_ref(...)]`,
648/// naming the tvm-ffi crate `tvm_ffi_crate`.
649pub(crate) fn expand_object_ref(
650    derive_input: &DeriveInput,
651    tvm_ffi_crate: &TokenStream2,
652) -> TokenStream2 {
653    let struct_name = derive_input.ident.clone();
654    let options = object_ref_options(derive_input);
655    let tvm_ffi_crate = &match &options.tvm_ffi_crate {
656        Some(path) => quote!(#path),
657        None => tvm_ffi_crate.clone(),
658    };
659
660    // The `ObjectArc<T>` slot is the first field; its name is the struct's choice
661    // (`data` in this crate, `data_` in stubgen output, so that reflected fields
662    // named `data` keep their name).
663    let (data_id, data_ty) = match &derive_input.data {
664        syn::Data::Struct(s) => s
665            .fields
666            .iter()
667            .next()
668            .and_then(|f| Some((f.ident.clone()?, f.ty.clone()))),
669        _ => panic!("derive only works for structs"),
670    }
671    .expect("First field must be `<name>: ObjectArc<T>`");
672    // Spell the object type as written, so rustdoc shows `Target = FooObj`.
673    let object_ty = object_arc_argument(&data_ty)
674        .unwrap_or_else(|| quote!(<#data_ty as ::core::ops::Deref>::Target));
675
676    let mut expanded = quote! {
677        unsafe impl #tvm_ffi_crate::object::ObjectRefCore for #struct_name {
678            type ContainerType = #object_ty;
679            #[inline]
680            fn data(this: &Self) -> &#tvm_ffi_crate::object::ObjectArc<Self::ContainerType> {
681                &this.#data_id
682            }
683            #[inline]
684            fn into_data(this: Self) -> #tvm_ffi_crate::object::ObjectArc<Self::ContainerType> {
685                this.#data_id
686            }
687            #[inline]
688            unsafe fn from_data(
689                data: #tvm_ffi_crate::object::ObjectArc<Self::ContainerType>
690            ) -> Self {
691                Self { #data_id: data }
692            }
693        }
694
695        impl #struct_name {
696            /// The reference to a new object, for the type's constructors.
697            #[allow(dead_code)]
698            #[inline(always)]
699            fn from_object(object: #object_ty) -> Self {
700                Self { #data_id: #tvm_ffi_crate::object::ObjectArc::new(object) }
701            }
702        }
703
704        // The handle is exactly its object pointer, which an FFI cell's
705        // payload word holds, so an `Array` cell can lend `&Self` in place.
706        const _: () = {
707            type ContainerType = <#struct_name as #tvm_ffi_crate::object::ObjectRefCore>::ContainerType;
708            ::core::assert!(
709                ::core::mem::size_of::<#struct_name>()
710                    == ::core::mem::size_of::<#tvm_ffi_crate::object::ObjectArc<ContainerType>>()
711                    && ::core::mem::offset_of!(#struct_name, #data_id) == 0,
712                "an ObjectRef must consist of its ObjectArc alone"
713            );
714        };
715
716        // SAFETY: one `ObjectArc` (asserted above), checked by `check_any_strict`.
717        #tvm_ffi_crate::impl_object_ref_downcast_ref!(unsafe #struct_name);
718
719        // As `str`, `Path` and `[T]` do, so `impl AsRef<Base>` bounds accept the
720        // base itself; upcasts add `AsRef` to each base.
721        impl ::core::convert::AsRef<#struct_name> for #struct_name {
722            #[inline(always)]
723            fn as_ref(&self) -> &Self {
724                self
725            }
726        }
727
728        impl ::std::convert::From<&#struct_name> for #struct_name {
729            #[inline]
730            fn from(value: &#struct_name) -> Self {
731                ::core::clone::Clone::clone(value)
732            }
733        }
734    };
735    // The default `AnyCompatible`: the object type check alone, which
736    // `match_any!` may route by.
737    if !options.skip_any_compatible {
738        expanded.extend(quote! {
739            unsafe impl #tvm_ffi_crate::object::ObjectRefAnyCompatible for #struct_name {
740                const MATCH_ANY_EXACT: bool = true;
741            }
742        });
743        if derive_input.generics.params.is_empty() {
744            expanded.extend(quote! {
745                unsafe impl #tvm_ffi_crate::object::PlainObjectRef for #struct_name {}
746            });
747        }
748    }
749    // Field reads through the reference, as C++ `operator->`.
750    if !options.skip_deref {
751        expanded.extend(quote! {
752            impl ::core::ops::Deref for #struct_name {
753                type Target = #object_ty;
754                #[inline(always)]
755                fn deref(&self) -> &Self::Target {
756                    &self.#data_id
757                }
758            }
759        });
760    }
761    // The runtime type key and object address, so `Result<_, Self>` from
762    // `downcast` can be unwrapped.
763    if !options.skip_debug {
764        expanded.extend(quote! {
765            impl ::core::fmt::Debug for #struct_name {
766                fn fmt(&self, f: &mut ::core::fmt::Formatter<'_>) -> ::core::fmt::Result {
767                    #tvm_ffi_crate::object::fmt_object_ref(self, f)
768                }
769            }
770        });
771    }
772    // The runtime `repr`, as `AnyView` formats it.
773    if !options.skip_display {
774        expanded.extend(quote! {
775            impl ::core::fmt::Display for #struct_name {
776                fn fmt(&self, f: &mut ::core::fmt::Formatter<'_>) -> ::core::fmt::Result {
777                    let object = &**#tvm_ffi_crate::object::ObjectRefCore::data(self);
778                    ::core::fmt::Display::fmt(&#tvm_ffi_crate::AnyView::from_object(object), f)
779                }
780            }
781        });
782    }
783    // `ObjectRef` itself skips these: `any.rs` implements them for it.
784    if struct_name != "ObjectRef" {
785        expanded.extend(quote! {
786            #tvm_ffi_crate::impl_any_compatible_conversions!(#struct_name);
787            // Every object reference upcasts to the root `ObjectRef`.
788            #tvm_ffi_crate::impl_object_ref_upcast!(
789                unsafe #struct_name => #tvm_ffi_crate::object::ObjectRef
790            );
791        });
792    }
793    // The root upcast is already above.
794    if let Some(parent) = options.parent.filter(|parent| !is_object_ref_root(parent)) {
795        if options.unsafe_parent {
796            // The object type must still derive from the parent's.
797            expanded.extend(quote! {
798                const _: () = {
799                    fn derives_from<T: #tvm_ffi_crate::object::ObjectDerivesFrom<A, I>, A, I>() {}
800                    let _ = derives_from::<
801                        <#struct_name as #tvm_ffi_crate::object::ObjectRefCore>::ContainerType,
802                        <#parent as #tvm_ffi_crate::object::ObjectRefCore>::ContainerType,
803                        _,
804                    >;
805                };
806                #tvm_ffi_crate::impl_object_ref_upcast!(unsafe #struct_name => #parent);
807            });
808        } else {
809            expanded.extend(quote! {
810                #tvm_ffi_crate::impl_object_ref_upcast!(#struct_name => #parent);
811            });
812        }
813    }
814    expanded
815}

817/// Whether `attr` is a `#[repr(...)]` that includes `C`.
818fn is_repr_c(attr: &syn::Attribute) -> bool {
819    let Ok(syn::Meta::List(list)) = attr.parse_meta() else {
820        return false;
821    };
822    list.path.is_ident("repr")
823        && list.nested.iter().any(|item| {
824            matches!(item, syn::NestedMeta::Meta(syn::Meta::Path(path)) if path.is_ident("C"))
825        })
826}

828/// `T` of a field type spelled `ObjectArc<T>`.
829fn object_arc_argument(ty: &syn::Type) -> Option<TokenStream2> {
830    let syn::Type::Path(path) = ty else {
831        return None;
832    };
833    let last = path.path.segments.last()?;
834    if last.ident != "ObjectArc" {
835        return None;
836    }
837    match &last.arguments {
838        syn::PathArguments::AngleBracketed(args) => match args.args.first()? {
839            syn::GenericArgument::Type(object) if args.args.len() == 1 => Some(quote!(#object)),
840            _ => None,
841        },
842        _ => None,
843    }
844}

846/// Whether `ty` names the root `ObjectRef`.
847fn is_object_ref_root(ty: &syn::Type) -> bool {
848    match ty {
849        syn::Type::Path(path) => path
850            .path
851            .segments
852            .last()
853            .is_some_and(|s| s.ident == "ObjectRef"),
854        _ => false,
855    }
856}

858/// The options of `#[object_ref(...)]` on an `ObjectRef` derive.
859#[derive(Default)]
860struct ObjectRefOptions {
861    tvm_ffi_crate: Option<syn::Path>,
862    parent: Option<syn::Type>,
863    unsafe_parent: bool,
864    skip_deref: bool,
865    skip_debug: bool,
866    skip_display: bool,
867    skip_any_compatible: bool,
868}
869
870/// One `#[object_ref(...)]` item: a flag, `[unsafe] parent = Type` or `crate = path`.
871enum ObjectRefOption {
872    Flag(syn::Ident),
873    Parent(Box<syn::Type>, bool),
874    Crate(syn::Path),
875}
876
877impl syn::parse::Parse for ObjectRefOption {
878    fn parse(input: syn::parse::ParseStream) -> syn::Result<Self> {
879        let is_unsafe = input.parse::<Option<syn::Token![unsafe]>>()?.is_some();
880        let name: syn::Ident = input.call(syn::Ident::parse_any)?;
881        if name == "parent" {
882            input.parse::<syn::Token![=]>()?;
883            return Ok(ObjectRefOption::Parent(Box::new(input.parse()?), is_unsafe));
884        }
885        if is_unsafe {
886            return Err(syn::Error::new(name.span(), "only `parent` takes `unsafe`"));
887        }
888        if name == "crate" {
889            input.parse::<syn::Token![=]>()?;
890            return Ok(ObjectRefOption::Crate(syn::Path::parse_mod_style(input)?));
891        }
892        Ok(ObjectRefOption::Flag(name))
893    }
894}
895
896fn object_ref_options(derive_input: &DeriveInput) -> ObjectRefOptions {
897    let mut options = ObjectRefOptions::default();
898    for attr in derive_input
899        .attrs
900        .iter()
901        .filter(|a| a.path.is_ident("object_ref"))
902    {
903        let items = attr
904            .parse_args_with(
905                syn::punctuated::Punctuated::<ObjectRefOption, syn::Token![,]>::parse_terminated,
906            )
907            .unwrap_or_else(|error| abort!(error.span(), "{}", error));
908        for item in items {
909            match item {
910                ObjectRefOption::Parent(parent, is_unsafe) => {
911                    options.parent = Some(*parent);
912                    options.unsafe_parent = is_unsafe;
913                }
914                ObjectRefOption::Crate(path) => options.tvm_ffi_crate = Some(path),
915                ObjectRefOption::Flag(flag) if flag == "skip_deref" => options.skip_deref = true,
916                ObjectRefOption::Flag(flag) if flag == "skip_debug" => options.skip_debug = true,
917                ObjectRefOption::Flag(flag) if flag == "skip_display" => {
918                    options.skip_display = true
919                }
920                ObjectRefOption::Flag(flag) if flag == "skip_any_compatible" => {
921                    options.skip_any_compatible = true
922                }
923                ObjectRefOption::Flag(flag) => abort!(
924                    flag,
925                    "unknown `object_ref` option; expected `[unsafe] parent = ...`, `skip_deref`, \
926                     `skip_debug`, `skip_display`, `skip_any_compatible` or `crate = ...`"
927                ),
928            }
929        }
930    }
931    options
932}

934/// The options of `#[object(...)]` on an `Object` derive.
935#[derive(Default)]
936struct ObjectOptions {
937    type_key: Option<syn::LitStr>,
938    type_index: Option<syn::Expr>,
939    type_final: bool,
940    immutable: bool,
941    register: bool,
942    rs_parent: bool,
943    child_slots: Option<syn::LitInt>,
944    child_slots_can_overflow: Option<syn::LitBool>,
945    s_eq_hash_kind: Option<syn::LitStr>,
946    reflection: Option<syn::Path>,
947}
948
949impl ObjectOptions {
950    /// Parses one `#[object(...)]` item: a flag, `name = value` or
951    /// `unsafe type_index = expr`.
952    fn parse_item(&mut self, input: syn::parse::ParseStream) -> syn::Result<()> {
953        let is_unsafe = input.parse::<Option<syn::Token![unsafe]>>()?.is_some();
954        // `final` is a keyword.
955        let name: syn::Ident = input.call(syn::Ident::parse_any)?;
956        if is_unsafe != (name == "type_index") {
957            let message = if is_unsafe {
958                "only `type_index` takes `unsafe`"
959            } else {
960                "expected `unsafe type_index = ...`: the type must be the one the runtime gives \
961                 that index"
962            };
963            return Err(syn::Error::new(name.span(), message));
964        }
965        match name.to_string().as_str() {
966            "final" => self.type_final = true,
967            "immutable" => self.immutable = true,
968            "register" => self.register = true,
969            "rs_parent" => self.rs_parent = true,
970            "type_key" => self.type_key = Some(parse_value(input)?),
971            "type_index" => self.type_index = Some(parse_value(input)?),
972            "child_slots" => self.child_slots = Some(parse_value(input)?),
973            "child_slots_can_overflow" => self.child_slots_can_overflow = Some(parse_value(input)?),
974            "s_eq_hash_kind" => self.s_eq_hash_kind = Some(parse_value(input)?),
975            "reflection" => self.reflection = Some(parse_value(input)?),
976            _ => {
977                return Err(syn::Error::new(
978                    name.span(),
979                    "unknown `object` option; expected `type_key = \"...\"`, \
980                     `unsafe type_index = ...`, `final`, `immutable`, `register`, `rs_parent`, \
981                     `child_slots = <n>`, `child_slots_can_overflow = <bool>`, \
982                     `s_eq_hash_kind = \"<kind>\"` or `reflection = <path>`",
983                ));
984            }
985        }
986        Ok(())
987    }
988}

990/// The `= value` of a `name = value` option.
991fn parse_value<T: syn::parse::Parse>(input: syn::parse::ParseStream) -> syn::Result<T> {
992    input.parse::<syn::Token![=]>()?;
993    input.parse()
994}

996fn object_options(derive_input: &DeriveInput) -> ObjectOptions {
997    let mut options = ObjectOptions::default();
998    for attr in derive_input
999        .attrs
1000        .iter()
1001        .filter(|a| a.path.is_ident("object"))
1002    {
1003        attr.parse_args_with(|input: syn::parse::ParseStream| {
1004            while !input.is_empty() {
1005                options.parse_item(input)?;
1006                if !input.is_empty() {
1007                    input.parse::<syn::Token![,]>()?;
1008                }
1009            }
1010            Ok(())
1011        })
1012        .unwrap_or_else(|error| abort!(error.span(), "{}", error));
1013    }
1014    options
1015}
