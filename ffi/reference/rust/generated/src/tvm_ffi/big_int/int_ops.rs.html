19//! Word-level integer algorithms behind the `BigInt` operators.
20//!
21//! Every fallback takes canonical two's-complement words, least-significant
22//! first, and returns a canonical `BigInt`. Word arithmetic wraps modulo 2^64
23//! on purpose, hence the explicit `wrapping_*`/`overflowing_*` forms. A result
24//! whose words are each computed once is written straight into a fresh object
25//! and pruned in place (`BigInt::build`); multiplication and division
26//! accumulate into a zeroed [`WordsBuf`] and finish with [`BigInt::from_words`].
27
28// The functions mirror C++ `detail::int_ops` in `include/tvm/ffi/big_int.h` one
29// for one; keep them in sync.
30#![allow(clippy::needless_range_loop)]
31

33use crate::error::{Error, OVERFLOW_ERROR, Result, VALUE_ERROR, ZERO_DIVISION_ERROR};
34use std::mem::MaybeUninit;

36const BASE: u64 = 1 << 32;
37
38/// Words kept on the stack before a result or scratch buffer spills to the heap.
39const INLINE_WORDS: usize = 32;
40
41/// Result or scratch words, zeroed in place on first use: up to `INLINE_WORDS` on
42/// the caller's stack, longer on the heap. Two phases avoid moving the array.
43struct WordsBuf {
44    inline: MaybeUninit<[i64; INLINE_WORDS]>,
45    heap: Vec<i64>,
46}
47
48impl WordsBuf {
49    #[inline(always)]
50    fn new() -> Self {
51        Self {
52            inline: MaybeUninit::uninit(),
53            heap: Vec::new(),
54        }
55    }
56
57    /// Zero `len` words in place and expose them; like `Vec`, exhaustion aborts.
58    #[inline]
59    fn zeroed(&mut self, len: usize) -> &mut [i64] {
60        if len <= INLINE_WORDS {
61            return self.inline_zeroed(len);
62        }
63        self.heap.resize(len, 0);
64        &mut self.heap
65    }
66
67    #[inline]
68    fn inline_zeroed(&mut self, len: usize) -> &mut [i64] {
69        debug_assert!(len <= INLINE_WORDS);
70        let words = self.inline.as_mut_ptr().cast::<i64>();
71        // Only these `len` words are ever exposed, and they are initialized here.
72        unsafe {
73            std::ptr::write_bytes(words, 0, len);
74            std::slice::from_raw_parts_mut(words, len)
75        }
76    }
77}

91/// Whether a nonempty signed-word view represents a negative integer.
92#[inline]
93pub(super) fn is_negative(x: &[i64]) -> bool {
94    x[x.len() - 1] < 0
95}

97/// The word that sign-extends `x` beyond its length.
98#[inline]
99pub(super) fn extension(x: &[i64]) -> u64 {
100    if is_negative(x) { u64::MAX } else { 0 }
101}
102
103/// Word `i` of `x`, sign-extended past its length.
104#[inline]
105pub(super) fn word_at(x: &[i64], i: usize, extension: u64) -> u64 {
106    x.get(i).map_or(extension, |&w| w as u64)
107}

109/// Length after pruning redundant sign words; zero words stay zero.
110pub(super) fn normalized_len(words: &[i64]) -> usize {
111    let mut size = words.len();
112    while size > 1 && words[size - 1] == (if words[size - 2] < 0 { -1 } else { 0 }) {
113        size -= 1;
114    }
115    size
116}

120/// Quotient of division rounded toward zero.
121#[inline]
122pub(super) fn trunc_div_i64(a: i64, b: i64) -> Result<Option<i64>> {
123    if b == 0 {
124        return Err(zero_division());
125    }
126    // i64::MIN / -1 is the only quotient that needs promotion.
127    Ok(a.checked_div(b))
128}
129
130/// Remainder of division rounded toward zero.
131#[inline]
132pub(super) fn trunc_mod_i64(a: i64, b: i64) -> Result<Option<i64>> {
133    if b == 0 {
134        return Err(zero_division());
135    }
136    // The remainder of i64::MIN / -1 is zero, which the wrapping form yields.
137    Ok(Some(a.wrapping_rem(b)))
138}
139
140/// Quotient of division rounded down.
141#[inline]
142pub(super) fn floor_div_i64(a: i64, b: i64) -> Result<Option<i64>> {
143    if b == 0 {
144        return Err(zero_division());
145    }
146    Ok(a.checked_div(b).map(|q| {
147        // A nonexact quotient of opposite signs rounds one lower.
148        if a.wrapping_rem(b) != 0 && ((a < 0) != (b < 0)) {
149            q - 1
150        } else {
151            q
152        }
153    }))
154}
155
156/// Remainder of division rounded down.
157#[inline]
158pub(super) fn floor_mod_i64(a: i64, b: i64) -> Result<Option<i64>> {
159    if b == 0 {
160        return Err(zero_division());
161    }
162    let r = a.wrapping_rem(b);
163    // Compensate for the quotient rounding down, preserving a = q*b + r.
164    Ok(Some(if r != 0 && ((a < 0) != (b < 0)) {
165        r + b
166    } else {
167        r
168    }))
169}

171/// Left shift, rejecting negative counts.
172#[inline]
173pub(super) fn shl_i64(a: i64, b: i64) -> Result<Option<i64>> {
174    if b < 0 {
175        return Err(negative_shift());
176    }
177    if b == 0 || a == 0 {
178        return Ok(Some(a));
179    }
180    // For positive a require a <= i64::MAX >> b; for negative a require a >= -2^(63-b).
181    if b >= 64 || a > (i64::MAX >> b) || a < -(1i64 << (63 - b)) {
182        return Ok(None);
183    }
184    Ok(Some(a << b))
185}

187/// Right shift with arithmetic sign extension; this operation cannot overflow.
188#[inline]
189pub(super) fn shr_i64(a: i64, b: i64) -> Result<i64> {
190    if b < 0 {
191        return Err(negative_shift());
192    }
193    Ok(if b >= 64 {
194        if a < 0 { -1 } else { 0 }
195    } else {
196        a >> b
197    })
198}

202/// Add signed integers.
203pub(super) fn add(a: &[i64], b: &[i64]) -> BigInt {
204    // One sign-extension word retains the carry out until normalization.
205    let size = a.len().max(b.len()) + 1;
206    let (ext_a, ext_b) = (extension(a), extension(b));
207    let mut carry = 0u64;
208    BigInt::build(size, |i| {
209        let (sum, c1) = word_at(a, i, ext_a).overflowing_add(word_at(b, i, ext_b));
210        let (result, c2) = sum.overflowing_add(carry);
211        carry = u64::from(c1 | c2);
212        result as i64
213    })
214}
215
216/// Subtract signed integers: a - b = a + !b + 1 after sign extension.
217pub(super) fn sub(a: &[i64], b: &[i64]) -> BigInt {
218    let size = a.len().max(b.len()) + 1;
219    let (ext_a, ext_b) = (extension(a), extension(b));
220    let mut carry = 1u64;
221    BigInt::build(size, |i| {
222        let (sum, c1) = word_at(a, i, ext_a).overflowing_add(!word_at(b, i, ext_b));
223        let (result, c2) = sum.overflowing_add(carry);
224        carry = u64::from(c1 | c2);
225        result as i64
226    })
227}
228
229/// Negate a signed integer: -x = !x + 1, with room for negating a minimum value.
230pub(super) fn negate(x: &[i64]) -> BigInt {
231    let size = x.len() + 1;
232    let ext = extension(x);
233    let mut carry = 1u64;
234    BigInt::build(size, |i| {
235        let word = !word_at(x, i, ext);
236        let (result, overflow) = word.overflowing_add(carry);
237        carry = u64::from(overflow);
238        result as i64
239    })
240}

242/// Negate two's-complement words in place, modulo their fixed width.
243pub(super) fn negate_in_place(words: &mut [i64]) {
244    let mut carry = true;
245    for word in words.iter_mut() {
246        let (sum, overflow) = (!(*word as u64)).overflowing_add(u64::from(carry));
247        *word = sum as i64;
248        carry = overflow;
249    }
250}

252/// Multiply signed integers.
253pub(super) fn mul(a: &[i64], b: &[i64]) -> BigInt {
254    let (na, nb) = (a.len(), b.len());
255    let mut buf = WordsBuf::new();
256    let data = buf.zeroed(na + nb);
257    for i in 0..na {
258        let x = u128::from(a[i] as u64);
259        let mut carry = 0u128;
260        for j in 0..nb {
261            // x*y + word + carry < 2^128: each term is below its width's maximum.
262            let product = x * u128::from(b[j] as u64) + u128::from(data[i + j] as u64) + carry;
263            data[i + j] = product as u64 as i64;
264            carry = product >> 64;
265        }
266        data[i + nb] = carry as u64 as i64;
267    }
268    // With stored unsigned A/B and 0/1 negative indicators neg_a/neg_b,
269    // a = A - neg_a*2^(64*n), b = B - neg_b*2^(64*m). Thus ab is AB minus
270    // neg_a*B*2^(64*n) and neg_b*A*2^(64*m); the remaining
271    // neg_a*neg_b*2^(64*(n+m)) vanishes modulo the allocated n+m words.
272    let subtract_shifted = |data: &mut [i64], x: &[i64], shift: usize| {
273        let mut borrow = 0u64;
274        for i in 0..x.len() {
275            let word = x[i] as u64;
276            let (rhs, o1) = word.overflowing_add(borrow);
277            let (result, o2) = (data[shift + i] as u64).overflowing_sub(rhs);
278            data[shift + i] = result as i64;
279            borrow = u64::from(o1 | o2);
280        }
281    };
282    if is_negative(a) {
283        subtract_shifted(data, b, na);
284    }
285    if is_negative(b) {
286        subtract_shifted(data, a, nb);
287    }
288    BigInt::from_words(data)
289}

291/// Apply a sign-extended bitwise operation.
292fn bitwise(a: &[i64], b: &[i64], op: impl Fn(u64, u64) -> u64) -> BigInt {
293    let size = a.len().max(b.len());
294    let (ext_a, ext_b) = (extension(a), extension(b));
295    BigInt::build(size, |i| {
296        op(word_at(a, i, ext_a), word_at(b, i, ext_b)) as i64
297    })
298}

317/// Convert a nonnegative shift count; `None` when it does not fit `usize`.
318fn shift_count(count: &[i64]) -> Result<Option<usize>> {
319    if is_negative(count) {
320        return Err(negative_shift());
321    }
322    if count[1..].iter().any(|&w| w != 0) {
323        return Ok(None);
324    }
325    Ok(usize::try_from(count[0] as u64).ok())
326}

328/// Shift a signed integer left, growing as needed.
329///
330/// Inlined so callers unwrap the `Result` in registers: moving a `BigInt` out of a
331/// returned `Result` reloads it in one 16-byte read that store forwarding misses.
332#[inline]
333pub(super) fn shl(x: &[i64], count: &[i64]) -> Result<BigInt> {
334    let shift = shift_count(count)?;
335    if x.len() == 1 && x[0] == 0 {
336        return Ok(BigInt::from_i64(0));
337    }
338    let too_large = || overflow("BigInt shift count is too large");
339    let shift = shift.ok_or_else(too_large)?;
340    let whole = shift / 64;
341    let part = (shift % 64) as u32;
342    let size = x
343        .len()
344        .checked_add(whole)
345        .and_then(|size| size.checked_add(1))
346        .ok_or_else(too_large)?;
347    let ext = extension(x);
348    BigInt::try_build(size, |i| {
349        // The low `whole` words are zero.
350        if i < whole {
351            return 0;
352        }
353        let j = i - whole;
354        let mut word = word_at(x, j, ext) << part;
355        if part != 0 && j != 0 {
356            word |= (x[j - 1] as u64) >> (64 - part);
357        }
358        word as i64
359    })
360    .ok_or_else(|| overflow("BigInt allocation is too large"))
361}

363/// Shift a signed integer right with sign extension; huge counts give 0 or -1.
364pub(super) fn shr(x: &[i64], count: &[i64]) -> Result<BigInt> {
365    let shift = match shift_count(count)? {
366        Some(shift) if shift / 64 < x.len() => shift,
367        _ => return Ok(BigInt::from_i64(if is_negative(x) { -1 } else { 0 })),
368    };
369    let whole = shift / 64;
370    let size = x.len() - whole;
371    let part = (shift % 64) as u32;
372    let ext = extension(x);
373    let result = BigInt::build(size, |i| {
374        let mut word = (x[i + whole] as u64) >> part;
375        // The last partial word draws its high bits from the sign extension.
376        if part != 0 {
377            word |= word_at(x, i + whole + 1, ext) << (64 - part);
378        }
379        word as i64
380    });
381    Ok(result)
382}

384// Absolute-value access without copying operands: negation carries through the
385// low zero words, negates the first nonzero word, then complements higher words.
386
387/// The first nonzero index, or `x.len()` for zero.
388fn first_nonzero(x: &[i64]) -> usize {
389    x.iter().position(|&w| w != 0).unwrap_or(x.len())
390}

392/// Raw word `i` of the absolute value of `x`; indices beyond the view read as zero.
393#[inline]
394fn abs_word(x: &[i64], first_nonzero: usize, i: usize) -> i64 {
395    if i >= x.len() {
396        return 0;
397    }
398    let word = x[i];
399    if !is_negative(x) {
400        return word;
401    }
402    // Negation maps [0,0,w,v] to [0,0,-w,!v] as modulo-2^64 word patterns.
403    if i < first_nonzero {
404        0
405    } else if i == first_nonzero {
406        word.wrapping_neg()
407    } else {
408        !word
409    }
410}

412/// Truncating quotient and remainder with an `i64` divisor.
413///
414/// The remainder is zero or has the dividend's sign.
415pub(super) fn div_rem_single(a: &[i64], b: i64) -> Result<(BigInt, BigInt)> {
416    if b == 0 {
417        return Err(zero_division());
418    }
419    let divisor = b.unsigned_abs();
420    let first_a = first_nonzero(a);
421    let nq = a.len() + 1;
422    let mut quotient = WordsBuf::new();
423    // The top word stays zero as the sign guard.
424    let q = quotient.zeroed(nq);
425    let mut remainder = 0u64;
426    let mut shift = 0u32;
427    let mut normalized = divisor;
428    if divisor > u64::from(u32::MAX) {
429        while normalized >> 63 == 0 {
430            normalized <<= 1;
431            shift += 1;
432        }
433    }
434    let divisor_high = normalized >> 32;
435    let divisor_low = normalized as u32 as u64;
436    for i in (0..a.len()).rev() {
437        let word = abs_word(a, first_a, i) as u64;
438        let quotient_word = if divisor <= u64::from(u32::MAX) {
439            // remainder < divisor < 2^32, so each brought-down half fits u64
440            // and yields a quotient half below 2^32 while restoring the bound.
441            let high = (remainder << 32) | (word >> 32);
442            let quotient_high = high / divisor;
443            remainder = high % divisor;
444            let low = (remainder << 32) | (word as u32 as u64);
445            let quotient_low = low / divisor;
446            remainder = low % divisor;
447            (quotient_high << 32) | quotient_low
448        } else {
449            // Normalize the carried 128-bit numerator without ever shifting by 64.
450            let high = if shift != 0 {
451                (remainder << shift) | (word >> (64 - shift))
452            } else {
453                remainder
454            };
455            let low = word << shift;
456            // Requires upper < normalized; returns a radix-2^32 quotient digit
457            // and a remainder below normalized.
458            let digit = |upper: u64, next: u64| -> (u64, u64) {
459                let mut estimate = upper / divisor_high;
460                let mut residual = upper % divisor_high;
461                // The normalized high digit bounds overestimation by two. Check
462                // the base before multiplying, and stop before a residual shift
463                // can overflow.
464                while estimate >= BASE || estimate * divisor_low > (residual << 32) + next {
465                    estimate -= 1;
466                    residual += divisor_high;
467                    if residual >= BASE {
468                        break;
469                    }
470                }
471                // Wrapped subtraction is exact: the true remainder is below normalized.
472                let rest = (upper << 32)
473                    .wrapping_add(next)
474                    .wrapping_sub(estimate.wrapping_mul(normalized));
475                (estimate, rest)
476            };
477            let upper = digit(high, low >> 32);
478            let lower = digit(upper.1, low as u32 as u64);
479            remainder = lower.1 >> shift;
480            (upper.0 << 32) | lower.0
481        };
482        q[i] = quotient_word as i64;
483    }
484    if is_negative(a) != (b < 0) {
485        negate_in_place(&mut q[..nq]);
486    }
487    // remainder < |b| <= 2^63 makes the conversion and signed negation safe.
488    let mut signed_remainder = remainder as i64;
489    if is_negative(a) {
490        signed_remainder = -signed_remainder;
491    }
492    Ok((BigInt::from_words(q), BigInt::from_i64(signed_remainder)))
493}

495/// Truncating quotient and remainder in one division pass.
496///
497/// The remainder is zero or has the dividend's sign.
498pub(super) fn div_rem(a: &[i64], b: &[i64]) -> Result<(BigInt, BigInt)> {
499    if b.len() == 1 {
500        return div_rem_single(a, b[0]);
501    }
502    let first_a = first_nonzero(a);
503    let first_b = first_nonzero(b);
504    // Count magnitude digits, dropping the signed representation's high zero guards.
505    let digit_count = |x: &[i64], first: usize| -> usize {
506        let mut words = x.len();
507        while words > 0 && abs_word(x, first, words - 1) == 0 {
508            words -= 1;
509        }
510        if words == 0 {
511            return 0;
512        }
513        let top = abs_word(x, first, words - 1) as u64;
514        2 * words - usize::from(top >> 32 == 0)
515    };
516    let m = digit_count(a, first_a);
517    let n = digit_count(b, first_b);
518    if m < n {
519        return Ok((BigInt::from_i64(0), BigInt::from_words(a)));
520    }
521
522    // Algorithm D in radix 2^32 keeps products and estimates in u64. Digits below
523    // are least-significant first; the scalar dispatch left at least two in b.
524    let top_word = abs_word(b, first_b, (n - 1) / 2) as u64;
525    let mut top_digit = (top_word >> (((n - 1) % 2) * 32)) as u32;
526    let mut shift = 0u32;
527    // Normalize both inputs until B[0] >= base/2, bounding the quotient estimate error.
528    while top_digit < (1u32 << 31) {
529        top_digit <<= 1;
530        shift += 1;
531    }
532    let nq = (m - n) / 2 + 2;
533    let nr = n.div_ceil(2) + 1;
534    // The quotient words come first, then the Algorithm D scratch, in one buffer.
535    let capacity = nq
536        .checked_add(m + 1)
537        .and_then(|words| words.checked_add(n))
538        .ok_or_else(|| overflow("BigInt division workspace is too large"))?;
539    let mut quotient = WordsBuf::new();
540    let mut remainder = WordsBuf::new();
541    let (q, scratch) = quotient.zeroed(capacity).split_at_mut(nq);
542    let (u, v) = scratch.split_at_mut(m + 1);
543    let r = remainder.zeroed(nr);
544    let normalize = |x: &[i64], first: usize, digits: usize, out: &mut [i64]| -> u32 {
545        let mut carry = 0u64;
546        for i in 0..digits {
547            let word = abs_word(x, first, i / 2) as u64;
548            let part = (word >> ((i % 2) * 32)) as u32 as u64;
549            let value = (part << shift) | carry;
550            out[i] = (value as u32) as i64;
551            carry = value >> 32;
552        }
553        carry as u32
554    };
555    // The dividend also keeps its shifted high carry.
556    u[m] = i64::from(normalize(a, first_a, m, &mut u[..m]));
557    normalize(b, first_b, n, &mut v[..n]);
558
559    let mut quotient_word = 0u64;
560    for j in (0..=(m - n)).rev() {
561        // Step 1: estimate q = A[0]A[1] / B[0] with A[0] = u[j+n], B[0] = v[n-1];
562        // A[0] <= B[0], and equality is clamped so q stays a radix digit.
563        let (mut estimate, mut residual) = if u[j + n] == v[n - 1] {
564            (BASE - 1, u[j + n - 1] as u64 + v[n - 1] as u64)
565        } else {
566            let numerator = ((u[j + n] as u64) << 32) | u[j + n - 1] as u64;
567            (numerator / v[n - 1] as u64, numerator % v[n - 1] as u64)
568        };
569        // Step 2: refine q with B[1] until q*B[1] <= r*base + A[2]; at most two
570        // decrements, and r < base guards the shift.
571        while residual < BASE && estimate * v[n - 2] as u64 > (residual << 32) + u[j + n - 2] as u64
572        {
573            estimate -= 1;
574            residual += v[n - 1] as u64;
575        }
576        // Step 3: subtract q * B from the window; q is correct or one too high.
577        let mut borrow = 0u64;
578        for i in 0..n {
579            // A radix-digit product plus the incoming borrow fits u64.
580            let product = estimate * v[i] as u64 + borrow;
581            let low = product as u32;
582            let old = u[j + i] as u32;
583            u[j + i] = i64::from(old.wrapping_sub(low));
584            borrow = (product >> 32) + u64::from(old < low);
585        }
586        let old = u[j + n] as u64;
587        // Only a borrow out of the full window means the result went negative.
588        let negative = old < borrow;
589        u[j + n] = i64::from(old.wrapping_sub(borrow) as u32);
590        if negative {
591            // Step 4: decrement q and add B back, restoring 0 <= R < divisor.
592            estimate -= 1;
593            let mut carry = 0u64;
594            for i in 0..n {
595                let sum = u[j + i] as u64 + v[i] as u64 + carry;
596                u[j + i] = i64::from(sum as u32);
597                carry = sum >> 32;
598            }
599            u[j + n] = i64::from((u[j + n] as u64 + carry) as u32);
600        }
601        // Pack descending quotient digits into complete words.
602        if j % 2 == 1 {
603            quotient_word = estimate << 32;
604        } else {
605            q[j / 2] = (quotient_word | estimate) as i64;
606        }
607    }
608    // Extract the remainder, undoing the normalization shift.
609    for i in 0..nr - 1 {
610        let digit = 2 * i;
611        let mut word = u[digit] as u64;
612        if digit + 1 < n {
613            word |= (u[digit + 1] as u64) << 32;
614        }
615        word >>= shift;
616        // Bring down bits from the next digit; shift zero never evaluates a shift by 64.
617        if shift != 0 && digit + 2 < n {
618            word |= (u[digit + 2] as u64) << (64 - shift);
619        }
620        r[i] = word as i64;
621    }
622    // The quotient takes the operand-sign XOR, a nonzero remainder the dividend's sign.
623    if is_negative(a) != is_negative(b) {
624        negate_in_place(q);
625    }
626    if is_negative(a) {
627        negate_in_place(r);
628    }
629    Ok((BigInt::from_words(q), BigInt::from_words(r)))
630}

632/// Quotient rounded toward zero.
633pub(super) fn trunc_div(a: &[i64], b: &[i64]) -> Result<BigInt> {
634    Ok(div_rem(a, b)?.0)
635}
636
637/// Remainder of truncating division: zero or with the dividend's sign.
638pub(super) fn trunc_mod(a: &[i64], b: &[i64]) -> Result<BigInt> {
639    if b.len() == 1 {
640        if b[0] == 0 {
641            return Err(zero_division());
642        }
643        let divisor = b[0].unsigned_abs();
644        if divisor <= u64::from(u32::MAX) {
645            // remainder < divisor < 2^32: each brought-down half fits u64 and the
646            // result is inline, so no quotient or allocation is needed.
647            let first = first_nonzero(a);
648            let mut remainder = 0u64;
649            for i in (0..a.len()).rev() {
650                let word = abs_word(a, first, i) as u64;
651                remainder = ((remainder << 32) | (word >> 32)) % divisor;
652                remainder = ((remainder << 32) | (word as u32 as u64)) % divisor;
653            }
654            let result = remainder as i64;
655            return Ok(BigInt::from_i64(if is_negative(a) {
656                -result
657            } else {
658                result
659            }));
660        }
661    }
662    Ok(div_rem(a, b)?.1)
663}

665/// Quotient rounded toward negative infinity.
666pub(super) fn floor_div(a: &[i64], b: &[i64]) -> Result<BigInt> {
667    let (quotient, remainder) = div_rem(a, b)?;
668    if !remainder.is_zero() && is_negative(a) != is_negative(b) {
669        // With a nonzero remainder, (q-1)*b + (r+b) preserves a while rounding down.
670        return Ok(sub(quotient.words(), &[1]));
671    }
672    Ok(quotient)
673}
674
675/// Remainder of floor division: zero or with the divisor's sign.
676pub(super) fn floor_mod(a: &[i64], b: &[i64]) -> Result<BigInt> {
677    let remainder = trunc_mod(a, b)?;
678    if !remainder.is_zero() && is_negative(a) != is_negative(b) {
679        // The floor remainder is r+b, which |r| < |b| keeps inline for a scalar divisor.
680        if let ([divisor], Some(r)) = (b, remainder.to_i64()) {
681            return Ok(BigInt::from_i64(r + divisor));
682        }
683        return Ok(add(remainder.words(), b));
684    }
685    Ok(remainder)
686}

688/// Convert to the nearest double, ties to even; `OverflowError` beyond the finite range.
689pub(super) fn to_f64(x: &[i64]) -> Result<f64> {
690    let first = first_nonzero(x);
691    let mut size = x.len();
692    while size > 0 && abs_word(x, first, size - 1) == 0 {
693        size -= 1;
694    }
695    if size == 0 {
696        return Ok(0.0);
697    }
698    let does_not_fit = || overflow("BigInt does not fit finite double");
699    if size > 16 {
700        return Err(does_not_fit());
701    }
702    let high = abs_word(x, first, size - 1) as u64;
703    let bits = (size - 1) * 64 + (64 - high.leading_zeros() as usize);
704    let shift = bits.saturating_sub(53);
705    let bit = |i: usize| ((abs_word(x, first, i / 64) as u64) >> (i % 64)) & 1;
706    let mut significand = 0u64;
707    for i in (shift..bits).rev() {
708        significand = (significand << 1) | bit(i);
709    }
710    if shift > 0 {
711        let sticky = (0..shift - 1).any(|i| bit(i) != 0);
712        if bit(shift - 1) == 1 && (sticky || significand & 1 == 1) {
713            significand += 1;
714        }
715    }
716    if bits == 1024 && significand == 1 << 53 {
717        return Err(does_not_fit());
718    }
719    let result = significand as f64 * 2f64.powi(shift as i32);
720    Ok(if is_negative(x) { -result } else { result })
721}

723/// Multiply magnitude words by `factor` and add `addend`, growing as needed.
724pub(super) fn mul_add_small(magnitude: &mut Vec<u64>, factor: u64, addend: u64) {
725    let mut carry = u128::from(addend);
726    for word in magnitude.iter_mut() {
727        let product = u128::from(*word) * u128::from(factor) + carry;
728        *word = product as u64;
729        carry = product >> 64;
730    }
731    if carry != 0 {
732        magnitude.push(carry as u64);
733    }
734}
