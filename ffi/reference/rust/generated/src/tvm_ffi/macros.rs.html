25/// Macro gto get the name of the function
26///
27/// # Usage
28/// Usage: function_name!()
29#[macro_export]
30macro_rules! function_name {
31    () => {{
32        // dummy function to get the name of the function
33        fn f() {}
34        fn type_name_of<T>(_: T) -> &'static str {
35            std::any::type_name::<T>()
36        }
37        let name = type_name_of(f);
38        // remove the f() from the name
39        &name[..name.len() - 3]
40    }};
41}

43/// Resolves a registered global [`Function`](crate::function::Function) by name
44/// and caches it for the lifetime of the process (lock-free after the first
45/// lookup). Each call site gets its own cache and the macro evaluates to a
46/// `&'static Function`. Panics — naming the function — if it is not registered.
47///
48/// Usage: `cached_global_func!("ffi.Map").call_packed(args)`
49#[macro_export]
50macro_rules! cached_global_func {
51    ($name:literal) => {{
52        static FUNC: std::sync::LazyLock<$crate::function::Function> =
53            std::sync::LazyLock::new(|| {
54                $crate::function::Function::get_global($name).unwrap_or_else(|_| {
55                    panic!(concat!("global function `", $name, "` is not registered"))
56                })
57            });
58        &*FUNC
59    }};
60}

62/// Implement zero-copy conversions from a derived object reference to one of
63/// its base object-reference types.
64///
65/// # Safety
66///
67/// Generated bindings must ensure that `target` is a registered base of
68/// `source` and that every `source` satisfies the target reference invariants.
69/// An incorrect declaration makes the generated safe conversion unsound.
70#[macro_export]
71macro_rules! impl_object_upcast {
72    ($($source:ty => $target:ty),+ $(,)?) => {
73        $(
74            impl ::std::convert::From<$source> for $target {
75                #[inline]
76                fn from(value: $source) -> Self {
77                    let data = <$source as $crate::object::ObjectRefCore>::into_data(value);
78                    // SAFETY: The macro declaration promises that `target` is
79                    // a registered base of `source`. Both references retain
80                    // the same allocation and only change its static view.
81                    let data = unsafe {
82                        $crate::object::ObjectArc::from_raw(
83                            $crate::object::ObjectArc::into_raw(data).cast::
84                                <<$target as $crate::object::ObjectRefCore>::ContainerType>(),
85                        )
86                    };
87                    // SAFETY: The macro declaration promises both the
88                    // container inheritance relation and every additional
89                    // invariant imposed by the target reference view.
90                    unsafe {
91                        <$target as $crate::object::ObjectRefCore>::from_data(data)
92                    }
93                }
94            }
95
96            impl ::std::convert::From<&$source> for $target {
97                #[inline]
98                fn from(value: &$source) -> Self {
99                    value.clone().into()
100                }
101            }
102
103        )+
104    };
105}

107/// Check the return code of the safe call
108///
109/// # Arguments
110/// * `ret_code` - The return code of the safe call
111///
112/// # Returns
113/// * `Result<(), Error>` - The result of the safe call
114/// Macro to check safe calls and automatically update traceback with file/line info
115///
116/// Usage: check_safe_call!(function(args))?;
117#[macro_export]
118macro_rules! check_safe_call {
119    ($expr:expr) => {{
120        let ret_code = $expr;
121        if ret_code == 0 {
122            Ok(())
123        } else {
124            let error = $crate::error::Error::from_raised();
125            Err(error)
126        }
127    }};
128}

130/// Create a new error with file/line info attached
131///
132/// This macro automatically appends file/line info to the traceback
133///
134/// # Arguments
135/// * `error_kind` - The kind of the error
136/// * `msg` - The message of the error
137/// * `args` - The posisble format arguments
138///
139/// # Returns
140/// * `Result<(), Error>` - The result of the safe call
141#[macro_export]
142macro_rules! bail {
143    ($error_kind:expr, $fmt:expr $(, $args:expr)* $(,)?) => {{
144        let context = format!(
145            "  File \"{}\", line {}, in {}\n",
146            file!(),
147            line!(),
148            $crate::function_name!()
149        );
150        return Err($crate::error::Error::new($error_kind, &format!($fmt $(, $args)*), &context));
151    }};
152}

154/// Create a new error with file/line info attached
155///
156/// This macro automatically appends file/line info to the traceback
157///
158/// # Arguments
159/// * `kind` - The kind of the error
160/// * `msg` - The message of the error
161/// * `args` - The posisble format arguments
162///
163/// # Returns
164/// * `Result<(), Error>` - The result of the safe call
165#[macro_export]
166macro_rules! ensure {
167    ($cond:expr, $error_kind:expr, $fmt:expr $(, $args:expr)* $(,)?) => {{
168        if !$cond {
169            $crate::bail!($error_kind, $fmt $(, $args)*);
170        }
171    }};

174/// Attach a context to a result if it is error
175///
176/// This macro automatically appends file/line info to the traceback
177///
178/// # Arguments
179/// * `error` - The error to attach the context to
180/// * `msg` - The message of the error
181///
182/// # Returns
183/// * `Result<(), Error>` - The result of the safe call
184#[macro_export]
185macro_rules! attach_context {
186    ($error:expr) => {{
187        match $error {
188            Ok(value) => Ok(value),
189            Err(error) => {
190                let context = format!(
191                    "  File \"{}\", line {}, in {}\n",
192                    file!(),
193                    line!(),
194                    $crate::function_name!()
195                );
196                Err(Error::with_appended_backtrace(error, &context))
197            }
198        }
199    }};
200}

309/// Macro to export a typed function as a C symbol that follows the tvm-ffi ABI
310///
311/// An error the function returns is raised to the caller. Report errors by
312/// returning them; a panic is also caught and raised, as an `InternalError`,
313/// since unwinding into the caller would abort the process, but panicking
314/// is discouraged.
315///
316/// # Arguments
317/// * `$name` - The name of the function
318/// * `$func` - The function to export
319///
320/// # Example
321/// ```rust
322/// use tvm_ffi::*;
323///
324/// fn add_one(x: i32) -> Result<i32> { Ok(x + 1) }
325///
326/// tvm_ffi_dll_export_typed_func!(add_one, add_one);
327/// ```
328#[macro_export]
329macro_rules! tvm_ffi_dll_export_typed_func {
330    ($name:ident, $func:expr) => {
331        $crate::macros::paste::paste! {
332            // `#[no_mangle]` is required so the symbol is preserved in a
333            // `cdylib` and matches the `__tvm_ffi_<name>` naming convention
334            // that `ffi.Module.load_from_file.<format>` looks up via
335            // `GetSymbolWithSymbolPrefix`. Without it, the linker strips the
336            // function from the output `.so`.
337            //
338            // Using plain `#[no_mangle]` (rather than `#[unsafe(no_mangle)]`,
339            // which would require rustc >= 1.82) keeps the crate buildable
340            // on older toolchains. Edition-2024 callers will see a
341            // deprecation warning, which is harmless.
342            //
343            // The path-qualified `$crate::tvm_ffi_sys::…` reference (rather
344            // than a bare `tvm_ffi_sys::…`) lets downstream crates use the
345            // macro without having to add `tvm-ffi-sys` to their own
346            // `[dependencies]`.
347            #[no_mangle]
348            pub unsafe extern "C" fn [<__tvm_ffi_ $name>](
349                _handle: *mut std::ffi::c_void,
350                args: *const $crate::tvm_ffi_sys::TVMFFIAny,
351                num_args: i32,
352                result: *mut $crate::tvm_ffi_sys::TVMFFIAny,
353            ) -> i32 {
354                let packed_args =
355                    std::slice::from_raw_parts(args as *const $crate::any::AnyView, num_args as usize);
356                let ret_value = match ::std::panic::catch_unwind(::std::panic::AssertUnwindSafe(|| {
357                    $crate::function_internal::call_packed_callable($func, packed_args)
358                })) {
359                    Ok(ret_value) => ret_value,
360                    Err(payload) => Err($crate::function_internal::panic_to_error(payload)),
361                };
362                match ret_value {
363                    Ok(value) => {
364                        *result = $crate::any::Any::into_raw_ffi_any(value);
365                        0
366                    }
367                    Err(error) => {
368                        $crate::error::Error::set_raised(&error);
369                        -1
370                    }
371                }
372            }
373        }
374    };
375}

377///-----------------------------------------------------------
378/// into_typed_fn
379///
380/// Converts a generic `Function` into a typed function with compile-time
381/// argument count and type checking. This macro provides a convenient way
382/// to create type-safe wrappers around TVM functions.
383///
384/// # Arguments
385/// * `$f` - The function identifier to convert
386/// * `$trait` - The trait type (typically `Fn`)
387/// * `($t0, $t1, ...)` - The argument types
388/// * `$ret_ty` - The return type
389///
390/// # Example
391/// ```rust
392/// use tvm_ffi::*;
393///
394/// let func = Function::from_typed(|x: i32, y: i32| -> Result<i32> { Ok(x + y) });
395/// let typed_func = into_typed_fn!(func, Fn(i32, &i32) -> Result<i32>);
396/// let result = typed_func(10, &20).unwrap(); // Returns 30
397/// assert_eq!(result, 30);
398/// ```
399/// Note that the `into_typed_fn!` macro can specify arguments to be passed either
400/// by reference or by value in the argument list.
401/// We recommend passing by reference for ObjectRef types such as Tensor.
402/// Since the ffi mechanism requires us to pass arguments by reference.
403///
404/// # Supported Argument Counts
405/// This macro supports functions with 0 to 8 arguments.
406///-----------------------------------------------------------
407#[macro_export]
408macro_rules! into_typed_fn {
409    // Case for 0 arguments
410    ($f:expr, $trait:ident() -> $ret_ty:ty) => {{
411        let _f = $f;
412        move || -> $ret_ty { Ok(_f.call_tuple_with_len::<0, _>(())?.try_into()?) }
413    }};
414    // Case for 1 argument
415    ($f:expr, $trait:ident($t0:ty) -> $ret_ty:ty) => {{
416        let _f = $f;
417        move |a0: $t0| -> $ret_ty {
418            use $crate::function_internal::IntoArgHolderTuple;
419            let tuple_args = (a0,).into_arg_holder_tuple();
420            Ok(_f.call_tuple_with_len::<1, _>(tuple_args)?.try_into()?)
421        }
422    }};
423    // Case for 2 arguments
424    ($f:expr, $trait:ident($t0:ty, $t1:ty) -> $ret_ty:ty) => {{
425        let _f = $f;
426        move |a0: $t0, a1: $t1| -> $ret_ty {
427            use $crate::function_internal::IntoArgHolderTuple;
428            let tuple_args = (a0, a1).into_arg_holder_tuple();
429            Ok(_f.call_tuple_with_len::<2, _>(tuple_args)?.try_into()?)
430        }
431    }};
432    // Case for 3 arguments
433    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty) -> $ret_ty:ty) => {{
434        let _f = $f;
435        move |a0: $t0, a1: $t1, a2: $t2| -> $ret_ty {
436            use $crate::function_internal::IntoArgHolderTuple;
437            let tuple_args = (a0, a1, a2).into_arg_holder_tuple();
438            Ok(_f.call_tuple_with_len::<3, _>(tuple_args)?.try_into()?)
439        }
440    }};
441    // Case for 4 arguments
442    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty) -> $ret_ty:ty) => {{
443        let _f = $f;
444        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3| -> $ret_ty {
445            use $crate::function_internal::IntoArgHolderTuple;
446            let tuple_args = (a0, a1, a2, a3).into_arg_holder_tuple();
447            Ok(_f.call_tuple_with_len::<4, _>(tuple_args)?.try_into()?)
448        }
449    }};
450    // Case for 5 arguments
451    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty) -> $ret_ty:ty) => {{
452        let _f = $f;
453        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4| -> $ret_ty {
454            use $crate::function_internal::IntoArgHolderTuple;
455            let tuple_args = (a0, a1, a2, a3, a4).into_arg_holder_tuple();
456            Ok(_f.call_tuple_with_len::<5, _>(tuple_args)?.try_into()?)
457        }
458    }};
459    // Case for 6 arguments
460    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty, $t5:ty) -> $ret_ty:ty) => {{
461        let _f = $f;
462        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4, a5: $t5| -> $ret_ty {
463            use $crate::function_internal::IntoArgHolderTuple;
464            let tuple_args = (a0, a1, a2, a3, a4, a5).into_arg_holder_tuple();
465            Ok(_f.call_tuple_with_len::<6, _>(tuple_args)?.try_into()?)
466        }
467    }};
468    // Case for 7 arguments
469    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty, $t5:ty, $t6:ty)
470        -> $ret_ty:ty) => {{
471        let _f = $f;
472        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4, a5: $t5, a6: $t6| -> $ret_ty {
473            use $crate::function_internal::IntoArgHolderTuple;
474            let tuple_args = (a0, a1, a2, a3, a4, a5, a6).into_arg_holder_tuple();
475            Ok(_f.call_tuple_with_len::<7, _>(tuple_args)?.try_into()?)
476        }
477    }};
478    // Case for 8 arguments
479    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty, $t5:ty, $t6:ty, $t7:ty)
480        -> $ret_ty:ty) => {{
481        let _f = $f;
482        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4, a5: $t5, a6: $t6, a7: $t7| -> $ret_ty {
483            use $crate::function_internal::IntoArgHolderTuple;
484            let tuple_args = (a0, a1, a2, a3, a4, a5, a6, a7).into_arg_holder_tuple();
485            Ok(_f.call_tuple_with_len::<8, _>(tuple_args)?.try_into()?)
486        }
487    }};
