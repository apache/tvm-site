39/// The root object type, `ffi.Object`: just the object header.
40///
41/// An object type defined in Rust starts with its parent's object, which for
42/// a direct subtype is `base: Object`.
43#[repr(C)]
44pub struct Object {
45    header: TVMFFIObject,
46}
47
48impl Object {
49    /// A new object header, for the `base` field of a new object.
50    #[allow(clippy::new_without_default)]
51    pub fn new() -> Self {
52        Self {
53            header: TVMFFIObject::new(),
54        }
55    }
56}

71/// A reference-counted pointer to an object `T`, as `Arc<T>`, counting with
72/// the object header so that every language shares the count.
73#[repr(C)]
74pub struct ObjectArc<T: ObjectCore> {
75    ptr: std::ptr::NonNull<T>,
76    _phantom: std::marker::PhantomData<T>,
77}
78
79impl ObjectArc<Object> {
80    /// Allocate a synchronous runtime payload that may contain borrowed Rust state.
81    ///
82    /// # Safety
83    /// `T` begins with a valid Object header and is ABI-compatible with the
84    /// registered `type_index`. The caller must keep this owner scoped to the
85    /// payload's borrows and ensure every temporary strong and weak reference
86    /// is released before those borrows end. Neither native handles nor this
87    /// erased owner may escape that synchronous scope. Destruction uses T's
88    /// exact layout and drop function, not the erased Object layout.
89    /// The erased owner and handles must remain on this thread unless the
90    /// concrete payload permits sharing or transfer; erasure grants neither.
91    pub(crate) unsafe fn new_opaque_unchecked<T>(data: T, type_index: i32) -> Self {
92        Self {
93            ptr: allocate_object(data, type_index).cast(),
94            _phantom: std::marker::PhantomData,
95        }
96    }
97}

99impl<T: ObjectCore> ObjectArc<T> {
100    /// Allocate an initialized object at its stable managed address.
101    pub fn new(data: T) -> Self {
102        // Resolve metadata before moving the payload into its allocation.
103        let type_index = T::type_index();
104        unsafe {
105            Self {
106                ptr: allocate_object(data, type_index),
107                _phantom: std::marker::PhantomData,
108            }
109        }
110    }
111
112    /// Allocate a stable object followed by its type's extra item storage.
113    pub fn new_with_extra_items<U>(data: T) -> Self
114    where
115        T: ObjectCoreWithExtraItems<ExtraItem = U>,
116    {
117        unsafe {
118            // ensure strict alignment requirements
119            // so we can have { T, U*extra_items } layout
120            assert_eq!(std::mem::align_of::<T>() % std::mem::align_of::<U>(), 0);
121            assert_eq!(std::mem::size_of::<T>() % std::mem::align_of::<U>(), 0);
122            let extra_items_count = T::extra_items_count(&data);
123            let layout = unsafe_::extra_items_layout::<T, U>(extra_items_count);
124            let raw_data_ptr = std::alloc::alloc(layout);
125            if raw_data_ptr.is_null() {
126                std::alloc::handle_alloc_error(layout);
127            }
128            let ptr = raw_data_ptr as *mut T;
129            std::ptr::write(ptr, data);
130            // now override the header directly
131            std::ptr::write(
132                ptr as *mut TVMFFIObject,
133                TVMFFIObject {
134                    combined_ref_count: AtomicU64::new(COMBINED_REF_COUNT_BOTH_ONE),
135                    type_index: T::type_index(),
136                    __padding: 0,
137                    deleter: Some(unsafe_::object_deleter_for_new_with_extra_items::<T, U>),
138                },
139            );
140            // move into the object arc ptr
141            Self {
142                ptr: std::ptr::NonNull::new_unchecked(ptr),
143                _phantom: std::marker::PhantomData,
144            }
145        }
146    }
147
148    /// Whether `this` holds a null pointer, as all-zero bytes do: no
149    /// `ObjectArc`, but the field of an object that a creator left undefined.
150    ///
151    /// See also [`AnyCompatible::is_undefined`].
152    ///
153    /// # Safety
154    ///
155    /// `this` points to an `ObjectArc<T>` or to all-zero bytes.
156    #[inline]
157    pub unsafe fn is_null(this: *const Self) -> bool {
158        (*std::ptr::addr_of!((*this).ptr).cast::<*const T>()).is_null()
159    }
160
161    /// Take ownership of a raw object pointer, as `Arc::from_raw`.
162    ///
163    /// # Arguments
164    /// * `ptr` - A pointer that owns one strong reference.
165    ///
166    /// # Returns
167    /// The handle that owns that reference.
168    ///
169    /// # Safety
170    /// `ptr` must own one strong reference to a live, stable managed allocation
171    /// whose dynamic type is compatible with `T` and whose payload is valid.
172    /// Ownership of that reference is transferred into the returned handle.
173    #[inline(always)]
174    pub unsafe fn from_raw(ptr: *const T) -> Self {
175        Self {
176            ptr: std::ptr::NonNull::new_unchecked(ptr as *mut T),
177            _phantom: std::marker::PhantomData,
178        }
179    }
180
181    /// Turn the handle into a raw pointer, as `Arc::into_raw`.
182    ///
183    /// # Arguments
184    /// * `this` - The handle.
185    ///
186    /// # Returns
187    /// The pointer, which owns the strong reference; pass it back to
188    /// [`ObjectArc::from_raw`] to release it.
189    #[inline(always)]
190    pub fn into_raw(this: Self) -> *const T {
191        let droped_this = std::mem::ManuallyDrop::new(this);
192        droped_this.ptr.as_ptr() as *const T
193    }
194
195    /// The raw pointer, without giving up ownership, as `Arc::as_ptr`.
196    ///
197    /// # Arguments
198    /// * `this` - The handle.
199    ///
200    /// # Returns
201    /// The non-owning pointer. Dereferencing it must respect existing borrows,
202    /// and it cannot outlive an owning handle.
203    #[inline]
204    pub fn as_ptr(this: &Self) -> *const T {
205        this.ptr.as_ptr() as *const T
206    }
207
208    /// The raw mutable pointer, without giving up ownership.
209    ///
210    /// # Arguments
211    /// * `this` - The handle.
212    ///
213    /// # Returns
214    /// The non-owning pointer. The handle need not be unique: writing through
215    /// it needs exclusive access to the object, and must not move or replace
216    /// the object or its header.
217    #[inline]
218    pub fn as_mut_ptr(this: &mut Self) -> *mut T {
219        this.ptr.as_ptr()
220    }
221
222    /// The number of strong references, as `Arc::strong_count`.
223    ///
224    /// # Arguments
225    /// * `this` - The handle.
226    #[inline]
227    pub fn strong_count(this: &Self) -> usize {
228        unsafe { unsafe_::strong_count(this.ptr.as_ptr().cast::<TVMFFIObject>()) }
229    }
230
231    /// Whether this is the only reference, so it may be mutated in place.
232    #[inline]
233    pub(crate) fn is_unique(this: &Self) -> bool {
234        unsafe { unsafe_::is_unique(this.ptr.as_ptr().cast::<TVMFFIObject>()) }
235    }
236
237    /// The number of weak references, as `Arc::weak_count`.
238    ///
239    /// # Arguments
240    /// * `this` - The handle.
241    #[inline]
242    pub fn weak_count(this: &Self) -> usize {
243        unsafe { unsafe_::weak_count(this.ptr.as_ptr().cast::<TVMFFIObject>()) }
244    }
245}

276/// An object type whose shared instances are never mutated, so `ObjectArc<T>` is
277/// `Send + Sync` when `T`'s fields are.
278///
279/// # Safety
280///
281/// No code may mutate an instance while it is shared: changes go through
282/// copy-on-write, in place only on a uniquely held instance (one strong
283/// reference, no weak ones). This holds for every type not registered as
284/// mutable (`_type_mutable`), in any language.
285pub unsafe trait ImmutableObject: ObjectCore {}

287// SAFETY: `T` is never mutated while shared, its fields are thread-safe, and the
288// reference count is atomic.
289unsafe impl<T: ImmutableObject + Send + Sync> Send for ObjectArc<T> {}
290// SAFETY: as for `Send`.
291unsafe impl<T: ImmutableObject + Send + Sync> Sync for ObjectArc<T> {}

303/// The native representation of an object-system type.
304///
305/// # Safety
306/// `Self` must have a compatible C layout with a valid `TVMFFIObject` prefix
307/// at offset zero. Its registered type, inheritance and payload must describe
308/// that layout correctly, including for foreign declarations.
309///
310/// A live managed object and its header must remain at their original allocation
311/// address until destruction. Safe APIs must not move, replace or expose mutable
312/// access to that header while managed; payload mutation must preserve its type
313/// invariants. Preallocation builders and values moved out during destruction
314/// are not live managed objects and cannot be borrowed as `AnyView`.
315pub unsafe trait ObjectCore: Sized + 'static {
316    /// The type key, such as `"ffi.Array"`.
317    const TYPE_KEY: &'static str;
318    /// Depth of this type in the object inheritance tree.
319    ///
320    /// The root [`Object`] has depth zero, and every registered subtype has
321    /// depth one greater than its parent. This value must be non-negative and
322    /// agree with the runtime type table entry for `Self`.
323    const TYPE_DEPTH: i32;
324    /// Whether every instance of this type has exactly `Self::type_index()`.
325    ///
326    /// A final type has no separately registered object-system subtype.
327    #[doc(hidden)]
328    const TYPE_FINAL: bool = false;
329    /// The runtime type index of this type.
330    fn type_index() -> i32;
331    /// The object header, mutably.
332    ///
333    /// # Arguments
334    /// * `this` - The object.
335    ///
336    /// # Safety
337    /// The caller must have exclusive access to the header and preserve the
338    /// managed allocation's address, reference counts, deleter and dynamic type.
339    unsafe fn object_header_mut(this: &mut Self) -> &mut TVMFFIObject;
340}

342/// An object type whose allocation is followed by a run of extra items, such
343/// as the dimensions of a [`Shape`](crate::Shape).
344///
345/// # Safety
346/// Every allocation of `Self` must be followed by `extra_items_count` initialized
347/// `ExtraItem`s, as `ObjectArc::new_with_extra_items` lays them out.
348pub unsafe trait ObjectCoreWithExtraItems: ObjectCore {
349    /// The type of the items that follow the object.
350    type ExtraItem;
351    /// The number of extra items.
352    fn extra_items_count(this: &Self) -> usize;
353    /// The extra items.
354    ///
355    /// # Safety
356    /// `this` must be an allocation that carries its extra items.
357    unsafe fn extra_items(this: &Self) -> &[Self::ExtraItem] {
358        let extra_items_ptr = (this as *const Self as *const u8).add(std::mem::size_of::<Self>());
359        std::slice::from_raw_parts(
360            extra_items_ptr as *const Self::ExtraItem,
361            Self::extra_items_count(this),
362        )
363    }
364    /// The extra items, mutably.
365    ///
366    /// # Safety
367    /// `this` must be an allocation that carries its extra items.
368    unsafe fn extra_items_mut(this: &mut Self) -> &mut [Self::ExtraItem] {
369        let extra_items_ptr = (this as *mut Self as *mut u8).add(std::mem::size_of::<Self>());
370        std::slice::from_raw_parts_mut(
371            extra_items_ptr as *mut Self::ExtraItem,
372            Self::extra_items_count(this),
373        )
374    }
375}

377/// A managed object reference, such as [`ObjectRef`](struct@ObjectRef), `Array<T>`
378/// or a `#[derive(ObjectRef)]` type, with identity ([`same_as`](Self::same_as))
379/// and strict casts that check the runtime type and never convert; upcasts are
380/// `From`/`Into`.
381///
382/// # Casts
383///
384/// * [`downcast_ref`](Self::downcast_ref) borrows, strictly, as std's
385///   `dyn Any::downcast_ref` does.
386/// * [`downcast`](Self::downcast) takes the value, strictly, and hands it back
387///   on a mismatch, as `Box<dyn Any>::downcast` does.
388/// * `TryFrom` takes the value and returns a `TypeError`, for `?`. Between
389///   references it only checks; from an [`Any`](crate::Any) it converts by
390///   the FFI rules, such as an integer into a float.
391///
392/// # Safety
393///
394/// `data`, `into_data`, and `from_data` must preserve the same object
395/// allocation and form an ownership-preserving round trip. That allocation must
396/// start with a valid `TVMFFIObject` header whose registered object-range
397/// runtime type index correctly describes its layout and inheritance.
398///
399/// When `Self` also implements [`AnyCompatible`], `copy_to_any_view` must
400/// produce a non-owning view, while `move_to_any` must transfer ownership of
401/// the same object pointer and dynamic type index. `move_from_any_after_check`
402/// must be able to reclaim that owned representation exactly once, and a true
403/// `check_any_strict` result must guarantee that both after-check constructors
404/// are valid for it.
405pub unsafe trait ObjectRefCore: Sized + Clone + AnyCompatible {
406    /// The object type the reference points to.
407    type ContainerType: ObjectCore;
408    /// Borrow the object's handle.
409    fn data(this: &Self) -> &ObjectArc<Self::ContainerType>;
410    /// Take the object's handle.
411    fn into_data(this: Self) -> ObjectArc<Self::ContainerType>;
412
413    /// Make a reference from the object's handle.
414    ///
415    /// # Safety
416    ///
417    /// In addition to containing a valid `ContainerType` allocation, `data`
418    /// must satisfy every semantic invariant imposed by `Self`. This matters
419    /// for zero-state views that share a container type but accept only a
420    /// subset of its values, such as a reference whose
421    /// [`ObjectRefAnyCompatible::extra_check`] rejects some objects.
422    unsafe fn from_data(data: ObjectArc<Self::ContainerType>) -> Self;
423
424    /// Whether two references point to the same object.
425    #[inline]
426    fn same_as<Other: ObjectRefCore>(&self, other: &Other) -> bool {
427        ObjectArc::as_ptr(Self::data(self)).cast::<()>()
428            == ObjectArc::as_ptr(Other::data(other)).cast::<()>()
429    }
430
431    /// Borrow the underlying object as `T` when it is one.
432    ///
433    /// `T` is an object type such as `MyPairObj`, or a managed reference such
434    /// as `MyPair`: `&ObjectRef` borrows as `&MyPair` when its object is a
435    /// `MyPairObj`. Unrelated types return `None`. Unlike [`downcast`](Self::downcast), this
436    /// method neither consumes the reference nor changes the object's reference
437    /// count. The returned borrow cannot outlive `self`.
438    /// See [`crate::Any::downcast_ref`] for an example.
439    #[inline(always)]
440    fn downcast_ref<T: crate::type_traits::DowncastRefFromObject>(&self) -> Option<&T> {
441        // SAFETY: `ObjectArc` is exactly one non-null object pointer, and
442        // ObjectRefCore guarantees a live object allocation.
443        unsafe {
444            let place = &*(Self::data(self) as *const ObjectArc<Self::ContainerType>
445                as *const *mut TVMFFIObject);
446            T::downcast_ref_from_object(place)
447        }
448    }
449
450    /// Take `self` as `B`, a subtype or a container such as `Array<i64>` whose
451    /// elements are checked too, when its object is one, without copying it.
452    /// On a mismatch the reference is returned unchanged.
453    ///
454    /// ```
455    /// use tvm_ffi::{object::ObjectRef, Array, ObjectRefCore};
456    /// let object: ObjectRef = Array::from_vec(vec![1_i64]).into();
457    /// let object = object.downcast::<Array<f64>>().unwrap_err();
458    /// let array: Array<i64> = object.try_into()?;
459    /// assert_eq!(array.len(), 1);
460    /// # Ok::<(), tvm_ffi::Error>(())
461    /// ```
462    #[inline(always)]
463    fn downcast<B>(self) -> std::result::Result<B, Self>
464    where
465        B: ObjectRefCore,
466    {
467        let mut any_data = TVMFFIAny::new();
468        unsafe {
469            // Keep ownership in `self` while the target check runs, so a
470            // mismatch can hand it back and a panic unwinds normally.
471            Self::copy_to_any_view(&self, &mut any_data);
472            debug_assert!(any_data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32);
473            // SAFETY: ObjectRefCore's contract requires its AnyCompatible
474            // representation to contain a valid object-range type index.
475            std::hint::assert_unchecked(
476                any_data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32,
477            );
478            if B::check_any_strict(&any_data) {
479                // Transfer ownership only after the borrowed representation has
480                // passed the target's complete hierarchy/container check.
481                Self::move_to_any(self, &mut any_data);
482                Ok(B::move_from_any_after_check(&mut any_data))
483            } else {
484                Err(self)
485            }
486        }
487    }
488}

490/// The [`AnyCompatible`] implementation of an object reference, with every
491/// hook defaulted: every implementer is `AnyCompatible`.
492///
493/// `#[derive(ObjectRef)]` implements it with the defaults. A typed reference
494/// that accepts only some objects of its type, or converts other values,
495/// implements it by hand and overrides only what differs, such as
496/// [`extra_check`](Self::extra_check) and
497/// [`convert_fallback`](Self::convert_fallback) below. A type with another
498/// representation, such as [`Array`](crate::Array), implements
499/// `AnyCompatible` instead.
500///
501/// # Safety
502///
503/// `Self` is exactly one [`ObjectArc<Self::ContainerType>`](ObjectArc). The
504/// hooks are safe and cannot break the [`AnyCompatible`] contract.
505///
506/// # Examples
507///
508/// ```
509/// use tvm_ffi::derive::ObjectRef;
510/// use tvm_ffi::{rs_object, Any, AnyView, Object, ObjectArc, ObjectRefAnyCompatible};
511///
512/// #[rs_object("my_ext.MyPair", final)]
513/// pub struct MyPairObj {
514///     base: Object,
515///     pub first: i64,
516///     pub second: i64,
517/// }
518///
519/// #[derive(Clone, ObjectRef)]
520/// pub struct MyPair {
521///     data: ObjectArc<MyPairObj>,
522/// }
523///
524/// /// A `MyPair` whose `first` is positive.
525/// #[derive(Clone, ObjectRef)]
526/// #[object_ref(skip_any_compatible)]
527/// pub struct PositivePair {
528///     data: ObjectArc<MyPairObj>,
529/// }
530///
531/// unsafe impl ObjectRefAnyCompatible for PositivePair {
532///     fn extra_check(pair: &MyPairObj) -> bool {
533///         pair.first > 0
534///     }
535///
536///     // A positive integer `n` converts to the pair `(n, n)`.
537///     fn convert_fallback(value: AnyView<'_>) -> Option<Self> {
538///         let n = i64::try_from(value).ok().filter(|n| *n > 0)?;
539///         Some(Self::from_object(MyPairObj::new(Object::new(), n, n)))
540///     }
541/// }
542///
543/// let negative = Any::from(MyPair::from_object(MyPairObj::new(Object::new(), -1, 2)));
544/// assert!(negative.downcast_ref::<PositivePair>().is_none());
545/// assert!(PositivePair::try_from(negative).is_err());
546/// assert_eq!(PositivePair::try_from(Any::from(3_i64))?.second, 3);
547/// # Ok::<(), tvm_ffi::Error>(())
548/// ```
549pub unsafe trait ObjectRefAnyCompatible: ObjectRefCore {
550    /// Whether no hook is overridden, so that `match_any!` may route a final
551    /// object type by its type index alone; only `#[derive(ObjectRef)]` sets
552    /// it. See [`AnyCompatible::MATCH_ANY_EXACT`].
553    #[doc(hidden)]
554    const MATCH_ANY_EXACT: bool = false;
555
556    /// Whether `obj`, an instance of `ContainerType`, is a value of `Self`.
557    /// Every strict and converting cast checks it, after the object type.
558    #[inline]
559    fn extra_check(obj: &Self::ContainerType) -> bool {
560        let _ = obj;
561        true
562    }
563
564    /// Convert a value that is not a `Self` object, such as an integer that
565    /// stands for a constant; `None` rejects it. Only converting casts (`TryFrom<Any>`,
566    /// typed function arguments) call it; `downcast_ref`, `downcast` and
567    /// `match_any!` never do.
568    #[inline]
569    fn convert_fallback(value: AnyView<'_>) -> Option<Self> {
570        let _ = value;
571        None
572    }
573
574    /// The type name in conversion errors and type schemas; by default the
575    /// object's type key. See [`AnyCompatible::type_str`].
576    fn custom_type_str() -> String {
577        Self::ContainerType::TYPE_KEY.into()
578    }
579
580    /// The name of the mismatched `value` in conversion errors; by default its
581    /// runtime type key. See [`AnyCompatible::get_mismatch_type_info`].
582    fn custom_mismatch_type_info(value: AnyView<'_>) -> String {
583        crate::type_traits::runtime_type_key(value.type_index())
584    }
585}

587// The object cell encoding, with the strict check `is_instance_of` then
588// `extra_check`, and `convert_fallback` after it in converting casts.
589unsafe impl<T: ObjectRefAnyCompatible> AnyCompatible for T {
590    // An overridden hook makes the type inexact; a non-final object type always is.
591    const MATCH_ANY_EXACT: bool = <T as ObjectRefAnyCompatible>::MATCH_ANY_EXACT
592        && <T::ContainerType as ObjectCore>::TYPE_FINAL;
593    const MATCH_ANY_STRICT_EXACT: bool = <T as AnyCompatible>::MATCH_ANY_EXACT;
594
595    #[inline]
596    fn match_any_exact_type_index() -> i32 {
597        T::ContainerType::type_index()
598    }
599
600    #[inline]
601    unsafe fn is_undefined(field: *const Self) -> bool {
602        // `Self` is one `ObjectArc`, so its null form is the undefined field.
603        ObjectArc::is_null(field.cast::<ObjectArc<T::ContainerType>>())
604    }

606    #[inline(always)]
607    unsafe fn copy_to_any_view(src: &Self, data: &mut TVMFFIAny) {
608        write_object_cell(ObjectArc::as_ptr(T::data(src)), data);
609    }
610
611    #[inline(always)]
612    unsafe fn move_to_any(src: Self, data: &mut TVMFFIAny) {
613        write_object_cell(ObjectArc::into_raw(T::into_data(src)), data);
614    }
615
616    #[inline(always)]
617    unsafe fn check_any_strict(data: &TVMFFIAny) -> bool {
618        // An instance of `ContainerType` points to one, so `extra_check` may read it.
619        is_instance_of::<T::ContainerType>(data.type_index)
620            && T::extra_check(&*data.data_union.v_obj.cast::<T::ContainerType>())
621    }
622
623    #[inline]
624    unsafe fn copy_from_any_view_after_check(data: &TVMFFIAny) -> Self {
625        // The view does not own the object, so the new reference takes one.
626        unsafe_::inc_ref(data.data_union.v_obj);
627        T::from_data(ObjectArc::from_raw(data.data_union.v_obj.cast()))
628    }
629
630    #[inline(always)]
631    unsafe fn move_from_any_after_check(data: &mut TVMFFIAny) -> Self {
632        T::from_data(ObjectArc::from_raw(data.data_union.v_obj.cast()))
633    }
634
635    #[inline]
636    unsafe fn try_cast_from_any_view(data: &TVMFFIAny) -> Result<Self, ()> {
637        if Self::check_any_strict(data) {
638            return Ok(Self::copy_from_any_view_after_check(data));
639        }
640        // A moved-from rvalue slot is consumed only by an exact object match.
641        if data.type_index == TypeIndex::kTVMFFIObjectRValueRef as i32 {
642            return Err(());
643        }
644        T::convert_fallback(AnyView::from_raw_ffi_any(*data)).ok_or(())
645    }
646
647    fn get_mismatch_type_info(data: &TVMFFIAny) -> String {
648        // SAFETY: the caller's cell is valid for the borrow.
649        T::custom_mismatch_type_info(unsafe { AnyView::from_raw_ffi_any(*data) })
650    }
651
652    fn type_str() -> String {
653        T::custom_type_str()
654    }
655}

657/// A value that converts into the object reference `T`, owned or borrowed.
658///
659/// An owned reference converts into an owned `T`, and a borrowed one borrows
660/// as `&T` without changing a reference count, as [`IntoIterator`] consumes a
661/// `Vec` but borrows from a `&Vec`. [`unsafe_impl_object_ref_upcast!`](crate::unsafe_impl_object_ref_upcast)
662/// implements it for each derived reference and its reference into the base,
663/// and `#[derive(ObjectRef)]` for the type itself. A function that only reads
664/// the value borrows the output with `.borrow()`; one that keeps it takes
665/// ownership with [`into_owned_object_ref`](Self::into_owned_object_ref), which clones only a
666/// borrowed reference.
667///
668/// # Examples
669///
670/// ```
671/// use std::borrow::Borrow;
672/// use tvm_ffi::derive::ObjectRef;
673/// use tvm_ffi::object::{IntoObjectRef, ObjectRef, ObjectRefCore};
674/// use tvm_ffi::{rs_object, Object, ObjectArc};
675///
676/// #[rs_object("my_ext.MyPair", final)]
677/// pub struct MyPairObj {
678///     base: Object,
679///     pub first: i64,
680///     pub second: i64,
681/// }
682///
683/// #[derive(Clone, ObjectRef)]
684/// pub struct MyPair {
685///     data: ObjectArc<MyPairObj>,
686/// }
687///
688/// fn same(a: impl IntoObjectRef<ObjectRef>, b: &ObjectRef) -> bool {
689///     a.into_object_ref().borrow().same_as(b)
690/// }
691///
692/// let pair = MyPair::from_object(MyPairObj::new(Object::new(), 1, 2));
693/// let base: ObjectRef = pair.clone().into();
694/// assert!(same(&pair, &base)); // borrowed as `&ObjectRef`, no new reference
695/// assert!(same(pair.clone(), &base)); // moved into an `ObjectRef`
696///
697/// // A function that keeps the value takes ownership.
698/// fn keep(value: impl IntoObjectRef<ObjectRef>) -> ObjectRef {
699///     value.into_owned_object_ref()
700/// }
701/// assert!(keep(&pair).same_as(&base));
702/// ```
703pub trait IntoObjectRef<T> {
704    /// `T` for an owned value, or `&T` for a borrowed one.
705    type Output: std::borrow::Borrow<T> + Into<T>;
706
707    /// Convert into the reference `T`.
708    ///
709    /// # Returns
710    ///
711    /// The owned or borrowed reference.
712    fn into_object_ref(self) -> Self::Output;
713
714    /// Convert into an owned `T`, as [`Cow::into_owned`](std::borrow::Cow::into_owned)
715    /// does: an owned value moves, and only a borrowed reference is cloned.
716    ///
717    /// # Returns
718    ///
719    /// The owned reference.
720    #[inline(always)]
721    fn into_owned_object_ref(self) -> T
722    where
723        Self: Sized,
724    {
725        self.into_object_ref().into()
726    }
727}

730///
731/// The retained strong reference prevents the allocation address from being
732/// reused while the key is alive. This makes it suitable for identity-based
733/// maps without exposing raw object pointers to downstream code.

736    data: ObjectArc<Object>,
737}

744            unsafe_::inc_ref(ptr);
745            Self {
746                data: ObjectArc::from_raw(ptr.cast::<Object>()),
747            }
748        }
749    }

781/// Whether the runtime type index of an object is `Target` or one of its
782/// subtypes, by walking the registered type hierarchy.
783///
784/// Use it in the strict check of a reference type implemented by hand; the
785/// casts of [`ObjectRefCore`] call it already. An index in the object range
786/// must be registered.
787#[inline(always)]
788pub fn is_instance_of<Target: ObjectCore>(object_type_index: i32) -> bool {
789    // Every registered type records its depth in the single inheritance tree
790    // and its ancestors, so the check is O(1): an ancestor `Target` appears in
791    // the candidate's ancestor array exactly at `Target`'s depth.
792    let target_type_index = Target::type_index();
793    if object_type_index == target_type_index {
794        return true;
795    }
796    // A final type cannot have a separately registered subtype. Keep common
797    // borrowed checks, such as `IntImmObj`, to one integer comparison.
798    if Target::TYPE_FINAL {
799        return false;
800    }
801    let object_begin = TypeIndex::kTVMFFIStaticObjectBegin as i32;
802    // Only object types participate in the type hierarchy.
803    if object_type_index < object_begin || target_type_index < object_begin {
804        return false;
805    }
806    // Only the root Object type has depth zero.
807    if const { Target::TYPE_DEPTH == 0 } {
808        return true;
809    }
810    // Parent indices are always smaller than their descendants.
811    if object_type_index < target_type_index {
812        return false;
813    }
814    unsafe {
815        let object_info = TVMFFIGetTypeInfo(object_type_index);
816        if object_info.is_null() {
817            return false;
818        }
819        let target_depth = Target::TYPE_DEPTH;
820        if (*object_info).type_depth <= target_depth {
821            return false;
822        }
823        let ancestor = *(*object_info).type_acenstors.add(target_depth as usize);
824        !ancestor.is_null() && (*ancestor).type_index == target_type_index
825    }
826}

828/// Format a managed reference as its runtime type key and object address, as
829/// the `Debug` implementation of `#[derive(ObjectRef)]` does.
830///
831/// Call it from `Debug` for a reference type implemented by hand, such as a
832/// typed view over another reference, so that `downcast(..).unwrap()` and
833/// `assert_eq!` can print it.
834///
835/// ```
836/// use tvm_ffi::object::{fmt_object_ref, ObjectRef};
837/// use tvm_ffi::Array;
838/// struct Shown(ObjectRef);
839/// impl std::fmt::Debug for Shown {
840///     fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
841///         fmt_object_ref(&self.0, f)
842///     }
843/// }
844/// let shown = Shown(Array::from_vec(vec![1_i64]).into());
845/// assert!(format!("{shown:?}").starts_with("ffi.Array@"));
846/// ```
847pub fn fmt_object_ref<T: ObjectRefCore>(
848    this: &T,
849    f: &mut std::fmt::Formatter<'_>,
850) -> std::fmt::Result {
851    let object = ObjectArc::as_ptr(T::data(this)).cast::<TVMFFIObject>();
852    // SAFETY: `this` keeps the object, and so its header, alive.
853    let type_index = unsafe { (*object).type_index };
854    write!(f, "{}@{:p}", type_key_of_index(type_index), object)
855}

910/// Write the object cell of the live object `ptr`.
911#[inline(always)]
912unsafe fn write_object_cell<T: ObjectCore>(ptr: *const T, data: &mut TVMFFIAny) {
913    let object = ptr as *mut TVMFFIObject;
914    data.type_index = (*object).type_index;
915    // SAFETY: a live object's header holds its registered object-range type index.
916    std::hint::assert_unchecked(data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32);
917    data.small_str_len = 0;
918    data.data_union.v_obj = object;

921// Allocate and initialize the normal FFI header while preserving the concrete
922// payload's alignment and drop/layout deleter. The payload starts with an Object.
923unsafe fn allocate_object<T>(data: T, type_index: i32) -> NonNull<T> {
924    let layout = std::alloc::Layout::new::<T>();
925    let allocation = std::alloc::alloc(layout);
926    if allocation.is_null() {
927        std::alloc::handle_alloc_error(layout);
928    }
929    let ptr = allocation.cast::<T>();
930    std::ptr::write(ptr, data);
931    std::ptr::write(
932        ptr.cast::<TVMFFIObject>(),
933        TVMFFIObject {
934            combined_ref_count: AtomicU64::new(COMBINED_REF_COUNT_BOTH_ONE),
935            type_index,
936            __padding: 0,
937            deleter: Some(unsafe_::object_deleter_for_new::<T>),
938        },
939    );
940    NonNull::new_unchecked(ptr)
941}

943/// Unsafe reference-count operations on raw object headers.
944#[doc(hidden)]
945pub mod unsafe_ {
946    use tvm_ffi_sys::{
947        COMBINED_REF_COUNT_BOTH_ONE, COMBINED_REF_COUNT_MASK_U32, COMBINED_REF_COUNT_STRONG_ONE,
948        COMBINED_REF_COUNT_WEAK_ONE,
949    };
950
951    use std::ffi::c_void;
952    use std::sync::atomic::{fence, Ordering};
953    use tvm_ffi_sys::TVMFFIObject;
954    use tvm_ffi_sys::TVMFFIObjectDeleterFlagBitMask::{
955        kTVMFFIObjectDeleterFlagBitMaskBoth, kTVMFFIObjectDeleterFlagBitMaskStrong,
956        kTVMFFIObjectDeleterFlagBitMaskWeak,
957    };
958
959    /// Increase the strong reference count of `handle`.
960    #[inline]
961    pub unsafe fn inc_ref(handle: *mut TVMFFIObject) {
962        let obj = &*handle;
963        obj.combined_ref_count.fetch_add(1, Ordering::Relaxed);
964    }
965
966    /// Decrease the strong reference count of `handle`, deleting the object when it was the last.
967    #[inline]
968    pub(crate) unsafe fn dec_ref(handle: *mut TVMFFIObject) {
969        // Release publishes this owner's writes; the Acquire fence before the deleter pairs with it.
970        let old_combined_count = (*handle)
971            .combined_ref_count
972            .fetch_sub(COMBINED_REF_COUNT_STRONG_ONE, Ordering::Release);
973        if old_combined_count == COMBINED_REF_COUNT_BOTH_ONE {
974            fence(Ordering::Acquire);
975            if let Some(deleter) = (*handle).deleter {
976                deleter(
977                    handle.cast::<c_void>(),
978                    kTVMFFIObjectDeleterFlagBitMaskBoth as i32,
979                );
980            }
981        } else if (old_combined_count & COMBINED_REF_COUNT_MASK_U32)
982            == COMBINED_REF_COUNT_STRONG_ONE
983        {
984            // slow path, there is still a weak reference left
985            // need to run two phase decrement
986            fence(Ordering::Acquire);
987            if let Some(deleter) = (*handle).deleter {
988                deleter(
989                    handle.cast::<c_void>(),
990                    kTVMFFIObjectDeleterFlagBitMaskStrong as i32,
991                );
992            }
993            let old_weak_count = (*handle)
994                .combined_ref_count
995                .fetch_sub(COMBINED_REF_COUNT_WEAK_ONE, Ordering::Release);
996            if old_weak_count == COMBINED_REF_COUNT_WEAK_ONE {
997                fence(Ordering::Acquire);
998                if let Some(deleter) = (*handle).deleter {
999                    deleter(
1000                        handle.cast::<c_void>(),
1001                        kTVMFFIObjectDeleterFlagBitMaskWeak as i32,
1002                    );
1003                }
1004            }
1005        }
1006    }
1007
1008    #[inline]
1009    pub(crate) unsafe fn strong_count(handle: *mut TVMFFIObject) -> usize {
1010        let obj = &*handle;
1011        (obj.combined_ref_count.load(Ordering::Relaxed) & COMBINED_REF_COUNT_MASK_U32) as usize
1012    }
1013
1014    /// Whether `handle` is the only strong reference, so it may be mutated in
1015    /// place. Any weak reference rejects it, as a weak handle could upgrade and
1016    /// observe the write.
1017    #[inline]
1018    pub(crate) unsafe fn is_unique(handle: *mut TVMFFIObject) -> bool {
1019        // Acquire pairs with the releasing decrement: other owners' accesses precede an in-place write.
1020        (*handle).combined_ref_count.load(Ordering::Acquire) == COMBINED_REF_COUNT_BOTH_ONE
1021    }
1022
1023    #[inline]
1024    pub(crate) unsafe fn weak_count(handle: *mut TVMFFIObject) -> usize {
1025        let obj = &*handle;
1026        (obj.combined_ref_count.load(Ordering::Relaxed) >> 32) as usize
1027    }
1028
1029    /// Generic object deleter for objects allocated through Rust's global allocator.
1030    pub(crate) unsafe extern "C" fn object_deleter_for_new<T>(ptr: *mut c_void, flags: i32) {
1031        let obj = ptr as *mut T;
1032        if flags & kTVMFFIObjectDeleterFlagBitMaskStrong as i32 != 0 {
1033            std::ptr::drop_in_place(obj);
1034        }
1035        if flags & kTVMFFIObjectDeleterFlagBitMaskWeak as i32 != 0 {
1036            std::alloc::dealloc(ptr as *mut u8, std::alloc::Layout::new::<T>());
1037        }
1038    }
1039
1040    /// Layout of a `T` followed by `count` extra items, kept at least one word past the
1041    /// header: once a weak reference outlives the data, the deleter stores `count` there.
1042    pub(crate) fn extra_items_layout<T, U>(count: usize) -> std::alloc::Layout {
1043        let size = std::mem::size_of::<T>() + count * std::mem::size_of::<U>();
1044        let min_size = std::mem::size_of::<TVMFFIObject>() + std::mem::size_of::<u64>();
1045        std::alloc::Layout::from_size_align(size.max(min_size), std::mem::align_of::<T>()).unwrap()
1046    }
1047
1048    pub(crate) unsafe extern "C" fn object_deleter_for_new_with_extra_items<T, U>(
1049        ptr: *mut c_void,
1050        flags: i32,
1051    ) where
1052        T: super::ObjectCoreWithExtraItems<ExtraItem = U>,
1053    {
1054        let obj = ptr as *mut T;
1055        if flags == kTVMFFIObjectDeleterFlagBitMaskBoth as i32 {
1056            let extra_items_count = T::extra_items_count(&(*obj));
1057            std::ptr::drop_in_place(obj);
1058            std::alloc::dealloc(
1059                ptr as *mut u8,
1060                extra_items_layout::<T, U>(extra_items_count),
1061            );
1062        } else {
1063            // Weak references keep using the header, so the weak phase finds the item
1064            // count in the first word after it.
1065            let count_slot = (ptr as *mut u8).add(std::mem::size_of::<TVMFFIObject>()) as *mut u64;
1066            if flags & kTVMFFIObjectDeleterFlagBitMaskStrong as i32 != 0 {
1067                let extra_items_count = T::extra_items_count(&(*obj));
1068                std::ptr::drop_in_place(obj);
1069                std::ptr::write(count_slot, extra_items_count as u64);
1070            }
1071            if flags & kTVMFFIObjectDeleterFlagBitMaskWeak as i32 != 0 {
1072                let extra_items_count = std::ptr::read(count_slot) as usize;
1073                std::alloc::dealloc(
1074                    ptr as *mut u8,
1075                    extra_items_layout::<T, U>(extra_items_count),
1076                );
1077            }
1078        }
1079    }
