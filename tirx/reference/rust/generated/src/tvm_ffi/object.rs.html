42/// Arc-like wrapper for Object that allows shared ownership
43///
44/// \tparam T The type of the object to be wrapped
45#[repr(C)]
46pub struct ObjectArc<T: ObjectCore> {
47    ptr: std::ptr::NonNull<T>,
48    _phantom: std::marker::PhantomData<T>,

54/// The native representation of an object-system type.
55///
56/// # Safety
57/// `Self` must have a compatible C layout with a valid `TVMFFIObject` prefix
58/// at offset zero. Its registered type, inheritance and payload must describe
59/// that layout correctly, including for foreign declarations.
60///
61/// A live managed object and its header must remain at their original allocation
62/// address until destruction. Safe APIs must not move, replace or expose mutable
63/// access to that header while managed; payload mutation must preserve its type
64/// invariants. Preallocation builders and values moved out during destruction
65/// are not live managed objects and cannot be borrowed as `AnyView`.
66pub unsafe trait ObjectCore: Sized + 'static {
67    /// the type key of the object
68    const TYPE_KEY: &'static str;
69    /// Depth of this type in the object inheritance tree.
70    ///
71    /// The root [`Object`] has depth zero, and every registered subtype has
72    /// depth one greater than its parent. This value must be non-negative and
73    /// agree with the runtime type table entry for `Self`.
74    const TYPE_DEPTH: i32;
75    /// Whether every instance of this type has exactly `Self::type_index()`.
76    ///
77    /// A final type has no separately registered object-system subtype.
78    #[doc(hidden)]
79    const TYPE_FINAL: bool = false;
80    // return the type index of the object
81    fn type_index() -> i32;
82    /// Return the object header
83    /// This function is implemented as a static function so
84    ///
85    /// # Arguments
86    /// * `this` - The object to get the header
87    ///
88    /// # Returns
89    /// * `&mut TVMFFIObject` - The object header
90    /// \return The object header
91    ///
92    /// # Safety
93    /// The caller must have exclusive access to the header and preserve the
94    /// managed allocation's address, reference counts, deleter and dynamic type.
95    unsafe fn object_header_mut(this: &mut Self) -> &mut TVMFFIObject;
96}

98/// Traits for objects with extra items that follows the object
99///
100/// This extra trait can be helpful to implement array types and string types
101pub unsafe trait ObjectCoreWithExtraItems: ObjectCore {
102    /// type of extra items storage that follows the object
103    type ExtraItem;
104    /// Return the number of extra items
105    fn extra_items_count(this: &Self) -> usize;
106    /// Return the extra items data pointer
107    unsafe fn extra_items(this: &Self) -> &[Self::ExtraItem] {
108        let extra_items_ptr = (this as *const Self as *const u8).add(std::mem::size_of::<Self>());
109        std::slice::from_raw_parts(
110            extra_items_ptr as *const Self::ExtraItem,
111            Self::extra_items_count(this),
112        )
113    }
114    /// Return the extra items data pointer
115    unsafe fn extra_items_mut(this: &mut Self) -> &mut [Self::ExtraItem] {
116        let extra_items_ptr = (this as *mut Self as *mut u8).add(std::mem::size_of::<Self>());
117        std::slice::from_raw_parts_mut(
118            extra_items_ptr as *mut Self::ExtraItem,
119            Self::extra_items_count(this),
120        )
121    }
122}

124/// Traits to specify core operations of ObjectRef
125///
126/// used by the ffi Any system and not user facing
127///
128/// We mark as unsafe since it moves out the internal of the ObjectRef
129///
130/// # Safety
131///
132/// `data`, `into_data`, and `from_data` must preserve the same object
133/// allocation and form an ownership-preserving round trip. That allocation must
134/// start with a valid `TVMFFIObject` header whose registered object-range
135/// runtime type index correctly describes its layout and inheritance.
136///
137/// When `Self` also implements [`AnyCompatible`], `copy_to_any_view` must
138/// produce a non-owning view, while `move_to_any` must transfer ownership of
139/// the same object pointer and dynamic type index. `move_from_any_after_check`
140/// must be able to reclaim that owned representation exactly once, and a true
141/// `check_any_strict` result must guarantee that both after-check constructors
142/// are valid for it.
143pub unsafe trait ObjectRefCore: Sized + Clone {
144    type ContainerType: ObjectCore;
145    fn data(this: &Self) -> &ObjectArc<Self::ContainerType>;
146    fn into_data(this: Self) -> ObjectArc<Self::ContainerType>;
147
148    /// Construct a reference view from an owning container handle.
149    ///
150    /// # Safety
151    ///
152    /// In addition to containing a valid `ContainerType` allocation, `data`
153    /// must satisfy every semantic invariant imposed by `Self`. This matters
154    /// for zero-state views that share a container type but accept only a
155    /// subset of its values, such as a typed expression view.
156    unsafe fn from_data(data: ObjectArc<Self::ContainerType>) -> Self;
157
158    /// Return whether two object references point to the same allocation.
159    #[inline]
160    fn same_as<Other: ObjectRefCore>(&self, other: &Other) -> bool {
161        unsafe {
162            ObjectArc::as_raw(Self::data(self)).cast::<()>()
163                == ObjectArc::as_raw(Other::data(other)).cast::<()>()
164        }
165    }
166
167    /// Borrow the underlying object as `T` when its runtime type is `T` or a subtype.
168    ///
169    /// `T` is an object representation type such as `FooObj`, rather than its
170    /// managed reference `Foo`. Unrelated object types return `None`.
171    /// Unlike [`ObjectRefCast::try_cast`], this method neither consumes the
172    /// reference nor changes the object's reference count. The returned borrow
173    /// cannot outlive `self`.
174    /// See [`crate::Any::downcast_ref`] for an example.
175    #[inline(always)]
176    fn downcast_ref<T: ObjectCore>(&self) -> Option<&T> {
177        let object = unsafe { ObjectArc::as_raw(Self::data(self)) };
178        let type_index = unsafe { (*object.cast::<TVMFFIObject>()).type_index };
179        if !is_instance_of::<T>(type_index) {
180            return None;
181        }
182        // SAFETY: ObjectRefCore guarantees a live object allocation, the type
183        // check establishes T's layout, and the borrow cannot outlive self.
184        Some(unsafe { &*object.cast::<T>() })
185    }
186}

188/// An owning, hashable identity key for an FFI object.
189///
190/// The retained strong reference prevents the allocation address from being
191/// reused while the key is alive. This makes it suitable for identity-based
192/// maps without exposing raw object pointers to downstream code.
193#[derive(Clone)]
194pub struct ObjectIdentity {
195    data: ObjectArc<Object>,
196}
197
198impl ObjectIdentity {
199    /// Retain the allocation referenced by `value` as an identity key.
200    pub fn of<T: ObjectRefCore>(value: &T) -> Self {
201        unsafe {
202            let ptr = ObjectArc::as_raw(T::data(value)) as *mut TVMFFIObject;
203            unsafe_::inc_ref(ptr);
204            Self {
205                data: ObjectArc::from_raw(ptr.cast::<Object>()),
206            }
207        }
208    }
209
210    #[inline]
211    fn as_ptr(&self) -> *const TVMFFIObject {
212        unsafe { ObjectArc::as_raw(&self.data).cast::<TVMFFIObject>() }
213    }
214}

217    #[inline]
218    fn eq(&self, other: &Self) -> bool {
219        self.as_ptr() == other.as_ptr()
220    }
221}

240/// Check whether a runtime type index refers to `Target` or one of its
241/// subtypes.
242///
243/// The subtype relation lives in the process-wide type table maintained by the
244/// tvm-ffi library: every registered type records its depth in the single
245/// inheritance tree together with the chain of its ancestors. The check is
246/// O(1) — if `target` really is an ancestor, it must appear in the candidate's
247/// ancestor array exactly at `target`'s depth.
248///
249/// This is a hidden support function for derive-generated object checks. Object
250/// indices in the registered range must refer to entries in the runtime type
251/// table.
252#[doc(hidden)]
253#[inline(always)]
254pub fn is_instance_of<Target: ObjectCore>(object_type_index: i32) -> bool {
255    let target_type_index = Target::type_index();
256    if object_type_index == target_type_index {
257        return true;
258    }
259    // A final type cannot have a separately registered subtype. Keep common
260    // borrowed checks, such as `IntImmObj`, to one integer comparison.
261    if Target::TYPE_FINAL {
262        return false;
263    }
264    let object_begin = TypeIndex::kTVMFFIStaticObjectBegin as i32;
265    // Only object types participate in the type hierarchy.
266    if object_type_index < object_begin || target_type_index < object_begin {
267        return false;
268    }
269    // Only the root Object type has depth zero.
270    if const { Target::TYPE_DEPTH == 0 } {
271        return true;
272    }
273    // Parent indices are always smaller than their descendants.
274    if object_type_index < target_type_index {
275        return false;
276    }
277    unsafe {
278        let object_info = TVMFFIGetTypeInfo(object_type_index);
279        if object_info.is_null() {
280            return false;
281        }
282        let target_depth = Target::TYPE_DEPTH;
283        if (*object_info).type_depth <= target_depth {
284            return false;
285        }
286        let ancestor = *(*object_info).type_acenstors.add(target_depth as usize);
287        !ancestor.is_null() && (*ancestor).type_index == target_type_index
288    }
289}

291/// Runtime-checked casting between arbitrary `ObjectRef` types.
292///
293/// The cast uses the target's [`AnyCompatible::check_any_strict`] implementation,
294/// mirroring the semantics of `ObjectRef::as<T>` in C++. This supports both
295/// object hierarchies and parameterized object containers.
296///
297/// This trait is blanket-implemented for every [`ObjectRefCore`] type that is
298/// also [`AnyCompatible`].
299pub trait ObjectRefCast: ObjectRefCore + AnyCompatible {
300    /// Consume `self` and rewrap the underlying object as `B` without copying.
301    #[inline(always)]
302    fn try_cast<B>(self) -> crate::error::Result<B>
303    where
304        B: ObjectRefCore + AnyCompatible,
305    {
306        let mut any_data = TVMFFIAny::new();
307        unsafe {
308            // Keep ownership in `self` while the target check runs. This makes
309            // the failure and panic paths unwind normally instead of stranding
310            // an owned object inside a raw TVMFFIAny.
311            Self::copy_to_any_view(&self, &mut any_data);
312            debug_assert!(any_data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32);
313            // SAFETY: ObjectRefCore's contract requires its AnyCompatible
314            // representation to contain a valid object-range type index.
315            std::hint::assert_unchecked(
316                any_data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32,
317            );
318
319            if B::check_any_strict(&any_data) {
320                // Transfer ownership only after the borrowed representation has
321                // passed the target's complete hierarchy/container check.
322                Self::move_to_any(self, &mut any_data);
323                Ok(B::move_from_any_after_check(&mut any_data))
324            } else {
325                let msg = format!(
326                    "Cannot convert from type `{}` to `{}`",
327                    B::get_mismatch_type_info(&any_data),
328                    B::type_str()
329                );
330                Err(crate::error::Error::new(crate::error::TYPE_ERROR, &msg, ""))
331            }
332        }
333    }
334}

348/// Unsafe operations on object
349#[doc(hidden)]
350pub mod unsafe_ {
351    use tvm_ffi_sys::{
352        COMBINED_REF_COUNT_BOTH_ONE, COMBINED_REF_COUNT_MASK_U32, COMBINED_REF_COUNT_STRONG_ONE,
353        COMBINED_REF_COUNT_WEAK_ONE,
354    };
355
356    use std::ffi::c_void;
357    use std::sync::atomic::{fence, Ordering};
358    use tvm_ffi_sys::TVMFFIObject;
359    use tvm_ffi_sys::TVMFFIObjectDeleterFlagBitMask::{
360        kTVMFFIObjectDeleterFlagBitMaskBoth, kTVMFFIObjectDeleterFlagBitMaskStrong,
361        kTVMFFIObjectDeleterFlagBitMaskWeak,
362    };
363
364    /// Increase the strong reference count of the object
365    ///
366    /// This function is same as TVMFFIObjectIncRef but implemented natively in Rust
367    ///
368    /// # Arguments
369    /// * `obj` - The object to increase the reference count
370    #[inline]
371    pub unsafe fn inc_ref(handle: *mut TVMFFIObject) {
372        let obj = &*handle;
373        obj.combined_ref_count.fetch_add(1, Ordering::Relaxed);
374    }
375
376    /// Decrease the strong reference count of the object
377    ///
378    /// This function is same as TVMFFIObjectDecRef but implemented natively in Rust
379    ///
380    /// # Arguments
381    /// * `obj` - The object to decrease the reference count
382    #[inline]
383    pub(crate) unsafe fn dec_ref(handle: *mut TVMFFIObject) {
384        let old_combined_count = (*handle)
385            .combined_ref_count
386            .fetch_sub(COMBINED_REF_COUNT_STRONG_ONE, Ordering::Relaxed);
387        if old_combined_count == COMBINED_REF_COUNT_BOTH_ONE {
388            if let Some(deleter) = (*handle).deleter {
389                fence(Ordering::Acquire);
390                deleter(
391                    handle.cast::<c_void>(),
392                    kTVMFFIObjectDeleterFlagBitMaskBoth as i32,
393                );
394            }
395        } else if (old_combined_count & COMBINED_REF_COUNT_MASK_U32)
396            == COMBINED_REF_COUNT_STRONG_ONE
397        {
398            // slow path, there is still a weak reference left
399            // need to run two phase decrement
400            fence(Ordering::Acquire);
401            if let Some(deleter) = (*handle).deleter {
402                deleter(
403                    handle.cast::<c_void>(),
404                    kTVMFFIObjectDeleterFlagBitMaskStrong as i32,
405                );
406            }
407            let old_weak_count = (*handle)
408                .combined_ref_count
409                .fetch_sub(COMBINED_REF_COUNT_WEAK_ONE, Ordering::Release);
410            if old_weak_count == COMBINED_REF_COUNT_WEAK_ONE {
411                fence(Ordering::Acquire);
412                if let Some(deleter) = (*handle).deleter {
413                    deleter(
414                        handle.cast::<c_void>(),
415                        kTVMFFIObjectDeleterFlagBitMaskWeak as i32,
416                    );
417                }
418            }
419        }
420    }
421
422    #[inline]
423    pub(crate) unsafe fn strong_count(handle: *mut TVMFFIObject) -> usize {
424        let obj = &*handle;
425        (obj.combined_ref_count.load(Ordering::Relaxed) & COMBINED_REF_COUNT_MASK_U32) as usize
426    }
427
428    #[inline]
429    pub(crate) unsafe fn weak_count(handle: *mut TVMFFIObject) -> usize {
430        let obj = &*handle;
431        (obj.combined_ref_count.load(Ordering::Relaxed) >> 32) as usize
432    }
433
434    /// Generic object deleter for objects allocated through Rust's global allocator.
435    pub(crate) unsafe extern "C" fn object_deleter_for_new<T>(ptr: *mut c_void, flags: i32) {
436        let obj = ptr as *mut T;
437        if flags & kTVMFFIObjectDeleterFlagBitMaskStrong as i32 != 0 {
438            std::ptr::drop_in_place(obj);
439        }
440        if flags & kTVMFFIObjectDeleterFlagBitMaskWeak as i32 != 0 {
441            std::alloc::dealloc(ptr as *mut u8, std::alloc::Layout::new::<T>());
442        }
443    }
444
445    /// Layout of a `T` followed by `count` extra items, kept at least one word past the
446    /// header: once a weak reference outlives the data, the deleter stores `count` there.
447    pub(crate) fn extra_items_layout<T, U>(count: usize) -> std::alloc::Layout {
448        let size = std::mem::size_of::<T>() + count * std::mem::size_of::<U>();
449        let min_size = std::mem::size_of::<TVMFFIObject>() + std::mem::size_of::<u64>();
450        std::alloc::Layout::from_size_align(size.max(min_size), std::mem::align_of::<T>()).unwrap()
451    }
452
453    pub(crate) unsafe extern "C" fn object_deleter_for_new_with_extra_items<T, U>(
454        ptr: *mut c_void,
455        flags: i32,
456    ) where
457        T: super::ObjectCoreWithExtraItems<ExtraItem = U>,
458    {
459        let obj = ptr as *mut T;
460        if flags == kTVMFFIObjectDeleterFlagBitMaskBoth as i32 {
461            let extra_items_count = T::extra_items_count(&(*obj));
462            std::ptr::drop_in_place(obj);
463            std::alloc::dealloc(
464                ptr as *mut u8,
465                extra_items_layout::<T, U>(extra_items_count),
466            );
467        } else {
468            // Weak references keep using the header, so the weak phase finds the item
469            // count in the first word after it.
470            let count_slot = (ptr as *mut u8).add(std::mem::size_of::<TVMFFIObject>()) as *mut u64;
471            if flags & kTVMFFIObjectDeleterFlagBitMaskStrong as i32 != 0 {
472                let extra_items_count = T::extra_items_count(&(*obj));
473                std::ptr::drop_in_place(obj);
474                std::ptr::write(count_slot, extra_items_count as u64);
475            }
476            if flags & kTVMFFIObjectDeleterFlagBitMaskWeak as i32 != 0 {
477                let extra_items_count = std::ptr::read(count_slot) as usize;
478                std::alloc::dealloc(
479                    ptr as *mut u8,
480                    extra_items_layout::<T, U>(extra_items_count),
481                );
482            }
483        }
484    }
485}

516// Allocate and initialize the normal FFI header while preserving the concrete
517// payload's alignment and drop/layout deleter. The payload starts with an Object.
518unsafe fn allocate_object<T>(data: T, type_index: i32) -> NonNull<T> {
519    let layout = std::alloc::Layout::new::<T>();
520    let allocation = std::alloc::alloc(layout);
521    if allocation.is_null() {
522        std::alloc::handle_alloc_error(layout);
523    }
524    let ptr = allocation.cast::<T>();
525    std::ptr::write(ptr, data);
526    std::ptr::write(
527        ptr.cast::<TVMFFIObject>(),
528        TVMFFIObject {
529            combined_ref_count: AtomicU64::new(COMBINED_REF_COUNT_BOTH_ONE),
530            type_index,
531            __padding: 0,
532            deleter: Some(unsafe_::object_deleter_for_new::<T>),
533        },
534    );
535    NonNull::new_unchecked(ptr)
536}

538impl ObjectArc<Object> {
539    /// Allocate a synchronous runtime payload that may contain borrowed Rust state.
540    ///
541    /// # Safety
542    /// `T` begins with a valid Object header and is ABI-compatible with the
543    /// registered `type_index`. The caller must keep this owner scoped to the
544    /// payload's borrows and ensure every temporary strong and weak reference
545    /// is released before those borrows end. Neither native handles nor this
546    /// erased owner may escape that synchronous scope. Destruction uses T's
547    /// exact layout and drop function, not the erased Object layout.
548    /// The erased owner and handles must remain on this thread unless the
549    /// concrete payload permits sharing or transfer; erasure grants neither.
550    pub(crate) unsafe fn new_opaque_unchecked<T>(data: T, type_index: i32) -> Self {
551        Self {
552            ptr: allocate_object(data, type_index).cast(),
553            _phantom: std::marker::PhantomData,
554        }
555    }
556}

558impl<T: ObjectCore> ObjectArc<T> {
559    /// Allocate an initialized object at its stable managed address.
560    pub fn new(data: T) -> Self {
561        // Resolve metadata before moving the payload into its allocation.
562        let type_index = T::type_index();
563        unsafe {
564            Self {
565                ptr: allocate_object(data, type_index),
566                _phantom: std::marker::PhantomData,
567            }
568        }
569    }
570
571    /// Allocate a stable object followed by its type's extra item storage.
572    pub fn new_with_extra_items<U>(data: T) -> Self
573    where
574        T: ObjectCoreWithExtraItems<ExtraItem = U>,
575    {
576        unsafe {
577            // ensure strict alignment requirements
578            // so we can have { T, U*extra_items } layout
579            assert_eq!(std::mem::align_of::<T>() % std::mem::align_of::<U>(), 0);
580            assert_eq!(std::mem::size_of::<T>() % std::mem::align_of::<U>(), 0);
581            let extra_items_count = T::extra_items_count(&data);
582            let layout = unsafe_::extra_items_layout::<T, U>(extra_items_count);
583            let raw_data_ptr = std::alloc::alloc(layout);
584            if raw_data_ptr.is_null() {
585                std::alloc::handle_alloc_error(layout);
586            }
587            let ptr = raw_data_ptr as *mut T;
588            std::ptr::write(ptr, data);
589            // now override the header directly
590            std::ptr::write(
591                ptr as *mut TVMFFIObject,
592                TVMFFIObject {
593                    combined_ref_count: AtomicU64::new(COMBINED_REF_COUNT_BOTH_ONE),
594                    type_index: T::type_index(),
595                    __padding: 0,
596                    deleter: Some(unsafe_::object_deleter_for_new_with_extra_items::<T, U>),
597                },
598            );
599            // move into the object arc ptr
600            Self {
601                ptr: std::ptr::NonNull::new_unchecked(ptr as *mut T),
602                _phantom: std::marker::PhantomData,
603            }
604        }
605    }
606
607    /// Whether `this` holds a null pointer, as all-zero bytes do: no
608    /// `ObjectArc`, but the field of an object that a creator left undefined.
609    ///
610    /// # Safety
611    ///
612    /// `this` points to an `ObjectArc<T>` or to all-zero bytes.
613    #[doc(hidden)]
614    #[inline]
