33// Type registration: writes to the runtime's type table from Rust, and the
34// check that a key registered elsewhere matches its Rust declaration.
35
36/// Held around every write to the type table from Rust, so that registrations
37/// from Rust do not race each other: the runtime does not lock its type table
38/// and leaves that to callers.
39static TYPE_REGISTRATION: std::sync::Mutex<()> = std::sync::Mutex::new(());
40
41/// Holds the lock around every write to the type table from Rust. A panic in
42/// another registration leaves the table consistent, so a poisoned lock is
43/// still taken.
44#[doc(hidden)]
45pub fn lock_type_registration() -> std::sync::MutexGuard<'static, ()> {
46    TYPE_REGISTRATION
47        .lock()
48        .unwrap_or_else(std::sync::PoisonError::into_inner)
49}

51/// Panics unless `type_index`, the index `#[derive(Object)]` registered or
52/// bound for `T`'s key, describes `T`: a key registered by another library or
53/// another type must have the same parent and depth and, if it has metadata,
54/// the same size, or for a `binding` no larger, since an opaque binding
55/// declares only its parent.
56///
57/// # Safety
58///
59/// `type_index` is registered and `parent` is the type index of `T::Parent`.
60#[doc(hidden)]
61pub unsafe fn assert_registered_type_matches<T: ObjectCore>(
62    type_index: i32,
63    parent: i32,
64    binding: bool,
65) {
66    let info = &*TVMFFIGetTypeInfo(type_index);
67    let depth = T::TYPE_DEPTH;
68    let same_parent = info.type_depth == depth
69        && depth > 0
70        && (**info.type_acenstors.add(depth as usize - 1)).type_index == parent;
71    let size = std::mem::size_of::<T>();
72    let same_size = info.metadata.is_null() || {
73        let registered = (*info.metadata).total_size as usize;
74        size == registered || (binding && size < registered)
75    };
76    assert!(
77        same_parent && same_size,
78        "Type key {} is already registered with another parent, depth or size",
79        T::TYPE_KEY
80    );
81}

83/// The root object type, `ffi.Object`: just the object header.
84///
85/// An object type defined in Rust starts with its parent's object, which for
86/// a direct subtype is `base_: Object`.
87#[repr(C)]
88pub struct Object {
89    header: TVMFFIObject,
90}

92impl Object {
93    /// A new object header, for the `base_` field of a new object.
94    #[allow(clippy::new_without_default)]
95    pub fn new() -> Self {
96        Self {
97            header: TVMFFIObject::new(),
98        }
99    }
100}

102unsafe impl ObjectCore for Object {
103    type Parent = ();
104    const TYPE_KEY: &'static str = "ffi.Object";
105    const TYPE_DEPTH: i32 = 0;

108        TypeIndex::kTVMFFIStaticObjectBegin as i32

116/// A reference-counted pointer to an object `T`, as `Arc<T>`, counting with
117/// the object header so that every language shares the count.
118#[repr(C)]
119pub struct ObjectArc<T: ObjectCore> {
120    ptr: std::ptr::NonNull<T>,
121    _phantom: std::marker::PhantomData<T>,
122}
123
124impl ObjectArc<Object> {
125    /// Allocate a synchronous runtime payload that may contain borrowed Rust state.
126    ///
127    /// # Safety
128    /// `T` begins with a valid Object header and is ABI-compatible with the
129    /// registered `type_index`. The caller must keep this owner scoped to the
130    /// payload's borrows and ensure every temporary strong and weak reference
131    /// is released before those borrows end. Neither native handles nor this
132    /// erased owner may escape that synchronous scope. Destruction uses T's
133    /// exact layout and drop function, not the erased Object layout.
134    /// The erased owner and handles must remain on this thread unless the
135    /// concrete payload permits sharing or transfer; erasure grants neither.
136    pub(crate) unsafe fn new_opaque_unchecked<T>(data: T, type_index: i32) -> Self {
137        Self {
138            ptr: allocate_object(data, type_index).cast(),
139            _phantom: std::marker::PhantomData,
140        }
141    }
142}
143
144impl<T: ObjectCore> ObjectArc<T> {
145    /// Allocate an initialized object at its stable managed address.
146    pub fn new(data: T) -> Self {
147        // Resolve metadata before moving the payload into its allocation.
148        let type_index = T::type_index();
149        unsafe {
150            Self {
151                ptr: allocate_object(data, type_index),
152                _phantom: std::marker::PhantomData,
153            }
154        }
155    }
156
157    /// Allocate a stable object followed by its type's extra item storage.
158    ///
159    /// # Safety
160    /// The extra items start uninitialized: the caller writes each of them,
161    /// through [`ObjectCoreWithExtraItems::extra_items_ptr`] of
162    /// [`as_mut_ptr`](Self::as_mut_ptr), before the object is read or shared.
163    pub unsafe fn new_with_extra_items<U>(data: T) -> Self
164    where
165        T: ObjectCoreWithExtraItems<ExtraItem = U>,
166    {
167        unsafe {
168            // ensure strict alignment requirements
169            // so we can have { T, U*extra_items } layout
170            assert_eq!(std::mem::align_of::<T>() % std::mem::align_of::<U>(), 0);
171            assert_eq!(std::mem::size_of::<T>() % std::mem::align_of::<U>(), 0);
172            let extra_items_count = T::extra_items_count(&data);
173            let layout = unsafe_::extra_items_layout::<T, U>(extra_items_count);
174            let raw_data_ptr = std::alloc::alloc(layout);
175            if raw_data_ptr.is_null() {
176                std::alloc::handle_alloc_error(layout);
177            }
178            let ptr = raw_data_ptr as *mut T;
179            std::ptr::write(ptr, data);
180            // now override the header directly
181            std::ptr::write(
182                ptr as *mut TVMFFIObject,
183                TVMFFIObject {
184                    combined_ref_count: AtomicU64::new(COMBINED_REF_COUNT_BOTH_ONE),
185                    type_index: T::type_index(),
186                    __padding: 0,
187                    deleter: Some(unsafe_::object_deleter_for_new_with_extra_items::<T, U>),
188                },
189            );
190            // move into the object arc ptr
191            Self {
192                ptr: std::ptr::NonNull::new_unchecked(ptr),
193                _phantom: std::marker::PhantomData,
194            }
195        }
196    }
197
198    /// Whether `this` holds a null pointer, as all-zero bytes do: no
199    /// `ObjectArc`, but the field of an object that a creator left undefined.
200    ///
201    /// See also [`AnyCompatible::is_undefined`].
202    ///
203    /// # Safety
204    ///
205    /// `this` points to an `ObjectArc<T>` or to all-zero bytes.
206    #[inline]
207    pub(crate) unsafe fn is_null(this: *const Self) -> bool {
208        (*std::ptr::addr_of!((*this).ptr).cast::<*const T>()).is_null()
209    }
210
211    /// Take ownership of a raw object pointer, as `Arc::from_raw`.
212    ///
213    /// # Arguments
214    /// * `ptr` - A pointer that owns one strong reference.
215    ///
216    /// # Returns
217    /// The handle that owns that reference.
218    ///
219    /// # Safety
220    /// `ptr` must own one strong reference to a live, stable managed allocation
221    /// whose dynamic type is compatible with `T` and whose payload is valid.
222    /// Ownership of that reference is transferred into the returned handle.
223    #[inline(always)]
224    pub unsafe fn from_raw(ptr: *const T) -> Self {
225        Self {
226            ptr: std::ptr::NonNull::new_unchecked(ptr as *mut T),
227            _phantom: std::marker::PhantomData,
228        }
229    }
230
231    /// Turn the handle into a raw pointer, as `Arc::into_raw`.
232    ///
233    /// # Arguments
234    /// * `this` - The handle.
235    ///
236    /// # Returns
237    /// The pointer, which owns the strong reference; pass it back to
238    /// [`ObjectArc::from_raw`] to release it.
239    #[inline(always)]
240    pub fn into_raw(this: Self) -> *const T {
241        let droped_this = std::mem::ManuallyDrop::new(this);
242        droped_this.ptr.as_ptr() as *const T
243    }
244
245    /// The raw pointer, without giving up ownership, as `Arc::as_ptr`.
246    ///
247    /// # Arguments
248    /// * `this` - The handle.
249    ///
250    /// # Returns
251    /// The non-owning pointer. Dereferencing it must respect existing borrows,
252    /// and it cannot outlive an owning handle.
253    #[inline]
254    pub fn as_ptr(this: &Self) -> *const T {
255        this.ptr.as_ptr() as *const T
256    }
257
258    /// The raw mutable pointer, without giving up ownership.
259    ///
260    /// # Arguments
261    /// * `this` - The handle.
262    ///
263    /// # Returns
264    /// The non-owning pointer. The handle need not be unique: writing through
265    /// it needs exclusive access to the object, and must not move or replace
266    /// the object or its header.
267    #[inline]
268    pub fn as_mut_ptr(this: &mut Self) -> *mut T {
269        this.ptr.as_ptr()
270    }
271
272    /// The number of strong references, as `Arc::strong_count`.
273    ///
274    /// # Arguments
275    /// * `this` - The handle.
276    #[inline]
277    pub fn strong_count(this: &Self) -> usize {
278        unsafe { unsafe_::strong_count(this.ptr.as_ptr().cast::<TVMFFIObject>()) }
279    }
280
281    /// Whether this is the only reference, strong and with no weak one, so the
282    /// object may be mutated in place.
283    ///
284    /// # Arguments
285    /// * `this` - The handle.
286    #[inline]
287    pub fn is_unique(this: &Self) -> bool {
288        unsafe { unsafe_::is_unique(this.ptr.as_ptr().cast::<TVMFFIObject>()) }
289    }
290
291    /// The number of weak references, as `Arc::weak_count`.
292    ///
293    /// # Arguments
294    /// * `this` - The handle.
295    #[inline]
296    pub fn weak_count(this: &Self) -> usize {
297        unsafe { unsafe_::weak_count(this.ptr.as_ptr().cast::<TVMFFIObject>()) }
298    }
299}

330/// An object type whose shared instances are never mutated, so `ObjectArc<T>` is
331/// `Send + Sync` when `T`'s fields are.
332///
333/// Implement it with the `immutable` option of [`rs_object`](crate::rs_object)
334/// or `#[type_immutable]` on [`derive(Object)`](crate::derive::Object), which
335/// reject `writable` fields.
336///
337/// # Safety
338///
339/// No code may mutate an instance while it is shared: changes go through
340/// copy-on-write, in place only on a uniquely held instance (one strong
341/// reference, no weak ones). This holds for every type not registered as
342/// mutable (`_type_mutable`), in any language. Every subclass must also be
343/// immutable while shared, with thread-safe fields.
344pub unsafe trait ImmutableObject: ObjectCore {}
345
346// SAFETY: `T` is never mutated while shared, its fields are thread-safe, and the
347// reference count is atomic.
348unsafe impl<T: ImmutableObject + Send + Sync> Send for ObjectArc<T> {}
349// SAFETY: as for `Send`.
350unsafe impl<T: ImmutableObject + Send + Sync> Sync for ObjectArc<T> {}

362/// The native representation of an object-system type.
363///
364/// # Safety
365/// `Self` must have a compatible C layout with a valid `TVMFFIObject` prefix
366/// at offset zero, and embed a `Self::Parent` there unless it is the root.
367/// Its registered type, inheritance and payload must describe that layout
368/// correctly, including for foreign declarations.
369///
370/// A live managed object and its header must remain at their original allocation
371/// address until destruction. Safe APIs must not move, replace or expose mutable
372/// access to that header while managed; payload mutation must preserve its type
373/// invariants. Preallocation builders and values moved out during destruction
374/// are not live managed objects and cannot be borrowed as `AnyView`.
375#[diagnostic::on_unimplemented(message = "`{Self}` is not an object type")]
376pub unsafe trait ObjectCore: Sized + 'static {
377    /// The parent's object type, which `Self` embeds at offset zero; `()` for
378    /// the root [`Object`].
379    type Parent;
380    /// The type key, such as `"ffi.Array"`.
381    const TYPE_KEY: &'static str;
382    /// Depth of this type in the object inheritance tree.
383    ///
384    /// The root [`Object`] has depth zero, and every registered subtype has
385    /// depth one greater than its parent. This value must be non-negative and
386    /// agree with the runtime type table entry for `Self`.
387    const TYPE_DEPTH: i32;
388    /// Whether every instance of this type has exactly `Self::type_index()`.
389    ///
390    /// A final type has no separately registered object-system subtype.
391    #[doc(hidden)]
392    const TYPE_FINAL: bool = false;
393    /// The runtime type index of this type.
394    fn type_index() -> i32;
395    /// Whether [`type_index`](Self::type_index) can return now: false while
396    /// the key of a binding, this type or an ancestor, is not registered
397    /// because its library is not loaded yet.
398    #[doc(hidden)]
399    fn type_index_available() -> bool {
400        true
401    }
402    /// The object header, mutably.
403    ///
404    /// # Arguments
405    /// * `this` - The object.
406    ///
407    /// # Safety
408    /// The caller must have exclusive access to the header and preserve the
409    /// managed allocation's address, reference counts, deleter and dynamic type.
410    unsafe fn object_header_mut(this: &mut Self) -> &mut TVMFFIObject;
411}

413// Object hierarchy: a compile-time check that one object type derives from
414// another, which `#[object_ref(parent = ...)]` relies on.
415
416/// The object type `Self` is `A` or derives from it through its chain of
417/// [`ObjectCore::Parent`]s, so a `Self` is a valid `A`.
418///
419/// `Depth` is the number of parent steps from `Self` to `A`, a type from
420/// [`marker`] that the compiler infers.
421///
422/// # Safety
423///
424/// The blanket implementations are the only sound ones: do not implement it.
425#[diagnostic::on_unimplemented(message = "`{Self}` does not derive from `{A}`")]
426pub unsafe trait ObjectDerivesFrom<A, Depth> {}
427
428/// Inferred depths for [`ObjectDerivesFrom`].
429pub mod marker {
430    /// `A` is the object type itself.
431    pub enum Itself {}
432    /// `A` is reached through one more [`Parent`](super::ObjectCore::Parent).
433    pub struct Parent<Depth>(std::marker::PhantomData<fn() -> Depth>);
434}
435
436// SAFETY: every type is itself.
437unsafe impl<T: ObjectCore> ObjectDerivesFrom<T, marker::Itself> for T {}
438
439// SAFETY: `T` embeds `T::Parent` at offset zero, so whatever the parent
440// derives from, `T` does.
441unsafe impl<T: ObjectCore, A, Depth> ObjectDerivesFrom<A, marker::Parent<Depth>> for T where
442    T::Parent: ObjectDerivesFrom<A, Depth>
443{
444}

446/// An object type whose allocation is followed by a run of extra items, such
447/// as the dimensions of a [`Shape`](crate::Shape).
448///
449/// The items lie past `Self`, so a pointer to them is derived from a pointer
450/// to the whole allocation, such as [`ObjectArc::as_ptr`], never from a
451/// `&Self`, which covers `Self` only.
452///
453/// # Safety
454/// Every allocation of `Self` is followed by `extra_items_count` `ExtraItem`s,
455/// as [`ObjectArc::new_with_extra_items`] lays them out. The count stays the
456/// same for the object's life, and the items are never dropped.
457pub unsafe trait ObjectCoreWithExtraItems: ObjectCore {
458    /// The type of the items that follow the object.
459    type ExtraItem;
460    /// The number of extra items.
461    fn extra_items_count(this: &Self) -> usize;
462    /// A pointer to the first extra item.
463    ///
464    /// # Safety
465    /// `this` points to an allocation that carries its extra items, and its
466    /// provenance covers them.
467    unsafe fn extra_items_ptr(this: *mut Self) -> *mut Self::ExtraItem {
468        unsafe { this.cast::<u8>().add(std::mem::size_of::<Self>()).cast() }
469    }
470    /// The extra items.
471    ///
472    /// # Safety
473    /// As [`extra_items_ptr`](Self::extra_items_ptr), with the items
474    /// initialized and not written for `'a`.
475    unsafe fn extra_items<'a>(this: *const Self) -> &'a [Self::ExtraItem] {
476        unsafe {
477            std::slice::from_raw_parts(
478                Self::extra_items_ptr(this.cast_mut()),
479                Self::extra_items_count(&*this),
480            )
481        }
482    }
483}

485/// A managed object reference, such as [`ObjectRef`](struct@ObjectRef), `Array<T>`
486/// or a `#[derive(ObjectRef)]` type, with identity ([`same_as`](Self::same_as))
487/// and strict casts that check the runtime type and never convert; upcasts are
488/// `From`/`Into`.
489///
490/// # Casts
491///
492/// * [`downcast_ref`](Self::downcast_ref) borrows, strictly, as std's
493///   `dyn Any::downcast_ref` does.
494/// * [`downcast`](Self::downcast) takes the value, strictly, and hands it back
495///   on a mismatch, as `Box<dyn Any>::downcast` does.
496/// * `TryFrom` takes the value and returns a `TypeError`, for `?`. Between
497///   references it only checks; from an [`Any`](crate::Any) it converts by
498///   the FFI rules, such as an integer into a float.
499///
500/// # Safety
501///
502/// `data`, `into_data`, and `from_data` must preserve the same object
503/// allocation and form an ownership-preserving round trip. That allocation must
504/// start with a valid `TVMFFIObject` header whose registered object-range
505/// runtime type index correctly describes its layout and inheritance.
506///
507/// When `Self` also implements [`AnyCompatible`], `copy_to_any_view` must
508/// produce a non-owning view, while `move_to_any` must transfer ownership of
509/// the same object pointer and dynamic type index. `move_from_any_after_check`
510/// must be able to reclaim that owned representation exactly once, and a true
511/// `check_any_strict` result must guarantee that both after-check constructors
512/// are valid for it.
513pub unsafe trait ObjectRefCore: Sized + Clone + AnyCompatible {
514    /// The object type the reference points to.
515    type ContainerType: ObjectCore;
516    /// Borrow the object's handle.
517    fn data(this: &Self) -> &ObjectArc<Self::ContainerType>;
518    /// Take the object's handle.
519    fn into_data(this: Self) -> ObjectArc<Self::ContainerType>;
520
521    /// Make a reference from the object's handle.
522    ///
523    /// # Safety
524    ///
525    /// In addition to containing a valid `ContainerType` allocation, `data`
526    /// must satisfy every semantic invariant imposed by `Self`. This matters
527    /// for zero-state views that share a container type but accept only a
528    /// subset of its values, such as a reference whose
529    /// [`ObjectRefAnyCompatible::extra_check`] rejects some objects.
530    unsafe fn from_data(data: ObjectArc<Self::ContainerType>) -> Self;
531
532    /// Whether two references point to the same object.
533    #[inline]
534    fn same_as<Other: ObjectRefCore>(&self, other: &Other) -> bool {
535        ObjectArc::as_ptr(Self::data(self)).cast::<()>()
536            == ObjectArc::as_ptr(Other::data(other)).cast::<()>()
537    }
538
539    /// Borrow the underlying object as `T` when it is one.
540    ///
541    /// `T` is an object type such as `MyPairObj`, or a managed reference such
542    /// as `MyPair`: `&ObjectRef` borrows as `&MyPair` when its object is a
543    /// `MyPairObj`. Unrelated types return `None`. Unlike [`downcast`](Self::downcast), this
544    /// method neither consumes the reference nor changes the object's reference
545    /// count. The returned borrow cannot outlive `self`.
546    /// See [`crate::Any::downcast_ref`] for an example.
547    #[inline(always)]
548    fn downcast_ref<T: crate::type_traits::DowncastRefFromObject>(&self) -> Option<&T> {
549        // SAFETY: `ObjectArc` is exactly one non-null object pointer, and
550        // ObjectRefCore guarantees a live object allocation.
551        unsafe {
552            let place = &*(Self::data(self) as *const ObjectArc<Self::ContainerType>
553                as *const *mut TVMFFIObject);
554            T::downcast_ref_from_object(place)
555        }
556    }
557
558    /// Take `self` as `B`, a subtype or a container such as `Array<i64>` whose
559    /// elements are checked too, when its object is one, without copying it.
560    /// On a mismatch the reference is returned unchanged.
561    ///
562    /// ```
563    /// use tvm_ffi::{object::ObjectRef, Array, ObjectRefCore};
564    /// let object: ObjectRef = Array::from_vec(vec![1_i64]).into();
565    /// let object = object.downcast::<Array<f64>>().unwrap_err();
566    /// let array: Array<i64> = object.try_into()?;
567    /// assert_eq!(array.len(), 1);
568    /// # Ok::<(), tvm_ffi::Error>(())
569    /// ```
570    #[inline(always)]
571    fn downcast<B>(self) -> std::result::Result<B, Self>
572    where
573        B: ObjectRefCore,
574    {
575        let mut any_data = TVMFFIAny::new();
576        unsafe {
577            // Keep ownership in `self` while the target check runs, so a
578            // mismatch can hand it back and a panic unwinds normally.
579            Self::copy_to_any_view(&self, &mut any_data);
580            debug_assert!(any_data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32);
581            // SAFETY: ObjectRefCore's contract requires its AnyCompatible
582            // representation to contain a valid object-range type index.
583            std::hint::assert_unchecked(
584                any_data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32,
585            );
586            if B::check_any_strict(&any_data) {
587                // Transfer ownership only after the borrowed representation has
588                // passed the target's complete hierarchy/container check.
589                Self::move_to_any(self, &mut any_data);
590                Ok(B::move_from_any_after_check(&mut any_data))
591            } else {
592                Err(self)
593            }
594        }
595    }
596}

598/// The [`AnyCompatible`] implementation of an object reference, with every
599/// hook defaulted: every implementer is `AnyCompatible`.
600///
601/// `#[derive(ObjectRef)]` implements it with the defaults. A typed reference
602/// that accepts only some objects of its type, or converts other values,
603/// implements it by hand and overrides only what differs, such as
604/// [`extra_check`](Self::extra_check) and
605/// [`convert_fallback`](Self::convert_fallback) below. A type with another
606/// representation, such as [`Array`](crate::Array), implements
607/// `AnyCompatible` instead.
608///
609/// # Safety
610///
611/// `Self` is exactly one [`ObjectArc<Self::ContainerType>`](ObjectArc). The
612/// hooks do not modify an `ffi.List` or `ffi.Dict` that holds the value they
613/// examine, whose cell a container read borrows while they run; otherwise
614/// they cannot break the [`AnyCompatible`] contract.
615///
616/// # Examples
617///
618/// ```
619/// use tvm_ffi::derive::ObjectRef;
620/// use tvm_ffi::{rs_object, Any, AnyView, Object, ObjectArc};
621/// use tvm_ffi::object::ObjectRefAnyCompatible;
622///
623/// #[rs_object("my_ext.MyPair", final)]
624/// pub struct MyPairObj {
625///     base_: Object,
626///     pub first: i64,
627///     pub second: i64,
628/// }
629///
630/// #[derive(Clone, ObjectRef)]
631/// pub struct MyPair {
632///     data: ObjectArc<MyPairObj>,
633/// }
634///
635/// /// A `MyPair` whose `first` is positive.
636/// #[derive(Clone, ObjectRef)]
637/// #[object_ref(skip_any_compatible)]
638/// pub struct PositivePair {
639///     data: ObjectArc<MyPairObj>,
640/// }
641///
642/// unsafe impl ObjectRefAnyCompatible for PositivePair {
643///     fn extra_check(pair: &MyPairObj) -> bool {
644///         pair.first > 0
645///     }
646///
647///     // A positive integer `n` converts to the pair `(n, n)`.
648///     fn convert_fallback(value: AnyView<'_>) -> Option<Self> {
649///         let n = i64::try_from(value).ok().filter(|n| *n > 0)?;
650///         Some(Self::from_object(MyPairObj::new(Object::new(), n, n)))
651///     }
652/// }
653///
654/// let negative = Any::from(MyPair::from_object(MyPairObj::new(Object::new(), -1, 2)));
655/// assert!(negative.downcast_ref::<PositivePair>().is_none());
656/// assert!(PositivePair::try_from(negative).is_err());
657/// assert_eq!(PositivePair::try_from(Any::from(3_i64))?.second, 3);
658/// # Ok::<(), tvm_ffi::Error>(())
659/// ```
660pub unsafe trait ObjectRefAnyCompatible: ObjectRefCore {
661    /// Whether no hook is overridden, so that `match_any!` may route a final
662    /// object type by its type index alone; only `#[derive(ObjectRef)]` sets
663    /// it. See [`AnyCompatible::MATCH_ANY_EXACT`].
664    #[doc(hidden)]
665    const MATCH_ANY_EXACT: bool = false;
666
667    /// Whether `obj`, an instance of `ContainerType`, is a value of `Self`.
668    /// Every strict and converting cast checks it, after the object type.
669    #[inline]
670    fn extra_check(obj: &Self::ContainerType) -> bool {
671        let _ = obj;
672        true
673    }
674
675    /// Convert a value that is not a `Self` object, such as an integer that
676    /// stands for a constant; `None` rejects it. Only converting casts (`TryFrom<Any>`,
677    /// typed function arguments) call it; `downcast_ref`, `downcast` and
678    /// `match_any!` never do.
679    #[inline]
680    fn convert_fallback(value: AnyView<'_>) -> Option<Self> {
681        let _ = value;
682        None
683    }
684
685    /// The type name in conversion errors and type schemas; by default the
686    /// object's type key. See [`AnyCompatible::type_str`].
687    fn custom_type_str() -> String {
688        Self::ContainerType::TYPE_KEY.into()
689    }
690
691    /// The name of the mismatched `value` in conversion errors; by default its
692    /// runtime type key. See [`AnyCompatible::get_mismatch_type_info`].
693    fn custom_mismatch_type_info(value: AnyView<'_>) -> String {
694        crate::type_traits::runtime_type_key(value.type_index())
695    }
696}

698/// An object reference that accepts every object of its type and promises
699/// nothing more, so any object type that derives from it may claim it as
700/// `#[object_ref(parent = ...)]`. `#[derive(ObjectRef)]` implements it for a
701/// non-generic reference with the default checks.
702///
703/// # Safety
704///
705/// Every object of `Self::ContainerType`, or of a type derived from it, is a
706/// valid `Self`.
707#[doc(hidden)]
708#[diagnostic::on_unimplemented(
709    message = "`{Self}` promises more than its object type (generic or custom checks)",
710    note = "write `#[object_ref(unsafe parent = ...)]` and keep that promise for every value"
711)]
712pub unsafe trait PlainObjectRef: ObjectRefCore {}
713
714// The object cell encoding, with the strict check `is_instance_of` then
715// `extra_check`, and `convert_fallback` after it in converting casts.
716unsafe impl<T: ObjectRefAnyCompatible> AnyCompatible for T {
717    // An overridden hook makes the type inexact; a non-final object type always is.
718    const MATCH_ANY_EXACT: bool = <T as ObjectRefAnyCompatible>::MATCH_ANY_EXACT
719        && <T::ContainerType as ObjectCore>::TYPE_FINAL;
720    const MATCH_ANY_STRICT_EXACT: bool = <T as AnyCompatible>::MATCH_ANY_EXACT;
721
722    #[inline]
723    fn match_any_exact_type_index() -> i32 {
724        T::ContainerType::type_index()
725    }
726
727    #[inline]
728    unsafe fn is_undefined(field: *const Self) -> bool {
729        // `Self` is one `ObjectArc`, so its null form is the undefined field.
730        ObjectArc::is_null(field.cast::<ObjectArc<T::ContainerType>>())
731    }
732
733    #[inline(always)]
734    unsafe fn copy_to_any_view(src: &Self, data: &mut TVMFFIAny) {
735        write_object_cell(ObjectArc::as_ptr(T::data(src)), data);
736    }
737
738    #[inline(always)]
739    unsafe fn move_to_any(src: Self, data: &mut TVMFFIAny) {
740        write_object_cell(ObjectArc::into_raw(T::into_data(src)), data);
741    }
742
743    #[inline(always)]
744    unsafe fn check_any_strict(data: &TVMFFIAny) -> bool {
745        // An instance of `ContainerType` points to one, so `extra_check` may read it.
746        is_instance_of::<T::ContainerType>(data.type_index)
747            && T::extra_check(&*data.data_union.v_obj.cast::<T::ContainerType>())
748    }
749
750    #[inline]
751    unsafe fn copy_from_any_view_after_check(data: &TVMFFIAny) -> Self {
752        // The view does not own the object, so the new reference takes one.
753        unsafe_::inc_ref(data.data_union.v_obj);
754        T::from_data(ObjectArc::from_raw(data.data_union.v_obj.cast()))
755    }
756
757    #[inline(always)]
758    unsafe fn move_from_any_after_check(data: &mut TVMFFIAny) -> Self {
759        T::from_data(ObjectArc::from_raw(data.data_union.v_obj.cast()))
760    }
761
762    #[inline]
763    unsafe fn try_cast_from_any_view(data: &TVMFFIAny) -> Result<Self, ()> {
764        if Self::check_any_strict(data) {
765            return Ok(Self::copy_from_any_view_after_check(data));
766        }
767        // A moved-from rvalue slot is consumed only by an exact object match.
768        if data.type_index == TypeIndex::kTVMFFIObjectRValueRef as i32 {
769            return Err(());
770        }
771        T::convert_fallback(AnyView::from_raw_ffi_any(*data)).ok_or(())
772    }
773
774    fn get_mismatch_type_info(data: &TVMFFIAny) -> String {
775        // SAFETY: the caller's cell is valid for the borrow.
776        T::custom_mismatch_type_info(unsafe { AnyView::from_raw_ffi_any(*data) })
777    }
778
779    fn type_str() -> String {
780        T::custom_type_str()
781    }
782}

784/// A value that converts into the object reference `T`, owned or borrowed.
785///
786/// An owned reference converts into an owned `T`, and a borrowed one borrows
787/// as `&T` without changing a reference count, as [`IntoIterator`] consumes a
788/// `Vec` but borrows from a `&Vec`. [`impl_object_ref_upcast!`](crate::impl_object_ref_upcast)
789/// implements it for each derived reference and its reference into the base,
790/// and `#[derive(ObjectRef)]` for the type itself. A function that only reads
791/// the value borrows the output with `.borrow()`; one that keeps it takes
792/// ownership with [`into_owned_object_ref`](Self::into_owned_object_ref), which clones only a
793/// borrowed reference.
794///
795/// # Examples
796///
797/// ```
798/// use std::borrow::Borrow;
799/// use tvm_ffi::derive::ObjectRef;
800/// use tvm_ffi::object::{IntoObjectRef, ObjectRef, ObjectRefCore};
801/// use tvm_ffi::{rs_object, Object, ObjectArc};
802///
803/// #[rs_object("my_ext.MyPair", final)]
804/// pub struct MyPairObj {
805///     base_: Object,
806///     pub first: i64,
807///     pub second: i64,
808/// }
809///
810/// #[derive(Clone, ObjectRef)]
811/// pub struct MyPair {
812///     data: ObjectArc<MyPairObj>,
813/// }
814///
815/// fn same(a: impl IntoObjectRef<ObjectRef>, b: &ObjectRef) -> bool {
816///     a.into_object_ref().borrow().same_as(b)
817/// }
818///
819/// let pair = MyPair::from_object(MyPairObj::new(Object::new(), 1, 2));
820/// let base: ObjectRef = pair.clone().into();
821/// assert!(same(&pair, &base)); // borrowed as `&ObjectRef`, no new reference
822/// assert!(same(pair.clone(), &base)); // moved into an `ObjectRef`
823///
824/// // A function that keeps the value takes ownership.
825/// fn keep(value: impl IntoObjectRef<ObjectRef>) -> ObjectRef {
826///     value.into_owned_object_ref()
827/// }
828/// assert!(keep(&pair).same_as(&base));
829/// ```
830pub trait IntoObjectRef<T> {
831    /// `T` for an owned value, or `&T` for a borrowed one.
832    type Output: std::borrow::Borrow<T> + Into<T>;
833
834    /// Convert into the reference `T`.
835    ///
836    /// # Returns
837    ///
838    /// The owned or borrowed reference.
839    fn into_object_ref(self) -> Self::Output;
840
841    /// Convert into an owned `T`, as [`Cow::into_owned`](std::borrow::Cow::into_owned)
842    /// does: an owned value moves, and only a borrowed reference is cloned.
843    ///
844    /// # Returns
845    ///
846    /// The owned reference.
847    #[inline(always)]
848    fn into_owned_object_ref(self) -> T
849    where
850        Self: Sized,
851    {
852        self.into_object_ref().into()
853    }
854}

908/// Whether the runtime type index of an object is `Target` or one of its
909/// subtypes, by walking the registered type hierarchy.
910///
911/// Use it in the strict check of a reference type implemented by hand; the
912/// casts of [`ObjectRefCore`] call it already. An index in the object range
913/// must be registered.
914#[inline(always)]
915pub fn is_instance_of<Target: ObjectCore>(object_type_index: i32) -> bool {
916    // Every registered type records its depth in the single inheritance tree
917    // and its ancestors, so the check is O(1): an ancestor `Target` appears in
918    // the candidate's ancestor array exactly at `Target`'s depth.
919    let target_type_index = Target::type_index();
920    if object_type_index == target_type_index {
921        return true;
922    }
923    // A final type cannot have a separately registered subtype. Keep common
924    // borrowed checks, such as `IntImmObj`, to one integer comparison.
925    if Target::TYPE_FINAL {
926        return false;
927    }
928    let object_begin = TypeIndex::kTVMFFIStaticObjectBegin as i32;
929    // Only object types participate in the type hierarchy.
930    if object_type_index < object_begin || target_type_index < object_begin {
931        return false;
932    }
933    // Only the root Object type has depth zero.
934    if const { Target::TYPE_DEPTH == 0 } {
935        return true;
936    }
937    // Parent indices are always smaller than their descendants.
938    if object_type_index < target_type_index {
939        return false;
940    }
941    unsafe {
942        let object_info = TVMFFIGetTypeInfo(object_type_index);
943        if object_info.is_null() {
944            return false;
945        }
946        let target_depth = Target::TYPE_DEPTH;
947        if (*object_info).type_depth <= target_depth {
948            return false;
949        }
950        let ancestor = *(*object_info).type_acenstors.add(target_depth as usize);
951        !ancestor.is_null() && (*ancestor).type_index == target_type_index
952    }
953}

955/// Format a managed reference as its runtime type key and object address, as
956/// the `Debug` implementation of `#[derive(ObjectRef)]` does.
957///
958/// Call it from `Debug` for a reference type implemented by hand, such as a
959/// typed view over another reference, so that `downcast(..).unwrap()` and
960/// `assert_eq!` can print it.
961///
962/// ```
963/// use tvm_ffi::object::{fmt_object_ref, ObjectRef};
964/// use tvm_ffi::Array;
965/// struct Shown(ObjectRef);
966/// impl std::fmt::Debug for Shown {
967///     fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
968///         fmt_object_ref(&self.0, f)
969///     }
970/// }
971/// let shown = Shown(Array::from_vec(vec![1_i64]).into());
972/// assert!(format!("{shown:?}").starts_with("ffi.Array@"));
973/// ```
974pub fn fmt_object_ref<T: ObjectRefCore>(
975    this: &T,
976    f: &mut std::fmt::Formatter<'_>,
977) -> std::fmt::Result {
978    let object = ObjectArc::as_ptr(T::data(this)).cast::<TVMFFIObject>();
979    // SAFETY: `this` keeps the object, and so its header, alive.
980    let type_index = unsafe { (*object).type_index };
981    write!(f, "{}@{:p}", type_key_of_index(type_index), object)
982}

1033/// Write the object cell of the live object `ptr`.
1034#[inline(always)]
1035unsafe fn write_object_cell<T: ObjectCore>(ptr: *const T, data: &mut TVMFFIAny) {
1036    let object = ptr as *mut TVMFFIObject;
1037    data.type_index = (*object).type_index;
1038    // SAFETY: a live object's header holds its registered object-range type index.
1039    std::hint::assert_unchecked(data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32);
1040    data.small_str_len = 0;

1044// Allocate and initialize the normal FFI header while preserving the concrete
1045// payload's alignment and drop/layout deleter. The payload starts with an Object.
1046unsafe fn allocate_object<T>(data: T, type_index: i32) -> NonNull<T> {
1047    let layout = std::alloc::Layout::new::<T>();
1048    let allocation = std::alloc::alloc(layout);
1049    if allocation.is_null() {
1050        std::alloc::handle_alloc_error(layout);
1051    }
1052    let ptr = allocation.cast::<T>();
1053    std::ptr::write(ptr, data);
1054    std::ptr::write(
1055        ptr.cast::<TVMFFIObject>(),
1056        TVMFFIObject {
1057            combined_ref_count: AtomicU64::new(COMBINED_REF_COUNT_BOTH_ONE),
1058            type_index,
1059            __padding: 0,
1060            deleter: Some(unsafe_::object_deleter_for_new::<T>),
1061        },
1062    );
1063    NonNull::new_unchecked(ptr)
1064}

1066/// Unsafe reference-count operations on raw object headers.
1067#[doc(hidden)]
1068pub mod unsafe_ {
1069    use tvm_ffi_sys::{
1070        COMBINED_REF_COUNT_BOTH_ONE, COMBINED_REF_COUNT_MASK_U32, COMBINED_REF_COUNT_STRONG_ONE,
1071        COMBINED_REF_COUNT_WEAK_ONE,
1072    };
1073
1074    use std::ffi::c_void;
1075    use std::sync::atomic::{fence, Ordering};
1076    use tvm_ffi_sys::TVMFFIObject;
1077    use tvm_ffi_sys::TVMFFIObjectDeleterFlagBitMask::{
1078        kTVMFFIObjectDeleterFlagBitMaskBoth, kTVMFFIObjectDeleterFlagBitMaskStrong,
1079        kTVMFFIObjectDeleterFlagBitMaskWeak,
1080    };
1081
1082    /// Increase the strong reference count of `handle`.
1083    #[inline]
1084    pub unsafe fn inc_ref(handle: *mut TVMFFIObject) {
1085        let obj = &*handle;
1086        obj.combined_ref_count.fetch_add(1, Ordering::Relaxed);
1087    }
1088
1089    /// Decrease the strong reference count of `handle`, deleting the object when it was the last.
1090    #[inline]
1091    pub(crate) unsafe fn dec_ref(handle: *mut TVMFFIObject) {
1092        // Release publishes this owner's writes; the Acquire fence before the deleter pairs with it.
1093        let old_combined_count = (*handle)
1094            .combined_ref_count
1095            .fetch_sub(COMBINED_REF_COUNT_STRONG_ONE, Ordering::Release);
1096        if old_combined_count == COMBINED_REF_COUNT_BOTH_ONE {
1097            fence(Ordering::Acquire);
1098            if let Some(deleter) = (*handle).deleter {
1099                deleter(
1100                    handle.cast::<c_void>(),
1101                    kTVMFFIObjectDeleterFlagBitMaskBoth as i32,
1102                );
1103            }
1104        } else if (old_combined_count & COMBINED_REF_COUNT_MASK_U32)
1105            == COMBINED_REF_COUNT_STRONG_ONE
1106        {
1107            // slow path, there is still a weak reference left
1108            // need to run two phase decrement
1109            fence(Ordering::Acquire);
1110            if let Some(deleter) = (*handle).deleter {
1111                deleter(
1112                    handle.cast::<c_void>(),
1113                    kTVMFFIObjectDeleterFlagBitMaskStrong as i32,
1114                );
1115            }
1116            let old_weak_count = (*handle)
1117                .combined_ref_count
1118                .fetch_sub(COMBINED_REF_COUNT_WEAK_ONE, Ordering::Release);
1119            if old_weak_count == COMBINED_REF_COUNT_WEAK_ONE {
1120                fence(Ordering::Acquire);
1121                if let Some(deleter) = (*handle).deleter {
1122                    deleter(
1123                        handle.cast::<c_void>(),
1124                        kTVMFFIObjectDeleterFlagBitMaskWeak as i32,
1125                    );
1126                }
1127            }
1128        }
1129    }
1130
1131    #[inline]
1132    pub(crate) unsafe fn strong_count(handle: *mut TVMFFIObject) -> usize {
1133        let obj = &*handle;
1134        (obj.combined_ref_count.load(Ordering::Relaxed) & COMBINED_REF_COUNT_MASK_U32) as usize
1135    }
1136
1137    /// Whether `handle` is the only strong reference, so it may be mutated in
1138    /// place. Any weak reference rejects it, as a weak handle could upgrade and
1139    /// observe the write.
1140    #[inline]
1141    pub(crate) unsafe fn is_unique(handle: *mut TVMFFIObject) -> bool {
1142        // Acquire pairs with the releasing decrement: other owners' accesses precede an in-place write.
1143        (*handle).combined_ref_count.load(Ordering::Acquire) == COMBINED_REF_COUNT_BOTH_ONE
1144    }
1145
1146    #[inline]
1147    pub(crate) unsafe fn weak_count(handle: *mut TVMFFIObject) -> usize {
1148        let obj = &*handle;
1149        (obj.combined_ref_count.load(Ordering::Relaxed) >> 32) as usize
1150    }
1151
1152    /// Generic object deleter for objects allocated through Rust's global allocator.
1153    pub(crate) unsafe extern "C" fn object_deleter_for_new<T>(ptr: *mut c_void, flags: i32) {
1154        let obj = ptr as *mut T;
1155        if flags & kTVMFFIObjectDeleterFlagBitMaskStrong as i32 != 0 {
1156            std::ptr::drop_in_place(obj);
1157        }
1158        if flags & kTVMFFIObjectDeleterFlagBitMaskWeak as i32 != 0 {
1159            std::alloc::dealloc(ptr as *mut u8, std::alloc::Layout::new::<T>());
1160        }
1161    }
1162
1163    /// Layout of a `T` followed by `count` extra items, kept at least one word past the
1164    /// header: once a weak reference outlives the data, the deleter stores `count` there.
1165    pub(crate) fn extra_items_layout<T, U>(count: usize) -> std::alloc::Layout {
1166        let size = std::mem::size_of::<T>() + count * std::mem::size_of::<U>();
1167        let min_size = std::mem::size_of::<TVMFFIObject>() + std::mem::size_of::<u64>();
1168        std::alloc::Layout::from_size_align(size.max(min_size), std::mem::align_of::<T>()).unwrap()
1169    }
1170
1171    pub(crate) unsafe extern "C" fn object_deleter_for_new_with_extra_items<T, U>(
1172        ptr: *mut c_void,
1173        flags: i32,

1193            }
1194            if flags & kTVMFFIObjectDeleterFlagBitMaskWeak as i32 != 0 {
1195                let extra_items_count = std::ptr::read(count_slot) as usize;
1196                std::alloc::dealloc(
1197                    ptr as *mut u8,
1198                    extra_items_layout::<T, U>(extra_items_count),
