25/// A borrowed, type-erased FFI value.
26///
27/// `AnyView` holds any FFI value, such as an integer, a string or an object,
28/// without owning it: it does not retain objects, and its lifetime `'a` keeps
29/// it from outliving the value it borrows. It is `Copy` and is the argument
30/// type of FFI calls. Convert it into an [`Any`] to own the value.
31///
32/// # Examples
33///
34/// ```
35/// use tvm_ffi::{Any, AnyView, Array};
36///
37/// let array = Array::from_vec(vec![1_i64, 2, 3]);
38/// let view = AnyView::from(&array);
39/// assert_eq!(view.downcast_ref::<Array<i64>>().unwrap().len(), 3);
40///
41/// let owned = Any::from(view);
42/// assert_eq!(Array::<i64>::try_from(owned)?.len(), 3);
43/// # Ok::<(), tvm_ffi::Error>(())
44/// ```
45#[derive(Copy, Clone)]
46#[repr(C)]
47pub struct AnyView<'a> {
48    data: TVMFFIAny,
49    /// needs to explicit mark lifetime to avoid lifetime mismatch
50    _phantom: std::marker::PhantomData<&'a ()>,
51}

53/// An owned, type-erased FFI value.
54///
55/// `Any` holds any FFI value, such as an integer, a string or an object, and
56/// owns it: cloning an `Any` that holds an object retains the object, and
57/// dropping it releases the object. Any [`AnyCompatible`] value converts into
58/// it with `From`; `TryFrom` (converting), [`downcast_ref`](Self::downcast_ref)
59/// (borrowing) and [`match_any!`](crate::match_any) read it back. Borrow it as an [`AnyView`] to pass it without retaining.
60///
61/// # Examples
62///
63/// ```
64/// use tvm_ffi::Any;
65///
66/// let value = Any::from(42_i64);
67/// assert_eq!(i64::try_from(&value)?, 42);
68/// assert!(tvm_ffi::String::try_from(&value).is_err());
69///
70/// let text = Any::from(tvm_ffi::String::from("hello"));
71/// assert_eq!(text.downcast_ref::<tvm_ffi::String>().unwrap().as_str(), "hello");
72/// # Ok::<(), tvm_ffi::Error>(())
73/// ```
74#[repr(C)]
75pub struct Any {
76    data: TVMFFIAny,
77}

82impl<'a> AnyView<'a> {
83    /// Create a view of `None`, the empty FFI value.
84    ///
85    /// # Returns
86    ///
87    /// A view holding `None`.
88    pub fn new() -> Self {
89        Self {
90            data: TVMFFIAny::new(),
91            _phantom: std::marker::PhantomData,
92        }
93    }
94
95    /// Borrow a live managed object without acquiring ownership.
96    #[doc(hidden)]
97    #[inline(always)]
98    pub fn from_object<T: object::ObjectCore>(value: &'a T) -> Self {
99        let pointer = std::ptr::from_ref(value).cast::<tvm_ffi_sys::TVMFFIObject>();
100        // ObjectCore guarantees the prefix and stable managed allocation. A
101        // builder or a moved-out destructor value is not a live managed object.
102        let header = unsafe { &*pointer };
103        assert!(
104            header.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32
105                && header.deleter.is_some()
106                && header
107                    .combined_ref_count
108                    .load(std::sync::atomic::Ordering::Relaxed)
109                    & tvm_ffi_sys::COMBINED_REF_COUNT_MASK_U32
110                    != 0,
111            "cannot borrow an unmanaged or dead object as AnyView"
112        );
113        let mut data = TVMFFIAny::new();
114        data.type_index = header.type_index;
115        data.data_union.v_obj = pointer.cast_mut();
116        // The source borrow bounds the view; the dynamic type comes from the
117        // allocation even when value borrows a base-class prefix.
118        unsafe { Self::from_raw_ffi_any(data) }
119    }
120
121    /// Return the runtime type index of the viewed value.
122    ///
123    /// Indices below `TypeIndex::kTVMFFIStaticObjectBegin` are the built-in
124    /// non-object types; the others are object types.
125    ///
126    /// # Returns
127    ///
128    /// The type index.
129    #[inline(always)]
130    pub fn type_index(&self) -> i32 {
131        self.data.type_index
132    }
133
134    /// Borrow the raw ABI representation.
135    /// Copying it does not retain referenced data or extend its lifetime.
136    #[inline(always)]
137    pub fn as_raw_ffi_any(&self) -> &TVMFFIAny {
138        &self.data
139    }
140
141    /// Construct a borrowed view from its ABI representation.
142    ///
143    /// # Safety
144    ///
145    /// `data.type_index` must describe the payload, and the caller must keep
146    /// every resource it references alive for the view's complete lifetime.
147    ///
148    /// `kTVMFFIObjectRValueRef` carries a further obligation: `v_ptr` must
149    /// point to a writable slot the caller uniquely owns, holding one strong
150    /// reference. Owning the view takes that reference and writes null back
151    /// through the pointer, so the view must be converted at most once and no
152    /// other access to the slot may overlap the conversion.
153    #[inline(always)]
154    pub unsafe fn from_raw_ffi_any(data: TVMFFIAny) -> Self {
155        Self {
156            data,
157            _phantom: std::marker::PhantomData,
158        }
159    }
160
161    /// Borrow the viewed value as `T` when it holds one, without copying it.
162    ///
163    /// `T` is an object type such as `ArrayObj`, a managed reference such as
164    /// `Array<i64>`, or a value stored in place such as `String` or `i64`. The
165    /// check is strict and never converts, unlike `TryFrom`.
166    /// The borrow does not change any reference count and is tied to this view,
167    /// which holds its own copy of the value's cell.
168    ///
169    /// # Returns
170    ///
171    /// The borrowed value, or `None`.
172    ///
173    /// See also [`Any::downcast_ref`].
174    #[inline(always)]
175    pub fn downcast_ref<T: crate::type_traits::DowncastRefFromAny>(&self) -> Option<&T> {
176        // SAFETY: the view holds a valid cell for as long as it is borrowed.
177        unsafe { T::downcast_ref_from_any(&self.data) }
178    }
179
180    /// Get the strong count of the underlying object for testing/debugging purposes
181    ///
182    /// If the underlying object is not ref counted, return None
183    pub fn debug_strong_count(&self) -> Option<usize> {
184        unsafe {
185            if self.data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32 {
186                Some(object::unsafe_::strong_count(self.data.data_union.v_obj))
187            } else {
188                None
189            }
190        }
191    }
192}

254impl Any {
255    /// Create an `Any` holding `None`, the empty FFI value.
256    ///
257    /// # Returns
258    ///
259    /// An `Any` holding `None`.
260    #[inline(always)]
261    pub fn new() -> Self {
262        Self {
263            data: TVMFFIAny::new(),
264        }
265    }
266
267    /// Return the runtime type index of the stored value.
268    ///
269    /// Indices below `TypeIndex::kTVMFFIStaticObjectBegin` are the built-in
270    /// non-object types; the others are object types.
271    ///
272    /// # Returns
273    ///
274    /// The type index.
275    #[inline(always)]
276    pub fn type_index(&self) -> i32 {
277        self.data.type_index
278    }
279
280    /// Borrow the raw ABI representation, which this value owns.
281    /// Copying it does not retain referenced data or extend its lifetime.
282    #[inline(always)]
283    pub fn as_raw_ffi_any(&self) -> &TVMFFIAny {
284        &self.data
285    }
286    /// Borrow the stored value as `T` when it holds one, without copying it.
287    ///
288    /// `T` is an object type such as `ArrayObj`, a managed reference such as
289    /// `Array<i64>`, or a value stored in place such as `String` or `i64`. The
290    /// check is strict and never converts, unlike `TryFrom`.
291    /// This neither consumes `self` nor changes any reference count, and the
292    /// returned reference cannot outlive `self`.
293    ///
294    /// # Returns
295    ///
296    /// The borrowed object, or `None`.
297    ///
298    /// ```
299    /// use tvm_ffi::{collections::array::ArrayObj, Any, Array};
300    /// let value = Any::from(Array::from_vec(vec![1_i64, 2]));
301    /// assert_eq!(value.downcast_ref::<ArrayObj>().unwrap().size, 2);
302    /// assert_eq!(value.downcast_ref::<Array<i64>>().unwrap()[1], 2);
303    /// assert!(value.downcast_ref::<i64>().is_none());
304    /// ```
305    ///
306    /// See also [`match_any!`](crate::match_any).
307    #[inline]
308    pub fn downcast_ref<T: crate::type_traits::DowncastRefFromAny>(&self) -> Option<&T> {
309        // SAFETY: the owned cell stays valid for as long as `self` is borrowed.
310        unsafe { T::downcast_ref_from_any(&self.data) }
311    }
312
313    /// Return a pointer to the raw ABI representation, for a runtime call that
314    /// writes an owned result into it, as C++ `Any` out-parameters are filled.
315    ///
316    /// Writing through the pointer is unsafe: the cell must afterwards hold a
317    /// value this `Any` may own and release, and the previous value must have
318    /// been released or moved out.
319    #[inline(always)]
320    pub fn as_raw_ffi_any_mut(&mut self) -> *mut TVMFFIAny {
321        &mut self.data
322    }
323
324    /// Give up ownership of the value as its raw ABI representation, as
325    /// `Box::into_raw` does. Rebuild it with [`from_raw_ffi_any`](Self::from_raw_ffi_any)
326    /// to release it.
327    #[inline(always)]
328    pub fn into_raw_ffi_any(this: Self) -> TVMFFIAny {
329        let this = std::mem::ManuallyDrop::new(this);
330        this.data
331    }
332
333    /// Take ownership of a raw ABI representation.
334    ///
335    /// # Safety
336    ///
337    /// `data.type_index` must describe the payload, and the caller must hand
338    /// over the ownership `data` carries, such as one strong reference for an
339    /// object, so it is released exactly once, by this `Any`.
340    #[inline(always)]
341    pub unsafe fn from_raw_ffi_any(data: TVMFFIAny) -> Self {
342        Self { data }
343    }
344
345    /// Get the strong count of the underlying object for testing/debugging purposes
346    ///
347    /// If the underlying object is not ref counted, return None
348    pub fn debug_strong_count(&self) -> Option<usize> {
349        unsafe {
350            if self.data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32 {
351                Some(object::unsafe_::strong_count(self.data.data_union.v_obj))
352            } else {
353                None
354            }
355        }
356    }
357}

369        if self.data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32 {

396/// Whether a `TVMFFIAny` cell owns everything it holds, so its owning form is a
397/// bitwise copy. Mirrors C++ `detail::InplaceConvertAnyViewToAny`.
398#[inline]
399pub(crate) fn is_plain_inline(type_index: i32) -> bool {
400    type_index < TypeIndex::kTVMFFIRawStr as i32
401        || type_index == TypeIndex::kTVMFFISmallStr as i32
402        || type_index == TypeIndex::kTVMFFISmallBytes as i32
403        || type_index == TypeIndex::kTVMFFIUnchanged as i32
404}
405
406// convert AnyView to Any
407impl From<AnyView<'_>> for Any {
408    #[inline]
409    fn from(value: AnyView<'_>) -> Self {
410        let data = value.data;
411        // Owning a borrowed object is the same incref `Any::clone` does below.
412        if data.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32 {
413            unsafe { object::unsafe_::inc_ref(data.data_union.v_obj) };
414            return Self { data };
415        }
416        if is_plain_inline(data.type_index) {
417            return Self { data };
418        }
419        // What is left borrows foreign storage and needs the runtime.
420        any_view_to_owned_via_runtime(data)
421    }
422}
423
424/// Out of line so the inlined conversions above stay leaf code, without the
425/// stack frame and unwind path this call needs.
426#[cold]
427#[inline(never)]
428fn any_view_to_owned_via_runtime(view: TVMFFIAny) -> Any {
429    unsafe {
430        let mut data = TVMFFIAny::new();
431        crate::check_safe_call!(TVMFFIAnyViewToOwnedAny(&view, &mut data)).unwrap();
432        Any { data }
433    }
434}

447impl<'a, T: AnyCompatible> TryFrom<AnyView<'a>> for TryFromTemp<T> {
448    type Error = crate::error::Error;
449    #[inline]
450    fn try_from(value: AnyView<'a>) -> Result<Self, Self::Error> {
451        unsafe {
452            if T::check_any_strict(&value.data) {
453                Ok(TryFromTemp::new(T::copy_from_any_view_after_check(
454                    &value.data,
455                )))
456            } else {
457                T::try_cast_from_any_view(&value.data)
458                    .map_err(|_| {
459                        let msg = format!(
460                            "Cannot convert from type `{}` to `{}`",
461                            T::get_mismatch_type_info(&value.data),
462                            T::type_str()
463                        );
464                        crate::error::Error::new(crate::error::TYPE_ERROR, &msg, "")
465                    })
466                    .map(TryFromTemp::new)
467            }
468        }
469    }
470}

533/// ArgTryFromAnyView: Helper for function argument passing

535pub(crate) trait ArgTryFromAnyView: Sized + crate::type_traits::TypeSchema {
536    fn try_from_any_view(value: &AnyView, arg_index: usize) -> Result<Self, Error>;
537}
538
539impl<T: AnyCompatible> ArgTryFromAnyView for T {
540    fn try_from_any_view(value: &AnyView, arg_index: usize) -> Result<Self, Error> {
541        unsafe {
542            if T::check_any_strict(&value.data) {
543                Ok(T::copy_from_any_view_after_check(&value.data))
544            } else {
545                T::try_cast_from_any_view(&value.data).map_err(|_| {
546                    let msg = format!(
547                        "Argument #{}: Cannot convert from type `{}` to `{}`",
548                        arg_index,
549                        T::get_mismatch_type_info(&value.data),
550                        T::type_str()
551                    );
552                    crate::error::Error::new(crate::error::TYPE_ERROR, &msg, "")
553                })
554            }
555        }
556    }
