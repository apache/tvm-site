19//! [`Map`], an immutable hash map whose lookups run in the FFI runtime.
20//!
21//! Converting an `Any` into a `Map<K, V>` checks every entry; when an entry
22//! is not a `K`/`V` as is, the entries are converted one by one into a new map.
23
24// The `MapObj` header mirrors C++ `MapBaseObj`, so `Map::len` reads `size`
25// directly (no FFI). The hash-table storage itself is an implementation detail,
26// so lookups and iteration are delegated to the global functions the C++ runtime
27// registers (`ffi.MapGetItemOrMissing`, `ffi.MapForwardIterFunctor`, ...): there is no
28// Map-specific C ABI, and a Rust re-implementation would have to replicate the
29// hashing (`AnyHash`/`AnyEqual`) and dense/small probing, just as the Python
30// bindings also delegate.
31

47#[inline]
48pub(crate) fn element_view<T: ContainerElement>(value: &T) -> AnyView<'_> {
49    unsafe {
50        let mut data = TVMFFIAny::new();
51        T::container_copy_to_any_view(value, &mut data);
52        AnyView::from_raw_ffi_any(data)
53    }
54}

56/// The address of the runtime's missing-value sentinel, which
57/// `ffi.MapGetItemOrMissing` returns for an absent key. The sentinel lives for
58/// the whole process, so its address identifies it.
59static MISSING_OBJECT: ColdInitLazyLock<usize> = ColdInitLazyLock::new(|| {
60    let missing = Function::get_global("ffi.GetInvalidObject")
61        .and_then(|f| f.call_packed(&[]))
62        .expect("ffi.GetInvalidObject failed");
63    unsafe { missing.as_raw_ffi_any().data_union.v_obj as usize }
64});
65
66fn element_from_any<T: ContainerElement>(value: Any) -> Result<T> {
67    unsafe {
68        if T::container_check_any_strict(value.as_raw_ffi_any()) {
69            let mut value = std::mem::ManuallyDrop::new(value);
70            return Ok(T::container_move_from_any_after_check(
71                &mut *value.as_raw_ffi_any_mut(),
72            ));
73        }
74        T::container_try_cast_from_any_view(value.as_raw_ffi_any()).map_err(|()| {
75            let message = format!(
76                "Cannot convert from type `{}` to `{}`",
77                T::container_get_mismatch_type_info(value.as_raw_ffi_any()),
78                T::container_type_str()
79            );
80            Error::new(crate::error::TYPE_ERROR, &message, "")
81        })
82    }
83}

85/// The object behind a [`Map`]. The header fields are laid out as the C++
86/// `MapBaseObj`, so [`Map::len`] reads `size` without an FFI call; the storage
87/// `data` points to stays opaque.
88//
89// This is a partial mirror by design: `MapBaseObj` has one more trailing field
90// after `slots_` (a `data_deleter_` function pointer), omitted here because this
91// binding is read-only — it only reads fields up to `size`, which precede it,
92// and never allocates a `MapObj` itself (the C++ runtime allocates the larger
93// `Small`/`DenseMapBaseObj` subclass), so the trailing field is never touched.
94// The layout also assumes `TVM_FFI_DEBUG_WITH_ABI_CHANGE` is off (the default):
95// with that debug flag set, `MapBaseObj` gains a *leading* `state_marker` field
96// that would shift every offset below.
97#[repr(C)]
98#[derive(Object)]
99#[type_key = "ffi.Map"]
100#[type_index(unsafe TypeIndex::kTVMFFIMap)]
101pub struct MapObj {
102    pub(crate) object: Object,
103    /// Pointer to the (opaque) key/value storage region (`MapBaseObj::data_`).
104    pub data: *mut core::ffi::c_void,
105    /// Number of entries (`MapBaseObj::size_`).
106    pub size: u64,
107    /// Number of hash slots; the MSB is a small-map tag (`MapBaseObj::slots_`).
108    pub slots: u64,
109}

111/// An immutable, reference-counted map from `K` to `V`, passed across
112/// languages as `ffi.Map`.
113///
114/// Cloning a `Map` is cheap: it shares the same entries.
115///
116/// # Examples
117///
118/// ```
119/// use tvm_ffi::{Map, String};
120///
121/// let map: Map<String, i64> = [(String::from("a"), 1), (String::from("b"), 2)]
122///     .into_iter()
123///     .collect();
124/// assert_eq!(map.len(), 2);
125/// assert_eq!(map.get(&String::from("b"))?, Some(2));
126/// assert_eq!(map.get(&String::from("c"))?, None);
127/// # Ok::<(), tvm_ffi::Error>(())
128/// ```
129#[repr(C)]
130pub struct Map<K, V> {
131    data: ObjectArc<MapObj>,
132    _marker: PhantomData<(K, V)>,
133}
134
135// SAFETY: a `Map` is immutable, so sharing it is as safe as sharing its entries.
136unsafe impl<K: Send + Sync, V: Send + Sync> Send for Map<K, V> {}
137unsafe impl<K: Send + Sync, V: Send + Sync> Sync for Map<K, V> {}

214impl<K, V> Map<K, V>
215where
216    K: ContainerElement,
217    V: ContainerElement,
218{
219    /// Creates an empty map.
220    pub fn new() -> Self {
221        Self::from_pairs(&[]).expect("ffi.Map() failed to construct an empty map")
222    }
223
224    /// Builds a map by calling the runtime's `ffi.Map` constructor with a flattened
225    /// `[k0, v0, k1, v1, ...]` argument list.
226    fn from_pairs(pairs: &[(K, V)]) -> Result<Self> {
227        let mut args: Vec<AnyView<'_>> = Vec::with_capacity(pairs.len() * 2);
228        for (k, v) in pairs {
229            args.push(element_view(k));
230            args.push(element_view(v));
231        }
232        let result = crate::cached_global_func!("ffi.Map").call_packed(&args)?;
233        Self::try_from(result)
234    }
235
236    /// Returns the number of entries.
237    pub fn len(&self) -> usize {
238        self.size as usize
239    }
240
241    /// Returns `true` if the map contains no entries.
242    pub fn is_empty(&self) -> bool {
243        self.len() == 0
244    }
245
246    /// Returns whether `key` is present, propagating any FFI error (e.g. a key
247    /// the runtime cannot hash) rather than panicking.
248    fn try_contains_key(&self, key: &K) -> Result<bool> {
249        let result = crate::cached_global_func!("ffi.MapCount")
250            .call_packed(&[AnyView::from(self), element_view(key)])?;
251        Ok(i64::try_from(result)? != 0)
252    }
253
254    /// In debug builds, panics if `K` does not match the map's actual key type,
255    /// checked against one stored key. Compiles to nothing in release, so
256    /// [`Map::get`] and [`Map::contains_key`] stay one FFI call there.
257    ///
258    /// A pure key lookup *hashes* the key rather than retrieving one, so a wrong
259    /// `K` simply fails to match and reads as "absent" — unlike [`Map::iter`],
260    /// which retrieves keys and surfaces the mismatch. This assertion catches
261    /// that misuse in dev/tests without taxing release lookups. Only meaningful
262    /// on a miss (a hit already proves `K` matched a stored key).
263    #[inline]
264    fn debug_assert_key_type(&self) {
265        #[cfg(debug_assertions)]
266        {
267            if !self.is_empty() {
268                let functor = self.iter_functor();
269                let first_key = functor
270                    .call_packed(&[AnyView::from(&0i64)])
271                    .expect("map iterator: reading current key failed");
272                assert!(
273                    unsafe { K::container_check_any_strict(first_key.as_raw_ffi_any()) },
274                    "Map lookup: key type `{}` does not match the map's stored key type",
275                    std::any::type_name::<K>(),
276                );
277            }
278        }
279    }
280
281    /// Returns whether `key` is present.
282    ///
283    /// # Errors
284    ///
285    /// Returns an error if the runtime cannot look `key` up, such as when it
286    /// is not hashable.
287    ///
288    /// The result for a `key` of another type than the map's keys is
289    /// unspecified, as for [`Map::get`].
290    pub fn contains_key(&self, key: &K) -> Result<bool> {
291        let present = self.try_contains_key(key)?;
292        if !present {
293            self.debug_assert_key_type();
294        }
295        Ok(present)
296    }
297
298    /// Looks up `key`, returning `Ok(None)` if it is absent.
299    ///
300    /// # Errors
301    ///
302    /// Returns an error if the runtime cannot look `key` up, such as when it
303    /// is not hashable, or if the value is not a `V`.
304    ///
305    /// The result for a `key` of another type than the map's keys is
306    /// unspecified: it currently reads as absent, since a lookup hashes the key
307    /// instead of reading a stored one. Debug builds assert against it.
308    pub fn get(&self, key: &K) -> Result<Option<V>> {
309        let result = crate::cached_global_func!("ffi.MapGetItemOrMissing")
310            .call_packed(&[AnyView::from(self), element_view(key)])?;
311        let raw = result.as_raw_ffi_any();
312        if raw.type_index >= TypeIndex::kTVMFFIStaticObjectBegin as i32
313            && unsafe { raw.data_union.v_obj } as usize == *MISSING_OBJECT
314        {
315            self.debug_assert_key_type();
316            return Ok(None);
317        }
318        let value = element_from_any(result)?;
319        Ok(Some(value))
320    }
321
322    /// Returns an iterator over the `(key, value)` pairs of the map.
323    pub fn iter(&self) -> Iter<K, V> {
324        Iter(self.make_iter(|f| (iter_read::<K>(f, 0, "key"), iter_read::<V>(f, 1, "value"))))
325    }
326
327    /// Returns an iterator over the keys of the map.
328    pub fn keys(&self) -> Keys<K> {
329        Keys(self.make_iter(|f| iter_read::<K>(f, 0, "key")))
330    }
331
332    /// Returns an iterator over the values of the map.
333    pub fn values(&self) -> Values<V> {
334        Values(self.make_iter(|f| iter_read::<V>(f, 1, "value")))
335    }
336
337    /// Builds a `MapIter` whose `read` extracts each entry as `T`. The
338    /// forward-iteration functor is requested only for a non-empty map, so
339    /// iterating an empty map makes no FFI call and allocates no functor.
340    fn make_iter<T>(&self, read: fn(&Function) -> T) -> MapIter<T> {
341        let remaining = self.len();
342        MapIter {
343            functor: (remaining != 0).then(|| self.iter_functor()),
344            remaining,
345            _keepalive: self.data.clone(),
346            read,
347        }
348    }
349
350    /// Obtains a fresh stateful forward-iteration functor from the runtime.
351    fn iter_functor(&self) -> Function {
352        let result = crate::cached_global_func!("ffi.MapForwardIterFunctor")
353            .call_packed(&[AnyView::from(self)])
354            .expect("ffi.MapForwardIterFunctor call failed");
355        Function::try_from(result).expect("ffi.MapForwardIterFunctor returned a non-function")
356    }
357
358    /// Reads all entries as raw `(Any, Any)` pairs without converting to
359    /// `K`/`V`, propagating FFI failures instead of panicking (unlike the
360    /// iterator API): this backs strict checks and the converting fallback in
361    /// `try_cast_from_any_view`, which run during argument decoding where a
362    /// panic could unwind across the C ABI.
363    fn try_raw_entries(&self) -> Result<Vec<(Any, Any)>> {
364        forward_entries(
365            crate::cached_global_func!("ffi.MapForwardIterFunctor"),
366            AnyView::from(self),
367            self.len(),
368        )
369    }
370
371    /// Converts raw entries one by one into a new map.
372    fn from_raw_entries(entries: Vec<(Any, Any)>) -> Result<Self> {
373        let mut pairs = Vec::with_capacity(entries.len());
374        for (k, v) in entries {
375            pairs.push((element_from_any::<K>(k)?, element_from_any::<V>(v)?));
376        }
377        Self::from_pairs(&pairs)
378    }
379}
380
381/// The `len` entries of the map or dict `container`, read with the iterator
382/// functor that `iter_functor` makes for it.
383fn forward_entries(
384    iter_functor: &Function,
385    container: AnyView,
386    len: usize,
387) -> Result<Vec<(Any, Any)>> {
388    let mut entries = Vec::with_capacity(len);
389    let mut remaining = len;
390    if remaining == 0 {
391        return Ok(entries);
392    }
393    let functor = iter_functor
394        .call_packed(&[container])
395        .and_then(Function::try_from)?;
396    loop {
397        let k = functor.call_packed(&[AnyView::from(&0i64)])?;
398        let v = functor.call_packed(&[AnyView::from(&1i64)])?;
399        entries.push((k, v));
400        remaining -= 1;
401        if remaining == 0 {
402            return Ok(entries);
403        }
404        functor.call_packed(&[AnyView::from(&2i64)])?;
405    }
406}

511// --- Iterators ---
512//
513// The functor from `ffi.MapForwardIterFunctor` does not keep the map alive, so
514// each iterator carries an `ObjectArc<MapObj>` keepalive. Functor command codes:
515// `0` = read key, `1` = read value, `2` = advance.
516//
517// These are `ExactSizeIterator`s, so a key/value not matching the declared
518// `K`/`V` panics rather than ending iteration early (which would drop entries).
519// Use [`Map::get`], which returns `Err` on a type mismatch, when the element
520// types are uncertain.
521
522/// Reads the functor's current key (`command` 0) or value (`command` 1) as `T`,
523/// panicking on a type mismatch (see the note above on `ExactSizeIterator`).
524fn iter_read<T: ContainerElement>(functor: &Function, command: i64, kind: &str) -> T {
525    let any = functor
526        .call_packed(&[AnyView::from(&command)])
527        .expect("map iterator: reading current element failed");
528    element_from_any(any)
529        .unwrap_or_else(|_| panic!("map iterator: {kind} does not match the map's {kind} type"))
530}
531
532/// Consumes one entry: decrements `remaining` and advances the functor unless
533/// the map is now exhausted. Callers must guard `remaining > 0` (every `next`
534/// returns early when it hits 0), so the decrement can never underflow.
535fn iter_advance(functor: &Function, remaining: &mut usize) {
536    debug_assert!(
537        *remaining > 0,
538        "iter_advance called with no remaining entries"
539    );
540    *remaining -= 1;
541    if *remaining > 0 {
542        functor
543            .call_packed(&[AnyView::from(&2i64)])
544            .expect("map iterator: advancing failed");
545    }
546}

548/// Forward iterator over a [`Map`], yielding `T` per entry.
549struct MapIter<T> {
550    /// `None` for an empty map (no functor is requested); `Some` while at least
551    /// one entry remains to be yielded.
552    functor: Option<Function>,
553    remaining: usize,
554    _keepalive: ObjectArc<MapObj>,
555    /// Reads the current entry, only the key or value the iterator yields.
556    read: fn(&Function) -> T,
557}
558
559impl<T> Iterator for MapIter<T> {
560    type Item = T;
561
562    fn next(&mut self) -> Option<T> {
563        if self.remaining == 0 {
564            return None;
565        }
566        // `remaining > 0` implies `functor` is `Some` (see `Map::make_iter`).
567        let functor = self
568            .functor
569            .as_ref()
570            .expect("non-empty map iterator has a functor");
571        let item = (self.read)(functor);
572        iter_advance(functor, &mut self.remaining);
573        Some(item)
574    }
575
576    fn size_hint(&self) -> (usize, Option<usize>) {
577        (self.remaining, Some(self.remaining))
578    }
579}

583macro_rules! map_iterator {
584    ($(#[$doc:meta])* $name:ident<$($param:ident),+>, $item:ty) => {
585        $(#[$doc])*
586        pub struct $name<$($param),+>(MapIter<$item>);
587
588        impl<$($param),+> Iterator for $name<$($param),+> {
589            type Item = $item;
590
591            #[inline]
592            fn next(&mut self) -> Option<$item> {
593                self.0.next()
594            }
595
596            #[inline]
597            fn size_hint(&self) -> (usize, Option<usize>) {
598                self.0.size_hint()
599            }

602        impl<$($param),+> ExactSizeIterator for $name<$($param),+> {}
603    };

633
634unsafe impl<K, V> AnyCompatible for Map<K, V>
635where
636    K: ContainerElement,
637    V: ContainerElement,
638{
639    #[inline]
640    unsafe fn is_undefined(field: *const Self) -> bool {
641        ObjectArc::is_null(std::ptr::addr_of!((*field).data))
642    }
643
644    fn type_str() -> String {
645        format!(
646            "Map<{}, {}>",
647            K::container_type_str(),
648            V::container_type_str()
649        )
650    }
651
652    fn type_schema() -> String {
653        crate::type_traits::type_schema(
654            <MapObj as crate::ObjectCore>::TYPE_KEY,
655            &[K::container_type_schema(), V::container_type_schema()],
656        )
657    }
658
659    unsafe fn check_any_strict(data: &TVMFFIAny) -> bool {
660        // Mirrors C++ `CheckAnyStrict` (map_base.h): every entry must strictly
661        // match `K`/`V`. An FFI failure reads as "no match" — this fn has no
662        // error channel and must not panic (see `try_raw_entries`).
663        if data.type_index != TypeIndex::kTVMFFIMap as i32 {
664            return false;
665        }
666        let map = <Self as AnyCompatible>::copy_from_any_view_after_check(data);
667        match map.try_raw_entries() {
668            Ok(entries) => entries.iter().all(|(k, v)| unsafe {
669                K::container_check_any_strict(k.as_raw_ffi_any())
670                    && V::container_check_any_strict(v.as_raw_ffi_any())
671            }),
672            Err(_) => false,
673        }
674    }
675
676    unsafe fn match_constraints(data: &TVMFFIAny) -> bool {
677        let map = <Self as AnyCompatible>::copy_from_any_view_after_check(data);
678        match map.try_raw_entries() {
679            Ok(entries) => entries.iter().all(|(k, v)| unsafe {
680                K::container_match_constraints(k.as_raw_ffi_any())
681                    && V::container_match_constraints(v.as_raw_ffi_any())
682            }),
683            Err(_) => false,
684        }
685    }
686
687    unsafe fn copy_to_any_view(src: &Self, data: &mut TVMFFIAny) {
688        data.type_index = TypeIndex::kTVMFFIMap as i32;
689        data.data_union.v_obj = ObjectArc::as_ptr(Self::data(src)) as *mut TVMFFIObject;
690        data.small_str_len = 0;
691    }
692
693    unsafe fn move_to_any(src: Self, data: &mut TVMFFIAny) {
694        data.type_index = TypeIndex::kTVMFFIMap as i32;
695        data.data_union.v_obj = ObjectArc::into_raw(Self::into_data(src)) as *mut TVMFFIObject;
696        data.small_str_len = 0;
697    }
698
699    unsafe fn copy_from_any_view_after_check(data: &TVMFFIAny) -> Self {
700        let ptr = data.data_union.v_obj as *const MapObj;
701        crate::object::unsafe_::inc_ref(ptr as *mut TVMFFIObject);
702        Self::from_data(ObjectArc::from_raw(ptr))
703    }
704
705    unsafe fn move_from_any_after_check(data: &mut TVMFFIAny) -> Self {
706        let ptr = data.data_union.v_obj as *const MapObj;
707        let obj = Self::from_data(ObjectArc::from_raw(ptr));
708        data.type_index = TypeIndex::kTVMFFINone as i32;
709        data.data_union.v_int64 = 0;
710        obj
711    }
712
713    unsafe fn try_cast_from_any_view(data: &TVMFFIAny) -> Result<Self, ()> {
714        if data.type_index == TypeIndex::kTVMFFIDict as i32 {
715            // A dict is mutable, so it converts by copy, as C++ `Map` accepts it.
716            // `ffi.Dict` shares the `MapBaseObj` header.
717            let len = (*(data.data_union.v_obj as *const MapObj)).size as usize;
718            let entries = forward_entries(
719                crate::cached_global_func!("ffi.DictForwardIterFunctor"),
720                AnyView::from_raw_ffi_any(*data),
721                len,
722            )
723            .map_err(|_| ())?;
724            return Self::from_raw_entries(entries).map_err(|_| ());
725        }
726        if data.type_index != TypeIndex::kTVMFFIMap as i32 {
727            return Err(());
728        }
729
730        // Fast path: if all entries match strictly, we can just copy the reference.
731        if <Self as AnyCompatible>::check_any_strict(data) {
732            return Ok(<Self as AnyCompatible>::copy_from_any_view_after_check(
733                data,
734            ));
735        }
736
737        // Slow path: try to convert entry by entry into a new map, as C++
738        // `TryCastFromAnyView` does.
739        let src = <Self as AnyCompatible>::copy_from_any_view_after_check(data);
740        let entries = src.try_raw_entries().map_err(|_| ())?;
741        Self::from_raw_entries(entries).map_err(|_| ())
742    }
743}
