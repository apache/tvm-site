19//! The derive macros of object types and object references.
20
21/// Bind an object type that another library, usually C++, registers: derive
22/// [`ObjectCore`](crate::ObjectCore) for a `#[repr(C)]` object struct whose
23/// first field is its parent, so that a pointer to the object is one to its
24/// parent. Define a new type in Rust with [`rs_object`](macro@crate::rs_object)
25/// instead.
26///
27/// `#[type_key = "..."]` names the type. `type_index()` takes the index the
28/// defining library registered for the key, and panics if that library is not
29/// loaded yet, so a binding never registers a key with the wrong child slots.
30/// Such a panic is final: `type_index()` of the type, and of every type under
31/// it, keeps panicking after the library loads, so call it only once the
32/// library is loaded. It also panics unless the registered key has the type's
33/// parent and depth and, when the type table records a size, a size no
34/// smaller than the type's. `#[type_index(unsafe ...)]` gives the type a
35/// static index instead; the `unsafe` marks the caller's promise that the
36/// runtime gives that index to this type.
37///
38/// A binding that declares only part of its object must not be constructed:
39/// keep its fields private.
40///
41/// * `#[type_final]` - The type has no subtypes, as C++ `_type_final`.
42/// * `#[type_immutable]` - Implement
43///   [`ImmutableObject`](crate::object::ImmutableObject), as for a C++ type
44///   not registered as `_type_mutable`; every subtype must be immutable too.
45/// * `#[type_rs_parent]` - Implement `ShallowCopy` and
46///   [`UnsafeInit`](crate::reflection::UnsafeInit), so an `rs_object` can
47///   derive from the type. Each field other than the parent must be `Clone`
48///   and implement [`AnyCompatible`](crate::AnyCompatible), be an
49///   [`Any`](crate::Any) or have no drop glue.
50pub use tvm_ffi_macros::Object;
51
52/// Declare an object reference: the managed, reference-counted handle to an
53/// object type, as `Arc<FooObj>` is to `FooObj`.
54///
55/// The struct holds one field, `ObjectArc<FooObj>`. The derive implements
56/// [`ObjectRefCore`](crate::ObjectRefCore), the default
57/// [`ObjectRefAnyCompatible`](crate::object::ObjectRefAnyCompatible) (conversions to
58/// and from [`Any`](crate::Any), in-place downcasts, typed function
59/// arguments), the upcast to [`ObjectRef`](crate::object::ObjectRef),
60/// `Deref<Target = FooObj>` for field reads, `Debug`, `Display` (the runtime
61/// `repr`, as [`AnyView`](crate::AnyView) formats it), and a private
62/// `fn from_object(object: FooObj) -> Self` for the type's constructors.
63///
64/// # Options
65///
66/// `#[object_ref(...)]` takes:
67///
68/// * `parent = Ref` - The reference to the parent's object, or to a further
69///   ancestor's; `Self` converts into it. The object type must embed that
70///   object type through its chain of parents
71///   ([`ObjectDerivesFrom`](crate::object::ObjectDerivesFrom)), so a wrong parent does not
72///   compile. Any other ancestor takes
73///   [`impl_object_ref_upcast!`](crate::impl_object_ref_upcast).
74/// * `unsafe parent = Ref` - As `parent`, for a reference that promises more
75///   than its object type: a generic one (`Array<T>`, `Map<K, V>`,
76///   `Optional<T>`) or one with a custom `ObjectRefAnyCompatible`. The caller
77///   promises that every value of `Self` is a valid `Ref`; `parent` alone
78///   does not compile for such a reference.
79/// * `skip_deref` - No `Deref`, for a reference whose object is internal.
80/// * `skip_debug` - `Debug` is written by hand.
81/// * `skip_display` - `Display` is written by hand.
82/// * `skip_any_compatible` - `ObjectRefAnyCompatible` is written by hand,
83///   for a typed reference with a stricter check or fallback conversions.
84/// * `crate = path` - Where the generated code finds tvm-ffi, as for
85///   [`rs_object`](macro@crate::rs_object).
86///
87/// # Examples
88///
89/// ```
90/// use tvm_ffi::derive::ObjectRef;
91/// use tvm_ffi::{rs_object, Object, ObjectArc};
92///
93/// #[rs_object("my_ext.MyPair")]
94/// pub struct MyPairObj {
95///     base_: Object,
96///     pub first: i64,
97///     pub second: i64,
98/// }
99///
100/// #[derive(Clone, ObjectRef)]
101/// pub struct MyPair {
102///     data: ObjectArc<MyPairObj>,
103/// }
104///
105/// #[rs_object("my_ext.MyTriple", final)]
106/// pub struct MyTripleObj {
107///     base_: MyPairObj,
108///     pub third: i64,
109/// }
110///
111/// #[derive(Clone, ObjectRef)]
112/// #[object_ref(parent = MyPair)]
113/// pub struct MyTriple {
114///     data: ObjectArc<MyTripleObj>,
115/// }
116///
117/// impl MyTriple {
118///     pub fn new(first: i64, second: i64, third: i64) -> Self {
119///         let pair = MyPairObj::new(Object::new(), first, second);
120///         Self::from_object(MyTripleObj::new(pair, third))
121///     }
122/// }
123///
124/// let triple = MyTriple::new(1, 2, 3);
125/// assert_eq!(triple.first + triple.third, 4);
126/// let _pair: MyPair = triple.into();
127/// ```
128///
129/// A `MyPairObj` does not embed an `ArrayObj`, so it cannot claim `Array` as
130/// its parent:
131///
132/// ```compile_fail,E0277
133/// use tvm_ffi::derive::ObjectRef;
134/// use tvm_ffi::{rs_object, Array, Object, ObjectArc};
135///
136/// #[rs_object("my_ext.MyPair")]
137/// pub struct MyPairObj {
138///     base_: Object,
139///     pub first: i64,
140/// }
141///
142/// #[derive(Clone, ObjectRef)]
143/// #[object_ref(unsafe parent = Array<i64>)]
144/// pub struct MyPair {
145///     data: ObjectArc<MyPairObj>,
146/// }
147/// ```
148pub use tvm_ffi_macros::ObjectRef;
