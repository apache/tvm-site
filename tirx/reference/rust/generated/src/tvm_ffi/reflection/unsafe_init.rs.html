20//! Objects made with every field undefined, for reflection to set the fields,
21//! as C++ makes an object with `UnsafeInit`.
22//!
23//! A C++ object made with `UnsafeInit` holds undefined (null) references, which
24//! the field setters overwrite and the destructor skips. A Rust reference is
25//! never null, so a Rust object cannot hold one; instead [`unsafe_init`]
26//! allocates the object zeroed, and a field holding all-zero bytes that are no
27//! value of its type ([`AnyCompatible::is_undefined`]) is undefined. The field
28//! setter writes an undefined field without dropping it, the field getter reads
29//! it as None, and the object's deleter drops only the defined fields, all
30//! through raw pointers, so no undefined field is ever viewed as its type.
31
32use std::alloc::Layout;
33use std::ffi::c_void;
34use std::panic::{catch_unwind, AssertUnwindSafe};
35use std::sync::atomic::AtomicU64;
36
37use crate::object::{Object, ObjectArc, ObjectCore, ObjectRef, ObjectRefCore};
38use crate::tvm_ffi_sys::TVMFFIObjectDeleterFlagBitMask::{
39    kTVMFFIObjectDeleterFlagBitMaskStrong, kTVMFFIObjectDeleterFlagBitMaskWeak,
40};
41use crate::tvm_ffi_sys::{TVMFFIObject, TVMFFIObjectHandle, COMBINED_REF_COUNT_BOTH_ONE};
42use crate::type_traits::AnyCompatible;
43use crate::{Error, INTERNAL_ERROR};

45/// An object type whose object can start with every field undefined, for
46/// reflection to set the fields: C++ `UnsafeInit`.
47///
48/// The creator that `#[rs_object]` registers for its type makes the object with
49/// [`unsafe_init`], which a JSON graph read back, `__ffi_init__` and
50/// `__ffi_new__` then fill. `#[rs_object]` implements it for its object, so the
51/// parent's object must implement it too: [`Object`] does, and so does the
52/// object of a non-final C++ type bound by `tvm-ffi-stubgen`.
53///
54/// `__ffi_init__` leaves an `init_off` field as the creator made it, so such a
55/// field's type must take all-zero bytes as a value, as a scalar, `Option<T>`
56/// or `Any` does; the type's registration fails otherwise.
57///
58/// # Safety
59///
60/// `drop_defined_fields` drops each defined field of the object and no
61/// undefined one (as [`drop_if_defined`] does), the parent's object's included.
62pub unsafe trait UnsafeInit: ObjectCore {
63    /// Drops the defined fields of the object at `this`, the parent's object's
64    /// included, and leaves the undefined ones.
65    ///
66    /// # Safety
67    ///
68    /// `this` points to an object that [`unsafe_init`] made, which is not used
69    /// afterwards.
70    #[doc(hidden)]
71    unsafe fn drop_defined_fields(this: *mut Self);
72}
73
74// The header holds nothing to drop.
75unsafe impl UnsafeInit for Object {
76    unsafe fn drop_defined_fields(_this: *mut Self) {}
77}
78
79/// Drops the field at `field` unless it is undefined.
80///
81/// # Safety
82///
83/// `field` points to a field of an object, defined or undefined, which is not
84/// used afterwards.
85#[doc(hidden)]
86#[inline]
87pub unsafe fn drop_if_defined<F: AnyCompatible>(field: *mut F) {
88    unsafe {
89        if !F::is_undefined(field) {
90            std::ptr::drop_in_place(field);
91        }
92    }
93}

95/// An object of `T` with every field undefined, as C++
96/// `make_object<T>(UnsafeInit{})` makes one, for reflection to set the fields
97/// through their setters.
98///
99/// The [`ObjectRef`] views only the object's header, which is valid all along.
100///
101/// # Safety
102///
103/// Each field that all-zero bytes leave undefined
104/// ([`AnyCompatible::is_undefined`]) is set before the object is viewed as a
105/// `T` (by a cast of the reference) or the field is read as its type. Until
106/// then, only the reflection's field setters and getters reach the fields.
107pub unsafe fn unsafe_init<T: UnsafeInit>() -> ObjectRef {
108    let layout = Layout::new::<T>();
109    unsafe {
110        let ptr = std::alloc::alloc_zeroed(layout);
111        if ptr.is_null() {
112            std::alloc::handle_alloc_error(layout);
113        }
114        std::ptr::write(
115            ptr.cast::<TVMFFIObject>(),
116            TVMFFIObject {
117                combined_ref_count: AtomicU64::new(COMBINED_REF_COUNT_BOTH_ONE),
118                type_index: T::type_index(),
119                __padding: 0,
120                deleter: Some(unsafe_init_deleter::<T>),
121            },
122        );
123        // SAFETY: `ObjectRef` is the view of any object.
124        ObjectRef::from_data(ObjectArc::from_raw(ptr.cast::<Object>()))
125    }
126}
127
128/// The deleter of an object that [`unsafe_init`] made, which may go before
129/// reflection sets every field, as when a JSON graph lacks one.
130unsafe extern "C" fn unsafe_init_deleter<T: UnsafeInit>(ptr: *mut c_void, flags: i32) {
131    unsafe {
132        if flags & kTVMFFIObjectDeleterFlagBitMaskStrong as i32 != 0 {
133            T::drop_defined_fields(ptr.cast::<T>());
134        }
135        if flags & kTVMFFIObjectDeleterFlagBitMaskWeak as i32 != 0 {
136            std::alloc::dealloc(ptr.cast::<u8>(), Layout::new::<T>());
137        }
138    }
139}

141/// The creator of `T` (`TVMFFITypeMetadata::creator`) that `#[rs_object]`
142/// registers, as C++ `ObjectCreatorUnsafeInit<T>`: the owned handle of an
143/// object that [`unsafe_init`] made.
144///
145/// # Safety
146///
147/// `result` is valid for writes.
148#[doc(hidden)]
149pub unsafe extern "C" fn unsafe_init_creator<T: UnsafeInit>(
150    result: *mut TVMFFIObjectHandle,
151) -> i32 {
152    let made = catch_unwind(AssertUnwindSafe(|| unsafe {
153        let object = unsafe_init::<T>();
154        let handle = ObjectArc::into_raw(ObjectRef::into_data(object));
155        std::ptr::write(result, handle as TVMFFIObjectHandle);
156    }));
157    match made {
158        Ok(()) => 0,
159        Err(_) => {
160            let message = format!("panic while creating an object of `{}`", T::TYPE_KEY);
161            Error::set_raised(&Error::new(INTERNAL_ERROR, &message, ""));
162            -1
163        }
164    }
165}

167#[cfg(test)]
168mod tests {
169    use std::mem::MaybeUninit;
170
171    use crate::tvm_ffi_sys::dlpack::{DLDevice, DLDeviceType};
172    use crate::type_traits::AnyCompatible;
173    use crate::{Array, BigInt, String};
174
175    fn zero_is_undefined<F: AnyCompatible>() -> bool {
176        let zero = MaybeUninit::<F>::zeroed();
177        unsafe { F::is_undefined(zero.as_ptr()) }
178    }
179
180    fn is_undefined<F: AnyCompatible>(value: F) -> bool {
181        unsafe { F::is_undefined(&value) }
182    }
183
184    #[test]
185    fn all_zero_bytes_are_undefined_where_they_are_no_value() {
186        assert!(zero_is_undefined::<Array<i64>>());
187        assert!(zero_is_undefined::<String>());
188        assert!(zero_is_undefined::<BigInt>());
189        assert!(zero_is_undefined::<DLDevice>());
190
191        assert!(!zero_is_undefined::<i64>());
192        assert!(!zero_is_undefined::<bool>());
193        // `None`, through the niche of a reference or a device, or through a tag.
194        assert!(!zero_is_undefined::<Option<Array<i64>>>());
195        assert!(!zero_is_undefined::<Option<DLDevice>>());
196        assert!(!zero_is_undefined::<Option<String>>());
197    }
198
199    #[test]
200    fn values_are_defined() {
201        let cpu = DLDevice {
202            device_type: DLDeviceType::kDLCPU,
203            device_id: 0,
204        };
205        assert!(!is_undefined(Array::<i64>::new()));
206        assert!(!is_undefined(String::from("")));
207        assert!(!is_undefined(BigInt::from(0_i64)));
208        assert!(!is_undefined(cpu));
209        assert!(!is_undefined(Some(cpu)));
210        assert!(!is_undefined(None::<DLDevice>));
211        assert!(!is_undefined(None::<Array<i64>>));
212    }
