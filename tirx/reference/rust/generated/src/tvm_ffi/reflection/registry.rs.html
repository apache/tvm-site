20//! Registration of the reflection of an object type defined in Rust.
21//!
22//! [`ObjectDef`] fills the type table entries that every language reads: the
23//! type metadata, one record per field with C ABI getter and setter, one
24//! record per method, and the `__ffi_shallow_copy__` and `__ffi_type_final__`
25//! type attributes; for an `#[rs_object]` type also the creator, and from it
26//! the `__ffi_new__` and `__ffi_init__` type attributes. Through them
27//! structural equality, hashing and mutation, `repr`, deep copy, JSON graphs
28//! and Python's generated classes reach the fields and methods of a Rust type,
29//! and construct it.
30//!
31//! Only [`rs_object`](crate::rs_object) creates an [`ObjectDef`], which
32//! guarantees that each registered field's type and offset match. Its
33//! `reflection = path` function receives the builder to register methods with
34//! [`ObjectDef::def`] and [`ObjectDef::def_static`].
35

56/// A copy of an object's own fields under a fresh object header, which
57/// `__ffi_shallow_copy__` returns; fields that are references share their
58/// objects, as `Clone` of an `Arc` does.
59///
60/// [`rs_object`](crate::rs_object) and `#[type_rs_parent]` on
61/// [`derive(Object)`](crate::derive::Object) implement it by copying the
62/// parent with its `ShallowCopy` and cloning every other field. An
63/// `rs_object` needs its parent's object to implement it.
64pub trait ShallowCopy {
65    /// The copy, under a fresh object header.
66    fn shallow_copy(&self) -> Self;
67}
68
69impl ShallowCopy for Object {
70    fn shallow_copy(&self) -> Self {
71        Object::new()
72    }
73}

75/// How the structural engines treat instances of a type.
76#[derive(Clone, Copy, Debug, PartialEq, Eq)]
77#[non_exhaustive]
78pub enum SEqHashKind {
79    /// Rejected by structural equality and hashing; the default.
80    Unsupported,
81    /// Compared and hashed field by field.
82    TreeNode,
83    /// A free variable: bound by position inside a definition region, else by identity.
84    FreeVar,
85    /// Compared field by field, and once equal, mapped so later occurrences compare by mapping.
86    DAGNode,
87    /// A tree node whose identity is checked first, as a fast path.
88    ConstTreeNode,
89    /// Compared and hashed by identity only.
90    UniqueInstance,
91}
92
93impl SEqHashKind {
94    fn raw(self) -> i32 {
95        let kind = match self {
96            SEqHashKind::Unsupported => TVMFFISEqHashKind::kTVMFFISEqHashKindUnsupported,
97            SEqHashKind::TreeNode => TVMFFISEqHashKind::kTVMFFISEqHashKindTreeNode,
98            SEqHashKind::FreeVar => TVMFFISEqHashKind::kTVMFFISEqHashKindFreeVar,
99            SEqHashKind::DAGNode => TVMFFISEqHashKind::kTVMFFISEqHashKindDAGNode,
100            SEqHashKind::ConstTreeNode => TVMFFISEqHashKind::kTVMFFISEqHashKindConstTreeNode,
101            SEqHashKind::UniqueInstance => TVMFFISEqHashKind::kTVMFFISEqHashKindUniqueInstance,
102        };
103        kind as i32
104    }
105}

107/// The reflection flags of one field, combined with `|`.
108#[derive(Clone, Copy, Debug, PartialEq, Eq)]
109pub struct FieldFlags(i64);
110
111impl FieldFlags {
112    /// A read-only field that takes part in structural equality and hashing (`def_ro`).
113    pub const NONE: Self = Self(0);
114    /// Assignable through reflection, as from Python (`def_rw`).
115    pub const WRITABLE: Self = Self::of(TVMFFIFieldFlagBitMask::kTVMFFIFieldFlagBitMaskWritable);
116    /// Skipped by structural equality and hashing (`SEqHashIgnore`).
117    pub const SEQ_HASH_IGNORE: Self =
118        Self::of(TVMFFIFieldFlagBitMask::kTVMFFIFieldFlagBitMaskSEqHashIgnore);
119    /// Entered as a pattern definition region (`SEqHashDefPattern`).
120    pub const SEQ_HASH_DEF_PATTERN: Self =
121        Self::of(TVMFFIFieldFlagBitMask::kTVMFFIFieldFlagBitMaskSEqHashDefPattern);
122    /// Entered as a simple definition region (`SEqHashDefSimple`).
123    pub const SEQ_HASH_DEF_SIMPLE: Self =
124        Self::of(TVMFFIFieldFlagBitMask::kTVMFFIFieldFlagBitMaskSEqHashDefSimple);
125    /// Left out of `repr`.
126    pub const REPR_OFF: Self = Self::of(TVMFFIFieldFlagBitMask::kTVMFFIFieldFlagBitMaskReprOff);
127    /// Left out of `__eq__`.
128    pub const COMPARE_OFF: Self =
129        Self::of(TVMFFIFieldFlagBitMask::kTVMFFIFieldFlagBitMaskCompareOff);
130    /// Left out of `__hash__`.
131    pub const HASH_OFF: Self = Self::of(TVMFFIFieldFlagBitMask::kTVMFFIFieldFlagBitMaskHashOff);
132    /// Not a constructor argument of the Python class.
133    pub const INIT_OFF: Self = Self::of(TVMFFIFieldFlagBitMask::kTVMFFIFieldFlagBitMaskInitOff);
134    /// A keyword-only constructor argument of the Python class.
135    pub const KW_ONLY: Self = Self::of(TVMFFIFieldFlagBitMask::kTVMFFIFieldFlagBitMaskKwOnly);
136
137    const fn of(bit: TVMFFIFieldFlagBitMask) -> Self {
138        Self(bit as i64)
139    }
140
141    /// The raw `TVMFFIFieldInfo::flags` value.
142    pub fn bits(self) -> i64 {
143        self.0
144    }
145}
146
147impl BitOr for FieldFlags {
148    type Output = Self;
149    fn bitor(self, rhs: Self) -> Self {
150        Self(self.0 | rhs.0)
151    }
152}

154/// Registers the reflection of the object type `N`: the metadata, each
155/// `#[reflect]` field, the methods and the type attributes.
156///
157/// [`rs_object`](crate::rs_object) creates one while it registers its type
158/// and passes it to the function its `reflection = path` option names, which
159/// registers methods with [`ObjectDef::def`] and [`ObjectDef::def_static`].
160///
161/// That function runs under the registration lock, before `N::type_index()`
162/// returns, so it must not call `type_index()` of `N` or of another Rust type
163/// that is not registered yet, nor register attributes with [`TypeAttrDef`]
164/// or [`EnumDef`](super::EnumDef), nor call
165/// [`TypeAttrColumn::ensure`](super::TypeAttrColumn::ensure): each takes the
166/// same lock, which is not reentrant. When `N` already has metadata in the type
167/// table, every call is a no-op; registration fails if a previous one stopped
168/// partway.
169///
170/// # Examples
171///
172/// ```
173/// use tvm_ffi::derive::ObjectRef;
174/// use tvm_ffi::reflection::{self, ObjectDef};
175/// use tvm_ffi::{rs_object, Object, ObjectArc, Result};
176///
177/// #[rs_object("my_ext.MyPair", final, reflection = MyPairObj::reflect_methods)]
178/// pub struct MyPairObj {
179///     base_: Object,
180///     pub first: i64,
181///     pub second: i64,
182/// }
183///
184/// #[derive(Clone, ObjectRef)]
185/// pub struct MyPair {
186///     data: ObjectArc<MyPairObj>,
187/// }
188///
189/// impl MyPairObj {
190///     fn reflect_methods(def: &mut ObjectDef<Self>) {
191///         def.def("sum", |p: MyPair| -> Result<i64> { Ok(p.first + p.second) }, "The sum.");
192///     }
193/// }
194///
195/// let pair = MyPair::from_object(MyPairObj::new(Object::new(), 1, 2));
196/// let sum = reflection::get_method("my_ext.MyPair", "sum")?;
197/// assert_eq!(i64::try_from(sum.call_tuple((&pair,))?)?, 3);
198/// # Ok::<(), tvm_ffi::Error>(())
199/// ```
200pub struct ObjectDef<N: ObjectCore> {
201    type_index: i32,
202    fresh: bool,
203    /// The creator registered in the metadata, from which `finish` registers
204    /// `__ffi_new__` and `__ffi_init__`.
205    creator: Option<TVMFFIObjectCreator>,
206    /// The first failed method registration, which `finish` reports.
207    error: Option<Error>,
208    method_names: Vec<String>,
209    _object: PhantomData<N>,
210}
211
212impl<N: ObjectCore> ObjectDef<N> {
213    /// Registers `N`'s metadata, unless the type already has some, while the
214    /// derive registers `N`'s type index, before `N::type_index()` returns.
215    ///
216    /// `kind` of `None` takes the nearest ancestor's kind. `creator` makes an
217    /// object of `N` for reflection to fill (`unsafe_init_creator::<N>` for an
218    /// `#[rs_object]`).
219    ///
220    /// # Safety
221    ///
222    /// The caller holds `lock_type_registration()` for the builder's lifetime, and
223    /// `type_index` is `N`'s. `creator` returns an owned handle of an object of
224    /// `N` that the field setters can fill.
225    unsafe fn registering(
226        type_index: i32,
227        kind: Option<SEqHashKind>,
228        creator: Option<TVMFFIObjectCreator>,
229    ) -> Result<Self> {
230        let info = TVMFFIGetTypeInfo(type_index);
231        crate::ensure!(
232            !info.is_null(),
233            INTERNAL_ERROR,
234            "no type info for `{}` (type index {type_index})",
235            N::TYPE_KEY
236        );
237        let fresh = (*info).metadata.is_null();
238        // `finish` writes `__ffi_type_final__` last, as C++ `ObjectDef` and
239        // Python `py_class` do: metadata without it is left by a registration
240        // that failed partway, or by one that does not publish finality.
241        crate::ensure!(
242            fresh
243                || super::TypeAttrColumn::lookup("__ffi_type_final__")
244                    .and_then(|column| column.get_raw_ffi_any(type_index))
245                    .is_some_and(|cell| cell.type_index != TVMFFITypeIndex::kTVMFFINone as i32),
246            INTERNAL_ERROR,
247            "`{}` has reflection metadata but no `__ffi_type_final__`: an earlier registration \
248             of it failed, or another language registered the key without it",
249            N::TYPE_KEY
250        );
251        if fresh {
252            let kind = match kind {
253                Some(kind) => kind.raw(),
254                None => inherited_kind(info),
255            };
256            let metadata = TVMFFITypeMetadata {
257                doc: TVMFFIByteArray::from_str(""),
258                creator,
259                total_size: size_of::<N>() as i32,
260                structural_eq_hash_kind: kind,
261            };
262            crate::check_safe_call!(TVMFFITypeRegisterMetadata(type_index, &metadata))?;
263        }
264        Ok(Self {
265            type_index,
266            fresh,
267            creator,
268            error: None,
269            method_names: Vec::new(),
270            _object: PhantomData,
271        })
272    }
273
274    /// Registers the field of type `F` at byte offset `offset` in `N` under
275    /// `name`, which must be unique in `N`, with its doc string (an empty one
276    /// records none).
277    ///
278    /// # Safety
279    ///
280    /// `offset` is `std::mem::offset_of!` of a field of `N` whose type is `F`:
281    /// the registered getter and setter read and write an `F` there.
282    #[doc(hidden)]
283    pub unsafe fn field<F: AnyCompatible>(
284        self,
285        name: &str,
286        doc: &str,
287        offset: usize,
288        flags: FieldFlags,
289    ) -> Result<Self> {
290        if !self.fresh {
291            return Ok(self);
292        }
293        // `__ffi_init__` leaves an `init_off` field as the creator made it, all-zero.
294        if self.creator.is_some() && flags.bits() & FieldFlags::INIT_OFF.bits() != 0 {
295            let zero = std::mem::MaybeUninit::<F>::zeroed();
296            crate::ensure!(
297                !unsafe { F::is_undefined(zero.as_ptr()) },
298                VALUE_ERROR,
299                "`init_off` field `{name}` of `{}` stays all-zero after `__ffi_init__`, which \
300                 is no value of `{}`",
301                N::TYPE_KEY,
302                F::type_str()
303            );
304        }
305        let type_str = F::type_str();
306        crate::ensure!(
307            F::has_field_layout(),
308            TYPE_ERROR,
309            "field `{name}` of `{}`: `Option` of a non-object value (`{type_str}`) has no FFI \
310             field layout; use `tvm_ffi::Optional<T>`",
311            N::TYPE_KEY
312        );
313        let metadata = type_schema_metadata(&F::type_schema());
314        let info = TVMFFIFieldInfo {
315            name: unsafe { TVMFFIByteArray::from_str(name) },
316            doc: unsafe { TVMFFIByteArray::from_str(doc) },
317            metadata: unsafe { TVMFFIByteArray::from_str(&metadata) },
318            flags: flags.bits(),
319            size: size_of::<F>() as i64,
320            alignment: align_of::<F>() as i64,
321            offset: offset as i64,
322            getter: Some(field_getter::<F>),
323            setter: field_setter::<F> as TVMFFIFieldSetter as *mut c_void,
324            default_value_or_factory: TVMFFIAny::new(),
325            field_static_type_index: field_static_type_index(&type_str),
326        };
327        // The table copies the strings before returning.
328        crate::check_safe_call!(unsafe { TVMFFITypeRegisterField(self.type_index, &info) })?;
329        Ok(self)
330    }
331
332    /// Registers `__ffi_shallow_copy__` from [`ShallowCopy`], which reflected
333    /// mutation (copy on write through the field setters) needs.
334    fn shallow_copy(self) -> Result<Self>
335    where
336        N: ShallowCopy,
337    {
338        if !self.fresh {
339            return Ok(self);
340        }
341        let shallow_copy = Function::from_typed(|object: ObjectRef| shallow_copy_any::<N>(&object));
342        register_type_attr(
343            self.type_index,
344            "__ffi_shallow_copy__",
345            Any::from(shallow_copy),
346        )?;
347        Ok(self)
348    }

350    /// Registers `func`, a typed function as for [`Function::from_typed`]
351    /// whose first parameter is the object, as the method `name` of `N`,
352    /// which must be unique in `N`, with its type schema and `doc`. A failed
353    /// registration is reported when the type's registration finishes.
354    pub fn def<F, I, O>(&mut self, name: &str, func: F, doc: &str) -> &mut Self
355    where
356        F: AsPackedCallable<I, O> + Send + Sync + 'static,
357    {
358        self.method(name, func, doc, false)
359    }
360
361    /// Registers `func` as the static method `name` of `N`, which takes no
362    /// object; otherwise as [`ObjectDef::def`].
363    pub fn def_static<F, I, O>(&mut self, name: &str, func: F, doc: &str) -> &mut Self
364    where
365        F: AsPackedCallable<I, O> + Send + Sync + 'static,
366    {
367        self.method(name, func, doc, true)
368    }
369
370    fn method<F, I, O>(&mut self, name: &str, func: F, doc: &str, is_static: bool) -> &mut Self
371    where
372        F: AsPackedCallable<I, O> + Send + Sync + 'static,
373    {
374        if !self.fresh || self.error.is_some() {
375            return self;
376        }
377        if self.method_names.iter().any(|n| n == name) {
378            self.error = Some(Error::new(
379                VALUE_ERROR,
380                &format!("duplicate method `{name}` of `{}`", N::TYPE_KEY),
381                "",
382            ));
383            return self;
384        }
385        self.method_names.push(name.to_string());
386        let metadata = type_schema_metadata(&F::type_schema());
387        let func = Function::from_typed(func);
388        let flags = if is_static {
389            TVMFFIFieldFlagBitMask::kTVMFFIFieldFlagBitMaskIsStaticMethod as i64
390        } else {
391            0
392        };
393        let mut method = TVMFFIAny::new();
394        // SAFETY: the strings outlive the call, which copies them, and the
395        // table takes its own reference to the function.
396        let registered = unsafe {
397            <Function as AnyCompatible>::copy_to_any_view(&func, &mut method);
398            let info = TVMFFIMethodInfo {
399                name: TVMFFIByteArray::from_str(name),
400                doc: TVMFFIByteArray::from_str(doc),
401                metadata: TVMFFIByteArray::from_str(&metadata),
402                flags,
403                method,
404            };
405            crate::check_safe_call!(TVMFFITypeRegisterMethod(self.type_index, &info))
406        };
407        // A static `__ffi_init__` is also the type's `__ffi_init__` attribute, which
408        // other languages call to construct it, as C++ `ObjectDef::def(init)` registers.
409        let registered = registered.and_then(|()| {
410            if is_static && name == "__ffi_init__" {
411                register_type_attr(self.type_index, name, Any::from(func))
412            } else {
413                Ok(())
414            }
415        });
416        if let Err(error) = registered {
417            self.error = Some(Error::new(
418                error.kind(),
419                &format!("registering method `{name}`: {}", error.message()),
420                "",
421            ));
422        }
423        self
424    }
425
426    /// Registers `__ffi_new__` from the creator and
427    /// `__ffi_init__` from the fields unless the type defines its own, then
428    /// `__ffi_type_final__`, which no other registered metadata exposes; and
429    /// reports the first failed method registration.
430    fn finish(self) -> Result<()> {
431        if let Some(error) = self.error {
432            return Err(error);
433        }
434        if !self.fresh {
435            return Ok(());
436        }
437        if let Some(creator) = self.creator {
438            // Objects from this creator start zero-filled, in the `UnsafeInit`
439            // state: whoever creates one, in any language, must set every
440            // field before it is used.
441            let new = Function::from_typed(move || -> Result<ObjectRef> {
442                let mut handle: TVMFFIObjectHandle = std::ptr::null_mut();
443                crate::check_safe_call!(unsafe { creator(&mut handle) })?;
444                // SAFETY: the creator returns an owned handle, which `ObjectRef` views.
445                Ok(unsafe { ObjectRef::from_data(ObjectArc::from_raw(handle.cast::<Object>())) })
446            });
447            register_type_attr(self.type_index, "__ffi_new__", Any::from(new))?;
448            // C++ `ObjectDef` looks `ffi._RegisterFFIInit` up the same way, in case
449            // it is not registered yet.
450            let explicit_init = self.method_names.iter().any(|name| name == "__ffi_init__");
451            if !explicit_init {
452                if let Ok(register_init) = Function::get_global("ffi._RegisterFFIInit") {
453                    register_init.call_tuple((self.type_index,))?;
454                }
455            }
456        }
457        register_type_attr(
458            self.type_index,
459            "__ffi_type_final__",
460            Any::from(N::TYPE_FINAL),
461        )
462    }
463}

465/// Registers the reflection of `N` for `#[derive(Object)]`: its metadata,
466/// then `fields` its fields, its `__ffi_shallow_copy__`, then `methods` its
467/// methods, then its type attributes.
468///
469/// # Safety
470///
471/// As `ObjectDef::registering`; `fields` registers fields of `N` as
472/// [`ObjectDef::field`] requires. Like every registration, it runs before
473/// other threads use the type or register types: during static
474/// initialization, or on the first `type_index()` of a type under a binding
475/// loaded later.
476#[doc(hidden)]
477pub unsafe fn register_object<N: ObjectCore + ShallowCopy>(
478    type_index: i32,
479    kind: Option<SEqHashKind>,
480    creator: Option<TVMFFIObjectCreator>,
481    fields: impl FnOnce(ObjectDef<N>) -> Result<ObjectDef<N>>,
482    methods: impl FnOnce(&mut ObjectDef<N>),
483) -> Result<()> {
484    let mut def = fields(ObjectDef::registering(type_index, kind, creator)?)?.shallow_copy()?;
485    methods(&mut def);
486    def.finish()
487}

489/// Registers type attributes of the object type `T`, each as soon as
490/// [`attr`](Self::attr) names it, for every language to read with
491/// [`TypeAttrColumn`](crate::reflection::TypeAttrColumn).
492///
493/// As in C++, attributes register during static initialization, such as in a
494/// [`static_init_block!`](crate::static_init_block), before other threads
495/// read them: registrations from Rust lock each other out, but readers take
496/// no lock.
497///
498/// # Examples
499///
500/// ```
501/// use tvm_ffi::reflection::TypeAttrDef;
502/// use tvm_ffi::reflection::TypeAttrColumn;
503/// use tvm_ffi::{rs_object, Object, ObjectCore};
504///
505/// #[rs_object("my_ext.MyPair")]
506/// pub struct MyPairObj {
507///     base_: Object,
508/// }
509///
510/// TypeAttrDef::<MyPairObj>::new().attr("my_ext.arity", 2_i64)?;
511///
512/// let column = TypeAttrColumn::lookup("my_ext.arity").unwrap();
513/// let arity: i64 = column.get(MyPairObj::type_index()).unwrap().try_into()?;
514/// assert_eq!(arity, 2);
515/// # Ok::<(), tvm_ffi::Error>(())
516/// ```
517pub struct TypeAttrDef<T: ObjectCore> {
518    _type: PhantomData<fn() -> T>,
519}
520
521impl<T: ObjectCore> TypeAttrDef<T> {
522    /// A builder for the type attributes of `T`.
523    #[allow(clippy::new_without_default)]
524    pub fn new() -> Self {
525        Self { _type: PhantomData }
526    }
527
528    /// Register the type attribute `name` of `T`.
529    ///
530    /// # Arguments
531    ///
532    /// * `name` - The name of the attribute.
533    /// * `value` - The value of the attribute, `Send + Sync` since every
534    ///   thread reads it.
535    ///
536    /// # Returns
537    ///
538    /// The builder, for the next attribute.
539    ///
540    /// # Errors
541    ///
542    /// Returns an error if `T` already has the attribute `name`, or a
543    /// `TypeError` if `value` is an opaque pointer: register a native hook
544    /// with [`native_fn_attr`](Self::native_fn_attr).
545    pub fn attr(self, name: &str, value: impl Into<Any> + Send + Sync) -> Result<Self> {
546        let value = value.into();
547        ensure_not_opaque_ptr(name, &value)?;
548        self.register(name, value)
549    }
550
551    /// Register the native hook `f` as the type attribute `name` of `T`.
552    ///
553    /// # Arguments
554    ///
555    /// * `name` - The name of the attribute.
556    /// * `f` - The hook.
557    ///
558    /// # Returns
559    ///
560    /// The builder, for the next attribute.
561    ///
562    /// # Errors
563    ///
564    /// Returns an error if `T` already has the attribute `name`.
565    ///
566    /// # Safety
567    ///
568    /// Every caller of the column `name` calls its hooks with exactly `Args`
569    /// and reads `R`: a native caller calls the function pointer unchecked.
570    pub unsafe fn native_fn_attr<Args, R>(
571        self,
572        name: &str,
573        f: NativeFunction<Args, R>,
574    ) -> Result<Self> {
575        // SAFETY: the caller vouches for the readers of `name`.
576        self.register(name, unsafe { f.into_any() })
577    }

579    fn register(self, name: &str, value: Any) -> Result<Self> {
580        // Before the lock, which a first `type_index()` takes to register `T`.
581        let type_index = T::type_index();
582        let _registering = lock_type_registration();
583        register_type_attr(type_index, name, value)?;
584        Ok(self)
585    }
586}

588/// Native callers of an attribute call an opaque pointer as a function, so
589/// only [`NativeFunction`] may store one, through an `unsafe` registration.
590pub(super) fn ensure_not_opaque_ptr(name: &str, value: &Any) -> Result<()> {
591    crate::ensure!(
592        value.type_index() != TVMFFITypeIndex::kTVMFFIOpaquePtr as i32,
593        TYPE_ERROR,
594        "attribute `{name}` cannot be an opaque pointer; register a native hook through \
595         `NativeFunction`"
596    );
597    Ok(())
598}

600/// The structural kind of the nearest ancestor of `info` with metadata.
601unsafe fn inherited_kind(info: *const crate::tvm_ffi_sys::TVMFFITypeInfo) -> i32 {
602    for depth in (0..(*info).type_depth).rev() {
603        let ancestor = *(*info).type_acenstors.add(depth as usize);
604        if !ancestor.is_null() && !(*ancestor).metadata.is_null() {
605            return (*(*ancestor).metadata).structural_eq_hash_kind;
606        }
607    }
608    SEqHashKind::Unsupported.raw()
609}

611/// `__ffi_shallow_copy__` of `N`: a copy of `object`, which must be an `N`, as a new object.
612fn shallow_copy_any<N: ObjectCore + ShallowCopy>(object: &ObjectRef) -> Result<Any> {
613    let Some(node) = object.downcast_ref::<N>() else {
614        return Err(Error::new(
615            TYPE_ERROR,
616            &format!(
617                "__ffi_shallow_copy__ of `{}` applied to a `{}`",
618                N::TYPE_KEY,
619                object_type_key(object)
620            ),
621            "",
622        ));
623    };
624    let copy = ObjectArc::new(node.shallow_copy());
625    // SAFETY: `ObjectRef` is the view of any object; the allocation keeps its type index.
626    let copy =
627        unsafe { ObjectRef::from_data(ObjectArc::from_raw(ObjectArc::into_raw(copy).cast())) };
628    Ok(Any::from(copy))
629}
630
631fn object_type_key(object: &ObjectRef) -> String {
632    let type_index =
633        unsafe { (*ObjectArc::as_ptr(ObjectRef::data(object)).cast::<TVMFFIObject>()).type_index };
634    let info = unsafe { TVMFFIGetTypeInfo(type_index) };
635    if info.is_null() {
636        return format!("type index {type_index}");
637    }
638    unsafe { (*info).type_key.as_str().to_string() }
639}

641/// The caller holds the registration lock.
642pub(super) fn register_type_attr(type_index: i32, name: &str, mut value: Any) -> Result<()> {
643    let name = unsafe { TVMFFIByteArray::from_str(name) };
644    // The table takes its own reference to the value.
645    let value = Any::as_raw_ffi_any_mut(&mut value);
646    crate::check_safe_call!(unsafe { TVMFFITypeRegisterAttr(type_index, &name, value) })
647}

649/// Reflection getter of a field of type `F`: writes an owned `Any` of the field's value.
650/// `field` is the field's address inside the object.
651unsafe extern "C" fn field_getter<F: AnyCompatible>(
652    field: *mut c_void,
653    result: *mut TVMFFIAny,
654) -> i32 {
655    let read = catch_unwind(AssertUnwindSafe(|| {
656        let field = field.cast::<F>();
657        // An undefined field reads as None, as a null `ObjectRef` does in C++.
658        let value = if F::is_undefined(field) {
659            Any::new()
660        } else {
661            (*field).to_any()
662        };
663        std::ptr::write(result, Any::into_raw_ffi_any(value));
664    }));
665    match read {
666        Ok(()) => 0,
667        Err(_) => {
668            let message = format!("panic while reading a field of type {}", F::type_str());
669            Error::set_raised(&Error::new(INTERNAL_ERROR, &message, ""));
670            -1
671        }
672    }
673}

675/// Reflection setter of a field of type `F`: replaces the field with `value` converted to `F`.
676/// `field` is the field's address inside the object.
677unsafe extern "C" fn field_setter<F: AnyCompatible>(
678    field: *mut c_void,
679    value: *const TVMFFIAny,
680) -> i32 {
681    let written = catch_unwind(AssertUnwindSafe(|| {
682        match F::try_cast_from_any_view(&*value) {
683            Ok(converted) => {
684                let field = field.cast::<F>();
685                // An undefined field holds no value to drop.
686                if F::is_undefined(field) {
687                    field.write(converted);
688                } else {
689                    *field = converted;
690                }
691                None
692            }
693            Err(()) => Some(Error::new(
694                TYPE_ERROR,
695                &format!(
696                    "expected a field value of type {}, got {}",
697                    F::type_str(),
698                    F::get_mismatch_type_info(&*value)
699                ),
700                "",
701            )),
702        }
703    }));
704    let error = match written {
705        Ok(None) => return 0,
706        Ok(Some(error)) => error,
707        Err(_) => Error::new(
708            INTERNAL_ERROR,
709            &format!("panic while writing a field of type {}", F::type_str()),
710            "",
711        ),
712    };
713    Error::set_raised(&error);
714    -1
715}

717/// The static type index of a field, from the Rust type's `type_str()` (the names
718/// `AnyCompatible` reports, such as `int`, `ffi.String`, `Array<my.Node>`).
719// As C++ `TypeTraits<T>::field_static_type_index` states it.
720pub(crate) fn field_static_type_index(type_str: &str) -> i32 {
721    use TVMFFITypeIndex::*;
722    /// The static index of a field holding any value (`kTVMFFIAny`).
723    const ANY: i32 = -1;
724    let origin = type_str.split('<').next().unwrap_or(type_str).trim();
725    match origin {
726        "None" => kTVMFFINone as i32,
727        "int" => kTVMFFIInt as i32,
728        "bool" => kTVMFFIBool as i32,
729        "float" => kTVMFFIFloat as i32,
730        "void*" => kTVMFFIOpaquePtr as i32,
731        "DataType" => kTVMFFIDataType as i32,
732        "Device" => kTVMFFIDevice as i32,
733        "DLTensor*" => kTVMFFIDLTensorPtr as i32,
734        // C++ `String`, `Bytes` and `BigInt` fields are typed `Any` statically (they have
735        // inline forms), as is `Optional<T>`.
736        "ffi.String" | "ffi.Bytes" | "BigInt" | "Optional" => ANY,
737        "ffi.Function" => kTVMFFIFunction as i32,
738        "ffi.Shape" => kTVMFFIShape as i32,
739        "Array" => kTVMFFIArray as i32,
740        "List" => kTVMFFIList as i32,
741        // Any other object reference, `Map` included, is typed `Object` statically.
742        _ => kTVMFFIStaticObjectBegin as i32,
743    }
744}
