41/// An arbitrary-precision signed integer.
42///
43/// Its layout is part of the stable FFI ABI, so a `BigInt` passes across
44/// languages without conversion. A value that fits `i64` is stored inline, and
45/// a wider value uses a heap object of its minimal two's-complement `i64`
46/// words, least-significant word first. Results are always in that canonical
47/// form, so equality is word-wise.
48///
49/// Arithmetic is unbounded and signed. `/` and `%` truncate toward zero;
50/// [`floor_div`](Self::floor_div) and [`floor_mod`](Self::floor_mod) round down.
51/// Bitwise operations sign-extend infinitely, and negative shift counts are errors.
52/// Like the primitive operators, `/`, `%`, `<<` and `>>` panic on those errors;
53/// the `try_*` methods return them instead, as ffi callbacks need since a panic
54/// cannot unwind across them.
55#[repr(C)]
56pub struct BigInt {
57    data: TVMFFIAny,
58}
59
60// SAFETY: a `BigInt` is immutable: an inline `i64` or an immutable big-integer object.
61unsafe impl Send for BigInt {}
62unsafe impl Sync for BigInt {}

64// BigIntObj for heap-allocated integers: the word count, then the canonical
65// two's-complement words as trailing items. Mirrors C++ `detail::BigIntObj`.
66#[repr(C)]
67#[derive(Object)]
68#[object(type_key = "ffi.BigInt", unsafe type_index = TypeIndex::kTVMFFIBigInt, final)]
69struct BigIntObj {
70    object: Object,
71    size: usize,
72}
73
74unsafe impl ObjectCoreWithExtraItems for BigIntObj {
75    type ExtraItem = i64;
76    #[inline]
77    /// Get the count of extra items (the logical word count)
78    fn extra_items_count(this: &Self) -> usize {
79        this.size
80    }
81}

83// `BigInt::try_build` allocates from the C allocator like C++
84// `make_inplace_array_object`, so it can prune the logical size in place (C++
85// `ShrinkSize`): `free` needs no layout, unlike Rust's global allocator.
86unsafe extern "C" {
87    fn malloc(size: usize) -> *mut c_void;
88    fn free(ptr: *mut c_void);
89}

91/// Release an object from `BigInt::try_build`: free the allocation; the words
92/// need no destructor.
93unsafe extern "C" fn delete_obj(ptr: *mut c_void, flags: i32) {
94    if flags & kTVMFFIObjectDeleterFlagBitMaskWeak as i32 != 0 {
95        unsafe { free(ptr) };
96    }
97}

99/// Layout of an object with `len` words; `None` beyond `isize::MAX` bytes.
100fn obj_layout(len: usize) -> Option<Layout> {
101    let words = Layout::array::<i64>(len).ok()?;
102    Some(Layout::new::<BigIntObj>().extend(words).ok()?.0)
103}

122    /// Take ownership of a canonical multiword object without an extra reference.
123    #[inline]
124    fn from_obj(obj: *mut BigIntObj) -> Self {
125        Self {
126            data: TVMFFIAny {
127                type_index: TypeIndex::kTVMFFIBigInt as i32,
128                small_str_len: 0,
129                data_union: TVMFFIAnyDataUnion {
130                    v_obj: obj.cast::<TVMFFIObject>(),
131                },
132            },
133        }
134    }

136    /// Create from two's-complement `i64` words, least-significant word first.
137    ///
138    /// Redundant sign-extension words are pruned and a value that fits `i64`
139    /// is stored inline; an empty slice is zero.
140    #[doc(hidden)]
141    #[inline]
142    pub fn from_words(words: &[i64]) -> Self {
143        let size = int_ops::normalized_len(words);
144        if size <= 1 {
145            return Self::from_i64(words.first().copied().unwrap_or(0));
146        }
147        // Allocated at its final length, so the crate's generic extra-items deleter fits.
148        unsafe {
149            let mut obj = ObjectArc::new_with_extra_items(BigIntObj {
150                object: Object::new(),
151                size,
152            });
153            let items = BigIntObj::extra_items_ptr(ObjectArc::as_mut_ptr(&mut obj));
154            std::ptr::copy_nonoverlapping(words.as_ptr(), items, size);
155            Self::from_obj(ObjectArc::into_raw(obj) as *mut BigIntObj)
156        }
157    }
158
159    /// Build from `len` words, word `i` being `word(i)` called in increasing order,
160    /// written straight into a fresh object whose redundant sign words are then
161    /// pruned in place; `None` when the object cannot be allocated.
162    #[inline]
163    fn try_build(len: usize, mut word: impl FnMut(usize) -> i64) -> Option<Self> {
164        let layout = obj_layout(len)?;
165        unsafe {
166            let obj = malloc(layout.size()).cast::<BigIntObj>();
167            if obj.is_null() {
168                return None;
169            }
170            let data = obj.add(1).cast::<i64>();
171            for i in 0..len {
172                data.add(i).write(word(i));
173            }
174            let size = int_ops::normalized_len(std::slice::from_raw_parts(data, len));
175            if size <= 1 {
176                let value = if size == 1 { *data } else { 0 };
177                free(obj.cast());
178                return Some(Self::from_i64(value));
179            }
180            obj.cast::<TVMFFIObject>().write(TVMFFIObject {
181                combined_ref_count: AtomicU64::new(COMBINED_REF_COUNT_BOTH_ONE),
182                type_index: TypeIndex::kTVMFFIBigInt as i32,
183                __padding: 0,
184                deleter: Some(delete_obj),
185            });
186            std::ptr::addr_of_mut!((*obj).size).write(size);
187            Some(Self::from_obj(obj))
188        }
189    }
190
191    /// Like `try_build`, but like `Vec`, exhaustion aborts.
192    #[inline]
193    fn build(len: usize, word: impl FnMut(usize) -> i64) -> Self {
194        Self::try_build(len, word).unwrap_or_else(|| {
195            std::alloc::handle_alloc_error(obj_layout(len).unwrap_or(Layout::new::<BigIntObj>()))
196        })
197    }

199    /// Borrow the canonical two's-complement words, least-significant word first.
200    ///
201    /// The slice is never empty and its last word carries the sign.
202    #[doc(hidden)]
203    pub fn words(&self) -> &[i64] {
204        unsafe {
205            if self.data.type_index == TypeIndex::kTVMFFIInt as i32 {
206                std::slice::from_ref(&self.data.data_union.v_int64)
207            } else {
208                BigIntObj::extra_items(self.data.data_union.v_obj as *const BigIntObj)
209            }
210        }
211    }

224    /// Return the value when it fits `i64`.
225    ///
226    /// See also `i64::try_from(&big)`, which returns an `OverflowError` instead.
227    #[inline]
228    pub fn to_i64(&self) -> Option<i64> {
229        if self.data.type_index == TypeIndex::kTVMFFIInt as i32 {
230            Some(unsafe { self.data.data_union.v_int64 })
231        } else {
232            None
233        }
234    }

236    /// Return the value when it fits `u64`.
237    pub fn to_u64(&self) -> Option<u64> {
238        match *self.words() {
239            [w0] => u64::try_from(w0).ok(),
240            [w0, 0] => Some(w0 as u64),
241            _ => None,
242        }
243    }
244
245    /// Return the value when it fits `i128`.
246    pub fn to_i128(&self) -> Option<i128> {
247        match *self.words() {
248            [w0] => Some(i128::from(w0)),
249            [w0, w1] => Some((i128::from(w1) << 64) | i128::from(w0 as u64)),
250            _ => None,
251        }
252    }
253
254    /// Return the value when it fits `u128`.
255    pub fn to_u128(&self) -> Option<u128> {
256        let combine = |w0: i64, w1: i64| (u128::from(w1 as u64) << 64) | u128::from(w0 as u64);
257        match *self.words() {
258            [w0] => u128::try_from(w0).ok(),
259            [w0, w1] if w1 >= 0 => Some(combine(w0, w1)),
260            [w0, w1, 0] => Some(combine(w0, w1)),
261            _ => None,
262        }
263    }

265    /// Convert to the nearest finite double, rounding ties to even.
266    ///
267    /// Raises `OverflowError` when the value rounds beyond the finite range.
268    pub fn to_f64(&self) -> Result<f64> {
269        match self.to_i64() {
270            Some(value) => Ok(value as f64),
271            None => int_ops::to_f64(self.words()),
272        }
273    }

285    /// Truncating quotient and remainder in one pass; raises `ZeroDivisionError`.
286    pub fn div_rem(&self, rhs: &BigInt) -> Result<(BigInt, BigInt)> {
287        if let (Some(x), Some(y)) = (self.to_i64(), rhs.to_i64())
288            && let Some(q) = int_ops::trunc_div_i64(x, y)?
289        {
290            return Ok((Self::from_i64(q), Self::from_i64(x.wrapping_rem(y))));
291        }
292        int_ops::div_rem(self.words(), rhs.words())
293    }

399impl TryFrom<f64> for BigInt {
400    type Error = Error;
401
402    /// Convert a finite double by truncating toward zero.
403    ///
404    /// NaN raises `ValueError` and infinity raises `OverflowError`.
405    fn try_from(value: f64) -> Result<Self> {
406        // The upper bound is exclusive: i64::MAX rounds up to 2^63 as a double.
407        const LOWER: f64 = i64::MIN as f64;
408        if (LOWER..-LOWER).contains(&value) {
409            return Ok(Self::from_i64(value as i64));
410        }
411        if value.is_nan() {
412            return Err(Error::new(VALUE_ERROR, "Cannot convert NaN to BigInt"));
413        }
414        if value.is_infinite() {
415            return Err(Error::new(
416                OVERFLOW_ERROR,
417                "Cannot convert infinity to BigInt",
418            ));
419        }
420        // |value| >= 2^63 is an integer: shift the 53-bit significand to its binary exponent.
421        let bits = value.to_bits();
422        let exponent = ((bits >> 52) & 0x7ff) as i64 - 1075;
423        let significand = ((bits & ((1u64 << 52) - 1)) | (1u64 << 52)) as i64;
424        let magnitude = Self::from_i64(significand) << exponent;
425        Ok(if value < 0.0 { -magnitude } else { magnitude })
426    }
427}

429impl FromStr for BigInt {
430    type Err = Error;
431
432    /// Parse an optionally signed decimal integer; anything else raises `ValueError`.
433    fn from_str(text: &str) -> Result<Self> {
434        if let Ok(value) = text.parse::<i64>() {
435            return Ok(Self::from_i64(value));
436        }
437        let (negative, digits) = match text.as_bytes().first() {
438            Some(b'-') => (true, &text[1..]),
439            Some(b'+') => (false, &text[1..]),
440            _ => (false, text),
441        };
442        if digits.is_empty() || !digits.bytes().all(|byte| byte.is_ascii_digit()) {
443            return Err(Error::new(VALUE_ERROR, "Invalid decimal string for BigInt"));
444        }
445        // Accumulate 19-digit chunks: magnitude = magnitude * 10^len + chunk.
446        let mut magnitude = Vec::new();
447        for chunk in digits.as_bytes().chunks(19) {
448            let value = chunk
449                .iter()
450                .fold(0u64, |acc, &digit| acc * 10 + u64::from(digit - b'0'));
451            int_ops::mul_add_small(&mut magnitude, 10u64.pow(chunk.len() as u32), value);
452        }
453        let mut words: Vec<i64> = magnitude.into_iter().map(|word| word as i64).collect();
454        words.push(0);
455        if negative {
456            int_ops::negate_in_place(&mut words);
457        }
458        Ok(Self::from_words(&words))
459    }
460}

498impl Ord for BigInt {
499    fn cmp(&self, other: &Self) -> Ordering {
500        let (a, b) = (self.words(), other.words());
501        let (negative_a, negative_b) = (int_ops::is_negative(a), int_ops::is_negative(b));
502        if negative_a != negative_b {
503            return if negative_a {
504                Ordering::Less
505            } else {
506                Ordering::Greater
507            };
508        }
509        // With equal signs, sign-extended words compare as unsigned from the top.
510        let extension = int_ops::extension(a);
511        (0..a.len().max(b.len()))
512            .rev()
513            .map(|i| int_ops::word_at(a, i, extension).cmp(&int_ops::word_at(b, i, extension)))
514            .find(|ordering| ordering.is_ne())
515            .unwrap_or(Ordering::Equal)
516    }
517}

526impl Display for BigInt {
527    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
528        if let Some(value) = self.to_i64() {
529            return Display::fmt(&value, f);
530        }
531        let words = self.words();
532        let negative = words[words.len() - 1] < 0;
533        let mut magnitude = words.to_vec();
534        if negative {
535            int_ops::negate_in_place(&mut magnitude);
536        }
537        // Peel base-10^19 chunks, least significant first, by long division.
538        const CHUNK: u128 = 10_000_000_000_000_000_000;
539        let mut chunks = Vec::new();
540        while let Some(top) = magnitude.iter().rposition(|&word| word != 0) {
541            magnitude.truncate(top + 1);
542            let mut remainder = 0u128;
543            for word in magnitude.iter_mut().rev() {
544                let current = (remainder << 64) | u128::from(*word as u64);
545                *word = (current / CHUNK) as i64;
546                remainder = current % CHUNK;
547            }
548            chunks.push(remainder as u64);
549        }
550        let mut digits = String::new();
551        let mut chunks = chunks.iter().rev();
552        // A heap value is never zero, so the leading chunk always exists.
553        write!(digits, "{}", chunks.next().unwrap_or(&0))?;
554        for chunk in chunks {
555            write!(digits, "{chunk:019}")?;
556        }
557        f.pad_integral(!negative, "", &digits)
558    }
559}

569// ---- Arithmetic: an inline fast path, then the word fallbacks.
570#[inline]
571fn add_impl(a: &BigInt, b: &BigInt) -> BigInt {
572    if let (Some(x), Some(y)) = (a.to_i64(), b.to_i64())
573        && let Some(result) = x.checked_add(y)
574    {
575        return BigInt::from_i64(result);
576    }
577    int_ops::add(a.words(), b.words())
578}

610#[inline]
611fn rem_impl(a: &BigInt, b: &BigInt) -> Result<BigInt> {
612    if let (Some(x), Some(y)) = (a.to_i64(), b.to_i64())
613        && let Some(result) = int_ops::trunc_mod_i64(x, y)?
614    {
615        return Ok(BigInt::from_i64(result));

698/// The operator form of a fallible operation panics like the primitive one would.
699macro_rules! define_op_or_panic {
700    ($($name:ident => $imp:ident),* $(,)?) => {
701        $(
702            #[inline]
703            fn $name(a: &BigInt, b: &BigInt) -> BigInt {
704                $imp(a, b).unwrap_or_else(|error| panic!("{}", error.message()))
705            }
706        )*
707    };
708}

711    div_or_panic => div_impl,
712    rem_or_panic => rem_impl,
713    shl_or_panic => shl_impl,
714    shr_or_panic => shr_impl,
715);

717/// Forward an operator to its implementation for owned and borrowed operands,
718/// with `i64` accepted on either side.
719macro_rules! forward_binop {
720    ($Trait:ident::$method:ident, $Assign:ident::$assign:ident, $imp:ident) => {
721        impl $Trait<&BigInt> for &BigInt {
722            type Output = BigInt;
723            #[inline]
724            fn $method(self, rhs: &BigInt) -> BigInt {
725                $imp(self, rhs)
726            }
727        }
728        impl $Trait<BigInt> for &BigInt {
729            type Output = BigInt;
730            #[inline]
731            fn $method(self, rhs: BigInt) -> BigInt {
732                $imp(self, &rhs)
733            }
734        }
735        impl $Trait<&BigInt> for BigInt {
736            type Output = BigInt;
737            #[inline]
738            fn $method(self, rhs: &BigInt) -> BigInt {
739                $imp(&self, rhs)
740            }
741        }
742        impl $Trait<BigInt> for BigInt {
743            type Output = BigInt;
744            #[inline]
745            fn $method(self, rhs: BigInt) -> BigInt {
746                $imp(&self, &rhs)
747            }
748        }
749        impl $Trait<i64> for &BigInt {
750            type Output = BigInt;
751            #[inline]
752            fn $method(self, rhs: i64) -> BigInt {
753                $imp(self, &BigInt::from_i64(rhs))
754            }
755        }
756        impl $Trait<i64> for BigInt {
757            type Output = BigInt;
758            #[inline]
759            fn $method(self, rhs: i64) -> BigInt {
760                $imp(&self, &BigInt::from_i64(rhs))
761            }
762        }
763        impl $Trait<&BigInt> for i64 {
764            type Output = BigInt;
765            #[inline]
766            fn $method(self, rhs: &BigInt) -> BigInt {
767                $imp(&BigInt::from_i64(self), rhs)
768            }
769        }
770        impl $Trait<BigInt> for i64 {
771            type Output = BigInt;
772            #[inline]
773            fn $method(self, rhs: BigInt) -> BigInt {
774                $imp(&BigInt::from_i64(self), &rhs)
775            }
776        }
777        impl $Assign<&BigInt> for BigInt {
778            #[inline]
779            fn $assign(&mut self, rhs: &BigInt) {
780                *self = $imp(self, rhs);
781            }
782        }
783        impl $Assign<BigInt> for BigInt {
784            #[inline]
785            fn $assign(&mut self, rhs: BigInt) {
786                *self = $imp(self, &rhs);
787            }
788        }
789        impl $Assign<i64> for BigInt {
790            #[inline]
791            fn $assign(&mut self, rhs: i64) {
792                *self = $imp(self, &BigInt::from_i64(rhs));
793            }
794        }
795    };
796}

841// ---- AnyCompatible.
842unsafe impl AnyCompatible for BigInt {
843    fn type_str() -> std::string::String {
844        "BigInt".to_string()
845    }
846
847    // A big integer is an inline integer or a big integer object, never None.
848    #[inline]
849    unsafe fn is_undefined(field: *const Self) -> bool {
850        unsafe { (*std::ptr::addr_of!((*field).data.type_index)) == TypeIndex::kTVMFFINone as i32 }
851    }
852
853    fn type_schema() -> std::string::String {
854        crate::type_traits::type_schema(<BigIntObj as crate::ObjectCore>::TYPE_KEY, &[])
855    }
856
857    unsafe fn copy_to_any_view(this: &Self, data: &mut TVMFFIAny) {
858        *data = this.data;
859    }
860
861    unsafe fn move_to_any(src: Self, data: &mut TVMFFIAny) {
862        *data = src.data;
863        std::mem::forget(src);
864    }
865
866    unsafe fn check_any_strict(data: &TVMFFIAny) -> bool {
867        data.type_index == TypeIndex::kTVMFFIInt as i32
868            || data.type_index == TypeIndex::kTVMFFIBigInt as i32
869    }
870
871    unsafe fn copy_from_any_view_after_check(data: &TVMFFIAny) -> Self {
872        if data.type_index == TypeIndex::kTVMFFIBigInt as i32 {
873            unsafe { unsafe_::inc_ref(data.data_union.v_obj) }
874        }
875        Self { data: *data }
876    }
877
878    unsafe fn move_from_any_after_check(data: &mut TVMFFIAny) -> Self {
879        Self { data: *data }
880    }
881
882    unsafe fn try_cast_from_any_view(data: &TVMFFIAny) -> std::result::Result<Self, ()> {
883        if unsafe { Self::check_any_strict(data) } {
884            Ok(unsafe { Self::copy_from_any_view_after_check(data) })
885        } else if data.type_index == TypeIndex::kTVMFFIBool as i32 {
886            Ok(unsafe { Self::from_i64(data.data_union.v_int64) })
887        } else {
888            Err(())
889        }
890    }
