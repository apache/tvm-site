20//! Enums shared across languages: each value is a named singleton of an
21//! object type whose parent is [`EnumObj`]. [`EnumRef`] looks values up;
22//! [`reflection::EnumDef`](crate::reflection::EnumDef) registers them.
23
24use crate::derive::Object;
25use crate::object::{Object, ObjectCore, ObjectRefCore};
26use crate::reflection::enum_def::{any_into, dict_get, enum_state};
27use crate::reflection::EnumAttrMap;
28use crate::type_traits::AnyCompatible;
29use crate::{AnyView, Error, Result, String, VALUE_ERROR};
30
31/// The object behind every enum value: its name and the index the registry
32/// assigned it. An enum type's object embeds it as its parent,
33/// `base_: EnumObj`.
34#[repr(C)]
35#[derive(Object)]
36#[type_key = "ffi.Enum"]
37#[type_immutable]
38pub struct EnumObj {
39    base_: Object,
40    // The registration order, which can differ between processes, so values
41    // are identified by name only.
42    #[allow(dead_code)]
43    index: i64,
44    name: String,
45}
46
47impl EnumObj {
48    pub(crate) fn new(index: i64, name: &str) -> Self {
49        Self {
50            base_: Object::new(),
51            index,
52            name: String::from(name),
53        }
54    }
55}

57/// A reference to an enum value: one named singleton of an enum type,
58/// registered once and shared with every language.
59///
60/// [`get`](Self::get) looks a value up by name. Each name has one value, so
61/// equality is identity: compare values with
62/// [`same_as`](ObjectRefCore::same_as).
63/// [`EnumDef`](crate::reflection::EnumDef) registers values and their
64/// attributes, and [`attr_map`](Self::attr_map) reads an attribute.
65///
66/// The stub generator implements it for the enum types it binds. A
67/// hand-written reference implements it when its object's parent is
68/// [`EnumObj`].
69///
70/// # Examples
71///
72/// ```
73/// use tvm_ffi::derive::{Object, ObjectRef};
74/// use tvm_ffi::reflection::EnumDef;
75/// use tvm_ffi::{EnumRef, ObjectArc, ObjectRefCore};
76/// use tvm_ffi::enum_ref::EnumObj;
77///
78/// #[repr(C)]
79/// #[derive(Object)]
80/// #[type_key = "my_ext.Color"]
81/// #[type_register]
82/// #[type_final]
83/// #[type_immutable]
84/// pub struct ColorObj {
85///     base_: EnumObj,
86/// }
87///
88/// #[derive(Clone, ObjectRef)]
89/// pub struct Color {
90///     data: ObjectArc<ColorObj>,
91/// }
92///
93/// impl EnumRef for Color {}
94///
95/// EnumDef::<Color>::new("red")?;
96///
97/// let red = Color::get("red")?;
98/// assert_eq!(red.name(), "red");
99/// assert!(red.same_as(&Color::get("red")?));
100/// # Ok::<(), tvm_ffi::Error>(())
101/// ```
102pub trait EnumRef: ObjectRefCore {
103    /// Look up the value registered as `name`.
104    ///
105    /// # Arguments
106    ///
107    /// * `name` - The name of the value.
108    ///
109    /// # Returns
110    ///
111    /// The value, the same object on every call.
112    ///
113    /// # Errors
114    ///
115    /// Returns a `ValueError` if the enum type has no value named `name`.
116    fn get(name: &str) -> Result<Self> {
117        let key = String::from(name);
118        let found = match enum_state(Self::ContainerType::type_index())? {
119            Some(state) => dict_get(AnyView::from(&state.indexes), AnyView::from(&key))?,
120            None => None,
121        };
122        let Some(value) = found else {
123            return Err(Error::new(
124                VALUE_ERROR,
125                &format!(
126                    "Enum `{}` has no value named `{name}`",
127                    Self::ContainerType::TYPE_KEY
128                ),
129                "",
130            ));
131        };
132        any_into(value)
133    }
134
135    /// The name of the value, unique within its enum type.
136    ///
137    /// # Panics
138    ///
139    /// Panics if the object of `Self` does not have [`EnumObj`] as an
140    /// ancestor, which only an incorrect `impl EnumRef` allows.
141    fn name(&self) -> &str {
142        let object = self
143            .downcast_ref::<EnumObj>()
144            .expect("an `EnumRef` object has `EnumObj` as an ancestor");
145        object.name.as_str()
146    }
147
148    /// The attribute `name` of the values of the enum type, or `None` when no
149    /// value has it.
150    ///
151    /// The map holds the registry's column, so it also sees the attributes set
152    /// after it was read. See also
153    /// [`EnumDef::set_attr`](crate::reflection::EnumDef::set_attr).
154    ///
155    /// # Arguments
156    ///
157    /// * `name` - The name of the attribute.
158    ///
159    /// # Returns
160    ///
161    /// The attribute of every value that has it, read as `V`.
162    fn attr_map<V: AnyCompatible>(name: &str) -> Result<Option<EnumAttrMap<Self, V>>> {
163        EnumAttrMap::lookup(name)
164    }
165}
