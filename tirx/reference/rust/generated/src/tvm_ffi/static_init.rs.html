20//! Static init blocks: code that runs when the program or library that holds it loads. The
21//! blocks of a program or library are gathered at link time, wherever in it they are, and a
22//! constructor runs them when it loads. Once Python loads a library built on this crate, by any
23//! means, it sees what the blocks registered, such as global functions. A block that several
24//! libraries of a process have, as each links its own copy of a crate they share, runs once. See
25//! [`static_init_block!`](crate::static_init_block).

37/// Run code when the program or library that holds it loads, the Rust form of
38/// `TVM_FFI_STATIC_INIT_BLOCK`. The body is the body of a function that returns
39/// [`Result<()>`](crate::Result): it registers what the library defines, such as global functions,
40/// and stops at its first error, which `?` returns. A body whose statements all end in `;` returns
41/// `Ok(())`; one that ends in an expression returns it, a `Result<()>`.
42///
43/// ```
44/// use tvm_ffi::{Function, Result};
45///
46/// fn add_one(x: i64) -> Result<i64> {
47///     Ok(x + 1)
48/// }
49///
50/// tvm_ffi::static_init_block! {
51///     Function::register_global_typed("my_ext.add_one", add_one, "Add one to an integer.")?;
52/// }
53///
54/// // Registered when the program loaded.
55/// let add_one = Function::get_global("my_ext.add_one")?;
56/// assert_eq!(i64::try_from(add_one.call_tuple((1i64,))?)?, 2);
57/// # Ok::<(), tvm_ffi::Error>(())
58/// ```
59///
60/// A block runs when its program or library loads: it adds a constructor, which the loader runs,
61/// and the first constructor runs every block of the program or library, in the order of their
62/// modules and lines. Python sees what the blocks of a library registered once it loads the
63/// library, by any means, and the library stays loaded for the process, as the registrations keep
64/// its code. An error cannot reach a caller when the library loads: it is printed, and
65/// [`run_static_init_blocks`](crate::run_static_init_blocks) reports it again. Where the loader
66/// does not run the constructor, call [`run_static_init_blocks`](crate::run_static_init_blocks)
67/// first. A crate that a library depends on but never names is not linked into it, nor are its
68/// blocks: name it, as `use my_crate as _;` does.
69///
70/// A block runs once per process. Every library built on the crate links its own copy of the
71/// crates it uses, so a block of a crate that several libraries link is in each of them; the first
72/// library that runs it registers what it defines, and the others skip it. A block of another
73/// version of the crate, or whose text differs, is another block, and registering what it defines
74/// again is an error. A change to the body of a function the block calls does not make another
75/// block: rebuild every library that links a crate after changing it.
76#[macro_export]
77macro_rules! static_init_block {
78    ($($body:tt)*) => {
79        const _: () = {
80            #[$crate::static_init::linkme::distributed_slice($crate::static_init::STATIC_INIT_BLOCKS)]
81            #[linkme(crate = $crate::static_init::linkme)]
82            static BLOCK: $crate::static_init::StaticInitBlock =
83                $crate::static_init::StaticInitBlock::__new(
84                    module_path!(),
85                    line!(),
86                    match option_env!("CARGO_PKG_VERSION") {
87                        Some(version) => version,
88                        None => "",
89                    },
90                    stringify!($($body)*),
91                    __tvm_ffi_static_init_block_body,
92                );
93            // A function item rather than a closure, to carry the attribute: a body that diverges,
94            // such as one that panics, makes the conversion of its value unreachable.
95            #[allow(unreachable_code)]
96            fn __tvm_ffi_static_init_block_body() -> $crate::Result<()> {
97                $crate::static_init::IntoStaticInitResult::into_static_init_result({ $($body)* })
98            }
99            #[used]
100            #[cfg_attr(target_os = "linux", link_section = ".init_array")]
101            #[cfg_attr(target_os = "macos", link_section = "__DATA,__mod_init_func")]
102            #[cfg_attr(windows, link_section = ".CRT$XCU")]
103            static RUN_AT_LOAD: extern "C" fn() = $crate::static_init::__run_at_load;
104        };
105    };
106}

108/// The value of the body of a [`static_init_block!`](crate::static_init_block), converted to the
109/// outcome of the block: `()`, of a body whose statements all end in `;`, is `Ok(())`, a
110/// [`Result<()>`](crate::Result) is itself, and a body that diverges, as one that panics or ends
111/// in `return`, has no value to convert.
112#[doc(hidden)]
113#[diagnostic::on_unimplemented(
114    message = "a static init block returns `Result<()>`, not `{Self}`",
115    label = "the value of the block",
116    note = "end the last statement of the block with `;`, or return a `Result<()>`"
117)]
118pub trait IntoStaticInitResult {
119    fn into_static_init_result(self) -> Result<()>;
120}
121
122impl IntoStaticInitResult for () {
123    fn into_static_init_result(self) -> Result<()> {
124        Ok(())
125    }
126}
127
128impl IntoStaticInitResult for Result<()> {
129    fn into_static_init_result(self) -> Result<()> {
130        self
131    }
132}

134/// The never type `!`, which stable Rust names only as the return type of a function pointer.
135/// The value of a body that diverges is `!`, or `()` where an edition before 2024 falls back to it.
136#[doc(hidden)]
137pub trait FnOutput {
138    type Output;
139}

142    type Output = T;
143}
144
145#[doc(hidden)]
146pub type Never = <fn() -> ! as FnOutput>::Output;

148impl IntoStaticInitResult for Never {
149    fn into_static_init_result(self) -> Result<()> {
150        self
151    }

154/// A [`static_init_block!`](crate::static_init_block), gathered at link time.
155#[doc(hidden)]
156pub struct StaticInitBlock {
157    /// The module and line of the block.
158    module: &'static str,
159    line: u32,
160    /// The version of the block's crate and a hash of the block's text, which with the module and
161    /// line identify the block in every library that links the crate.
162    version: &'static str,
163    text_hash: u64,
164    body: fn() -> Result<()>,
165    /// The outcome of the block, which runs once: the error's text if it failed.
166    outcome: OnceLock<std::result::Result<(), std::string::String>>,
167}
168
169/// The [`static_init_block!`](crate::static_init_block)s of the program or library, gathered at
170/// link time from the whole of it.
171#[doc(hidden)]
172#[linkme::distributed_slice]
173pub static STATIC_INIT_BLOCKS: [StaticInitBlock];
174
175impl StaticInitBlock {
176    #[doc(hidden)]
177    pub const fn __new(
178        module: &'static str,
179        line: u32,
180        version: &'static str,
181        text: &str,
182        body: fn() -> Result<()>,
183    ) -> Self {
184        let text_hash = text_hash(text);
185        Self {
186            module,
187            line,
188            version,
189            text_hash,
190            body,
191            outcome: OnceLock::new(),
192        }
193    }
194
195    /// The identity of the block in the process, `module:line@version#hash`: the same in every
196    /// library that links the block's crate, and another one if the crate version or the block's
197    /// text differs.
198    #[doc(hidden)]
199    pub fn identity(&self) -> std::string::String {
200        format!(
201            "{}:{}@{}#{:016x}",
202            self.module, self.line, self.version, self.text_hash
203        )
204    }
205
206    /// The name of the global function that marks the block run in the process,
207    /// `ffi.static_init_block.<identity>`. The libraries of a process share the global function
208    /// table, and their statics are their own.
209    #[doc(hidden)]
210    pub fn marker(&self) -> std::string::String {
211        format!("ffi.static_init_block.{}", self.identity())
212    }
213
214    /// Run the block if it has not run. The caller that runs a failing block gets its error;
215    /// later ones get an error that names the block.
216    #[doc(hidden)]
217    pub fn run(&self) -> Result<()> {
218        let mut failure = None;
219        let outcome = self.outcome.get_or_init(|| {
220            self.run_once().map_err(|error| {
221                let text = format!("{}: {}", error.kind(), error.message());
222                failure = Some(error);
223                text
224            })
225        });
226        match (failure, outcome) {
227            (Some(error), _) => Err(error),
228            (None, Ok(())) => Ok(()),
229            (None, Err(text)) => {
230                let message = format!("The static init block at {} failed: {text}", self.place());
231                Err(Error::new(INTERNAL_ERROR, &message, ""))
232            }
233        }
234    }
235
236    /// Run the block unless another library of the process has run it. Every library links its
237    /// own copy of the crates it is built from, so the same block can be in several, and what it
238    /// defines is registered once, by the first library that runs it. The block is marked once it
239    /// succeeds, so that one that fails fails in every library. A panic in the body is its error.
240    fn run_once(&self) -> Result<()> {
241        let marker = self.marker();
242        if Function::get_global(&marker).is_err() {
243            catch_unwind(self.body).unwrap_or_else(|payload| Err(panic_to_error(payload)))?;
244            Function::register_global(&marker, Function::from_packed(|_| Ok(Any::default())))?;
245        }
246        Ok(())
247    }
248
249    /// Where the block is, for an error.
250    fn place(&self) -> std::string::String {
251        format!("{}:{}", self.module, self.line)
252    }
253}

255/// The FNV-1a hash of `text` without its whitespace, which does not depend on how the compiler
256/// spaces the tokens of a block it stringifies.
257const fn text_hash(text: &str) -> u64 {
258    let bytes = text.as_bytes();
259    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
260    let mut i = 0;
261    while i < bytes.len() {
262        if !bytes[i].is_ascii_whitespace() {
263            hash = (hash ^ bytes[i] as u64).wrapping_mul(0x0100_0000_01b3);
264        }
265        i += 1;
266    }
267    hash
268}

270/// Run every block of this program or library that has not run, in the order of their modules
271/// and lines, and return those that failed, each with its first error.
272fn run_blocks() -> Vec<(&'static StaticInitBlock, Error)> {
273    let mut blocks: Vec<&StaticInitBlock> = STATIC_INIT_BLOCKS.iter().collect();
274    blocks.sort_by_key(|block| (block.module, block.line));
275    blocks
276        .into_iter()
277        .filter_map(|block| block.run().err().map(|e| (block, e)))
278        .collect()
279}

281/// Run every [`static_init_block!`](crate::static_init_block) of this program or library that has
282/// not run, in the order of their modules and lines; running again does nothing. Every block is
283/// tried, and the error lists those that failed. A block that another library of the process has
284/// run, from the same version of its crate and the same text, is skipped: libraries that link the
285/// same crate register what its blocks define once.
286///
287/// The blocks run when their program or library loads, so this reports the errors of those that
288/// failed, which could not reach a caller then: a library exports it, with
289/// [`tvm_ffi_dll_export_typed_func!`](crate::tvm_ffi_dll_export_typed_func), for its loader to
290/// call. Where the loader does not run the blocks, call it before using what they define.
291pub fn run_static_init_blocks() -> Result<()> {
292    let mut failures = run_blocks();
293    if failures.len() <= 1 {
294        return failures.pop().map_or(Ok(()), |(_, error)| Err(error));
295    }
296    let mut message = format!("{} static init blocks failed:", failures.len());
297    for (block, error) in &failures {
298        message.push_str(&format!(
299            "\n  {}: {}: {}",
300            block.place(),
301            error.kind(),
302            error.message()
303        ));
304    }
305    Err(Error::new(failures[0].1.kind(), &message, ""))
306}

308/// The constructor of every [`static_init_block!`](crate::static_init_block), which the loader
309/// runs when the program or library loads: the first one keeps the library loaded and runs every
310/// block of it. An error cannot reach a caller here, so it is printed; [`run_static_init_blocks`]
311/// reports it again.
312#[doc(hidden)]
313pub extern "C" fn __run_at_load() {
314    static LOADED: Once = Once::new();
315    LOADED.call_once(|| {
316        let library = keep_loaded().unwrap_or_else(|| "static_init_block!".to_string());
317        match catch_unwind(run_blocks) {
318            Ok(failures) => {
319                for (block, error) in failures {
320                    eprintln!(
321                        "{library}: the static init block at {} failed: {}: {}",
322                        block.place(),
323                        error.kind(),
324                        error.message()
325                    );
326                }
327            }
328            Err(_) => eprintln!("{library}: running the static init blocks panicked"),
329        }
330    });
331}

333/// Keep the library that holds this code loaded for the rest of the process, as what its blocks
334/// register keeps its code: unloading it, as `tvm_ffi.load_module` does when Python drops the
335/// module, would leave the registrations dangling. Returns the path of the library.
336#[cfg(any(target_os = "linux", target_os = "macos"))]
337fn keep_loaded() -> Option<std::string::String> {
338    use std::ffi::{c_char, c_int, CStr};
339
340    #[repr(C)]
341    struct DlInfo {
342        fname: *const c_char,
343        fbase: *mut c_void,
344        sname: *const c_char,
345        saddr: *mut c_void,
346    }
347    extern "C" {
348        fn dladdr(addr: *const c_void, info: *mut DlInfo) -> c_int;
349        fn dlopen(filename: *const c_char, flags: c_int) -> *mut c_void;
350    }
351    // RTLD_LAZY | RTLD_NOLOAD | RTLD_NODELETE.
352    #[cfg(target_os = "linux")]
353    const FLAGS: c_int = 0x1 | 0x4 | 0x1000;
354    #[cfg(target_os = "macos")]
355    const FLAGS: c_int = 0x1 | 0x10 | 0x80;
356
357    let null = std::ptr::null_mut();
358    let mut info = DlInfo {
359        fname: null as _,
360        fbase: null,
361        sname: null as _,
362        saddr: null,
363    };
364    // SAFETY: `dladdr` fills `info` for an address of this library, whose path stays valid while
365    // the library is loaded; `dlopen` of a loaded library only marks it, and its handle is kept.
366    unsafe {
367        if dladdr(keep_loaded as *const c_void, &mut info) == 0 || info.fname.is_null() {
368            return None;
369        }
370        dlopen(info.fname, FLAGS);
371        Some(CStr::from_ptr(info.fname).to_string_lossy().into_owned())
372    }
373}

375/// Keep the library that holds this code loaded for the rest of the process, as what its blocks
376/// register keeps its code. Returns the path of the library.
377#[cfg(windows)]
378fn keep_loaded() -> Option<std::string::String> {
379    use std::ffi::OsString;
380    use std::os::windows::ffi::OsStringExt;
381
382    type Module = *mut c_void;
383    #[link(name = "kernel32")]
384    extern "system" {
385        fn GetModuleHandleExW(flags: u32, module_name: *const u16, module: *mut Module) -> i32;
386        fn GetModuleFileNameW(module: Module, filename: *mut u16, size: u32) -> u32;
387    }
388    // GET_MODULE_HANDLE_EX_FLAG_PIN | GET_MODULE_HANDLE_EX_FLAG_FROM_ADDRESS: the module that
389    // holds the address stays loaded until the process exits.
390    const FLAGS: u32 = 0x1 | 0x4;
391
392    let mut module: Module = std::ptr::null_mut();
393    let mut path = vec![0u16; 32768];
394    // SAFETY: `GetModuleHandleExW` takes an address of this library as the name, and
395    // `GetModuleFileNameW` writes at most `path.len()` units into `path`.
396    unsafe {
397        let address = keep_loaded as *const c_void as *const u16;
398        if GetModuleHandleExW(FLAGS, address, &mut module) == 0 || module.is_null() {
399            return None;
400        }
401        let len = GetModuleFileNameW(module, path.as_mut_ptr(), path.len() as u32) as usize;
402        if len == 0 {
403            return None;
404        }
405        Some(
406            OsString::from_wide(&path[..len.min(path.len())])
407                .to_string_lossy()
408                .into_owned(),
409        )
410    }
411}
