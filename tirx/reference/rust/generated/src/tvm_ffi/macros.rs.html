27/// The path of the enclosing function, as `function_name!()`.
28#[doc(hidden)]
29#[macro_export]
30macro_rules! function_name {
31    () => {{
32        // dummy function to get the name of the function
33        fn f() {}
34        fn type_name_of<T>(_: T) -> &'static str {
35            std::any::type_name::<T>()
36        }
37        let name = type_name_of(f);
38        // remove the f() from the name
39        &name[..name.len() - 3]
40    }};
41}

43/// Look up a registered global [`Function`](crate::Function) by name once, and
44/// keep it for the process: a `&'static Function`.
45///
46/// Each call site has its own cache, which later calls read without a lock.
47/// Use it for a function that must be registered, such as a built-in one; it
48/// panics, naming the function, if the function is not. See also
49/// [`Function::get_global`](crate::Function::get_global), which returns an
50/// error instead.
51///
52/// # Examples
53///
54/// ```
55/// use tvm_ffi::{cached_global_func, Array};
56///
57/// let array = cached_global_func!("ffi.Array").call_tuple((1_i64, 2_i64))?;
58/// assert_eq!(Array::<i64>::try_from(array)?.len(), 2);
59/// # Ok::<(), tvm_ffi::Error>(())
60/// ```
61#[macro_export]
62macro_rules! cached_global_func {
63    ($name:literal) => {{
64        static FUNC: $crate::extra::cold_init_lazy_lock::ColdInitLazyLock<
65            $crate::function::Function,
66        > = $crate::extra::cold_init_lazy_lock::ColdInitLazyLock::new(|| {
67            $crate::function::Function::get_global($name).unwrap_or_else(|_| {
68                panic!(concat!("global function `", $name, "` is not registered"))
69            })
70        });
71        &*FUNC
72    }};
73}

75/// Turn the return code of a safe call into a `Result`, taking the raised
76/// error when it is nonzero: `check_safe_call!(function(args))?`.
77#[doc(hidden)]
78#[macro_export]
79macro_rules! check_safe_call {
80    ($expr:expr) => {{
81        let ret_code = $expr;
82        if ret_code == 0 {
83            Ok(())
84        } else {
85            let error = $crate::error::Error::from_raised();
86            Err(error)
87        }
88    }};
89}

91/// Return an [`Error`](crate::Error) of a kind with a formatted message from
92/// the current function.
93///
94/// `bail!(kind, fmt, args...)` formats the message as `format!` does and
95/// records the source location in the backtrace.
96///
97/// # Arguments
98///
99/// * `kind` - The kind of the error, such as [`VALUE_ERROR`](crate::VALUE_ERROR).
100/// * `fmt` - The format string of the message.
101/// * `args` - The format arguments.
102///
103/// # Returns
104///
105/// Returns `Err` of the error from the enclosing function.
106///
107/// # Examples
108///
109/// ```
110/// use tvm_ffi::{bail, Result, INDEX_ERROR};
111///
112/// fn get(items: &[i64], index: usize) -> Result<i64> {
113///     if index >= items.len() {
114///         bail!(INDEX_ERROR, "index {} out of range for length {}", index, items.len());
115///     }
116///     Ok(items[index])
117/// }
118///
119/// assert_eq!(get(&[1, 2], 1)?, 2);
120/// assert_eq!(get(&[1, 2], 5).unwrap_err().message(), "index 5 out of range for length 2");
121/// # Ok::<(), tvm_ffi::Error>(())
122/// ```
123#[macro_export]
124macro_rules! bail {
125    ($error_kind:expr, $fmt:expr $(, $args:expr)* $(,)?) => {{
126        let context = format!(
127            "  File \"{}\", line {}, in {}\n",
128            file!(),
129            line!(),
130            $crate::function_name!()
131        );
132        return Err($crate::error::Error::new($error_kind, &format!($fmt $(, $args)*), &context));
133    }};
134}

136/// Return an [`Error`](crate::Error) from the current function unless a
137/// condition holds.
138///
139/// `ensure!(cond, kind, fmt, args...)` is [`bail!`](crate::bail)`(kind, fmt,
140/// args...)` when `cond` is false.
141///
142/// # Arguments
143///
144/// * `cond` - The condition that must hold.
145/// * `kind` - The kind of the error.
146/// * `fmt` - The format string of the message.
147/// * `args` - The format arguments.
148///
149/// # Returns
150///
151/// Returns `Err` of the error from the enclosing function when `cond` is
152/// false, and does nothing otherwise.
153///
154/// # Examples
155///
156/// ```
157/// use tvm_ffi::{ensure, Result, VALUE_ERROR};
158///
159/// fn sqrt(x: f64) -> Result<f64> {
160///     ensure!(x >= 0.0, VALUE_ERROR, "sqrt of negative number {}", x);
161///     Ok(x.sqrt())
162/// }
163///
164/// assert_eq!(sqrt(4.0)?, 2.0);
165/// assert!(sqrt(-1.0).is_err());
166/// # Ok::<(), tvm_ffi::Error>(())
167/// ```
168#[macro_export]
169macro_rules! ensure {
170    ($cond:expr, $error_kind:expr, $fmt:expr $(, $args:expr)* $(,)?) => {{
171        if !$cond {
172            $crate::bail!($error_kind, $fmt $(, $args)*);
173        }
174    }};
175}

177/// Append the current source location to the backtrace of an `Err` result,
178/// and pass an `Ok` through.
179///
180/// # Arguments
181///
182/// * `result` - The result to attach the context to.
183///
184/// # Returns
185///
186/// The result, with the location appended to an error's backtrace.
187///
188/// # Examples
189///
190/// ```
191/// use tvm_ffi::{attach_context, bail, Result, RUNTIME_ERROR};
192///
193/// fn inner() -> Result<()> {
194///     bail!(RUNTIME_ERROR, "failed");
195/// }
196///
197/// fn outer() -> Result<()> {
198///     attach_context!(inner())
199/// }
200///
201/// assert!(outer().unwrap_err().backtrace().contains("outer"));
202/// ```
203#[macro_export]
204macro_rules! attach_context {
205    ($error:expr) => {{
206        match $error {
207            Ok(value) => Ok(value),
208            Err(error) => {
209                let context = format!(
210                    "  File \"{}\", line {}, in {}\n",
211                    file!(),
212                    line!(),
213                    $crate::function_name!()
214                );
215                Err($crate::error::Error::with_appended_backtrace(
216                    error, &context,
217                ))
218            }
219        }
220    }};
221}

227/// Implement zero-copy conversions from a derived object reference to one of
228/// its base object-reference types: `From<Source>` and `From<&Source>` into
229/// `Target`, `AsRef<Target>`, which borrows a `&Source` as `&Target`
230/// without touching the reference count, and
231/// [`IntoObjectRef<Target>`](crate::object::IntoObjectRef) for `Source` and `&Source`.
232/// It also implements the strict downcast back, `TryFrom<Target>` and
233/// `TryFrom<&Target>` for `Source`, which return a `TypeError` naming both types.
234///
235/// # Safety
236///
237/// The declaration is trusted, as an `unsafe impl` is, hence the `unsafe_`
238/// name: `target` must be a registered base of `source`, and every `source`
239/// must satisfy the target reference invariants. An incorrect declaration makes
240/// the generated safe conversions unsound. The Rust stub generator writes these
241/// from the registered type hierarchy.
242#[macro_export]
243macro_rules! unsafe_impl_object_ref_upcast {
244    ($($source:ty => $target:ty),+ $(,)?) => {
245        $(
246            impl ::std::convert::From<$source> for $target {
247                #[inline]
248                fn from(value: $source) -> Self {
249                    let data = <$source as $crate::object::ObjectRefCore>::into_data(value);
250                    // SAFETY: The macro declaration promises that `target` is
251                    // a registered base of `source`. Both references retain
252                    // the same allocation and only change its static view.
253                    let data = unsafe {
254                        $crate::object::ObjectArc::from_raw(
255                            $crate::object::ObjectArc::into_raw(data).cast::
256                                <<$target as $crate::object::ObjectRefCore>::ContainerType>(),
257                        )
258                    };
259                    // SAFETY: The macro declaration promises both the
260                    // container inheritance relation and every additional
261                    // invariant imposed by the target reference view.
262                    unsafe {
263                        <$target as $crate::object::ObjectRefCore>::from_data(data)
264                    }
265                }
266            }
267
268            impl ::std::convert::From<&$source> for $target {
269                #[inline]
270                fn from(value: &$source) -> Self {
271                    value.clone().into()
272                }
273            }
274
275            impl $crate::object::IntoObjectRef<$target> for $source {
276                type Output = $target;
277                #[inline(always)]
278                fn into_object_ref(self) -> $target {
279                    ::core::convert::Into::into(self)
280                }
281            }
282
283            impl<'a> $crate::object::IntoObjectRef<$target> for &'a $source {
284                type Output = &'a $target;
285                #[inline(always)]
286                fn into_object_ref(self) -> &'a $target {
287                    ::core::convert::AsRef::<$target>::as_ref(self)
288                }
289            }
290
291            impl ::core::convert::AsRef<$target> for $source {
292                #[inline(always)]
293                fn as_ref(&self) -> &$target {
294                    const _: () = assert!(
295                        ::core::mem::size_of::<$source>() == ::core::mem::size_of::<$target>()
296                            && ::core::mem::align_of::<$source>()
297                                == ::core::mem::align_of::<$target>(),
298                        "an object reference upcast must keep the reference's layout"
299                    );
300                    // SAFETY: both references are exactly one object pointer, and the
301                    // macro declaration promises that every `source` object is a valid
302                    // `target`, so a `&source` is a valid `&target`.
303                    unsafe { &*(self as *const $source as *const $target) }
304                }
305            }
306
307            // The strict downcast back, as a `TryFrom` for `?`: a reference
308            // never converts, so this only checks the object's runtime type.
309            impl ::core::convert::TryFrom<$target> for $source {
310                type Error = $crate::error::Error;
311                #[inline]
312                fn try_from(value: $target) -> ::core::result::Result<Self, Self::Error> {
313                    $crate::object::ObjectRefCore::downcast::<$source>(value)
314                        .map_err($crate::error::Error::downcast_failed::<$source>)
315                }
316            }
317
318            impl ::core::convert::TryFrom<&$target> for $source {
319                type Error = $crate::error::Error;
320                #[inline]
321                fn try_from(value: &$target) -> ::core::result::Result<Self, Self::Error> {
322                    match $crate::object::ObjectRefCore::downcast_ref::<$source>(value) {
323                        ::core::option::Option::Some(found) => {
324                            ::core::result::Result::Ok(::core::clone::Clone::clone(found))
325                        }
326                        ::core::option::Option::None => ::core::result::Result::Err(
327                            $crate::error::Error::downcast_failed::<$source>(
328                                ::core::clone::Clone::clone(value),
329                            ),
330                        ),
331                    }
332                }
333            }
334        )+
335    };
336}

338/// Implement the in-place `downcast_ref` borrows (`array[i]`, `Any::downcast_ref`,
339/// [`ObjectRefCore::downcast_ref`](crate::ObjectRefCore::downcast_ref)) for each
340/// listed object reference type. [`#[derive(ObjectRef)]`](crate::derive::ObjectRef)
341/// emits it; a reference that implements `ObjectRefCore` by hand calls it.
342///
343/// # Safety
344///
345/// Each type is `repr(transparent)` over one object pointer, and its
346/// [`AnyCompatible::check_any_strict`](crate::AnyCompatible::check_any_strict)
347/// accepts exactly the objects that are a valid value of it.
348///
349/// # Examples
350///
351/// ```
352/// use tvm_ffi::derive::ObjectRef;
353/// use tvm_ffi::{rs_object, unsafe_impl_object_ref_downcast_ref, Any};
354/// use tvm_ffi::{Object, ObjectArc, ObjectRefAnyCompatible, ObjectRefCore};
355///
356/// #[rs_object("my_ext.MyPair", final)]
357/// pub struct MyPairObj {
358///     base_: Object,
359///     pub first: i64,
360///     pub second: i64,
361/// }
362///
363/// #[derive(Clone, ObjectRef)]
364/// pub struct MyPair {
365///     data: ObjectArc<MyPairObj>,
366/// }
367///
368/// /// A `MyPair` whose `first` is positive.
369/// #[derive(Clone)]
370/// #[repr(transparent)]
371/// pub struct PositivePair(ObjectArc<MyPairObj>);
372///
373/// unsafe impl ObjectRefCore for PositivePair {
374///     type ContainerType = MyPairObj;
375///     fn data(this: &Self) -> &ObjectArc<MyPairObj> {
376///         &this.0
377///     }
378///     fn into_data(this: Self) -> ObjectArc<MyPairObj> {
379///         this.0
380///     }
381///     unsafe fn from_data(data: ObjectArc<MyPairObj>) -> Self {
382///         Self(data)
383///     }
384/// }
385///
386/// unsafe impl ObjectRefAnyCompatible for PositivePair {
387///     fn extra_check(pair: &MyPairObj) -> bool {
388///         pair.first > 0
389///     }
390/// }
391/// unsafe_impl_object_ref_downcast_ref!(PositivePair);
392///
393/// let pair = Any::from(MyPair::from_object(MyPairObj::new(Object::new(), 1, 2)));
394/// assert!(pair.downcast_ref::<PositivePair>().is_some());
395/// ```
396#[macro_export]
397macro_rules! unsafe_impl_object_ref_downcast_ref {
398    ($($name:ty),+ $(,)?) => {
399        $(
400            // SAFETY: the macro declaration promises that `$name` is one object
401            // pointer whose `check_any_strict` is its type check.
402            unsafe impl $crate::type_traits::RefFromAnyUnchecked for $name {
403                #[inline(always)]
404                unsafe fn ref_from_any_unchecked(
405                    cell: &$crate::tvm_ffi_sys::TVMFFIAny,
406                ) -> &Self {
407                    const {
408                        assert!(
409                            ::core::mem::size_of::<$name>()
410                                == ::core::mem::size_of::<*mut $crate::tvm_ffi_sys::TVMFFIObject>()
411                                && ::core::mem::align_of::<$name>()
412                                    == ::core::mem::align_of::<*mut $crate::tvm_ffi_sys::TVMFFIObject>(),
413                            "a reference downcast target must be exactly one object pointer"
414                        )
415                    };
416                    unsafe { &*(::core::ptr::addr_of!(cell.data_union) as *const Self) }
417                }
418            }
419
420            unsafe impl $crate::type_traits::DowncastRefFromAny for $name {
421                #[inline(always)]
422                unsafe fn downcast_ref_from_any(
423                    cell: &$crate::tvm_ffi_sys::TVMFFIAny,
424                ) -> ::core::option::Option<&Self> {
425                    unsafe { $crate::type_traits::downcast_ref_from_any_in_place::<Self>(cell) }
426                }
427            }
428
429            unsafe impl $crate::type_traits::DowncastRefFromObject for $name {
430                #[inline(always)]
431                unsafe fn downcast_ref_from_object(
432                    place: &*mut $crate::tvm_ffi_sys::TVMFFIObject,
433                ) -> ::core::option::Option<&Self> {
434                    unsafe { $crate::type_traits::downcast_ref_from_object_in_place::<Self>(place) }
435                }
436            }
437        )+
438    };
439}

589/// Macro to export a typed function as a C symbol that follows the tvm-ffi ABI
590///
591/// An error the function returns is raised to the caller. Report errors by
592/// returning them; a panic is also caught and raised, as an `InternalError`,
593/// since unwinding into the caller would abort the process, but panicking
594/// is discouraged.
595///
596/// With the `export-metadata` feature, the counterpart of C++
597/// `TVM_FFI_DLL_EXPORT_INCLUDE_METADATA`, the macro also exports the
598/// function's metadata, `{"type_schema":...}`, as `__tvm_ffi__metadata_<name>`,
599/// which [`Module::get_function_metadata`](crate::Module::get_function_metadata)
600/// and the stub generator read. A
601/// function's schema lists its parameter and return types; a callable whose
602/// signature is not known, such as a custom [`AsPackedCallable`], has the
603/// schema of an untyped function. The feature is off by default, so exports
604/// are minimal.
605///
606/// [`AsPackedCallable`]: crate::function_internal::AsPackedCallable
607///
608/// # Arguments
609/// * `$name` - The name of the function
610/// * `$func` - The function to export
611///
612/// # Example
613/// ```rust
614/// use tvm_ffi::*;
615///
616/// fn add_one(x: i32) -> Result<i32> { Ok(x + 1) }
617///
618/// tvm_ffi_dll_export_typed_func!(add_one, add_one);
619/// ```
620#[macro_export]
621macro_rules! tvm_ffi_dll_export_typed_func {
622    ($name:ident, $func:expr) => {
623        $crate::macros::paste::paste! {
624            // `no_mangle` keeps the symbol in a `cdylib` under the
625            // `__tvm_ffi_<name>` name that `Module` looks up.
626            //
627            // The path-qualified `$crate::tvm_ffi_sys::…` reference (rather
628            // than a bare `tvm_ffi_sys::…`) lets downstream crates use the
629            // macro without having to add `tvm-ffi-sys` to their own
630            // `[dependencies]`.
631            #[unsafe(no_mangle)]
632            pub unsafe extern "C" fn [<__tvm_ffi_ $name>](
633                _handle: *mut std::ffi::c_void,
634                args: *const $crate::tvm_ffi_sys::TVMFFIAny,
635                num_args: i32,
636                result: *mut $crate::tvm_ffi_sys::TVMFFIAny,
637            ) -> i32 {
638                let packed_args =
639                    std::slice::from_raw_parts(args as *const $crate::any::AnyView, num_args as usize);
640                let ret_value = match ::std::panic::catch_unwind(::std::panic::AssertUnwindSafe(|| {
641                    $crate::function_internal::call_packed_callable($func, packed_args)
642                })) {
643                    Ok(ret_value) => ret_value,
644                    Err(payload) => Err($crate::function_internal::panic_to_error(payload)),
645                };
646                match ret_value {
647                    Ok(value) => {
648                        *result = $crate::any::Any::into_raw_ffi_any(value);
649                        0
650                    }
651                    Err(error) => {
652                        $crate::error::Error::set_raised(&error);
653                        -1
654                    }
655                }
656            }
657        }
658        $crate::__tvm_ffi_dll_export_metadata!($name, $func);
659    };
660}

662/// Macro to export the doc string of a function exported with
663/// [`tvm_ffi_dll_export_typed_func!`], as C++ `TVM_FFI_DLL_EXPORT_TYPED_FUNC_DOC`
664/// does
665///
666/// With the `export-metadata` feature, the doc string is exported as

675/// * `$doc` - The doc string, a string literal

681/// fn add(a: i64, b: i64) -> Result<i64> { Ok(a + b) }

685/// ```

687macro_rules! tvm_ffi_dll_export_typed_func_doc {
688    ($name:ident, $doc:expr) => {
689        $crate::__tvm_ffi_dll_export_doc!($name, $doc);
690    };
691}

759///-----------------------------------------------------------
760/// into_typed_fn
761///
762/// Wrap a [`Function`](crate::Function) as a typed closure with the given
763/// signature: `into_typed_fn!(func, Fn(A, B) -> Result<R>)`.
764///
765/// The closure converts its arguments and the result, and checks the argument
766/// count at compile time; it takes 0 to 8 arguments. An argument may be taken
767/// by value or by reference; prefer a reference for an object reference such as
768/// a `Tensor`, which the caller then keeps.
769///
770/// # Examples
771///
772/// ```
773/// use tvm_ffi::*;
774///
775/// let func = Function::from_typed(|x: i32, y: i32| -> Result<i32> { Ok(x + y) });
776/// let add = into_typed_fn!(func, Fn(i32, &i32) -> Result<i32>);
777/// assert_eq!(add(10, &20)?, 30);
778/// # Ok::<(), tvm_ffi::Error>(())
779/// ```
780#[macro_export]
781macro_rules! into_typed_fn {
782    // Case for 0 arguments
783    ($f:expr, $trait:ident() -> $ret_ty:ty) => {{
784        let _f = $f;
785        move || -> $ret_ty {
786            _f.call_tuple_with_len::<0, _>(())?
787                .try_into()
788                .map_err(Into::into)
789        }
790    }};
791    // Case for 1 argument
792    ($f:expr, $trait:ident($t0:ty) -> $ret_ty:ty) => {{
793        let _f = $f;
794        move |a0: $t0| -> $ret_ty {
795            use $crate::function_internal::IntoArgHolderTuple;
796            let tuple_args = (a0,).into_arg_holder_tuple();
797            _f.call_tuple_with_len::<1, _>(tuple_args)?
798                .try_into()
799                .map_err(Into::into)
800        }
801    }};
802    // Case for 2 arguments
803    ($f:expr, $trait:ident($t0:ty, $t1:ty) -> $ret_ty:ty) => {{
804        let _f = $f;
805        move |a0: $t0, a1: $t1| -> $ret_ty {
806            use $crate::function_internal::IntoArgHolderTuple;
807            let tuple_args = (a0, a1).into_arg_holder_tuple();
808            _f.call_tuple_with_len::<2, _>(tuple_args)?
809                .try_into()
810                .map_err(Into::into)
811        }
812    }};
813    // Case for 3 arguments
814    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty) -> $ret_ty:ty) => {{
815        let _f = $f;
816        move |a0: $t0, a1: $t1, a2: $t2| -> $ret_ty {
817            use $crate::function_internal::IntoArgHolderTuple;
818            let tuple_args = (a0, a1, a2).into_arg_holder_tuple();

820                .try_into()

823    }};
824    // Case for 4 arguments
825    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty) -> $ret_ty:ty) => {{
826        let _f = $f;
827        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3| -> $ret_ty {
828            use $crate::function_internal::IntoArgHolderTuple;
829            let tuple_args = (a0, a1, a2, a3).into_arg_holder_tuple();
830            _f.call_tuple_with_len::<4, _>(tuple_args)?
831                .try_into()
832                .map_err(Into::into)
833        }
834    }};
835    // Case for 5 arguments
836    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty) -> $ret_ty:ty) => {{
837        let _f = $f;
838        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4| -> $ret_ty {
839            use $crate::function_internal::IntoArgHolderTuple;
840            let tuple_args = (a0, a1, a2, a3, a4).into_arg_holder_tuple();
841            _f.call_tuple_with_len::<5, _>(tuple_args)?
842                .try_into()
843                .map_err(Into::into)
844        }
845    }};
846    // Case for 6 arguments
847    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty, $t5:ty) -> $ret_ty:ty) => {{
848        let _f = $f;
849        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4, a5: $t5| -> $ret_ty {
850            use $crate::function_internal::IntoArgHolderTuple;
851            let tuple_args = (a0, a1, a2, a3, a4, a5).into_arg_holder_tuple();
852            _f.call_tuple_with_len::<6, _>(tuple_args)?
853                .try_into()
854                .map_err(Into::into)
855        }
856    }};
857    // Case for 7 arguments
858    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty, $t5:ty, $t6:ty)
859        -> $ret_ty:ty) => {{
860        let _f = $f;
861        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4, a5: $t5, a6: $t6| -> $ret_ty {
862            use $crate::function_internal::IntoArgHolderTuple;
863            let tuple_args = (a0, a1, a2, a3, a4, a5, a6).into_arg_holder_tuple();
864            _f.call_tuple_with_len::<7, _>(tuple_args)?
865                .try_into()
866                .map_err(Into::into)
867        }
868    }};

870    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty, $t5:ty, $t6:ty, $t7:ty)
871        -> $ret_ty:ty) => {{

876            _f.call_tuple_with_len::<8, _>(tuple_args)?
877                .try_into()
878                .map_err(Into::into)
879        }
880    }};
