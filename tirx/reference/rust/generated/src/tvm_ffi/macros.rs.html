26/// Macro gto get the name of the function
27///
28/// # Usage
29/// Usage: function_name!()
30#[doc(hidden)]
31#[macro_export]
32macro_rules! function_name {
33    () => {{
34        // dummy function to get the name of the function
35        fn f() {}
36        fn type_name_of<T>(_: T) -> &'static str {
37            std::any::type_name::<T>()
38        }
39        let name = type_name_of(f);
40        // remove the f() from the name
41        &name[..name.len() - 3]
42    }};
43}

45/// Look up a registered global [`Function`](crate::Function) by name once, and
46/// keep it for the process: a `&'static Function`.
47///
48/// Each call site has its own cache, which later calls read without a lock.
49/// Use it for a function that must be registered, such as a built-in one; it
50/// panics, naming the function, if the function is not. See also
51/// [`Function::get_global`](crate::Function::get_global), which returns an
52/// error instead.
53///
54/// # Examples
55///
56/// ```
57/// use tvm_ffi::{cached_global_func, Array};
58///
59/// let array = cached_global_func!("ffi.Array").call_tuple((1_i64, 2_i64))?;
60/// assert_eq!(Array::<i64>::try_from(array)?.len(), 2);
61/// # Ok::<(), tvm_ffi::Error>(())
62/// ```
63#[macro_export]
64macro_rules! cached_global_func {
65    ($name:literal) => {{
66        static FUNC: $crate::extra::cold_init_lazy_lock::ColdInitLazyLock<
67            $crate::function::Function,
68        > = $crate::extra::cold_init_lazy_lock::ColdInitLazyLock::new(|| {
69            $crate::function::Function::get_global($name).unwrap_or_else(|_| {
70                panic!(concat!("global function `", $name, "` is not registered"))
71            })
72        });
73        &*FUNC
74    }};
75}

77/// Implement zero-copy conversions from a derived object reference to one of
78/// its base object-reference types: `From<Source>` and `From<&Source>` into
79/// `Target`, `AsRef<Target>`, which borrows a `&Source` as `&Target`
80/// without touching the reference count, and
81/// [`IntoObjectRef<Target>`](crate::object::IntoObjectRef) for `Source` and `&Source`.
82/// It also implements the strict downcast back, `TryFrom<Target>` and
83/// `TryFrom<&Target>` for `Source`, which return a `TypeError` naming both types.
84///
85/// # Safety
86///
87/// The declaration is trusted, as an `unsafe impl` is, hence the `unsafe_`
88/// name: `target` must be a registered base of `source`, and every `source`
89/// must satisfy the target reference invariants. An incorrect declaration makes
90/// the generated safe conversions unsound. The Rust stub generator writes these
91/// from the C++ type hierarchy.
92#[macro_export]
93macro_rules! unsafe_impl_object_ref_upcast {
94    ($($source:ty => $target:ty),+ $(,)?) => {
95        $(
96            impl ::std::convert::From<$source> for $target {
97                #[inline]
98                fn from(value: $source) -> Self {
99                    let data = <$source as $crate::object::ObjectRefCore>::into_data(value);
100                    // SAFETY: The macro declaration promises that `target` is
101                    // a registered base of `source`. Both references retain
102                    // the same allocation and only change its static view.
103                    let data = unsafe {
104                        $crate::object::ObjectArc::from_raw(
105                            $crate::object::ObjectArc::into_raw(data).cast::
106                                <<$target as $crate::object::ObjectRefCore>::ContainerType>(),
107                        )
108                    };
109                    // SAFETY: The macro declaration promises both the
110                    // container inheritance relation and every additional
111                    // invariant imposed by the target reference view.
112                    unsafe {
113                        <$target as $crate::object::ObjectRefCore>::from_data(data)
114                    }
115                }
116            }
117
118            impl ::std::convert::From<&$source> for $target {
119                #[inline]
120                fn from(value: &$source) -> Self {
121                    value.clone().into()
122                }
123            }
124
125            impl $crate::object::IntoObjectRef<$target> for $source {
126                type Output = $target;
127                #[inline(always)]
128                fn into_object_ref(self) -> $target {
129                    ::core::convert::Into::into(self)
130                }
131            }
132
133            impl<'a> $crate::object::IntoObjectRef<$target> for &'a $source {
134                type Output = &'a $target;
135                #[inline(always)]
136                fn into_object_ref(self) -> &'a $target {
137                    ::core::convert::AsRef::<$target>::as_ref(self)
138                }
139            }
140
141            impl ::core::convert::AsRef<$target> for $source {
142                #[inline(always)]
143                fn as_ref(&self) -> &$target {
144                    const _: () = assert!(
145                        ::core::mem::size_of::<$source>() == ::core::mem::size_of::<$target>()
146                            && ::core::mem::align_of::<$source>()
147                                == ::core::mem::align_of::<$target>(),
148                        "an object reference upcast must keep the reference's layout"
149                    );
150                    // SAFETY: both references are exactly one object pointer, and the
151                    // macro declaration promises that every `source` object is a valid
152                    // `target`, so a `&source` is a valid `&target`.
153                    unsafe { &*(self as *const $source as *const $target) }
154                }
155            }
156
157            // The strict downcast back, as a `TryFrom` for `?`: a reference
158            // never converts, so this only checks the object's runtime type.
159            impl ::core::convert::TryFrom<$target> for $source {
160                type Error = $crate::error::Error;
161                #[inline]
162                fn try_from(value: $target) -> ::core::result::Result<Self, Self::Error> {
163                    $crate::object::ObjectRefCore::downcast::<$source>(value)
164                        .map_err($crate::error::Error::downcast_failed::<$source>)
165                }
166            }
167
168            impl ::core::convert::TryFrom<&$target> for $source {
169                type Error = $crate::error::Error;
170                #[inline]
171                fn try_from(value: &$target) -> ::core::result::Result<Self, Self::Error> {
172                    match $crate::object::ObjectRefCore::downcast_ref::<$source>(value) {
173                        ::core::option::Option::Some(found) => {
174                            ::core::result::Result::Ok(::core::clone::Clone::clone(found))
175                        }
176                        ::core::option::Option::None => ::core::result::Result::Err(
177                            $crate::error::Error::downcast_failed::<$source>(
178                                ::core::clone::Clone::clone(value),
179                            ),
180                        ),
181                    }
182                }
183            }
184        )+
185    };
186}

188/// Implement the in-place `downcast_ref` borrows (`array[i]`, `Any::downcast_ref`,
189/// [`ObjectRefCore::downcast_ref`](crate::ObjectRefCore::downcast_ref)) for each
190/// listed object reference type. [`#[derive(ObjectRef)]`](crate::derive::ObjectRef)
191/// emits it; a reference that implements `ObjectRefCore` by hand calls it.
192///
193/// # Safety
194///
195/// Each type is `repr(transparent)` over one object pointer, and its
196/// [`AnyCompatible::check_any_strict`](crate::AnyCompatible::check_any_strict)
197/// accepts exactly the objects that are a valid value of it.
198///
199/// # Examples
200///
201/// ```
202/// use tvm_ffi::derive::ObjectRef;
203/// use tvm_ffi::{rs_object, unsafe_impl_object_ref_downcast_ref, Any};
204/// use tvm_ffi::{Object, ObjectArc, ObjectRefAnyCompatible, ObjectRefCore};
205///
206/// #[rs_object("my_ext.MyPair", final)]
207/// pub struct MyPairObj {
208///     base: Object,
209///     pub first: i64,
210///     pub second: i64,
211/// }
212///
213/// #[derive(Clone, ObjectRef)]
214/// pub struct MyPair {
215///     data: ObjectArc<MyPairObj>,
216/// }
217///
218/// /// A `MyPair` whose `first` is positive.
219/// #[derive(Clone)]
220/// #[repr(transparent)]
221/// pub struct PositivePair(ObjectArc<MyPairObj>);
222///
223/// unsafe impl ObjectRefCore for PositivePair {
224///     type ContainerType = MyPairObj;
225///     fn data(this: &Self) -> &ObjectArc<MyPairObj> {
226///         &this.0
227///     }
228///     fn into_data(this: Self) -> ObjectArc<MyPairObj> {
229///         this.0
230///     }
231///     unsafe fn from_data(data: ObjectArc<MyPairObj>) -> Self {
232///         Self(data)
233///     }
234/// }
235///
236/// unsafe impl ObjectRefAnyCompatible for PositivePair {
237///     fn extra_check(pair: &MyPairObj) -> bool {
238///         pair.first > 0
239///     }
240/// }
241/// unsafe_impl_object_ref_downcast_ref!(PositivePair);
242///
243/// let pair = Any::from(MyPair::from_object(MyPairObj::new(Object::new(), 1, 2)));
244/// assert!(pair.downcast_ref::<PositivePair>().is_some());
245/// ```
246#[macro_export]
247macro_rules! unsafe_impl_object_ref_downcast_ref {
248    ($($name:ty),+ $(,)?) => {
249        $(
250            // SAFETY: the macro declaration promises that `$name` is one object
251            // pointer whose `check_any_strict` is its type check.
252            unsafe impl $crate::type_traits::RefFromAnyUnchecked for $name {
253                #[inline(always)]
254                unsafe fn ref_from_any_unchecked(
255                    cell: &$crate::tvm_ffi_sys::TVMFFIAny,
256                ) -> &Self {
257                    const {
258                        assert!(
259                            ::core::mem::size_of::<$name>()
260                                == ::core::mem::size_of::<*mut $crate::tvm_ffi_sys::TVMFFIObject>()
261                                && ::core::mem::align_of::<$name>()
262                                    == ::core::mem::align_of::<*mut $crate::tvm_ffi_sys::TVMFFIObject>(),
263                            "a reference downcast target must be exactly one object pointer"
264                        )
265                    };
266                    unsafe { &*(::core::ptr::addr_of!(cell.data_union) as *const Self) }
267                }
268            }
269
270            unsafe impl $crate::type_traits::DowncastRefFromAny for $name {
271                #[inline(always)]
272                unsafe fn downcast_ref_from_any(
273                    cell: &$crate::tvm_ffi_sys::TVMFFIAny,
274                ) -> ::core::option::Option<&Self> {
275                    unsafe { $crate::type_traits::downcast_ref_from_any_in_place::<Self>(cell) }
276                }
277            }
278
279            unsafe impl $crate::type_traits::DowncastRefFromObject for $name {
280                #[inline(always)]
281                unsafe fn downcast_ref_from_object(
282                    place: &*mut $crate::tvm_ffi_sys::TVMFFIObject,
283                ) -> ::core::option::Option<&Self> {
284                    unsafe { $crate::type_traits::downcast_ref_from_object_in_place::<Self>(place) }
285                }
286            }
287        )+
288    };
289}

291/// Check the return code of the safe call
292///
293/// # Arguments
294/// * `ret_code` - The return code of the safe call
295///
296/// # Returns
297/// * `Result<(), Error>` - The result of the safe call
298/// Macro to check safe calls and automatically update traceback with file/line info
299///
300/// Usage: check_safe_call!(function(args))?;
301#[doc(hidden)]
302#[macro_export]
303macro_rules! check_safe_call {
304    ($expr:expr) => {{
305        let ret_code = $expr;
306        if ret_code == 0 {
307            Ok(())
308        } else {
309            let error = $crate::error::Error::from_raised();
310            Err(error)
311        }
312    }};

315/// Return an [`Error`](crate::Error) of a kind with a formatted message from
316/// the current function.
317///
318/// `bail!(kind, fmt, args...)` formats the message as `format!` does and
319/// records the source location in the backtrace.
320///
321/// # Arguments
322///
323/// * `kind` - The kind of the error, such as [`VALUE_ERROR`](crate::VALUE_ERROR).
324/// * `fmt` - The format string of the message.
325/// * `args` - The format arguments.
326///
327/// # Returns
328///
329/// Returns `Err` of the error from the enclosing function.
330///
331/// # Examples
332///
333/// ```
334/// use tvm_ffi::{bail, Result, INDEX_ERROR};
335///
336/// fn get(items: &[i64], index: usize) -> Result<i64> {
337///     if index >= items.len() {
338///         bail!(INDEX_ERROR, "index {} out of range for length {}", index, items.len());
339///     }
340///     Ok(items[index])
341/// }
342///
343/// assert_eq!(get(&[1, 2], 1)?, 2);
344/// assert_eq!(get(&[1, 2], 5).unwrap_err().message(), "index 5 out of range for length 2");
345/// # Ok::<(), tvm_ffi::Error>(())
346/// ```
347#[macro_export]
348macro_rules! bail {
349    ($error_kind:expr, $fmt:expr $(, $args:expr)* $(,)?) => {{
350        let context = format!(
351            "  File \"{}\", line {}, in {}\n",
352            file!(),
353            line!(),
354            $crate::function_name!()
355        );
356        return Err($crate::error::Error::new($error_kind, &format!($fmt $(, $args)*), &context));
357    }};
358}

360/// Return an [`Error`](crate::Error) from the current function unless a
361/// condition holds.
362///
363/// `ensure!(cond, kind, fmt, args...)` is [`bail!`](crate::bail)`(kind, fmt,
364/// args...)` when `cond` is false.
365///
366/// # Arguments
367///
368/// * `cond` - The condition that must hold.
369/// * `kind` - The kind of the error.
370/// * `fmt` - The format string of the message.
371/// * `args` - The format arguments.
372///
373/// # Returns
374///
375/// Returns `Err` of the error from the enclosing function when `cond` is
376/// false, and does nothing otherwise.
377///
378/// # Examples
379///
380/// ```
381/// use tvm_ffi::{ensure, Result, VALUE_ERROR};
382///
383/// fn sqrt(x: f64) -> Result<f64> {
384///     ensure!(x >= 0.0, VALUE_ERROR, "sqrt of negative number {}", x);
385///     Ok(x.sqrt())
386/// }
387///
388/// assert_eq!(sqrt(4.0)?, 2.0);
389/// assert!(sqrt(-1.0).is_err());
390/// # Ok::<(), tvm_ffi::Error>(())
391/// ```
392#[macro_export]
393macro_rules! ensure {
394    ($cond:expr, $error_kind:expr, $fmt:expr $(, $args:expr)* $(,)?) => {{
395        if !$cond {
396            $crate::bail!($error_kind, $fmt $(, $args)*);
397        }
398    }};
399}

401/// Append the current source location to the backtrace of an `Err` result,
402/// and pass an `Ok` through.
403///
404/// # Arguments
405///
406/// * `result` - The result to attach the context to.
407///
408/// # Returns
409///
410/// The result, with the location appended to an error's backtrace.
411///
412/// # Examples
413///
414/// ```
415/// use tvm_ffi::{attach_context, bail, Result, RUNTIME_ERROR};
416///
417/// fn inner() -> Result<()> {
418///     bail!(RUNTIME_ERROR, "failed");
419/// }
420///
421/// fn outer() -> Result<()> {
422///     attach_context!(inner())
423/// }
424///
425/// assert!(outer().unwrap_err().backtrace().contains("outer"));
426/// ```
427#[macro_export]
428macro_rules! attach_context {
429    ($error:expr) => {{
430        match $error {
431            Ok(value) => Ok(value),
432            Err(error) => {
433                let context = format!(
434                    "  File \"{}\", line {}, in {}\n",
435                    file!(),
436                    line!(),
437                    $crate::function_name!()
438                );
439                Err($crate::error::Error::with_appended_backtrace(
440                    error, &context,
441                ))
442            }
443        }
444    }};
445}

595/// Macro to export a typed function as a C symbol that follows the tvm-ffi ABI
596///
597/// An error the function returns is raised to the caller. Report errors by
598/// returning them; a panic is also caught and raised, as an `InternalError`,
599/// since unwinding into the caller would abort the process, but panicking
600/// is discouraged.
601///
602/// With the `export-metadata` feature, the counterpart of C++
603/// `TVM_FFI_DLL_EXPORT_INCLUDE_METADATA`, the macro also exports the
604/// function's metadata, `{"type_schema":...}`, as `__tvm_ffi__metadata_<name>`,
605/// which `Module::GetFunctionMetadata` and the stub generator read. A
606/// function's schema lists its parameter and return types; a callable whose
607/// signature is not known, such as a custom [`AsPackedCallable`], has the
608/// schema of an untyped function. As in C++, the feature is off by default,
609/// so exports are minimal.
610///
611/// [`AsPackedCallable`]: crate::function_internal::AsPackedCallable
612///
613/// # Arguments
614/// * `$name` - The name of the function
615/// * `$func` - The function to export
616///
617/// # Example
618/// ```rust
619/// use tvm_ffi::*;
620///
621/// fn add_one(x: i32) -> Result<i32> { Ok(x + 1) }
622///
623/// tvm_ffi_dll_export_typed_func!(add_one, add_one);
624/// ```
625#[macro_export]
626macro_rules! tvm_ffi_dll_export_typed_func {
627    ($name:ident, $func:expr) => {
628        $crate::macros::paste::paste! {
629            // `#[no_mangle]` is required so the symbol is preserved in a
630            // `cdylib` and matches the `__tvm_ffi_<name>` naming convention
631            // that `ffi.Module.load_from_file.<format>` looks up via
632            // `GetSymbolWithSymbolPrefix`. Without it, the linker strips the
633            // function from the output `.so`.
634            //
635            // Using plain `#[no_mangle]` (rather than `#[unsafe(no_mangle)]`,
636            // which would require rustc >= 1.82) keeps the crate buildable
637            // on older toolchains. Edition-2024 callers will see a
638            // deprecation warning, which is harmless.
639            //
640            // The path-qualified `$crate::tvm_ffi_sys::…` reference (rather
641            // than a bare `tvm_ffi_sys::…`) lets downstream crates use the
642            // macro without having to add `tvm-ffi-sys` to their own
643            // `[dependencies]`.
644            #[no_mangle]
645            pub unsafe extern "C" fn [<__tvm_ffi_ $name>](
646                _handle: *mut std::ffi::c_void,
647                args: *const $crate::tvm_ffi_sys::TVMFFIAny,
648                num_args: i32,
649                result: *mut $crate::tvm_ffi_sys::TVMFFIAny,
650            ) -> i32 {
651                let packed_args =
652                    std::slice::from_raw_parts(args as *const $crate::any::AnyView, num_args as usize);
653                let ret_value = match ::std::panic::catch_unwind(::std::panic::AssertUnwindSafe(|| {
654                    $crate::function_internal::call_packed_callable($func, packed_args)
655                })) {
656                    Ok(ret_value) => ret_value,
657                    Err(payload) => Err($crate::function_internal::panic_to_error(payload)),
658                };
659                match ret_value {
660                    Ok(value) => {
661                        *result = $crate::any::Any::into_raw_ffi_any(value);
662                        0
663                    }
664                    Err(error) => {
665                        $crate::error::Error::set_raised(&error);
666                        -1
667                    }
668                }
669            }
670        }
671        $crate::__tvm_ffi_dll_export_metadata!($name, $func);
672    };
673}

772///-----------------------------------------------------------
773/// into_typed_fn
774///
775/// Converts a generic `Function` into a typed function with compile-time
776/// argument count and type checking. This macro provides a convenient way
777/// to create type-safe wrappers around TVM functions.
778///
779/// # Arguments
780/// * `$f` - The function identifier to convert
781/// * `$trait` - The trait type (typically `Fn`)
782/// * `($t0, $t1, ...)` - The argument types
783/// * `$ret_ty` - The return type
784///
785/// # Example
786/// ```rust
787/// use tvm_ffi::*;
788///
789/// let func = Function::from_typed(|x: i32, y: i32| -> Result<i32> { Ok(x + y) });
790/// let typed_func = into_typed_fn!(func, Fn(i32, &i32) -> Result<i32>);
791/// let result = typed_func(10, &20).unwrap(); // Returns 30
792/// assert_eq!(result, 30);
793/// ```
794/// Note that the `into_typed_fn!` macro can specify arguments to be passed either
795/// by reference or by value in the argument list.
796/// We recommend passing by reference for ObjectRef types such as Tensor.
797/// Since the ffi mechanism requires us to pass arguments by reference.
798///
799/// # Supported Argument Counts
800/// This macro supports functions with 0 to 8 arguments.
801///-----------------------------------------------------------
802#[macro_export]
803macro_rules! into_typed_fn {
804    // Case for 0 arguments
805    ($f:expr, $trait:ident() -> $ret_ty:ty) => {{
806        let _f = $f;
807        move || -> $ret_ty {
808            _f.call_tuple_with_len::<0, _>(())?
809                .try_into()
810                .map_err(Into::into)
811        }
812    }};
813    // Case for 1 argument
814    ($f:expr, $trait:ident($t0:ty) -> $ret_ty:ty) => {{
815        let _f = $f;
816        move |a0: $t0| -> $ret_ty {
817            use $crate::function_internal::IntoArgHolderTuple;
818            let tuple_args = (a0,).into_arg_holder_tuple();
819            _f.call_tuple_with_len::<1, _>(tuple_args)?
820                .try_into()
821                .map_err(Into::into)
822        }
823    }};
824    // Case for 2 arguments
825    ($f:expr, $trait:ident($t0:ty, $t1:ty) -> $ret_ty:ty) => {{
826        let _f = $f;
827        move |a0: $t0, a1: $t1| -> $ret_ty {
828            use $crate::function_internal::IntoArgHolderTuple;
829            let tuple_args = (a0, a1).into_arg_holder_tuple();
830            _f.call_tuple_with_len::<2, _>(tuple_args)?
831                .try_into()
832                .map_err(Into::into)
833        }
834    }};
835    // Case for 3 arguments
836    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty) -> $ret_ty:ty) => {{
837        let _f = $f;
838        move |a0: $t0, a1: $t1, a2: $t2| -> $ret_ty {
839            use $crate::function_internal::IntoArgHolderTuple;
840            let tuple_args = (a0, a1, a2).into_arg_holder_tuple();
841            _f.call_tuple_with_len::<3, _>(tuple_args)?
842                .try_into()
843                .map_err(Into::into)
844        }
845    }};
846    // Case for 4 arguments
847    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty) -> $ret_ty:ty) => {{
848        let _f = $f;
849        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3| -> $ret_ty {
850            use $crate::function_internal::IntoArgHolderTuple;
851            let tuple_args = (a0, a1, a2, a3).into_arg_holder_tuple();
852            _f.call_tuple_with_len::<4, _>(tuple_args)?
853                .try_into()
854                .map_err(Into::into)
855        }
856    }};
857    // Case for 5 arguments
858    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty) -> $ret_ty:ty) => {{
859        let _f = $f;
860        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4| -> $ret_ty {
861            use $crate::function_internal::IntoArgHolderTuple;
862            let tuple_args = (a0, a1, a2, a3, a4).into_arg_holder_tuple();
863            _f.call_tuple_with_len::<5, _>(tuple_args)?
864                .try_into()
865                .map_err(Into::into)
866        }
867    }};
868    // Case for 6 arguments
869    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty, $t5:ty) -> $ret_ty:ty) => {{
870        let _f = $f;
871        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4, a5: $t5| -> $ret_ty {
872            use $crate::function_internal::IntoArgHolderTuple;
873            let tuple_args = (a0, a1, a2, a3, a4, a5).into_arg_holder_tuple();
874            _f.call_tuple_with_len::<6, _>(tuple_args)?
875                .try_into()
876                .map_err(Into::into)
877        }
878    }};
879    // Case for 7 arguments
880    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty, $t5:ty, $t6:ty)
881        -> $ret_ty:ty) => {{
882        let _f = $f;
883        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4, a5: $t5, a6: $t6| -> $ret_ty {
884            use $crate::function_internal::IntoArgHolderTuple;
885            let tuple_args = (a0, a1, a2, a3, a4, a5, a6).into_arg_holder_tuple();
886            _f.call_tuple_with_len::<7, _>(tuple_args)?
887                .try_into()
888                .map_err(Into::into)
889        }
890    }};
891    // Case for 8 arguments
892    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty, $t5:ty, $t6:ty, $t7:ty)
893        -> $ret_ty:ty) => {{
894        let _f = $f;
895        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4, a5: $t5, a6: $t6, a7: $t7| -> $ret_ty {
896            use $crate::function_internal::IntoArgHolderTuple;
897            let tuple_args = (a0, a1, a2, a3, a4, a5, a6, a7).into_arg_holder_tuple();
898            _f.call_tuple_with_len::<8, _>(tuple_args)?
899                .try_into()
900                .map_err(Into::into)
901        }
902    }};
