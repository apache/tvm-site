25/// Macro gto get the name of the function
26///
27/// # Usage
28/// Usage: function_name!()
29#[macro_export]
30macro_rules! function_name {
31    () => {{
32        // dummy function to get the name of the function
33        fn f() {}
34        fn type_name_of<T>(_: T) -> &'static str {
35            std::any::type_name::<T>()
36        }
37        let name = type_name_of(f);
38        // remove the f() from the name
39        &name[..name.len() - 3]
40    }};
41}

43/// Resolves a registered global [`Function`](crate::function::Function) by name
44/// and caches it for the lifetime of the process (lock-free after the first
45/// lookup). Each call site gets its own cache and the macro evaluates to a
46/// `&'static Function`. Panics — naming the function — if it is not registered.
47///
48/// Usage: `cached_global_func!("ffi.Map").call_packed(args)`
49#[macro_export]
50macro_rules! cached_global_func {
51    ($name:literal) => {{
52        static FUNC: $crate::extra::cold_init_lazy_lock::ColdInitLazyLock<
53            $crate::function::Function,
54        > = $crate::extra::cold_init_lazy_lock::ColdInitLazyLock::new(|| {
55            $crate::function::Function::get_global($name).unwrap_or_else(|_| {
56                panic!(concat!("global function `", $name, "` is not registered"))
57            })
58        });
59        &*FUNC
60    }};
61}

63/// Implement zero-copy conversions from a derived object reference to one of
64/// its base object-reference types.
65///
66/// # Safety
67///
68/// Generated bindings must ensure that `target` is a registered base of
69/// `source` and that every `source` satisfies the target reference invariants.
70/// An incorrect declaration makes the generated safe conversion unsound.
71#[macro_export]
72macro_rules! impl_object_upcast {
73    ($($source:ty => $target:ty),+ $(,)?) => {
74        $(
75            impl ::std::convert::From<$source> for $target {
76                #[inline]
77                fn from(value: $source) -> Self {
78                    let data = <$source as $crate::object::ObjectRefCore>::into_data(value);
79                    // SAFETY: The macro declaration promises that `target` is
80                    // a registered base of `source`. Both references retain
81                    // the same allocation and only change its static view.
82                    let data = unsafe {
83                        $crate::object::ObjectArc::from_raw(
84                            $crate::object::ObjectArc::into_raw(data).cast::
85                                <<$target as $crate::object::ObjectRefCore>::ContainerType>(),
86                        )
87                    };
88                    // SAFETY: The macro declaration promises both the
89                    // container inheritance relation and every additional
90                    // invariant imposed by the target reference view.
91                    unsafe {
92                        <$target as $crate::object::ObjectRefCore>::from_data(data)
93                    }
94                }
95            }
96
97            impl ::std::convert::From<&$source> for $target {
98                #[inline]
99                fn from(value: &$source) -> Self {
100                    value.clone().into()
101                }
102            }
103
104        )+
105    };
106}

131/// Return an [`Error`](crate::Error) of a kind with a formatted message from
132/// the current function.
133///
134/// `bail!(kind, fmt, args...)` formats the message as `format!` does and
135/// records the source location in the backtrace.
136///
137/// # Arguments
138///
139/// * `kind` - The kind of the error, such as [`VALUE_ERROR`](crate::VALUE_ERROR).
140/// * `fmt` - The format string of the message.
141/// * `args` - The format arguments.
142///
143/// # Returns
144///
145/// Returns `Err` of the error from the enclosing function.
146///
147/// # Examples
148///
149/// ```
150/// use tvm_ffi::{bail, Result, INDEX_ERROR};
151///
152/// fn get(items: &[i64], index: usize) -> Result<i64> {
153///     if index >= items.len() {
154///         bail!(INDEX_ERROR, "index {} out of range for length {}", index, items.len());
155///     }
156///     Ok(items[index])
157/// }
158///
159/// assert_eq!(get(&[1, 2], 1)?, 2);
160/// assert_eq!(get(&[1, 2], 5).unwrap_err().message(), "index 5 out of range for length 2");
161/// # Ok::<(), tvm_ffi::Error>(())
162/// ```
163#[macro_export]
164macro_rules! bail {
165    ($error_kind:expr, $fmt:expr $(, $args:expr)* $(,)?) => {{
166        let context = format!(
167            "  File \"{}\", line {}, in {}\n",
168            file!(),
169            line!(),
170            $crate::function_name!()
171        );
172        return Err($crate::error::Error::new($error_kind, &format!($fmt $(, $args)*), &context));
173    }};
174}

176/// Return an [`Error`](crate::Error) from the current function unless a
177/// condition holds.
178///
179/// `ensure!(cond, kind, fmt, args...)` is [`bail!`](crate::bail)`(kind, fmt,
180/// args...)` when `cond` is false.
181///
182/// # Arguments
183///
184/// * `cond` - The condition that must hold.
185/// * `kind` - The kind of the error.
186/// * `fmt` - The format string of the message.
187/// * `args` - The format arguments.
188///
189/// # Returns
190///
191/// Returns `Err` of the error from the enclosing function when `cond` is
192/// false, and does nothing otherwise.
193///
194/// # Examples
195///
196/// ```
197/// use tvm_ffi::{ensure, Result, VALUE_ERROR};
198///
199/// fn sqrt(x: f64) -> Result<f64> {
200///     ensure!(x >= 0.0, VALUE_ERROR, "sqrt of negative number {}", x);
201///     Ok(x.sqrt())
202/// }
203///
204/// assert_eq!(sqrt(4.0)?, 2.0);
205/// assert!(sqrt(-1.0).is_err());
206/// # Ok::<(), tvm_ffi::Error>(())
207/// ```
208#[macro_export]
209macro_rules! ensure {
210    ($cond:expr, $error_kind:expr, $fmt:expr $(, $args:expr)* $(,)?) => {{
211        if !$cond {
212            $crate::bail!($error_kind, $fmt $(, $args)*);
213        }
214    }};
215}

217/// Append the current source location to the backtrace of an `Err` result,
218/// and pass an `Ok` through.
219///
220/// # Arguments
221///
222/// * `result` - The result to attach the context to.
223///
224/// # Returns
225///
226/// The result, with the location appended to an error's backtrace.
227///
228/// # Examples
229///
230/// ```
231/// use tvm_ffi::{attach_context, bail, Result, RUNTIME_ERROR};
232///
233/// fn inner() -> Result<()> {
234///     bail!(RUNTIME_ERROR, "failed");
235/// }
236///
237/// fn outer() -> Result<()> {
238///     attach_context!(inner())
239/// }
240///
241/// assert!(outer().unwrap_err().backtrace().contains("outer"));
242/// ```
243#[macro_export]
244macro_rules! attach_context {
245    ($error:expr) => {{
246        match $error {
247            Ok(value) => Ok(value),
248            Err(error) => {
249                let context = format!(
250                    "  File \"{}\", line {}, in {}\n",
251                    file!(),
252                    line!(),
253                    $crate::function_name!()
254                );
255                Err($crate::error::Error::with_appended_backtrace(error, &context))
256            }
257        }
258    }};
259}

368/// Macro to export a typed function as a C symbol that follows the tvm-ffi ABI
369///
370/// An error the function returns is raised to the caller. Report errors by
371/// returning them; a panic is also caught and raised, as an `InternalError`,
372/// since unwinding into the caller would abort the process, but panicking
373/// is discouraged.
374///
375/// With the `export-metadata` feature, the counterpart of C++
376/// `TVM_FFI_DLL_EXPORT_INCLUDE_METADATA`, the macro also exports the
377/// function's metadata, `{"type_schema":...}`, as `__tvm_ffi__metadata_<name>`,
378/// which `Module::GetFunctionMetadata` and the stub generator read. A
379/// function's schema lists its parameter and return types; a callable whose
380/// signature is not known, such as a custom [`AsPackedCallable`], has the
381/// schema of an untyped function. As in C++, the feature is off by default,
382/// so exports are minimal.
383///
384/// [`AsPackedCallable`]: crate::function_internal::AsPackedCallable
385///
386/// # Arguments
387/// * `$name` - The name of the function
388/// * `$func` - The function to export
389///
390/// # Example
391/// ```rust
392/// use tvm_ffi::*;
393///
394/// fn add_one(x: i32) -> Result<i32> { Ok(x + 1) }
395///
396/// tvm_ffi_dll_export_typed_func!(add_one, add_one);
397/// ```
398#[macro_export]
399macro_rules! tvm_ffi_dll_export_typed_func {
400    ($name:ident, $func:expr) => {
401        $crate::macros::paste::paste! {
402            // `#[no_mangle]` is required so the symbol is preserved in a
403            // `cdylib` and matches the `__tvm_ffi_<name>` naming convention
404            // that `ffi.Module.load_from_file.<format>` looks up via
405            // `GetSymbolWithSymbolPrefix`. Without it, the linker strips the
406            // function from the output `.so`.
407            //
408            // Using plain `#[no_mangle]` (rather than `#[unsafe(no_mangle)]`,
409            // which would require rustc >= 1.82) keeps the crate buildable
410            // on older toolchains. Edition-2024 callers will see a
411            // deprecation warning, which is harmless.
412            //
413            // The path-qualified `$crate::tvm_ffi_sys::…` reference (rather
414            // than a bare `tvm_ffi_sys::…`) lets downstream crates use the
415            // macro without having to add `tvm-ffi-sys` to their own
416            // `[dependencies]`.
417            #[no_mangle]
418            pub unsafe extern "C" fn [<__tvm_ffi_ $name>](
419                _handle: *mut std::ffi::c_void,
420                args: *const $crate::tvm_ffi_sys::TVMFFIAny,
421                num_args: i32,
422                result: *mut $crate::tvm_ffi_sys::TVMFFIAny,
423            ) -> i32 {
424                let packed_args =
425                    std::slice::from_raw_parts(args as *const $crate::any::AnyView, num_args as usize);
426                let ret_value = match ::std::panic::catch_unwind(::std::panic::AssertUnwindSafe(|| {
427                    $crate::function_internal::call_packed_callable($func, packed_args)
428                })) {
429                    Ok(ret_value) => ret_value,
430                    Err(payload) => Err($crate::function_internal::panic_to_error(payload)),
431                };
432                match ret_value {
433                    Ok(value) => {
434                        *result = $crate::any::Any::into_raw_ffi_any(value);
435                        0
436                    }
437                    Err(error) => {
438                        $crate::error::Error::set_raised(&error);
439                        -1
440                    }
441                }
442            }
443        }
444        $crate::__tvm_ffi_dll_export_metadata!($name, $func);
445    };
446}

448/// Macro to export the doc string of a function exported with
449/// [`tvm_ffi_dll_export_typed_func!`], as C++ `TVM_FFI_DLL_EXPORT_TYPED_FUNC_DOC`
450/// does
451///
452/// With the `export-metadata` feature, the doc string is exported as
453/// `__tvm_ffi__doc_<name>`, which `Module::GetFunctionDoc` and the stub
454/// generator read. Without it, as C++ without
455/// `TVM_FFI_DLL_EXPORT_INCLUDE_METADATA`, the macro exports nothing, but
456/// still checks that the doc string is a string constant. The macro never
457/// exports the function itself.
458///
459/// # Arguments
460/// * `$name` - The name the function is exported as
461/// * `$doc` - The doc string, a string literal
462///
463/// # Example
464/// ```rust
465/// use tvm_ffi::*;
466///
467/// fn add(a: i64, b: i64) -> Result<i64> { Ok(a + b) }
468///
469/// tvm_ffi_dll_export_typed_func!(add, add);
470/// tvm_ffi_dll_export_typed_func_doc!(add, "Add two integers and return the sum.");
471/// ```
472#[macro_export]
473macro_rules! tvm_ffi_dll_export_typed_func_doc {
474    ($name:ident, $doc:expr) => {
475        $crate::__tvm_ffi_dll_export_doc!($name, $doc);
476    };
477}

479// The metadata and doc getters the export macros emit with the
480// `export-metadata` feature. A `#[cfg]` in a `macro_rules!` expansion would
481// test the features of the crate that invokes the macro, so the feature
482// selects which definition of these helpers this crate compiles instead.
483
484#[cfg(feature = "export-metadata")]
485#[doc(hidden)]
486#[macro_export]
487macro_rules! __tvm_ffi_dll_export_metadata {
488    ($name:ident, $func:expr) => {
489        $crate::macros::paste::paste! {
490            #[no_mangle]
491            pub unsafe extern "C" fn [<__tvm_ffi__metadata_ $name>](
492                _handle: *mut std::ffi::c_void,
493                _args: *const $crate::tvm_ffi_sys::TVMFFIAny,
494                _num_args: i32,
495                result: *mut $crate::tvm_ffi_sys::TVMFFIAny,
496            ) -> i32 {
497                let metadata = match ::std::panic::catch_unwind(::std::panic::AssertUnwindSafe(|| {
498                    $crate::function_internal::exported_metadata(&$func)
499                })) {
500                    Ok(metadata) => metadata,
501                    Err(payload) => Err($crate::function_internal::panic_to_error(payload)),
502                };
503                $crate::function_internal::write_exported_str(metadata, result)
504            }
505        }
506    };
507}

545///-----------------------------------------------------------
546/// into_typed_fn
547///
548/// Converts a generic `Function` into a typed function with compile-time
549/// argument count and type checking. This macro provides a convenient way
550/// to create type-safe wrappers around TVM functions.
551///
552/// # Arguments
553/// * `$f` - The function identifier to convert
554/// * `$trait` - The trait type (typically `Fn`)
555/// * `($t0, $t1, ...)` - The argument types
556/// * `$ret_ty` - The return type
557///
558/// # Example
559/// ```rust
560/// use tvm_ffi::*;
561///
562/// let func = Function::from_typed(|x: i32, y: i32| -> Result<i32> { Ok(x + y) });
563/// let typed_func = into_typed_fn!(func, Fn(i32, &i32) -> Result<i32>);
564/// let result = typed_func(10, &20).unwrap(); // Returns 30
565/// assert_eq!(result, 30);
566/// ```
567/// Note that the `into_typed_fn!` macro can specify arguments to be passed either
568/// by reference or by value in the argument list.
569/// We recommend passing by reference for ObjectRef types such as Tensor.
570/// Since the ffi mechanism requires us to pass arguments by reference.
571///
572/// # Supported Argument Counts
573/// This macro supports functions with 0 to 8 arguments.
574///-----------------------------------------------------------
575#[macro_export]
576macro_rules! into_typed_fn {
577    // Case for 0 arguments
578    ($f:expr, $trait:ident() -> $ret_ty:ty) => {{
579        let _f = $f;
580        move || -> $ret_ty { Ok(_f.call_tuple_with_len::<0, _>(())?.try_into()?) }
581    }};
582    // Case for 1 argument
583    ($f:expr, $trait:ident($t0:ty) -> $ret_ty:ty) => {{
584        let _f = $f;
585        move |a0: $t0| -> $ret_ty {
586            use $crate::function_internal::IntoArgHolderTuple;
587            let tuple_args = (a0,).into_arg_holder_tuple();
588            Ok(_f.call_tuple_with_len::<1, _>(tuple_args)?.try_into()?)
589        }
590    }};
591    // Case for 2 arguments
592    ($f:expr, $trait:ident($t0:ty, $t1:ty) -> $ret_ty:ty) => {{
593        let _f = $f;
594        move |a0: $t0, a1: $t1| -> $ret_ty {
595            use $crate::function_internal::IntoArgHolderTuple;
596            let tuple_args = (a0, a1).into_arg_holder_tuple();
597            Ok(_f.call_tuple_with_len::<2, _>(tuple_args)?.try_into()?)
598        }
599    }};
600    // Case for 3 arguments
601    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty) -> $ret_ty:ty) => {{
602        let _f = $f;
603        move |a0: $t0, a1: $t1, a2: $t2| -> $ret_ty {
604            use $crate::function_internal::IntoArgHolderTuple;
605            let tuple_args = (a0, a1, a2).into_arg_holder_tuple();
606            Ok(_f.call_tuple_with_len::<3, _>(tuple_args)?.try_into()?)
607        }
608    }};
609    // Case for 4 arguments
610    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty) -> $ret_ty:ty) => {{
611        let _f = $f;
612        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3| -> $ret_ty {
613            use $crate::function_internal::IntoArgHolderTuple;
614            let tuple_args = (a0, a1, a2, a3).into_arg_holder_tuple();
615            Ok(_f.call_tuple_with_len::<4, _>(tuple_args)?.try_into()?)
616        }
617    }};
618    // Case for 5 arguments
619    ($f:expr, $trait:ident($t0:ty, $t1:ty, $t2:ty, $t3:ty, $t4:ty) -> $ret_ty:ty) => {{
620        let _f = $f;
621        move |a0: $t0, a1: $t1, a2: $t2, a3: $t3, a4: $t4| -> $ret_ty {
