20//! Native Rust structural visiting.
21//!
22//! Two public layers mirror the C++ API split:
23//!
24//! * [`Dispatcher`](crate::Dispatcher) + [`structural_visit`] — handlers drive
25//!   recursion through an explicit [`StructuralVisitorView`] context. Pass state belongs on
26//!   the dispatcher; `visitor.visit(self, &child)` reborrows it for recursive calls.
27//!   Unmatched values use default child traversal.
28//! * [`structural_walk`] — dispatcher or closure observers steer engine-owned
29//!   recursion through [`WalkResult`] (advance, skip, interrupt).
30//!
31//! Both layers thread definition-region state explicitly. Walk handlers opt in
32//! with a trailing [`DefRegionKind`]; visit handlers read it from their context
33//! and may override it for a child through [`StructuralVisitorView::with_def_region_kind`].
34//!
35//! # Safety of the implementation
36//!
37//! The public interface is safe; its implementation deliberately uses raw pointers.
38//! Each private runtime has a matching dispatcher vtable and a fixed dispatcher
39//! identity. The dispatcher stays alive in the outer call, and the engine's dispatcher
40//! pointer publishes the actual exclusive reborrow before native recursion, then
41//! restores the enclosing borrow on return. Views and native runtime handles cannot
42//! escape their synchronous calls. No reference to the whole runtime spans recursion,
43//! and owning ABI results transfer exactly one reference at each boundary.
44//!
45//! Callbacks must return failures through [`Result`] and must not panic. A panic
46//! reaching the non-unwinding `extern "C"` callback boundary aborts the process;
47//! visitor callbacks do not catch or transport panic payloads across native frames.
48
49// -----------------------------------------------------------------------------
50// Implementation overview
51// -----------------------------------------------------------------------------
52//
53// This file is organized into:
54// - Data structures and visiting capabilities.
55// - Outlined errors and private utility functions.
56// - Engine implementation and the public entry point.
57//
58// We use narrowly scoped unsafe code for native interoperation and efficient
59// traversal. Safety and ownership contracts are documented at each boundary.
60//
61// The main performance strategies are:
62// - Outline cold work to keep its temporary storage out of recursive frames.
63// - Use _raw helpers and private extern "C-unwind" boundaries for register
64//   transport on AArch64 while preserving Rust unwinding.
65// - Inline small conversions and forwarding code so ownership transfers and
66//   redundant cleanup can optimize away.
67//
68// Local Perf note comments explain individual implementation choices.

157/// Native state and default traversal for [`StructuralVisitEngine`].
158/// The engine owns allocation, scoped dispatch and results.
159///
160/// # Safety
161///
162/// `Self` must have a complete initialized C layout with [`StructuralVisitorObj`]
163/// at offset zero, matching its registered native type and every field hooks access.
164/// Native-mutated fields require interior mutability; shared references must not
165/// freeze them during recursion. Hooks must preserve the vtable and valid region state.
166///
167/// The base must not implement [`crate::ObjectCore`] or otherwise allow callers
168/// to retain the runtime. Hooks may access the runtime and borrowed traversal
169/// state only synchronously, on the calling thread.
170///
171/// `default_visit_raw` returns one owning ABI None/Error/VisitInterrupt and recurses only
172/// through the installed vtable, without a whole-base mutable reference spanning
173/// recursion. Failures use that result carrier; panics must not cross native callbacks.
174pub unsafe trait StructuralVisitorBase: Sized {
175    /// Initialize the complete base before the engine installs its runtime header.
176    fn new() -> Result<Self>;
177
178    /// Native type registered for this complete base layout.
179    fn type_index() -> i32;
180
181    /// Descend from a borrowed value without dispatching that value again.
182    /// Return one owning ABI None, Error, or VisitInterrupt value.
183    /// Children reenter the installed vtable. The ordinary implementation accesses
184    /// only the visitor prefix, so specialized bases may reuse it.
185    ///
186    /// # Safety
187    /// `runtime` is this live initialized base with a matching installed vtable.
188    /// The input and runtime remain borrowed for this synchronous call. The
189    /// caller has published the current exclusive dispatcher reborrow.
190    unsafe fn default_visit_raw(runtime: *mut Self, value: AnyView<'_>) -> TVMFFIAny;
191}
192
193// SAFETY: this prefix exactly matches the ordinary native visitor. Its mutable
194// region is interior mutable, and it exposes no safe ownership conversion.
195unsafe impl StructuralVisitorBase for StructuralVisitorObj {
196    #[inline(always)]
197    fn new() -> Result<Self> {
198        Ok(Self::default())
199    }
200
201    fn type_index() -> i32 {
202        static TYPE_INDEX: ColdInitLazyLock<i32> = ColdInitLazyLock::new(|| unsafe {
203            let key = TVMFFIByteArray::from_str(StructuralVisitorObj::TYPE_KEY);
204            let mut type_index = 0;
205            let return_code = TVMFFITypeKeyToIndex(&key, &mut type_index);
206            if return_code != 0 {
207                panic!(
208                    "ffi.StructuralVisitor is not registered: {}",
209                    Error::from_raised()
210                );
211            }
212            type_index
213        });
214        *TYPE_INDEX
215    }
216
217    #[inline(always)]
218    unsafe fn default_visit_raw(runtime: *mut Self, value: AnyView<'_>) -> TVMFFIAny {
219        Self::visit_children_raw(runtime, *value.as_raw_ffi_any())
220    }
221}

223/// Scoped Rust visitor runtime with a statically selected native base.
224///
225/// Enter through [`StructuralVisitEngine::visit_root`]. The engine owns the complete base
226/// and the dispatcher reborrow handle, while the dispatcher stays in the outer
227/// call. Engines and their owning handles are never exposed to callbacks.
228///
229/// Select a custom base with `StructuralVisitEngine::<Base>::visit_root(root, input)`.
230/// Handlers use `StructuralVisitorView<'_, Target, Base>` for recursion and
231/// base-state inspection; normalization preserves the actual target's identity.
232#[repr(C)]
233pub struct StructuralVisitEngine<Base = StructuralVisitorObj> {
234    base: Base,
235    dispatcher: *mut c_void,
236}

249/// Definition-region state active at the current value.
250///
251/// Reflected fields marked `SEqHashDefPattern` or
252/// `SEqHashDefSimple` override the inherited state for that field's
253/// complete recursive visit.
254#[repr(i32)]
255#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
256pub enum DefRegionKind {
257    /// The value is outside a definition region.
258    #[default]
259    None = 0,
260    /// Definitions apply recursively through the visited value.
261    Pattern = 1,
262    /// Definitions apply to the visited value alone; its type is walked as uses.
263    Simple = 2,
264}
265
266const _: () = {
267    assert!(DefRegionKind::None as i32 == TVMFFIDefRegionKind::kTVMFFIDefRegionKindNone as i32);
268    assert!(
269        DefRegionKind::Pattern as i32 == TVMFFIDefRegionKind::kTVMFFIDefRegionKindPattern as i32
270    );
271    assert!(DefRegionKind::Simple as i32 == TVMFFIDefRegionKind::kTVMFFIDefRegionKindSimple as i32);
272};

277    /// Entry points and visitor-layer calls return

304
305// Internal visit protocol matches the C++ owning TVMFFIAny result: None means
306// continue; VisitInterrupt (including None/Error payloads) and Error stop descent.
307// Each ABI ingress/import and egress/export transfers ownership exactly once.
308// Recursive drivers move this carrier unchanged; typed public APIs decode it.
309/// Recursive visiting and default child descent for a dispatcher.
310///
311/// Pass the handler's `self` to visit methods to recurse with the same state.
312pub struct StructuralVisitorView<'a, D, Base = StructuralVisitorObj> {
313    runtime: StructuralVisitorHandle,
314    _scope: PhantomData<(&'a mut (), fn(&mut D) -> Base, Rc<()>)>,
315}
316
317impl<D, Base: StructuralVisitorBase> StructuralVisitorView<'_, D, Base> {
318    /// Inspect the base state without retaining the scoped visitor runtime.
319    ///
320    /// The reference cannot outlive its view:
321    ///
322    /// ```compile_fail
323    /// use tvm_ffi::{StructuralVisitorObj, StructuralVisitorView};
324    /// fn retain<D>(view: &StructuralVisitorView<'_, D>) -> &'static StructuralVisitorObj {
325    ///     view.base()
326    /// }
327    /// ```
328    ///
329    /// The prefix cannot be converted into an owning FFI value:
330    ///
331    /// ```compile_fail
332    /// use tvm_ffi::{Any, StructuralVisitorView};
333    /// fn retain<D>(view: &StructuralVisitorView<'_, D>) -> Any {
334    ///     Any::from(view.base())
335    /// }
336    /// ```
337    #[inline(always)]
338    pub fn base(&self) -> &Base {
339        // SAFETY: the unsafe base contract permits shared inspection, including
340        // native mutation through interior-mutable fields during recursion.
341        unsafe { &*self.runtime.cast::<Base>() }
342    }
343
344    #[inline(always)]
345    pub fn def_region_kind(&self) -> DefRegionKind {
346        // SAFETY: engine scopes and trusted ABI hooks store valid enum values.
347        unsafe { std::mem::transmute(*(*self.runtime).def_region_mode.get()) }
348    }
349
350    /// Run operations under a scoped definition region.
351    /// An enclosing Pattern wins; the previous region is restored on all exits.
352    #[inline(always)]
353    pub fn with_def_region_kind<R>(
354        &mut self,
355        kind: DefRegionKind,
356        callback: impl FnOnce(&mut Self) -> R,
357    ) -> R {
358        StructuralVisitorObj::with_def_region_kind(self.runtime, kind, || callback(self))
359    }
360
361    /// Visit a value through the dispatcher.
362    #[inline(always)]
363    pub fn visit<'input, T>(&mut self, dispatch: &mut D, value: T) -> Result<Option<VisitInterrupt>>
364    where
365        T: Into<AnyView<'input>>,
366    {
367        let value: AnyView<'input> = value.into();
368        // SAFETY: the view ties this runtime to D; the check establishes the
369        // instance, and publishing uses the caller's actual exclusive reborrow.
370        unsafe {
371            visit_result_from_any(Any::from_raw_ffi_any(
372                StructuralVisitEngine::<Base>::visit_raw(
373                    self.runtime,
374                    dispatch,
375                    *value.as_raw_ffi_any(),
376                ),
377            ))
378        }
379    }
380
381    /// Run default descent through registered hooks or reflected fields.
382    /// Children re-enter the dispatcher; the current value does not.
383    #[inline(always)]
384    pub fn default_visit<'input, T>(
385        &self,
386        dispatch: &mut D,
387        value: T,
388    ) -> Result<Option<VisitInterrupt>>
389    where
390        T: Into<AnyView<'input>>,
391    {
392        // SAFETY: type/layout and scope come from the private view constructor;
393        // the supplied borrow must identify this runtime's dispatcher instance.
394        unsafe {
395            let context = StructuralVisitEngine::<Base>::checked_dispatcher(self.runtime, dispatch);
396            let value: AnyView<'input> = value.into();
397            visit_result_from_any(Any::from_raw_ffi_any(
398                StructuralVisitEngine::<Base>::default_visit_raw(
399                    self.runtime,
400                    context,
401                    *value.as_raw_ffi_any(),
402                ),
403            ))
404        }
405    }
406}

408impl<Base: StructuralVisitorBase> StructuralVisitEngine<Base> {
409    /// Normalize the input once and traverse using this engine's native base.
410    /// Named dispatchers keep their actual recursive view type. Callback and
411    /// tuple targets are their aggregate adapters; closures taking their own
412    /// recursive view type remain unsupported.
413    #[inline(always)]
414    pub fn visit_root<'input, R, H, M, D, O>(
415        root: R,
416        dispatcher: H,
417    ) -> Result<Option<VisitInterrupt>>
418    where
419        R: Into<AnyView<'input>>,
420        O: BorrowMut<D>,
421        for<'v, 'c, 'r> H: IntoDispatcher<
422                (AnyView<'v>, &'c mut StructuralVisitorView<'r, D, Base>),
423                Option<Result<Option<VisitInterrupt>>>,
424                M,
425                Target = D,
426                Output = O,
427            >,
428        for<'s, 'v, 'c, 'r> &'s mut D: crate::Dispatcher<
429                (AnyView<'v>, &'c mut StructuralVisitorView<'r, D, Base>),
430                Option<Result<Option<VisitInterrupt>>>,
431            >,
432    {
433        let mut storage = dispatcher.into_dispatcher();
434        let root = root.into();
435        visit_result_from_any(unsafe {
436            Any::from_raw_ffi_any(Self::run_visitor_raw::<D>(
437                *root.as_raw_ffi_any(),
438                storage.borrow_mut(),
439            ))
440        })
441    }
442}

553#[inline(always)]
554pub(crate) fn field_def_region(field: &TVMFFIFieldInfo, inherited: DefRegionKind) -> DefRegionKind {
555    // Precedence: a pattern region propagates; entering any kind inside it has no effect.
556    if inherited == DefRegionKind::Pattern {
557        DefRegionKind::Pattern
558    } else if field.flags & FLAG_SEQ_HASH_DEF_SIMPLE != 0 {
559        DefRegionKind::Simple
560    } else if field.flags & FLAG_SEQ_HASH_DEF_PATTERN != 0 {
561        DefRegionKind::Pattern
562    } else {
563        inherited
564    }
565}

608/// Visit every reflected field described by `info` and its ancestors in the
609/// same parent-to-child order as C++ `ForEachFieldInfoWithEarlyStop`.
610///
611/// # Safety
612///
613/// `info` must point to an immortal registered type-info record.
614pub(crate) unsafe fn for_each_field_info<B>(
615    info: *const crate::tvm_ffi_sys::TVMFFITypeInfo,
616    callback: &mut impl FnMut(&'static TVMFFIFieldInfo) -> ControlFlow<B>,
617) -> Option<B> {
618    // Ancestor slot 0 is the root Object. C++ starts at slot 1, walks toward
619    // the immediate parent, then visits the concrete type's own fields.
620    for depth in 1..(*info).type_depth {
621        let ancestor = *(*info).type_acenstors.offset(depth as isize);
622        if let Some(value) = visit_field_level(ancestor, callback) {
623            return Some(value);
624        }
625    }
626    visit_field_level(info, callback)
627}
628
629unsafe fn visit_field_level<B>(
630    info: *const crate::tvm_ffi_sys::TVMFFITypeInfo,
631    callback: &mut impl FnMut(&'static TVMFFIFieldInfo) -> ControlFlow<B>,
632) -> Option<B> {
633    if info.is_null() || (*info).fields.is_null() {
634        return None;
635    }
636    let fields = std::slice::from_raw_parts((*info).fields, (*info).num_fields as usize);
637    for field in fields {
638        // C reflection tables are immortal once registered.
639        let field: &'static TVMFFIFieldInfo = &*(field as *const TVMFFIFieldInfo);
640        if let ControlFlow::Break(value) = callback(field) {
641            return Some(value);
642        }
643    }
644    None
645}

651// Perf note: Raw carriers avoid materializing an owning Any on the success path.
652// Private C-unwind boundaries return this 16-byte carrier in x0/x1 on AArch64,
653// avoiding the hidden result slot observed with the Rust ABI while allowing
654// Rust cleanup to unwind. Native vtables remain extern "C".
655// Raw bits provide no ownership guard: each `_raw` result carries one owned ABI
656// value. Forward it exactly once or import it into Any before any operation
657// that can unwind or discard it, including destruction of an observer.
658impl StructuralVisitorObj {
659    /// Consume owning `result` from descent, adding borrowed `value` context on Error.
660    /// Internal callers supply valid tokens and keep the input live during annotation;
661    /// nonerrors transfer unchanged, and Error ownership is guarded through formatting.
662    #[inline(always)]
663    fn with_visit_context_raw(result: TVMFFIAny, value: TVMFFIAny) -> TVMFFIAny {
664        if result.type_index != TVMFFITypeIndex::kTVMFFIError as i32
665            || value.type_index < TVMFFITypeIndex::kTVMFFIStaticObjectBegin as i32
666        {
667            result
668        } else {
669            core::hint::cold_path();
670            // SAFETY: the Error tag establishes the owned object's representation.
671            unsafe {
672                let error = take_error(Any::from_raw_ffi_any(result));
673                Any::into_raw_ffi_any(
674                    annotate_visit_error(
675                        error,
676                        value.type_index,
677                        value.small_str_len,
678                        value.data_union,
679                    )
680                    .into(),
681                )
682            }
683        }
684    }
685
686    /// Descend through the native hook or reflected fields using this runtime.
687    ///
688    /// The dispatcher field must already hold the caller's actual reborrow.
689    /// Descent itself never accesses D: every child reenters the same visitor
690    /// callback, which restores its incoming pointer before the next sibling.
691    /// Returns one owning raw None/interrupt/Error token.
692    ///
693    /// # Safety
694    /// `runtime` must be this live engine and `value` a valid borrowed input whose
695    /// owner survives descent. Hooks may not retain the runtime or input borrow.
696    #[inline(always)]
697    unsafe fn visit_children_raw(runtime: StructuralVisitorHandle, value: TVMFFIAny) -> TVMFFIAny {
698        if let Some(attr) = structural_visit_column().get_raw_ffi_any(value.type_index) {
699            if attr.type_index != TVMFFITypeIndex::kTVMFFINone as i32 {
700                return Self::call_structural_visit_hook_raw(runtime, value, attr);
701            }
702        }
703
704        if value.type_index < TVMFFITypeIndex::kTVMFFIStaticObjectBegin as i32 {
705            TVMFFIAny::new()
706        } else {
707            Self::visit_reflected_fields_raw(
708                runtime,
709                value.type_index,
710                value.small_str_len,
711                value.data_union,
712            )
713        }
714    }
715
716    /// Reconstruct borrowed input from its scalar fields and visit registered fields
717    /// through live `runtime`. Return one owning None/interrupt/Error after local cleanup.
718    /// The private C-unwind ABI preserves Rust unwinding and AArch64 register returns;
719    /// public/native callback ABIs are unchanged.
720    ///
721    /// # Safety
722    /// The input fields and runtime must be valid for synchronous descent; registered
723    /// reflection metadata must satisfy its getter/offset contracts.
724    // Perf note: Keep reflection scratch outside recursive frames. This fallback
725    // can be frequent, so inline(never) is intentional without cold.
726    // The AArch64 C ABI returns TVMFFIAny in registers; C-unwind preserves
727    // Rust cleanup if reflected traversal panics inside this private helper.
728    #[inline(never)]
729    unsafe extern "C-unwind" fn visit_reflected_fields_raw(
730        runtime: StructuralVisitorHandle,
731        type_index: i32,
732        small_str_len: u32,
733        data_union: TVMFFIAnyDataUnion,
734    ) -> TVMFFIAny {
735        let value = TVMFFIAny {
736            type_index,
737            small_str_len,
738            data_union,
739        };
740        let type_info = unsafe { TVMFFIGetTypeInfo(value.type_index) };
741        if type_info.is_null() {
742            return Any::into_raw_ffi_any(unregistered_visit_type_error(value.type_index).into());
743        }
744        let seq_hash_kind = unsafe {
745            let metadata = (*type_info).metadata;
746            if metadata.is_null() {
747                TVMFFISEqHashKind::kTVMFFISEqHashKindUnsupported as i32
748            } else {
749                (*metadata).structural_eq_hash_kind
750            }
751        };
752        let kind = std::mem::transmute::<i32, DefRegionKind>(*(*runtime).def_region_mode.get());
753        let def_region_kind = free_var_child_region(kind, seq_hash_kind);
754        let object = unsafe { value.data_union.v_obj } as *mut u8;
755        let halted = unsafe {
756            for_each_field_info(type_info, &mut |field| {
757                let result = Self::visit_reflected_field(runtime, object, field, def_region_kind);
758                if result.type_index() == TVMFFITypeIndex::kTVMFFINone as i32 {
759                    ControlFlow::Continue(())
760                } else {
761                    ControlFlow::Break(result)
762                }
763            })
764        };
765        Any::into_raw_ffi_any(halted.unwrap_or_else(Any::new))
766    }
767
768    /// Visit one registered `field` in borrowed `object` using `inherited_region`.
769    /// Own getter output through recursive traversal and return an owning result.
770    ///
771    /// # Safety
772    /// `runtime` and `object` must be live; `field` must describe valid storage and a
773    /// registered getter for that object. Neither raw input transfers ownership.
774    unsafe fn visit_reflected_field(
775        runtime: StructuralVisitorHandle,
776        object: *mut u8,
777        field: &TVMFFIFieldInfo,
778        inherited_region: DefRegionKind,
779    ) -> Any {
780        if field.flags & FLAG_SEQ_HASH_IGNORE != 0 {
781            return Any::new();
782        }
783
784        let Some(getter) = field.getter else {
785            return Any::from(missing_field_getter_error("visitor", field.name.as_str()));
786        };
787        let address = object.offset(field.offset as isize) as *mut c_void;
788        // Own the getter result so partial writes and recursive borrows drop safely.
789        let mut child = Any::new();
790        if getter(address, Any::as_data_ptr(&mut child)) != 0 {
791            return with_field_error_context(Any::from(Error::from_raised()), field.name.as_str());
792        }
793
794        let borrowed = AnyView::from(&child);
795        let child_region = field_def_region(field, inherited_region);
796        with_field_error_context(
797            Self::with_def_region_kind(runtime, child_region, || {
798                Any::from_raw_ffi_any(Self::call_visitor_raw(runtime, *borrowed.as_raw_ffi_any()))
799            }),
800            field.name.as_str(),
801        )
802    }
803
804    /// Invoke the live visitor vtable on borrowed `value`, returning one owning token.
805    /// Internal callers keep runtime/input valid and publish the dispatcher reborrow.
806    /// None bypasses dispatch; native callbacks report failures without unwinding.
807    #[inline(always)]
808    fn call_visitor_raw(visitor: StructuralVisitorHandle, value: TVMFFIAny) -> TVMFFIAny {
809        if value.type_index == TVMFFITypeIndex::kTVMFFINone as i32 {
810            core::hint::cold_path();
811            return TVMFFIAny::new();
812        }
813        unsafe {
814            let callback = (*(*visitor).vtable).visit;
815            callback(visitor, AnyView::from_raw_ffi_any(value))
816        }
817    }
818
819    /// Invoke a native or packed hook with borrowed `visitor`, `value` and `hook`.
820    /// Return one owning None/Error/VisitInterrupt; packed results are checked.
821    ///
822    /// # Safety
823    /// The runtime and both borrowed inputs must be valid and live for this call;
824    /// the inputs transfer no ownership. Publish the current exclusive dispatcher
825    /// reborrow first. A native hook must be a non-null function with the visit ABI
826    /// accepting this runtime's complete layout. All hooks must obey
827    /// [`StructuralVisitorBase`]'s synchronous, same-thread, non-escaping contract.
828    #[inline(always)]
829    pub unsafe fn call_structural_visit_hook_raw(
830        visitor: *mut Self,
831        value: TVMFFIAny,
832        hook: TVMFFIAny,
833    ) -> TVMFFIAny {
834        unsafe {
835            if hook.type_index == TVMFFITypeIndex::kTVMFFIOpaquePtr as i32 {
836                let pointer = hook.data_union.v_ptr;
837                // Native pointer registration is an unsafe contract: a non-null
838                // callback with this ABI, borrowed runtime lifetime and owning result.
839                let hook: FStructuralVisit = std::mem::transmute(pointer);
840                hook(visitor, AnyView::from_raw_ffi_any(value))
841            } else {
842                core::hint::cold_path();
843                Self::call_structural_visit_hook_slow_raw(
844                    visitor,
845                    value.type_index,
846                    value.small_str_len,
847                    value.data_union,
848                    hook.type_index,
849                    hook.small_str_len,
850                    hook.data_union,
851                )
852            }
853        }
854    }
855
856    /// Reconstruct borrowed value/hook from their scalar fields for a packed call.
857    /// Internal callers keep `visitor` and both inputs live; no input transfers ownership.
858    /// Validate the owning packed result and release temporary owners before exporting
859    /// one raw token. Private C-unwind preserves Rust cleanup and AArch64 register
860    /// returns without changing the public or native callback ABI.
861    // Keep packed calls and malformed registrations out of the native-hook hot path.
862    // Outline uncommon packed-call scratch to keep native visit frames small.
863    // Match the native raw-result ABI (registers on AArch64), without adding
864    // an abort boundary around Rust packed-call cleanup.
865    #[cold]
866    #[inline(never)]
867    extern "C-unwind" fn call_structural_visit_hook_slow_raw(
868        visitor: StructuralVisitorHandle,
869        value_type_index: i32,
870        value_small_str_len: u32,
871        value_data: TVMFFIAnyDataUnion,
872        hook_type_index: i32,
873        hook_small_str_len: u32,
874        hook_data: TVMFFIAnyDataUnion,
875    ) -> TVMFFIAny {
876        // Construct addressable packed-call arguments only on this cold path.
877        let value = TVMFFIAny {
878            type_index: value_type_index,
879            small_str_len: value_small_str_len,
880            data_union: value_data,
881        };
882        let hook = TVMFFIAny {
883            type_index: hook_type_index,
884            small_str_len: hook_small_str_len,
885            data_union: hook_data,
886        };
887        // Packed hooks (including Python class methods) can return any well-formed
888        // Any through safe APIs. Their result tag needs validation before decoding.
889        let result = unsafe {
890            Function::try_from(AnyView::from_raw_ffi_any(hook)).and_then(|function| {
891                // Borrow the live runtime for this synchronous hook; it may not retain it.
892                let object = visitor.cast::<TVMFFIObject>();
893                let mut runtime = TVMFFIAny::new();
894                runtime.type_index = (*object).type_index;
895                runtime.data_union.v_obj = object;
896                function.call_packed(&[
897                    AnyView::from_raw_ffi_any(runtime),
898                    AnyView::from_raw_ffi_any(value),
899                ])
900            })
901        };
902        let result = match result {
903            Ok(value) => checked_hook_result(value),
904            Err(error) => error.into(),
905        };
906        unsafe { Any::into_raw_ffi_any(result) }
907    }
908
909    /// Run `callback` under requested `kind` on live borrowed `visitor`, preserving
910    /// Pattern precedence. Restore prior region on return or Rust unwind. Internal
911    /// callers keep the runtime live and retain no reference to its region field.
912    #[inline(always)]
913    fn with_def_region_kind<T>(
914        visitor: StructuralVisitorHandle,
915        kind: DefRegionKind,
916        callback: impl FnOnce() -> T,
917    ) -> T {
918        unsafe {
919            let previous = *(*visitor).def_region_mode.get();
920            if previous == kind as i32 || previous == DefRegionKind::Pattern as i32 {
921                // Inherited descent does not change state; Pattern keeps precedence.
922                return callback();
923            }
924            *(*visitor).def_region_mode.get() = kind as i32;
925            struct Restore {
926                visitor: StructuralVisitorHandle,
927                previous: i32,
928            }
929            impl Drop for Restore {
930                #[inline(always)]
931                fn drop(&mut self) {
932                    unsafe { *(*self.visitor).def_region_mode.get() = self.previous };
933                }
934            }
935            let _restore = Restore { visitor, previous };
936            callback()
937        }
938    }
939}

941// Generic runtime: only this implementation projects the dispatcher after Base.
942impl<Base: StructuralVisitorBase> StructuralVisitEngine<Base> {
943    /// Recover the Rust engine whose first field is this ABI base.
944    ///
945    /// # Safety
946    /// The handle must belong to the live, synchronous Rust traversal that created
947    /// this exact repr(C) engine. An arbitrary native visitor is not such an engine.
948    #[inline(always)]
949    unsafe fn from_handle(handle: StructuralVisitorHandle) -> *mut Self {
950        handle.cast()
951    }
952
953    /// Publish the current exclusive `dispatch` reborrow and visit borrowed `raw`.
954    /// Return one owning raw None/interrupt/Error token.
955    ///
956    /// # Safety
957    /// `runtime` must belong to this live `D` invocation and `raw` remain valid
958    /// through recursion; the caller cannot use competing dispatcher borrows.
959    #[inline(always)]
960    unsafe fn visit_raw<D>(
961        runtime: StructuralVisitorHandle,
962        dispatch: &mut D,
963        raw: TVMFFIAny,
964    ) -> TVMFFIAny {
965        let context = Self::checked_dispatcher(runtime, dispatch);
966        (*Self::from_handle(runtime)).dispatcher = context;
967        StructuralVisitorObj::call_visitor_raw(runtime, raw)
968    }
969
970    /// Check runtime identity and return the current `dispatch` reborrow pointer.
971    ///
972    /// # Safety
973    /// `runtime` must be the live engine for `D`; address equality does not establish
974    /// type or lifetime validity. Mismatch panics before a pointer is published.
975    // The caller has already established matching runtime/D types and scope.
976    // Read only the stored raw address: do not create a competing mutable borrow.
977    #[inline(always)]
978    unsafe fn checked_dispatcher<D>(
979        runtime: StructuralVisitorHandle,
980        dispatch: &mut D,
981    ) -> *mut c_void {
982        let current = dispatcher_context(dispatch);
983        if current != (*Self::from_handle(runtime)).dispatcher {
984            core::hint::cold_path();
985            wrong_visitor_dispatcher();
986        }
987        current
988    }
989
990    /// Build the native visitor callback for concrete dispatcher `D`. Its higher-ranked
991    /// view arguments restrict all borrows to the synchronous handler invocation.
992    #[inline(always)]
993    fn vtable<D>() -> &'static StructuralVisitorVTable
994    where
995        for<'s, 'v, 'c, 'r> &'s mut D: crate::Dispatcher<
996                (AnyView<'v>, &'c mut StructuralVisitorView<'r, D, Base>),
997                Option<Result<Option<VisitInterrupt>>>,
998            >,
999    {
1000        &StructuralVisitorVTable {
1001            visit: Self::rust_vtable_visit_raw::<D>,
1002        }
1003    }
1004
1005    /// Enter a native visitor callback with borrowed `value`; return one owning token.
1006    /// Restore the incoming dispatcher provenance before returning to the caller.
1007    ///
1008    /// # Safety
1009    /// `runtime` must have this `D` vtable and a live published reborrow. The erased
1010    /// ABI lifetime on `value` is valid only for this call. Callback failures use
1011    /// Result; Rust panics cannot cross this native C boundary.
1012    unsafe extern "C" fn rust_vtable_visit_raw<D>(
1013        runtime: StructuralVisitorHandle,
1014        value: AnyView<'static>,
1015    ) -> TVMFFIAny
1016    where
1017        for<'s, 'v, 'c, 'r> &'s mut D: crate::Dispatcher<
1018                (AnyView<'v>, &'c mut StructuralVisitorView<'r, D, Base>),
1019                Option<Result<Option<VisitInterrupt>>>,
1020            >,
1021    {
1022        let context = (*Self::from_handle(runtime)).dispatcher;
1023        let raw = *value.as_raw_ffi_any();
1024        let result = if raw.type_index == TVMFFITypeIndex::kTVMFFINone as i32 {
1025            core::hint::cold_path();
1026            TVMFFIAny::new()
1027        } else {
1028            let dispatch = &mut *context.cast::<D>();
1029            let mut visitor = Self::visitor_context::<D>(runtime);
1030            // SAFETY: the native caller keeps the input alive for this synchronous
1031            // dispatch; the higher-ranked callback cannot retain its borrowed view.
1032            match crate::Dispatcher::dispatch_tuple(
1033                &mut *dispatch,
1034                (AnyView::from_raw_ffi_any(raw), &mut visitor),
1035            ) {
1036                Some(result) => visit_result_raw(result, raw),
1037                None => Self::default_visit_raw(runtime, dispatcher_context(dispatch), raw),
1038            }
1039        };
1040        // The enclosing callback may resume using its ancestor mutable borrow.
1041        // Native hooks cannot restore Rust-only borrow provenance for us.
1042        (*Self::from_handle(runtime)).dispatcher = context;
1043        result
1044    }
1045
1046    /// Create the handler view for borrowed `runtime`, without retaining it.
1047    ///
1048    /// # Safety
1049    /// The live runtime's vtable must match D. Keep this view within the current
1050    /// higher-ranked handler call, never outliving the runtime or dispatcher.
1051    #[inline(always)]
1052    unsafe fn visitor_context<'a, D>(
1053        runtime: StructuralVisitorHandle,
1054    ) -> StructuralVisitorView<'a, D, Base> {
1055        StructuralVisitorView {
1056            runtime,
1057            _scope: PhantomData,
1058        }
1059    }
1060
1061    /// Publish `context` and descend through borrowed `raw`, then annotate errors.
1062    /// The raw return transfers one owning None/interrupt/Error token.
1063    ///
1064    /// # Safety
1065    /// `runtime` must be live and `context` its current exclusive dispatcher reborrow;
1066    /// `raw` remains borrowed and must survive the complete synchronous descent.
1067    #[inline(always)]
1068    unsafe fn default_visit_raw(
1069        runtime: StructuralVisitorHandle,
1070        context: *mut c_void,
1071        raw: TVMFFIAny,
1072    ) -> TVMFFIAny {
1073        // Context is the supplied exclusive reborrow. Publish it before descent;
1074        // the handle-only scanner never creates or uses a competing D reference.
1075        (*Self::from_handle(runtime)).dispatcher = context;
1076        StructuralVisitorObj::with_visit_context_raw(
1077            Base::default_visit_raw(runtime.cast(), AnyView::from_raw_ffi_any(raw)),
1078            raw,
1079        )
1080    }
1081
1082    /// Run the public visit/walk adapter with borrowed `root` and exclusive `dispatch`.
1083    /// Internal callers keep the root owner live. Own the result across engine Drop,
1084    /// including unwind, then transfer one raw None/interrupt/Error token. Hooks must
1085    /// release temporary runtime owners and cannot retain the scoped dispatcher.
1086    pub(super) fn run_visitor_raw<D>(root: TVMFFIAny, dispatch: &mut D) -> TVMFFIAny
1087    where
1088        for<'s, 'v, 'c, 'r> &'s mut D: crate::Dispatcher<
1089                (AnyView<'v>, &'c mut StructuralVisitorView<'r, D, Base>),
1090                Option<Result<Option<VisitInterrupt>>>,
1091            >,
1092    {
1093        let base = match Base::new() {
1094            Ok(base) => base,
1095            Err(error) => return unsafe { Any::into_raw_ffi_any(error.into()) },
1096        };
1097        let type_index = Base::type_index();
1098        // The managed runtime may borrow its dispatcher only for this call.
1099        // Trusted hooks release all temporary handles before the owner is dropped.
1100        let mut engine = unsafe {
1101            ObjectArc::<Object>::new_opaque_unchecked(
1102                StructuralVisitEngine {
1103                    base,
1104                    dispatcher: dispatcher_context(dispatch),
1105                },
1106                type_index,
1107            )
1108        };
1109        let handle = unsafe { ObjectArc::as_raw_mut(&mut engine).cast::<StructuralVisitorObj>() };
1110        // The engine and outer dispatcher storage stay stationary for the whole run.
1111        // Only raw field projections cross recursion; no whole-engine borrow survives.
1112        unsafe {
1113            (*handle).vtable = Self::vtable::<D>();
1114            *(*handle).def_region_mode.get() = DefRegionKind::None as i32;
1115        }
1116        // Keep the result owned while releasing the runtime, including on unwind.
1117        let result =
1118            unsafe { Any::from_raw_ffi_any(StructuralVisitorObj::call_visitor_raw(handle, root)) };
1119        // Trusted hooks have released every temporary strong/weak runtime reference.
1120        drop(engine);
1121        unsafe { Any::into_raw_ffi_any(result) }
1122    }
1123}

1129/// Visit `root` with an owned or borrowed [`Dispatcher`](crate::Dispatcher).
1130///
1131/// Handlers receive [`StructuralVisitorView`] for explicit recursion and keep state on the
1132/// dispatcher. Unmatched values use default child traversal. Pass `&mut D` to
1133/// keep the dispatcher and inspect its state after the visit. An owned dispatcher
1134/// is held in the outer call and uses the same borrowed traversal engine.
1135///
1136/// Callbacks must report failures through [`Result`] and must not panic: a panic
1137/// reaching the non-unwinding native callback boundary aborts the process.
1138///
1139/// # Arguments
1140///
1141/// * `root` - Value borrowed for traversal, convertible into an [`AnyView`].
1142/// * `dispatcher` - Owned or mutably borrowed dispatcher implementing [`IntoDispatcher`].
1143///   Matching handlers control recursion through [`StructuralVisitorView`];
1144///   unmatched values use default child traversal.
1145///
1146/// # Examples
1147///
1148/// Sum integer leaves while default traversal descends into the array:
1149///
1150/// ```
1151/// # pub use tvm_ffi::{extra, match_any, match_any_internal, Dispatcher};
1152/// # fn main() -> tvm_ffi::Result<()> {
1153/// use tvm_ffi::{
1154///     dispatch, structural_visit, AnyView, Array, Result, StructuralVisitorView,
1155///     VisitInterrupt,
1156/// };
1157///
1158/// struct Sum(i64);
1159/// #[dispatch(visit, (AnyView<'_>, &mut StructuralVisitorView<'_, Self>), Result<Option<VisitInterrupt>>)]
1160/// impl Sum {
1161///     fn visit_integer(
1162///         &mut self,
1163///         value: i64,
1164///         _: &mut StructuralVisitorView<'_, Self>,
1165///     ) -> Result<Option<VisitInterrupt>> {
1166///         self.0 += value;
1167///         Ok(None)
1168///     }
1169/// }
1170///
1171/// let root = Array::new(vec![1_i64, 2, 3]);
1172/// let mut sum = Sum(0);
1173/// assert!(structural_visit(&root, &mut sum)?.is_none());
1174/// assert_eq!(sum.0, 6);
1175/// # Ok(())
1176/// # }
1177/// ```
1178pub fn structural_visit<'input, R, H, M, D, O>(
1179    root: R,
1180    dispatcher: H,
1181) -> Result<Option<VisitInterrupt>>
1182where
1183    R: Into<AnyView<'input>>,
1184    O: BorrowMut<D>,
1185    for<'v, 'c, 'r> H: IntoDispatcher<
1186            (
1187                AnyView<'v>,
1188                &'c mut StructuralVisitorView<'r, D, StructuralVisitorObj>,
1189            ),
1190            Option<Result<Option<VisitInterrupt>>>,
1191            M,
1192            Target = D,
1193            Output = O,
1194        >,
1195    for<'s, 'v, 'c, 'r> &'s mut D: crate::Dispatcher<
1196            (
1197                AnyView<'v>,
1198                &'c mut StructuralVisitorView<'r, D, StructuralVisitorObj>,
1199            ),
1200            Option<Result<Option<VisitInterrupt>>>,
1201        >,
1202{
1203    StructuralVisitEngine::visit_root(root, dispatcher)
1204}

1206#[inline(always)]
1207pub(super) fn visit_result_from_any(result: Any) -> Result<Option<VisitInterrupt>> {
1208    let tag = result.type_index();
1209    if tag == TVMFFITypeIndex::kTVMFFINone as i32 {
1210        return Ok(None);
1211    }
1212    core::hint::cold_path();
1213    if tag == TVMFFITypeIndex::kTVMFFIError as i32 {
1214        // SAFETY: internal results obey the documented owning carrier protocol.
1215        Err(unsafe { take_error(result) })
1216    } else {
1217        // SAFETY: the owning result protocol established a non-null interrupt.
1218        // Transfer that same reference, including carriers shared by native hooks.
1219        use crate::type_traits::AnyCompatible;
