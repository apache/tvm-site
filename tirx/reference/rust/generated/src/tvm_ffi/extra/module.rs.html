19//! [`Module`], a loaded library of functions.
20
21use crate::derive::{Object, ObjectRef};
22use crate::error::Result;
23use crate::function::Function;
24use crate::object::{Object, ObjectArc};
25use crate::string::String;
26use tvm_ffi_sys::TVMFFITypeIndex as TypeIndex;
27
28//-----------------------------------------------------
29// Module
30//-----------------------------------------------------
31
32/// The object behind a [`Module`].
33#[repr(C)]
34#[derive(Object)]
35#[type_key = "ffi.Module"]
36#[type_index(unsafe TypeIndex::kTVMFFIModule)]
37pub struct ModuleObj {
38    object: Object,
39}
40
41/// A collection of functions, such as a dynamic library, passed across
42/// languages as `ffi.Module`.
43#[repr(C)]
44#[derive(ObjectRef, Clone)]
45#[object_ref(skip_deref)]
46pub struct Module {
47    data: ObjectArc<ModuleObj>,
48}
49
50impl Module {
51    /// Load a module from a dynamic library file.
52    ///
53    /// # Arguments
54    /// * `file_name` - The path of the library.
55    ///
56    /// # Returns
57    /// The module.
58    ///
59    /// # Errors
60    /// If the library cannot be loaded.
61    pub fn load_from_file<Str: AsRef<str>>(file_name: Str) -> Result<Module> {
62        let file_name = crate::string::String::from(file_name);
63        crate::cached_global_func!("ffi.ModuleLoadFromFile")
64            .call_tuple_with_len::<1, _>((file_name,))?
65            .try_into()
66    }
67
68    /// Look up a function of the module by name.
69    ///
70    /// Only this module is searched, not the modules it imports.
71    ///
72    /// # Arguments
73    /// * `name` - The name of the function.
74    ///
75    /// # Returns
76    /// The function.
77    ///
78    /// # Errors
79    ///
80    /// Returns an error if the module has no function `name`, as Python's
81    /// `Module.get_function` raises.
82    pub fn get_function<Str: AsRef<str>>(&self, name: Str) -> Result<Function> {
83        let name = crate::string::String::from(name);
84        crate::cached_global_func!("ffi.ModuleGetFunction")
85            .call_tuple_with_len::<3, _>((self, name, false))?
86            .try_into()
87    }
88
89    /// As [`get_function`](Self::get_function), but also searching the modules
90    /// this one imports, depth first in import order.
91    pub fn get_function_with_imports<Str: AsRef<str>>(&self, name: Str) -> Result<Function> {
92        let name = crate::string::String::from(name);
93        crate::cached_global_func!("ffi.ModuleGetFunction")
94            .call_tuple_with_len::<3, _>((self, name, true))?
95            .try_into()
96    }
97
98    /// Whether the module implements the function `name`.
99    ///
100    /// Only this module is searched, not the modules it imports.
101    ///
102    /// # Arguments
103    /// * `name` - The name of the function.
104    ///
105    /// # Returns
106    /// Whether the function exists.
107    pub fn implements_function<Str: AsRef<str>>(&self, name: Str) -> Result<bool> {
108        let name = String::from(name);
109        crate::cached_global_func!("ffi.ModuleImplementsFunction")
110            .call_tuple_with_len::<3, _>((self, name, false))?
111            .try_into()
112    }
113
114    /// As [`implements_function`](Self::implements_function), but also searching the modules
115    /// this one imports, depth first in import order.
116    pub fn implements_function_with_imports<Str: AsRef<str>>(&self, name: Str) -> Result<bool> {
117        let name = String::from(name);
118        crate::cached_global_func!("ffi.ModuleImplementsFunction")
119            .call_tuple_with_len::<3, _>((self, name, true))?
120            .try_into()
121    }
122
123    /// Get the metadata of a function the module exports.
124    ///
125    /// Only this module is searched, not the modules it imports.
126    ///
127    /// The metadata is a JSON object whose `type_schema` entry is the
128    /// function's signature, as an export with metadata enabled records it
129    /// (see [`tvm_ffi_dll_export_typed_func!`](crate::tvm_ffi_dll_export_typed_func)).
130    ///
131    /// # Arguments
132    /// * `name` - The name of the function.
133    ///
134    /// # Returns
135    /// The metadata as a JSON string, or `None` if the function has none.
136    pub fn get_function_metadata<Str: AsRef<str>>(&self, name: Str) -> Result<Option<String>> {
137        let name = String::from(name);
138        crate::cached_global_func!("ffi.ModuleGetFunctionMetadata")
139            .call_tuple_with_len::<3, _>((self, name, false))?
140            .try_into()
141    }
142
143    /// As [`get_function_metadata`](Self::get_function_metadata), but also searching the modules
144    /// this one imports, depth first in import order.
145    pub fn get_function_metadata_with_imports<Str: AsRef<str>>(
146        &self,
147        name: Str,
148    ) -> Result<Option<String>> {
149        let name = String::from(name);
150        crate::cached_global_func!("ffi.ModuleGetFunctionMetadata")
151            .call_tuple_with_len::<3, _>((self, name, true))?
152            .try_into()
153    }
154
155    /// Get the doc string of a function the module exports.
156    ///
157    /// Only this module is searched, not the modules it imports.
158    ///
159    /// The doc string is the one the export recorded (see
160    /// [`tvm_ffi_dll_export_typed_func_doc!`](crate::tvm_ffi_dll_export_typed_func_doc)).
161    ///
162    /// # Arguments
163    /// * `name` - The name of the function.
164    ///
165    /// # Returns
166    /// The doc string, or `None` if the function has none.
167    pub fn get_function_doc<Str: AsRef<str>>(&self, name: Str) -> Result<Option<String>> {
168        let name = String::from(name);
169        crate::cached_global_func!("ffi.ModuleGetFunctionDoc")
170            .call_tuple_with_len::<3, _>((self, name, false))?
171            .try_into()
172    }
173
174    /// As [`get_function_doc`](Self::get_function_doc), but also searching the modules
175    /// this one imports, depth first in import order.
176    pub fn get_function_doc_with_imports<Str: AsRef<str>>(
177        &self,
178        name: Str,
179    ) -> Result<Option<String>> {
180        let name = String::from(name);
181        crate::cached_global_func!("ffi.ModuleGetFunctionDoc")
182            .call_tuple_with_len::<3, _>((self, name, true))?
183            .try_into()
184    }
185}
