20//! Runtime type dispatch indexed by its tuple of arguments.
21
22/// Runtime type dispatch with a complete handler outcome.
23///
24/// Implementations generated by `#[dispatch(prefix, Args, R)]` use a borrowed receiver
25/// (`&T` or `&mut T`) as `Self` and use `Ret = Option<R>`. Only the outer
26/// `None` denotes no match: a matched error or inner `None` remains `Some(result)`
27/// and does not try another handler. Consuming the receiver reference preserves
28/// results borrowing it while ordinary method calls automatically borrow it.
29/// The argument tuple and complete return type select an implementation.
30#[diagnostic::on_unimplemented(
31    message = "this receiver does not implement Dispatcher for the requested argument tuple",
32    note = "use #[dispatch(prefix, (Carrier, ...), Ret)] with compatible typed handlers"
33)]
34pub trait Dispatcher<Args, Ret> {
35    fn dispatch_tuple(self, args: Args) -> Ret;
36}

72/// Inferred selectors for dispatcher normalization.
73///
74/// Entry points infer one marker from the input. Callback markers select argument
75/// matching and a leading argument count; tuple markers contain each link's marker.
76/// Normalization stores the input once, before traversal, without cloning it.
77pub mod marker {
78    use std::marker::PhantomData;
79    /// Store an owned dispatcher with the exact target argument and result types.
80    pub enum FromOwned {}
81    /// Store a mutable borrow of an exact-signature dispatcher.
82    pub enum FromBorrowed {}
83    /// Match a callback's first argument using `Shape`, then pass its first `N`
84    /// arguments. `N` ranges from one through four, including the matched value.
85    pub struct FromCallback<const N: usize, Shape>(PhantomData<fn() -> Shape>);
86    /// Try tuple links in order. `Markers` is their marker tuple, including nested
87    /// tuples; `Args` selects how arguments are reborrowed between links.
88    pub struct FromTuple<Markers, Args = CopyArgs>(PhantomData<fn() -> (Markers, Args)>);
89    /// Replay copyable observer arguments; this does not copy dispatcher storage.
90    pub enum CopyArgs {}
91    /// Sequentially reborrow a mutable second argument for each tuple link.
92    pub enum MutArg {}
93    /// Match and convert the first `AnyView` to the FFI value type `T`.
94    pub struct Value<T>(PhantomData<fn() -> T>);
95    /// Match a borrowed object node `&T` without retaining it, unlike the
96    /// by-value conversion selected by [`Value<T>`].
97    pub struct BorrowedObject<T>(PhantomData<fn() -> T>);
98    /// Pass the first `AnyView` without a runtime type filter.
99    pub enum AnyView {}
100    /// Match a mutation capability for `T`, retaining its scoped permission.
101    pub struct Mutation<T = crate::Any>(PhantomData<fn() -> T>);
102}

104/// Convert an input to a reusable [`Dispatcher`].
105///
106/// `Args` is the complete target argument tuple, `Ret` its complete result, and
107/// `M` the inferred input-form marker. `Target` is the actual dispatcher type;
108/// `Output` owns or borrows it and provides `BorrowMut<Target>`. Owned `D` and
109/// borrowed `&mut D` inputs both preserve target `D`; callbacks and tuples use an
110/// adapter as their target. This keeps recursive view types tied to the dispatcher.
111///
112/// Inputs may be owned dispatchers, mutable dispatcher borrows, callbacks, or
113/// nested tuples mixing these forms. Dispatchers require the exact argument and
114/// result signature. Callbacks may accept a leading prefix of one to four
115/// arguments; structural callbacks match an FFI value, an object-node borrow, or
116/// `AnyView`; mutation callbacks match a typed `MutationView`. Callable values
117/// and their shared or mutable borrows are stored as supplied, subject to their `FnMut` implementation. Only an outer `None` tries
118/// the next tuple link; a matched error or inner `None` stops the chain.
119///
120/// Supply the target argument and result types when calling the conversion directly;
121/// the input-form marker remains inferred.
122///
123/// ```
124/// use tvm_ffi::{Any, AnyView, Dispatcher, IntoDispatcher, Result};
125/// let callbacks = (|value: i64| Any::from(value + 1), |_value: AnyView<'_>| Any::new());
126/// let mut dispatcher = IntoDispatcher::<(AnyView<'_>,), Option<Result<Any>>, _>::into_dispatcher(callbacks);
127/// let result: Option<Result<Any>> = dispatcher.dispatch_tuple((AnyView::from(&41_i64),));
128/// let result = result.unwrap()?;
129/// assert_eq!(result.try_as::<i64>(), Some(42));
130/// # Ok::<(), tvm_ffi::Error>(())
131/// ```
132pub trait IntoDispatcher<Args, Ret, M> {
133    /// Actual dispatcher type, independent of its owned or borrowed storage.
134    type Target;
135    /// Storage normalized once at entry, retaining the target's identity.
136    type Output: BorrowMut<Self::Target>;
137    fn into_dispatcher(self) -> Self::Output;
138}

140/// Normalized callback or tuple storage; direct dispatchers retain their own type.
141#[doc(hidden)]
142pub struct DispatchAdapter<F, M> {
143    input: F,
144    marker: PhantomData<fn() -> M>,
145}

147impl<D, Args, Ret> IntoDispatcher<Args, Ret, marker::FromOwned> for D
148where
149    for<'s> &'s mut D: Dispatcher<Args, Ret>,
150{
151    type Target = D;
152    type Output = D;
153    #[inline(always)]
154    fn into_dispatcher(self) -> Self::Output {
155        self
156    }
157}

163    type Target = D;
164    type Output = &'d mut D;
165    #[inline(always)]
166    fn into_dispatcher(self) -> Self::Output {

168    }

203/// Implementation boundary shared by callback and tuple adapters.
204#[doc(hidden)]
205pub trait DispatchLink<Args, Ret, M> {
206    fn dispatch_link(&mut self, args: Args) -> Ret;
207}
208
209impl<F, Args, Ret, M> Dispatcher<Args, Ret> for &mut DispatchAdapter<F, M>
210where
211    F: DispatchLink<Args, Ret, M>,
212{
213    #[inline(always)]
214    fn dispatch_tuple(self, args: Args) -> Ret {
215        self.input.dispatch_link(args)
216    }
217}
218
219impl<D, Args, Ret> DispatchLink<Args, Ret, marker::FromOwned> for D
220where
221    for<'s> &'s mut D: Dispatcher<Args, Ret>,
222{
223    #[inline(always)]
224    fn dispatch_link(&mut self, args: Args) -> Ret {
225        self.dispatch_tuple(args)
226    }
227}
228impl<D, Args, Ret> DispatchLink<Args, Ret, marker::FromBorrowed> for &mut D
229where
230    for<'s> &'s mut D: Dispatcher<Args, Ret>,
231{
232    #[inline(always)]
233    fn dispatch_link(&mut self, args: Args) -> Ret {
234        (&mut **self).dispatch_tuple(args)
235    }
236}

238/// Operation-specific callback result conversion; matching and chaining are shared.

240pub trait CallbackResult<R> {
241    fn callback_result(self) -> R;
242}

244macro_rules! callback_prefix {
245    ($n:literal; ($($A:ident:$a:ident),*); ($($B:ident:$b:ident),*)) => {
246        impl<'v, F, T, O, R, $($A,)*>
247            DispatchLink<(AnyView<'v>, $($A,)*), Option<R>, marker::FromCallback<$n, marker::Value<T>>> for F
248        where F: FnMut(T, $($B,)*) -> O, T: crate::type_traits::AnyCompatible,
249              O: CallbackResult<R>,
250        {
251            #[inline(always)]
252            fn dispatch_link(&mut self, (value, $($a,)*): (AnyView<'v>, $($A,)*)) -> Option<R> {
253                $(let _ = &$a;)*
254                value.try_as::<T>().map(|typed| self(typed, $($b,)*).callback_result())
255            }
256        }
257        impl<'v, F, T, O, R, $($A,)*>
258            DispatchLink<(AnyView<'v>, $($A,)*), Option<R>, marker::FromCallback<$n, marker::BorrowedObject<T>>> for F
259        where F: for<'node> FnMut(&'node T, $($B,)*) -> O, T: crate::ObjectCore,
260              O: CallbackResult<R>,
261        {
262            #[inline(always)]
263            fn dispatch_link(&mut self, (value, $($a,)*): (AnyView<'v>, $($A,)*)) -> Option<R> {
264                $(let _ = &$a;)*
265                value.downcast_ref::<T>().map(|node| self(node, $($b,)*).callback_result())
266            }
267        }
268        impl<'v, F, O, R, $($A,)*>
269            DispatchLink<(AnyView<'v>, $($A,)*), Option<R>, marker::FromCallback<$n, marker::AnyView>> for F
270        where F: FnMut(AnyView<'v>, $($B,)*) -> O, O: CallbackResult<R>,
271        {
272            #[inline(always)]
273            fn dispatch_link(&mut self, (value, $($a,)*): (AnyView<'v>, $($A,)*)) -> Option<R> {
274                $(let _ = &$a;)*
275                Some(self(value, $($b,)*).callback_result())
276            }
277        }
278    };
279}

317// Concrete tuple storage bounds keep trait inference from recursively inventing
318// unknown adapter receiver layers. Tuple arity is independent of callback arity.
319macro_rules! dispatch_tuple_links {
320    ($(($F:ident, $M:ident, $idx:tt)),+) => {
321        impl<Args: Copy, R, $($F, $M,)+>
322            DispatchLink<Args, Option<R>, marker::FromTuple<($($M,)+)>> for ($($F,)+)
323        where $($F: DispatchLink<Args, Option<R>, $M>,)+
324        {
325            #[inline(always)]
326            fn dispatch_link(&mut self, args: Args) -> Option<R> {
327                $(if let Some(result) = self.$idx.dispatch_link(args) { return Some(result); })+
328                None
329            }
330        }
331        impl<'v, 'c, C, R, $($F, $M,)+>
332            DispatchLink<(AnyView<'v>, &'c mut C), Option<R>, marker::FromTuple<($($M,)+), marker::MutArg>> for ($($F,)+)
333        where $(for<'borrow> $F: DispatchLink<(AnyView<'v>, &'borrow mut C), Option<R>, $M>,)+
334        {
335            #[inline(always)]
336            fn dispatch_link(&mut self, (value, context): (AnyView<'v>, &'c mut C)) -> Option<R> {
337                $(if let Some(result) = self.$idx.dispatch_link((value, &mut *context)) { return Some(result); })+
338                None
339            }
340        }
341        impl<'v, 'c, C, R, $($F, $M,)+>
342            DispatchLink<(crate::MutationView<'v>, &'c mut C), Option<R>, marker::FromTuple<($($M,)+), marker::MutArg>> for ($($F,)+)
343        where $(for<'value, 'borrow> $F: DispatchLink<(crate::MutationView<'value>, &'borrow mut C), Option<R>, $M>,)+
344        {
345            #[inline(always)]
346            fn dispatch_link(&mut self, (mut value, context): (crate::MutationView<'v>, &'c mut C)) -> Option<R> {
347                $(if let Some(result) = self.$idx.dispatch_link((value.reborrow(), &mut *context)) { return Some(result); })+
348                None
349            }
350        }
351    };
352}
