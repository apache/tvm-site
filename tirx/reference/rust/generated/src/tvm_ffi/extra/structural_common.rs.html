57/// Consume an owning error carrier while keeping checked-conversion scratch
58/// inside the cold path instead of successful recursive traversal frames.
59///
60/// # Safety
61/// The fields must describe one owning TVMFFIAny that has not been consumed.
62// Perf note: Construct the addressable Any and perform checked Error conversion
63// inside this outline boundary, not in the successful recursive caller's frame.
64#[cold]
65#[inline(never)]
66pub(super) unsafe fn structural_error_from_raw(
67    type_index: i32,
68    small_str_len: u32,
69    data_union: TVMFFIAnyDataUnion,
70) -> Error {
71    let value = Any::from_raw_ffi_any(TVMFFIAny {
72        type_index,
73        small_str_len,
74        data_union,
75    });
76    match Error::try_from(value) {
77        Ok(error) | Err(error) => error,
78    }
79}

81// Perf note: cold marks these error-composition paths as uncommon; inlining
82// within error paths is compiler-selected. Use explicit inline(never) at boundaries

84/// Add one structural traversal frame to an error's backtrace.

87    Error::with_appended_backtrace(error, &format!("[native structural {operation}] {frame}\n"))

90/// Rust binding of C++ `VisitErrorContextObj`, the payload that
91/// `Error::extra_context` carries to name the nodes a failed visit unwound through.
92#[repr(C)]
93#[derive(Object)]
94#[type_key = "ffi.VisitErrorContext"]
95#[type_final]
96pub(crate) struct VisitErrorContextObj {
97    base: Object,
98    /// Visited nodes, innermost first; C++ `ObjectRef` is nullable.
99    reverse_visit_pattern: Array<Option<ObjectRef>>,
100    /// The `extra_context` the error carried before this context replaced it.
101    prev_error_context: Option<ObjectRef>,
102}
103
104/// Name the visited object in `raw` in the error's visit context, like C++
105/// `detail::UpdateVisitErrorContext`. Copy-on-write at each level: the Error, its
106/// context and the context's pattern are each copied only when shared, so an error
107/// owned by the unwinding frame is appended in place.
108#[cold]
109pub(crate) fn with_visit_error_context(mut error: Error, raw: TVMFFIAny) -> Error {
110    if raw.type_index < TVMFFITypeIndex::kTVMFFIStaticObjectBegin as i32 {
111        return error;
112    }
113    // SAFETY: the traversal input remains live during annotation. Conversion
114    // acquires ownership before the temporary view ends; raw is an object tag.
115    let Ok(node) = ObjectRef::try_from(unsafe { AnyView::from_raw_ffi_any(raw) }) else {
116        return error;
117    };
118    let as_context = |context: &ObjectRef| {
119        let object = ObjectArc::as_ptr(ObjectRef::data(context));
120        // SAFETY: `context` keeps its object alive; the type is checked first.
121        unsafe {
122            object::is_instance_of::<VisitErrorContextObj>(
123                (*object.cast::<TVMFFIObject>()).type_index,
124            )
125            .then(|| &*object.cast::<VisitErrorContextObj>())
126        }
127    };
128    // Callback, policy and default descent can report the same frame.
129    if let Some(context) = error.extra_context().and_then(as_context) {
130        if context
131            .reverse_visit_pattern
132            .iter()
133            .next_back()
134            .and_then(Option::as_ref)
135            .is_some_and(|last| last.same_as(&node))
136        {
137            return error;
138        }
139    }
140    let slot = error.extra_context_mut();
141    let previous = slot.take();
142    let context = match previous {
143        Some(previous) if as_context(&previous).is_some() => {
144            // SAFETY: `previous` was checked to be a `VisitErrorContextObj`.
145            let mut context = unsafe {
146                ObjectArc::from_raw(
147                    ObjectArc::into_raw(ObjectRef::into_data(previous))
148                        .cast::<VisitErrorContextObj>(),
149                )
150            };
151            if !ObjectArc::is_unique(&context) {
152                context = ObjectArc::new(VisitErrorContextObj {
153                    base: Object::new(),
154                    reverse_visit_pattern: context.reverse_visit_pattern.clone(),
155                    prev_error_context: context.prev_error_context.clone(),
156                });
157            }
158            // SAFETY: this handle is the context's only reference.
159            unsafe {
160                (*ObjectArc::as_mut_ptr(&mut context))
161                    .reverse_visit_pattern
162                    .push(Some(node))
163            };
164            context
165        }
166        previous => ObjectArc::new(VisitErrorContextObj {
167            base: Object::new(),
168            reverse_visit_pattern: Array::from_vec(vec![Some(node)]),
169            prev_error_context: previous,
170        }),
171    };
172    // SAFETY: every object starts with the root `Object` header.
173    *slot = Some(unsafe {
174        ObjectRef::from_data(ObjectArc::from_raw(
175            ObjectArc::into_raw(context).cast::<Object>(),
176        ))
177    });
178    error
179}

181// Generate the tuple arities supported by the standard library (1 through 12).
182macro_rules! impl_callback_chain_tuple_arities {
183    ($impl_chain:ident) => {
184        impl_callback_chain_tuple_arities!(
185            @prefixes $impl_chain;
186            [];
187            (F0, M0, 0),
188            (F1, M1, 1),
189            (F2, M2, 2),
190            (F3, M3, 3),
191            (F4, M4, 4),
192            (F5, M5, 5),
193            (F6, M6, 6),
194            (F7, M7, 7),
195            (F8, M8, 8),
196            (F9, M9, 9),
197            (F10, M10, 10),
198            (F11, M11, 11)
199        );
200    };
201    (@prefixes $impl_chain:ident; [$($prefix:tt)*];) => {};
202    (
203        @prefixes $impl_chain:ident;
204        [$($prefix:tt)*];
205        $next:tt $(, $rest:tt)*
206    ) => {
207        $impl_chain!($($prefix)* $next);
208        impl_callback_chain_tuple_arities!(
209            @prefixes $impl_chain;
210            [$($prefix)* $next,];
211            $($rest),*
212        );
213    };
214}

222/// Compare tags, inline metadata and payload bits without deep equality.
223///
224/// Object-backed values compare pointer identity. Both carriers must contain
225/// initialized payload bits; no pointee is accessed or ownership transferred.
226#[inline(always)]
227pub(crate) fn raw_ffi_any_same_as(lhs: TVMFFIAny, rhs: TVMFFIAny) -> bool {
228    lhs.type_index == rhs.type_index
229        && lhs.small_str_len == rhs.small_str_len
230        && unsafe { lhs.data_union.v_uint64 == rhs.data_union.v_uint64 }
231}

233/// Transfer an owning success or Error into exactly one raw result token.
234///
235/// This is the structural hook protocol: an Error tag represents failure, so
236/// decoding `Ok(Any::from(error))` also yields `Err(error)`. Callers that carry
237/// Error objects as ordinary values must keep a separate result channel.
238/// No payload is cloned, normalized, or released by this transfer.
239#[inline(always)]
240pub(crate) fn result_into_raw_ffi_any(result: Result<Any>) -> TVMFFIAny {
241    match result {
242        Ok(value) => Any::into_raw_ffi_any(value),
243        Err(error) => Any::into_raw_ffi_any(Any::from(error)),
244    }
245}
246
247/// Take ownership of one value returned by a structural traversal ABI hook.
248///
249/// # Safety
250///
251/// `raw` must contain one owning TVMFFIAny result that has not already been
252/// consumed. An Error object is converted into the Rust error channel, even
253/// if it was encoded from `Ok`; all other tags become an owning success.
254/// Borrowed views and rvalue slots must be normalized before this transfer.
255#[inline(always)]
256pub(crate) unsafe fn raw_ffi_any_into_result(raw: TVMFFIAny) -> Result<Any> {
257    if raw.type_index != TVMFFITypeIndex::kTVMFFIError as i32 {
258        return Ok(Any::from_raw_ffi_any(raw));
259    }
260    core::hint::cold_path();
261    Err(structural_error_from_raw(
262        raw.type_index,
263        raw.small_str_len,
264        raw.data_union,
265    ))
266}

268#[cfg(test)]
269mod tests {
270    use super::VisitErrorContextObj;
271    use crate::object::ObjectCore;
272
273    // The binding must match the layout C++ registers for `ffi.VisitErrorContext`.
274    #[test]
275    fn visit_error_context_layout_matches_registered_fields() {
276        let expected = [
277            (
278                "reverse_visit_pattern",
279                std::mem::offset_of!(VisitErrorContextObj, reverse_visit_pattern),
280            ),
281            (
282                "prev_error_context",
283                std::mem::offset_of!(VisitErrorContextObj, prev_error_context),
284            ),
285        ];
286        // SAFETY: the type is registered by the runtime; its field table lives as long.
287        let fields = unsafe {
288            let info = crate::tvm_ffi_sys::TVMFFIGetTypeInfo(VisitErrorContextObj::type_index());
289            assert!(!info.is_null());
290            std::slice::from_raw_parts((*info).fields, (*info).num_fields as usize)
291        };
292        assert_eq!(fields.len(), expected.len());
293        for (name, offset) in expected {
294            let field = fields
295                .iter()
296                .find(|field| field.name.as_str() == name)
297                .unwrap_or_else(|| panic!("`{name}` is not a registered field"));
298            assert_eq!(field.offset as usize, offset, "offset of `{name}`");
299            assert_eq!(
300                field.size as usize,
301                std::mem::size_of::<usize>(),
302                "size of `{name}`"
303            );
304        }
305    }
