30/// Consume an owning error carrier while keeping checked-conversion scratch
31/// inside the cold path instead of successful recursive traversal frames.
32///
33/// # Safety
34/// The fields must describe one owning TVMFFIAny that has not been consumed.
35#[cold]
36#[inline(never)]
37pub(super) unsafe fn structural_error_from_raw(
38    type_index: i32,
39    small_str_len: u32,
40    data_union: TVMFFIAnyDataUnion,
41) -> Error {
42    let value = Any::from_raw_ffi_any(TVMFFIAny {
43        type_index,
44        small_str_len,
45        data_union,
46    });
47    match Error::try_from(value) {
48        Ok(error) | Err(error) => error,
49    }
50}

52/// Return the registered type key for `type_index`.
53///
54/// When the runtime returns no type information, return `<type_index N>` for diagnostics.
55///
56/// # Arguments
57///
58/// * `type_index` - Runtime type index whose registered type key is requested.
59pub(crate) fn type_key_of(type_index: i32) -> String {
60    unsafe {
61        let info = TVMFFIGetTypeInfo(type_index);
62        if info.is_null() {
63            format!("<type_index {type_index}>")
64        } else {
65            (*info).type_key.as_str().to_string()
66        }
67    }
68}

76#[cold]
77pub(crate) fn with_visit_error_context(error: Error, raw: TVMFFIAny) -> Error {
78    if raw.type_index < TVMFFITypeIndex::kTVMFFIStaticObjectBegin as i32 {
79        return error;
80    }
81    let context = (|| {
82        let mut context_type = 0;
83        unsafe {
84            crate::check_safe_call!(TVMFFITypeKeyToIndex(
85                &TVMFFIByteArray::from_str("ffi.VisitErrorContext"),
86                &mut context_type,
87            ))
88            .ok()?;
89        }
90        let node = unsafe { borrowed_view(&raw) }.try_as::<object::ObjectRef>()?;
91        let mut previous = error.extra_context();
92        let mut nodes = Vec::new();
93        if let Some(prior) = previous.as_ref() {
94            if AnyView::from(prior).type_index() == context_type {
95                let object = &**object::ObjectRef::data(prior);
96                let records = FieldGetter::new(context_type, "reverse_visit_pattern")
97                    .ok()?
98                    .get_any(object)
99                    .ok()?;
100                let size = Function::get_global("ffi.ListSize")
101                    .ok()?
102                    .call_tuple((records.clone(),))
103                    .ok()?
104                    .try_as::<i64>()?;
105                let get_item = Function::get_global("ffi.ListGetItem").ok()?;
106                if size > 0 {
107                    let last = get_item
108                        .call_tuple((records.clone(), size - 1))
109                        .ok()?
110                        .try_as::<object::ObjectRef>();
111                    // Callback, policy and default descent can report the same frame.
112                    if last.is_some_and(|last| last.same_as(&node)) {
113                        return None;
114                    }
115                }
116                for i in 0..size {
117                    nodes.push(get_item.call_tuple((records.clone(), i)).ok()?);
118                }
119                previous = FieldGetter::new(context_type, "prev_error_context")
120                    .ok()?
121                    .get::<_, Option<object::ObjectRef>>(object)
122                    .ok()?;
123            }
124        }
125        nodes.push(Any::from(node));
126        let records = Function::get_global("ffi.List")
127            .ok()?
128            .call_packed(&nodes.iter().map(AnyView::from).collect::<Vec<_>>())
129            .ok()?;
130        Function::get_global("ffi.MakeObjectFromPackedArgs")
131            .ok()?
132            .call_tuple((
133                context_type,
134                crate::String::from("reverse_visit_pattern"),
135                records,
136                crate::String::from("prev_error_context"),
137                previous,
138            ))
139            .ok()?
140            .try_as::<object::ObjectRef>()
141    })();
142    // Diagnostic enrichment must not replace the original error on failure.
143    match context {
144        Some(context) => Error::new_with_cause_and_extra_context(
145            error.kind(),
146            error.message(),
147            error.backtrace(),
148            error.cause_chain().as_ref(),
149            Some(&context),
150        ),
151        None => error,
152    }
153}

155// Generate the tuple arities supported by the standard library (1 through 12).
156macro_rules! impl_callback_chain_tuple_arities {
157    ($impl_chain:ident) => {
158        impl_callback_chain_tuple_arities!(
159            @prefixes $impl_chain;
160            [];
161            (F0, M0, 0),
162            (F1, M1, 1),
163            (F2, M2, 2),
164            (F3, M3, 3),
165            (F4, M4, 4),
166            (F5, M5, 5),
167            (F6, M6, 6),
168            (F7, M7, 7),
169            (F8, M8, 8),
170            (F9, M9, 9),
171            (F10, M10, 10),
172            (F11, M11, 11)
173        );
174    };
175    (@prefixes $impl_chain:ident; [$($prefix:tt)*];) => {};
176    (
177        @prefixes $impl_chain:ident;
178        [$($prefix:tt)*];
179        $next:tt $(, $rest:tt)*
180    ) => {
181        $impl_chain!($($prefix)* $next);
182        impl_callback_chain_tuple_arities!(
183            @prefixes $impl_chain;
184            [$($prefix)* $next,];
185            $($rest),*
186        );
187    };
188}

192/// Copy a borrowed FFI value when its owning representation is unchanged.
193///
194/// Raw string/byte views and `ObjectRValueRef` return `None` because they need
195/// the runtime's normalization or move logic. A null object pointer also
196/// returns `None` instead of constructing an invalid owning value.
197#[inline(always)]
198pub(crate) fn try_to_owned_without_normalization(raw: TVMFFIAny) -> Option<Any> {
199    if is_plain_inline(raw.type_index) {
200        return Some(unsafe { Any::from_raw_ffi_any(raw) });
201    }
202    if raw.type_index >= TVMFFITypeIndex::kTVMFFIStaticObjectBegin as i32 {
203        let object = unsafe { raw.data_union.v_obj };
204        if object.is_null() {
205            return None;
206        }
207        unsafe { object::unsafe_::inc_ref(object) };
208        return Some(unsafe { Any::from_raw_ffi_any(raw) });
209    }
210    None
211}

215#[inline(always)]
216pub(crate) fn same_shallow(lhs: TVMFFIAny, rhs: TVMFFIAny) -> bool {
217    lhs.type_index == rhs.type_index
218        && lhs.small_str_len == rhs.small_str_len
219        && unsafe { lhs.data_union.v_uint64 == rhs.data_union.v_uint64 }
220}

222/// Borrow a traversal ABI cell for no longer than the cell's scope.
223///
224/// # Safety
225/// The cell must satisfy `AnyView::from_raw_ffi_any`'s requirements, including
226/// its unique, at-most-once conversion rule for rvalue slots. The traversal
227/// driver must keep the owning root, field, or replacement alive for the borrow.
228#[inline(always)]
229pub(crate) unsafe fn borrowed_view(raw: &TVMFFIAny) -> AnyView<'_> {
230    AnyView::from_raw_ffi_any(*raw)
231}
