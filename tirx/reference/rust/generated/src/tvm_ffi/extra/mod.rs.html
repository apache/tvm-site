40/// Borrow guarantees of the structural views, checked as `compile_fail` doctests.
41///
42/// `StructuralVisitorView::base` cannot outlive its view:
43/// ```compile_fail
44/// use tvm_ffi::extra::structural_visit::{StructuralVisitorObj, StructuralVisitorView};
45/// fn retain<D>(view: &StructuralVisitorView<'_, D>) -> &'static StructuralVisitorObj {
46///     view.base()
47/// }
48/// ```
49///
50/// The visitor prefix cannot be converted into an owning FFI value:
51///
52/// ```compile_fail,E0277
53/// use tvm_ffi::Any;
54/// use tvm_ffi::StructuralVisitorView;
55/// fn retain<D>(view: &StructuralVisitorView<'_, D>) -> Any {
56///     Any::from(view.base())
57/// }
58/// ```
59///
60/// A node borrowed through a copied `MutationView` cannot survive consumption of the view:
61/// ```compile_fail,E0505
62/// use tvm_ffi::MutationView;
63/// use tvm_ffi::StructuralMutatorView;
64/// fn invalid<D>(value: MutationView<'_>, dispatch: &mut D, mutator: &mut StructuralMutatorView<'_, D>) {
65///     let view = value.as_any_view();
66///     let copy = view;
67///     let node = copy.downcast_ref::<tvm_ffi::collections::array::ArrayObj>().unwrap();
68///     mutator.default_mutate(dispatch, value).unwrap();
69///     println!("{}", node.size);
70/// }
71/// ```
72///
73/// Moving the handle into a nested callback cannot bypass that borrow:
74/// ```compile_fail,E0505
75/// use tvm_ffi::{dispatch, structural_mutate, Any, MutationView, Result, UnchangedOr};
76/// use tvm_ffi::StructuralMutatorView;
77/// struct Inner<'a>(Option<MutationView<'a>>);
78/// #[dispatch(mutate, (MutationView<'_>, &mut StructuralMutatorView<'_, Self>), Result<UnchangedOr<Any>>)]
79/// impl<'a> Inner<'a> {
80///     fn mutate_bool(&mut self, _: bool, mutator: &mut StructuralMutatorView<'_, Self>) -> Result<UnchangedOr<Any>> {
81///         let pending = self.0.take().unwrap();
82///         mutator.default_mutate(self, pending)
83///     }
84/// }
85/// fn invalid(value: MutationView<'_>) {
86///     let view = value.as_any_view();
87///     let node = view.downcast_ref::<tvm_ffi::collections::array::ArrayObj>().unwrap();
88///     structural_mutate(true, Inner(Some(value))).unwrap();
89///     println!("{}", node.size);
90/// }
91/// ```
92///
93/// A copied inspection view also cannot survive low-level default mutation:
94/// ```compile_fail,E0505
95/// use tvm_ffi::{dispatch, Any, MutationView, Result, Unchanged, UnchangedOr};
96/// use tvm_ffi::StructuralMutatorView;
97/// struct Pass;
98/// #[dispatch(mutate, (MutationView<'_>, &mut StructuralMutatorView<'_, Self>), Result<UnchangedOr<Any>>)]
99/// impl Pass {
100///     fn mutate_any(&mut self, value: MutationView<'_>, mutator: &mut StructuralMutatorView<'_, Self>) -> Result<UnchangedOr<Any>> {
101///         let view = value.as_any_view();
102///         let copy = view;
103///         mutator.default_mutate(self, value).unwrap();
104///         println!("{}", copy.type_index());
105///         Ok(Unchanged.into())
106///     }
107/// }
108/// ```
109#[cfg(doctest)]
110pub struct StructuralViewBorrowChecks;
