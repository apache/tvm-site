19//! Rust bindings of TVM FFI, the open ABI and FFI for machine learning
20//! systems.
21//!
22//! Values cross the boundary as [`Any`] and [`AnyView`]. Objects, such as
23//! [`Array`], [`Map`], [`Tensor`] and types defined with [`rs_object`], are
24//! reference counted and shared with C++ and Python. A [`Function`] is
25//! callable from every language, and an [`Error`] raised in one arrives in
26//! the others.
27//!
28//! # Examples
29//!
30//! ```
31//! use tvm_ffi::{Array, Function, Result};
32//!
33//! let sum = |xs: Array<i64>| -> Result<i64> { Ok(xs.iter().sum()) };
34//! Function::register_global_typed("my_ext.sum", sum, "Sum the elements.")?;
35//!
36//! let sum = Function::get_global("my_ext.sum")?;
37//! let total = sum.call_tuple((Array::from_vec(vec![1_i64, 2, 3]),))?;
38//! assert_eq!(i64::try_from(total)?, 6);
39//! # Ok::<(), tvm_ffi::Error>(())
40//! ```

111/// Generate runtime type dispatch from the `prefix_*` methods of an impl.
112///
113/// `#[dispatch(prefix, (First, Args...), Ret)]` on an inherent impl makes its
114/// `prefix_*` methods handlers and adds a `dispatch(first, args...) ->
115/// Option<Ret>` method. `dispatch` calls the first handler, in source order,
116/// whose first parameter matches the runtime type of `first`, and returns
117/// `None` when no handler matches. The macro also implements [`Dispatcher`]
118/// for `&Self` or `&mut Self`, so the type can drive [`structural_walk`] and
119/// the other traversals, whose engines call it and run their own default
120/// recursion; no prefix has a reserved meaning. Handlers without an `inline`
121/// attribute are marked `#[inline(always)]`.
122///
123/// # Arguments
124///
125/// * `prefix` - The handler name prefix; `describe` selects `describe_int`,
126///   `describe_str` and so on.
127/// * `(First, Args...)` - The complete input tuple. `First`, the value to
128///   match, is `AnyView<'_>` or [`MutationView<'_>`], including aliases.
129///   `Args` are passed through: each handler takes a leading prefix of them
130///   without cloning, and need not take all of them.
131/// * `Ret` - The result type. A handler's result is converted with
132///   `Into<Ret>`. When both the handler's result and `Ret` are written
133///   `Result<..>`, the success value is converted instead, so a
134///   `Result<UnchangedOr<MyPair>>` handler fits a `Result<UnchangedOr<Any>>`
135///   dispatcher.
136///
137/// # Handlers
138///
139/// A handler's first parameter selects what it matches, as for
140/// [`match_any!`]: an owned FFI value such as `i64` or `Array<Any>`, converted
141/// strictly; `&Type` (such as `&Array<i64>` or `&tvm_ffi::String`) or
142/// `&NodeObj`, which borrow the value or the object node in place without
143/// changing a reference count; `MutationView<'_, T>`,
144/// which keeps the in-place mode; or `AnyView<'_>`, which
145/// matches every value and must be the last handler.
146///
147/// Only the outer `None` means that no handler matched: a handler error, or a
148/// `None` inside `Ret`, is final. Handlers share one receiver, `&self` or
149/// `&mut self`, and the same method generics and bounds; shared generics must
150/// occur in the receiver, the input tuple or `Ret`. Use the same lifetime name
151/// in an input and `Ret` to return a borrow of that input; elided output
152/// lifetimes borrow the receiver. Handler-level `cfg` attributes are not
153/// supported; configure the whole impl instead. Generate one dispatcher per
154/// type; when inference is ambiguous, give the expected result type.
155///
156/// # Examples
157///
158/// ```
159/// use tvm_ffi::{dispatch, AnyView};
160///
161/// struct Describer;
162///
163/// #[dispatch(describe, (AnyView<'_>,), String)]
164/// impl Describer {
165///     fn describe_int(&self, value: i64) -> String {
166///         format!("integer {value}")
167///     }
168///
169///     fn describe_str(&self, value: &tvm_ffi::String) -> String {
170///         format!("string {}", value.as_str())
171///     }
172/// }
173///
174/// let text = tvm_ffi::String::from("hi");
175/// assert_eq!(Describer.dispatch(AnyView::from(&1_i64)).as_deref(), Some("integer 1"));
176/// assert_eq!(Describer.dispatch(AnyView::from(&text)).as_deref(), Some("string hi"));
177/// assert_eq!(Describer.dispatch(AnyView::from(&1.5_f64)), None);
178/// ```
179///
180/// [`MutationView<'_>`]: crate::MutationView
181/// [`Dispatcher`]: crate::Dispatcher
182/// [`structural_walk`]: crate::structural_walk
183pub use tvm_ffi_macros::dispatch;

185/// Match the runtime type of a value, the way `match` matches an enum.
186///
187/// The value is an [`Any`], an [`AnyView`], a managed reference or a
188/// [`MutationView`]. It is evaluated once and taken by value; write `&value`
189/// to keep it. Arms are tried in order, and the first arm whose type matches
190/// runs; a guard that is false moves on to the next arm. The last arm must be
191/// an unguarded `_ => ...`, since new types can be registered at any time.
192///
193/// * `Type(name)` binds an owned value of `Type`, such as `i64` or
194///   `Array<Any>`.
195/// * `Type(ref name)` borrows a managed reference, and `NodeObj(ref name)`
196///   borrows the object node, without changing its reference count.
197/// * `MutationView<Type>(name)` moves the in-place mode into a
198///   typed view.
199///
200/// Matching is strict: an `i64` arm does not match a stored `f64`, and a value
201/// is never truncated. Type aliases and qualified paths may name `Type`. Typed
202/// `MutationView` arms do not match a `Map`, including a nested one; use
203/// `MutationView<Any>` for those. Ordinary `Map<K, V>` arms do match.
204///
205/// # Examples
206///
207/// ```
208/// use tvm_ffi::{match_any, Any, Array};
209///
210/// fn kind(value: &Any) -> String {
211///     match_any! {
212///         value {
213///             i64(n) => format!("integer {n}"),
214///             tvm_ffi::String(s) => format!("string {}", s.as_str()),
215///             Array<Any>(arr) => format!("array of {} items", arr.len()),
216///             _ => "something else".to_string(),
217///         }
218///     }
219/// }
220///
221/// assert_eq!(kind(&Any::from(42_i64)), "integer 42");
222/// assert_eq!(kind(&Any::from(Array::from_vec(vec![1_i64, 2, 3]))), "array of 3 items");
223/// assert_eq!(kind(&Any::from(1.5_f64)), "something else");
224/// ```
225pub use tvm_ffi_macros::match_any;

227/// Define an object type in Rust that every language can use.
228///
229/// `#[rs_object("type.Key", options...)]` registers the struct as an object
230/// type with the reflection of every field, so that structural equality and
231/// hashing, `repr`, deep copy and Python see the fields. The first field,
232/// `base`, is the parent type's object, embedded at the start of the struct:
233/// `base: Object` for a type under `ffi.Object`. Declare the managed reference
234/// with [`#[derive(ObjectRef)]`](derive::ObjectRef).
235///
236/// The macro adds to the struct:
237///
238/// * `#[repr(C)]` and `Deref` to the parent's object, so `obj.field` reads the
239///   fields of both.
240/// * `MyPairObj::new(base, fields...)`, which takes the fields in order: a
241///   scalar (`bool`, an integer or a float) as its own type, so that a literal
242///   infers as it, and any other field from anything that converts into it.
243///
244/// # Arguments
245///
246/// * `"type.Key"` - The type key.
247/// * `final` - The type has no subtypes.
248/// * `s_eq_hash_kind = "TreeNode"` - The [`SEqHashKind`] of structural
249///   equality; by default the parent's.
250/// * `child_slots = N`, `child_slots_can_overflow = bool` - Type indices kept
251///   for subtypes; 0 and `true` by default.
252/// * `reflection = path` - A `fn(&mut ObjectDef<Self>)` that registers the
253///   type's methods.
254/// * `crate = path` - Where the generated code finds tvm-ffi, for a crate that
255///   depends on it only through a re-export; by default `tvm_ffi`.
256///
257/// A field takes `#[reflect(writable, seq_hash_ignore, ...)]` for the
258/// [`FieldFlags`] of those names, and `name = "..."` or `doc = "..."` to
259/// register it under another name or with a doc string, by default its `///`
260/// comment.
261///
262/// # Requirements
263///
264/// A field's type must implement `AnyCompatible` and `Clone`; an `init_off`
265/// field's type must take all-zero bytes as a value. The parent's object must
266/// implement `ShallowCopy` and `UnsafeInit`, as `rs_object` types and non-final
267/// types bound by `tvm-ffi-stubgen` do. The type registers on its first
268/// `type_index()`, with its fields, methods, `__ffi_new__` and, unless the
269/// methods define it, `__ffi_init__`.
270///
271/// # Examples
272///
273/// ```
274/// use tvm_ffi::derive::ObjectRef;
275/// use tvm_ffi::{rs_object, Object, ObjectArc};
276///
277/// #[rs_object("my_ext.MyPair", final)]
278/// pub struct MyPairObj {
279///     base: Object,
280///     pub first: i64,
281///     #[reflect(seq_hash_ignore)]
282///     pub second: tvm_ffi::String,
283/// }
284///
285/// #[derive(Clone, ObjectRef)]
286/// pub struct MyPair {
287///     data: ObjectArc<MyPairObj>,
288/// }
289///
290/// let pair = MyPair::from_object(MyPairObj::new(Object::new(), 1, "one"));
291/// assert_eq!(pair.first, 1);
292/// assert_eq!(pair.second.as_str(), "one");
293/// ```
294pub use tvm_ffi_macros::rs_object;
