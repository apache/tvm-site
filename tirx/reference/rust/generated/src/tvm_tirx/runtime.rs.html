87/// Allocate an uninitialized tensor on `device`, in the memory scope `mem_scope` or the global one
88/// (`runtime.TensorAllocWithScope`).
89///
90/// # Errors
91///
92/// Returns an error if the allocator of `device` does not support `mem_scope`; the CPU one supports
93/// only the global scope.
94///
95/// # Examples
96///
97/// ```
98/// use tvm_tirx::ir::dtype;
99/// use tvm_tirx::runtime;
100/// use tvm_tirx::tvm_ffi::{DLDevice, DLDeviceType};
101///
102/// let cpu = DLDevice::new(DLDeviceType::kDLCPU, 0);
103/// let buffer = runtime::empty(&[2, 8], dtype::float(32), cpu, None)?;
104///
105/// let bytes = runtime::create_view(&buffer, &[64], dtype::int(8), 0)?;
106/// assert_eq!(bytes.shape(), &[64]);
107///
108/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
109/// ```
110pub fn empty(
111    shape: &[i64],
112    dtype: DLDataType,
113    device: DLDevice,
114    mem_scope: Option<&MemoryScope>,
115) -> Result<Tensor> {
116    tvm_ffi::cached_global_func!("runtime.TensorAllocWithScope")
117        .call_tuple((Shape::from(shape), dtype, device, mem_scope.cloned()))?
118        .try_into()
119}

121/// Create a view of the storage of `source`, `relative_byte_offset` bytes past its data, with
122/// another shape and data type (`runtime.TensorCreateView`).
123///
124/// # Errors
125///
126/// Returns an error if `source` is not contiguous or the view does not fit in its storage.
127pub fn create_view(
128    source: impl IntoObjectRef<Tensor>,
129    shape: &[i64],
130    dtype: DLDataType,
131    relative_byte_offset: u64,
132) -> Result<Tensor> {
133    tvm_ffi::cached_global_func!("runtime.TensorCreateView")
134        .call_tuple((
135            source.into_object_ref().borrow(),
136            Shape::from(shape),
137            dtype,
138            relative_byte_offset,
139        ))?
140        .try_into()
141}

143/// Return the attribute `name` of `device`, such as `exist` or `max_threads_per_block`, or `None`
144/// when the device does not report it (`runtime.GetDeviceAttr`).
145///
146/// # Errors
147///
148/// Returns an error if the runtime of the device is not enabled and `name` is not `exist`, which is
149/// then `false`.
150pub fn get_device_attr(device: DLDevice, name: &str) -> Result<Any> {
151    tvm_ffi::cached_global_func!("runtime.GetDeviceAttr").call_tuple((
152        device.device_type,
153        device.device_id,
154        String::from(name),
155    ))
156}

158/// Return whether the runtime of the device or target kind `target` (`"cpu"`, `"cuda"`,
159/// `"llvm"`...) is enabled (`runtime.RuntimeEnabled`).
160pub fn enabled(target: &str) -> Result<bool> {
161    tvm_ffi::cached_global_func!("runtime.RuntimeEnabled")
162        .call_tuple((String::from(target),))?
163        .try_into()
164}

166/// Return the number of threads the runtime thread pool uses (`runtime.NumThreads`).
167pub fn num_threads() -> Result<i32> {
168    tvm_ffi::cached_global_func!("runtime.NumThreads").call_tuple(())?.try_into()
169}

174/// Return the number of CUDA devices (`runtime.GetCudaDeviceCount`).

176/// # Errors
177///
178/// Returns an error if the CUDA runtime is not built or loaded.
179pub fn get_cuda_device_count() -> Result<i32> {
180    tvm_ffi::Function::get_global("runtime.GetCudaDeviceCount")?.call_tuple(())?.try_into()
181}
