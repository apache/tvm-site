21//! The context of tile instruction dispatch.
22//!
23//! A [`DispatchContext`] gives a lowering its target, thread scope, launch parameters and variable
24//! ranges, and the state shared between dispatches.
25//!
26//! # Examples
27//!
28//! ```
29//! use tvm_tirx::ir::{Evaluate, IntImm, PrimType, PrimVar, UNKNOWN_LOC};
30//! use tvm_tirx::target::Target;
31//! use tvm_tirx::tirx::tile::DispatchContext;
32//! use tvm_tirx::tirx::ThreadScope;
33//! use tvm_tirx::tvm_ffi::Map;
34//!
35//! let tx = PrimVar::new("tx", PrimType::int(32), UNKNOWN_LOC);
36//! let ctx = DispatchContext::new(
37//!     Target::new("cuda")?,
38//!     ThreadScope::get("cuda.warp")?,
39//!     [("threadIdx.x", tx, 32.into())],
40//!     &Map::new(), false, &Map::new(), &Map::new(), &Map::new(), &Map::new(),
41//!     "warp",
42//! )?;
43//!
44//! ctx.cache_set("my_ext.zero", &IntImm::int32(0))?;
45//! assert!(ctx.cache_get("my_ext.zero")?.is_some());
46//! ctx.add_init_stmt(Evaluate::new(1, UNKNOWN_LOC), false)?;
47//! # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
48//! ```

155impl DispatchContext {
156    /// Create the context of a dispatch for `target` at `thread_scope` (`tirx.DispatchContext`).
157    ///
158    /// `launch_params` maps each launch parameter to its variable and extent; `inter` and `intra`
159    /// map each axis to its `[extent, offset]` across and within teams.
160    ///
161    /// # Errors
162    ///
163    /// Returns an error if the backend of `thread_scope` is not the target's kind, or `scope_kind`
164    /// is neither empty nor the local name of `thread_scope`.
165    #[allow(clippy::too_many_arguments)]
166    pub fn new(
167        target: impl Into<Target>,
168        thread_scope: impl Into<ThreadScope>,
169        launch_params: impl IntoIterator<Item = (impl AsRef<str>, PrimVar, PrimExpr)>,
170        var_range_map: &Map<Var, Range>,
171        alloc_only: bool,
172        callbacks: &Map<String, ObjectRef>,
173        shared_state: &Map<String, ObjectRef>,
174        inter: &Map<String, Array<PrimExpr>>,
175        intra: &Map<String, Array<PrimExpr>>,
176        scope_kind: &str,
177    ) -> Result<Self> {
178        let (target, thread_scope) = (target.into(), thread_scope.into());
179        let launch_params = launch_params
180            .into_iter()
181            .map(|(name, var, extent)| {
182                (String::from(name), Array::from_vec(vec![Any::from(var), Any::from(extent)]))
183            })
184            .collect::<Map<String, Array<Any>>>();
185        let scope_kind = String::from(scope_kind);
186        tvm_ffi::cached_global_func!("tirx.DispatchContext")
187            .call_packed(&[
188                AnyView::from(&target),
189                AnyView::from(&thread_scope),
190                AnyView::from(&launch_params),
191                AnyView::from(var_range_map),
192                AnyView::from(&alloc_only),
193                AnyView::from(callbacks),
194                AnyView::from(shared_state),
195                AnyView::from(inter),
196                AnyView::from(intra),
197                AnyView::from(&scope_kind),
198            ])?
199            .try_into()
200    }
201
202    /// Request the allocation of the tensor variable `tensor`, placed at `allocated_addr` and
203    /// annotated with `annotations`; only while `alloc_only`
204    /// (`tirx.DispatchContextAddAllocTensor`).
205    pub fn add_alloc_tensor(
206        &self,
207        tensor: impl Into<Var>,
208        allocated_addr: impl IntoIterator<Item = impl IntoExpr<PrimExpr>>,
209        annotations: &Map<String, Any>,
210    ) -> Result<()> {
211        let tensor = tensor.into();
212        let allocated_addr = allocated_addr
213            .into_iter()
214            .map(IntoExpr::into_owned_expr)
215            .collect::<Result<Array<PrimExpr>>>()?;
216        tvm_ffi::cached_global_func!("tirx.DispatchContextAddAllocTensor").call_tuple((
217            self,
218            &tensor,
219            &allocated_addr,
220            annotations,
221        ))?;
222        Ok(())
223    }
224
225    /// Request `stmt` at the start of the host code if `host`, else of the kernel
226    /// (`tirx.DispatchContextAddInitStmt`).
227    pub fn add_init_stmt(&self, stmt: impl Into<Stmt>, host: bool) -> Result<()> {
228        tvm_ffi::cached_global_func!("tirx.DispatchContextAddInitStmt").call_tuple((
229            self,
230            &stmt.into(),
231            host,
232        ))?;
233        Ok(())
234    }
235
236    /// Request `stmt` right after the definition of the tensor variable `tensor`
237    /// (`tirx.DispatchContextAddPostTensorDefStmt`).
238    pub fn add_post_tensor_def_stmt(
239        &self,
240        tensor: impl Into<Var>,
241        stmt: impl Into<Stmt>,
242    ) -> Result<()> {
243        tvm_ffi::cached_global_func!("tirx.DispatchContextAddPostTensorDefStmt").call_tuple((
244            self,
245            &tensor.into(),
246            &stmt.into(),
247        ))?;
248        Ok(())
249    }
250
251    /// Return the object cached under `key` across dispatches, if any
252    /// (`tirx.DispatchContextSharedStateGet`).
253    pub fn cache_get(&self, key: &str) -> Result<Option<ObjectRef>> {
254        tvm_ffi::cached_global_func!("tirx.DispatchContextSharedStateGet")
255            .call_tuple((self, String::from(key)))?
256            .try_into()
257    }
258
259    /// Cache `value` under `key` across dispatches (`tirx.DispatchContextSharedStateSet`).
260    pub fn cache_set(&self, key: &str, value: &impl ObjectRefCore) -> Result<()> {
261        let key = String::from(key);
262        tvm_ffi::cached_global_func!("tirx.DispatchContextSharedStateSet").call_packed(&[
263            AnyView::from(self),
264            AnyView::from(&key),
265            AnyView::from(value),
266        ])?;
267        Ok(())
268    }
