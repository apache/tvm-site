20//! Constraint-aware structural walking: observe values with the facts in scope.
21
22use std::borrow::BorrowMut;
23use tvm_ffi::{
24    Any, AnyView, DefRegionKind, Dispatcher, IntoDispatcher, Result, VisitInterrupt, WalkOrder,
25    WalkResult,
26};
27
28use crate::sym::ConstraintContext;
29use super::structural_visit_with_constraint_context::{
30    structural_visit_with_constraint_context, StructuralVisitorWithConstraintContextView,
31};
32
33struct WalkVisitor<'a, D, const PRE: bool> {
34    observer: &'a mut D,
35}
36
37impl<'v, 'c, 'r, 'a, D, const PRE: bool>
38    Dispatcher<
39        (
40            AnyView<'v>,
41            &'c mut StructuralVisitorWithConstraintContextView<'r, WalkVisitor<'a, D, PRE>>,
42        ),
43        Option<Result<Option<VisitInterrupt>>>,
44    > for &mut WalkVisitor<'a, D, PRE>
45where
46    for<'s, 'b, 'ctx> &'s mut D: Dispatcher<
47        (AnyView<'b>, &'ctx ConstraintContext, DefRegionKind),
48        Option<Result<WalkResult>>,
49    >,
50{
51    #[inline(always)]
52    fn dispatch_tuple(
53        self,
54        (value, visitor): (
55            AnyView<'v>,
56            &'c mut StructuralVisitorWithConstraintContextView<'r, WalkVisitor<'a, D, PRE>>,
57        ),
58    ) -> Option<Result<Option<VisitInterrupt>>> {
59        Some(self.visit(value, visitor))
60    }
61}
62
63impl<D, const PRE: bool> WalkVisitor<'_, D, PRE>
64where
65    for<'s, 'v, 'c> &'s mut D:
66        Dispatcher<(AnyView<'v>, &'c ConstraintContext, DefRegionKind), Option<Result<WalkResult>>>,
67{
68    #[inline(always)]
69    fn visit(
70        &mut self,
71        value: AnyView<'_>,
72        visitor: &mut StructuralVisitorWithConstraintContextView<'_, Self>,
73    ) -> Result<Option<VisitInterrupt>> {
74        let kind = visitor.def_region_kind();
75        if !PRE {
76            if let Some(interrupt) = visitor.default_visit(self, value)? {
77                core::hint::cold_path();
78                return Ok(Some(interrupt));
79            }
80        }
81        // Post callbacks observe the restored context. The observer cannot access
82        // this runtime's visitor; a nested walk creates its own runtime. Appending
83        // mutates context contents, not this handle. End the field borrow before
84        // any default descent can replace it; observers may clone it to retain it.
85        let outcome = unsafe {
86            (&mut *self.observer).dispatch_tuple((
87                value,
88                visitor.base().constraints_for_observer(),
89                kind,
90            ))
91        };
92        let outcome = match outcome {
93            Some(result) => result?,
94            None => WalkResult::Advance,
95        };
96        match outcome {
97            WalkResult::Advance if PRE => visitor.default_visit(self, value),
98            WalkResult::Advance => Ok(None),
99            WalkResult::Skip => {
100                core::hint::cold_path();
101                Ok(None)
102            }
103            WalkResult::Interrupt => {
104                core::hint::cold_path();
105                Ok(Some(VisitInterrupt::with(Any::new())))
106            }
107            WalkResult::InterruptWith(payload) => {
108                core::hint::cold_path();
109                Ok(Some(VisitInterrupt::with(payload)))
110            }
111        }
112    }
113}

115/// Walk every value of `root` with `walker`, in `order`, with the facts in scope at each value.
116///
117/// A callback takes a leading prefix of `(value, &ConstraintContext, DefRegionKind)` and returns a
118/// [`WalkResult`]: in pre-order `Skip` prunes the children, and `Interrupt` or an error stops the
119/// walk. Callbacks must not append to the context or panic.
120///
121/// # Examples
122///
123/// ```
124/// use tvm_tirx::ir::{Evaluate, EvaluateObj, For, ForKind, PrimType, PrimVar, Var, UNKNOWN_LOC};
125/// use tvm_tirx::script::script;
126/// use tvm_tirx::sym::{ConstraintContext, RangeConstraintObj};
127/// use tvm_tirx::tirx::structural_walk_with_constraint_context as walk;
128/// use tvm_tirx::tvm_ffi::{ObjectRefCore, WalkOrder, WalkResult};
129///
130/// let i = PrimVar::new("i", PrimType::int(32), UNKNOWN_LOC);
131/// let body = Evaluate::new(&i, UNKNOWN_LOC);
132/// let root = For::new(&i, 0, 16, 1, ForKind::Default, body, None, UNKNOWN_LOC)?;
133///
134/// let mut ranges = Vec::new();
135/// let record = |_: &EvaluateObj, context: &ConstraintContext| {
136///     context.for_each(|fact| {
137///         if let Some(range) = fact.downcast_ref::<RangeConstraintObj>() {
138///             ranges.push(script(&range.range).unwrap());
139///         }
140///     });
141///     WalkResult::Advance
142/// };
143/// walk::structural_walk_with_constraint_context(&root, record, WalkOrder::PreOrder)?;
144/// assert_eq!(ranges, ["T.Range(0, 16)"]);
145/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
146/// ```
147pub fn structural_walk_with_constraint_context<'input, R, H, M, D, O>(
148    root: R,
149    walker: H,
150    order: WalkOrder,
151) -> Result<Option<VisitInterrupt>>
152where
153    R: Into<AnyView<'input>>,
154    O: BorrowMut<D>,
155    for<'v, 'c> H: IntoDispatcher<
156        (AnyView<'v>, &'c ConstraintContext, DefRegionKind),
157        Option<Result<WalkResult>>,
158        M,
159        Target = D,
160        Output = O,
161    >,
162    for<'s, 'v, 'c> &'s mut D:
163        Dispatcher<(AnyView<'v>, &'c ConstraintContext, DefRegionKind), Option<Result<WalkResult>>>,
164{
165    let mut storage = walker.into_dispatcher();
166    let observer = storage.borrow_mut();
167    let root = root.into();
168    match order {
169        WalkOrder::PreOrder => {
170            structural_visit_with_constraint_context(root, WalkVisitor::<D, true> { observer })
171        }
172        WalkOrder::PostOrder => {
173            structural_visit_with_constraint_context(root, WalkVisitor::<D, false> { observer })
174        }
175    }
176}
