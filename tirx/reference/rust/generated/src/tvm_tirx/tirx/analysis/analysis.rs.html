51/// Return whether `func` has no illegal direct memory access from the host
52/// (`tirx.analysis.VerifyMemory`).
53///
54/// A function without a target attribute, or not of the default calling convention, is not checked
55/// and passes.
56pub fn verify_memory(func: impl IntoArg<Function>) -> Result<bool> {
57    tvm_ffi::cached_global_func!("tirx.analysis.VerifyMemory")
58        .call_tuple((func.into_arg().borrow(),))?
59        .try_into()
60}

62/// Return the variables `node`, a statement or an expression, uses but neither defines nor finds in
63/// `defs` (`tirx.analysis.UndefinedVars`).
64///
65/// # Examples
66///
67/// ```
68/// use tvm_tirx::ir::{Expr, PrimType, PrimVar, Var, UNKNOWN_LOC};
69/// use tvm_tirx::tirx::analysis;
70/// use tvm_tirx::tvm_ffi::ObjectRefCore;
71///
72/// let x = PrimVar::new("x", PrimType::int(32), UNKNOWN_LOC);
73/// let y = PrimVar::new("y", PrimType::int(32), UNKNOWN_LOC);
74/// let sum: Expr = (&x + &y).try_into()?;
75/// let undefined = analysis::undefined_vars(sum, [&x])?;
76/// assert!(undefined.len() == 1 && undefined[0].same_as(&y));
77/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
78/// ```
79pub fn undefined_vars(
80    node: impl Into<Any>,
81    defs: impl IntoIterator<Item = impl IntoArg<Var>>,
82) -> Result<Array<Var>> {
83    let defs = defs.into_iter().map(IntoArg::into_owned).collect::<Array<Var>>();
84    tvm_ffi::cached_global_func!("tirx.analysis.UndefinedVars")
85        .call_tuple((&node.into(), &defs))?
86        .try_into()
87}

105/// Return the coefficients `c` of `expr = sum(vars[i] * c[i]) + c[n]`, each invariant of `vars`, or
106/// an empty array if `expr` has no such form (`tirx.analysis.DetectLinearEquation`).
107pub fn detect_linear_equation(
108    expr: impl IntoExpr<PrimExpr>,
109    vars: impl IntoIterator<Item = impl IntoArg<PrimVar>>,
110) -> Result<Array<PrimExpr>> {
111    let vars = vars.into_iter().map(IntoArg::into_owned).collect::<Array<PrimVar>>();
112    tvm_ffi::cached_global_func!("tirx.analysis.DetectLinearEquation")
113        .call_tuple((expr.into_expr()?.borrow(), &vars))?
114        .try_into()
115}
