20//! Constraint-aware structural mutation through the shared Rust engine.
21//!
22//! The native semantic base carries only lexical context state. Rust owns its
23//! dispatcher and remaps in `StructuralMutateEngine<Base>`; native constraint
24//! hooks recurse through that engine's vtable and collect facts from results.
25

42/// Complete native-compatible semantic base, without an owning object conversion.
43///
44/// Native hooks may replace the current context during descent through a shared
45/// view. All native-mutated state is interior mutable. This representation
46/// requires the matching native header/build, including the private scope flag.
47#[repr(C)]
48pub struct StructuralMutatorWithConstraintContextBase {
49    base: StructuralMutatorObj,
50    constraints: UnsafeCell<ConstraintContext>,
51    needs_child_context: UnsafeCell<bool>,
52}
53
54impl StructuralMutatorWithConstraintContextBase {
55    /// Clone the current live context without retaining a borrow of its field.
56    /// Use `snapshot()` on the returned handle to freeze its current prefix.
57    #[inline]
58    pub fn constraints(&self) -> ConstraintContext {
59        unsafe { (&*self.constraints.get()).clone() }
60    }
61
62    /// Borrow only while a mapper cannot recurse through this mutator.
63    /// End this borrow before default descent can replace the context field.
64    #[inline]
65    pub(super) unsafe fn constraints_for_observer(&self) -> &ConstraintContext {
66        &*self.constraints.get()
67    }
68}

70// SAFETY: repr(C) matches the complete native state-only semantic base. The
71// shared engine installs its vtable, owns remaps and publishes the dispatcher's
72// actual recursive reborrow. Native hooks borrow this runtime synchronously and
73// cannot retain it. No shared reference freezes the replaceable context/flag.
74// Each default call transfers exactly one owning replacement/Unchanged/Error.
75unsafe impl StructuralMutatorBase for StructuralMutatorWithConstraintContextBase {
76    fn new() -> Result<Self> {
77        Ok(Self {
78            base: StructuralMutatorObj::default(),
79            constraints: UnsafeCell::new(ConstraintContext::new()?),
80            needs_child_context: UnsafeCell::new(false),
81        })
82    }
83
84    fn type_index() -> i32 {
85        StructuralMutatorWithConstraintContextObj::type_index()
86    }
87
88    #[inline(always)]
89    unsafe fn default_mutate_raw(runtime: *mut Self, value: MutationView<'_>) -> TVMFFIAny {
90        static COPYING: ColdInitLazyLock<Option<TypeAttrColumn>> =
91            ColdInitLazyLock::new(|| TypeAttrColumn::new("__s_mutate_with_constraint_context__"));
92        static INPLACE: ColdInitLazyLock<Option<TypeAttrColumn>> = ColdInitLazyLock::new(|| {
93            TypeAttrColumn::new("__s_maybe_inplace_mutate_with_constraint_context__")
94        });
95        let borrowed = value.as_any_view();
96        let lookup = |column: Option<TypeAttrColumn>| {
97            column
98                .and_then(|column| column.get_raw_ffi_any(borrowed.type_index()))
99                .filter(|hook| hook.type_index != TVMFFITypeIndex::kTVMFFINone as i32)
100        };
101        let hook = if value.inplace_mode() == InplaceMode::Allow {
102            lookup(*INPLACE).filter(|_| {
103                // Permission alone is insufficient after a callback retains an
104                // alias. The supported owning count includes this temporary;
105                // release it before invoking a hook or entering native descent.
106                borrowed.try_as::<ObjectRef>().is_some_and(|owner| {
107                    let unique = ObjectArc::strong_count(ObjectRef::data(&owner)) == 2;
108                    drop(owner);
109                    unique
110                })
111            })
112        } else {
113            None
114        }
115        .or_else(|| lookup(*COPYING));
116        if let Some(hook) = hook {
117            // Copy the borrowed registry token; the shared helper acquires
118            // Function ownership before packed reentry and handles error context.
119            StructuralMutatorObj::call_structural_mutate_hook_raw(
120                runtime.cast(),
121                *borrowed.as_raw_ffi_any(),
122                hook,
123            )
124        } else {
125            // Prefix-only fallback uses installed remap/callback thunks, never
126            // an ordinary Rust engine's dispatcher offset.
127            StructuralMutatorObj::default_mutate_raw(runtime.cast(), value)
128        }
129    }
130}

132/// Scoped constraint mutator view. Pass the handler's actual `self` to recursive
133/// operations and use `view.base().constraints()` to inspect resulting premises.
134/// Callbacks follow the shared engine's non-escaping, non-unwinding contract.
135pub type StructuralMutatorWithConstraintContextView<'a, D> =
136    StructuralMutatorView<'a, D, StructuralMutatorWithConstraintContextBase>;
137
138/// Mutate using native lexical binding, predicate and range semantics.
139///
140/// The first matching handler owns descent. Use `view.default_mutate(self, value)`
141/// for constraint-aware default descent and `view.mutate(self, child)` to dispatch
142/// recursively. A moved unique root may reuse storage; retained roots copy.
143/// Specialized hooks obey the ordinary synchronous mutation ABI and registration
144/// trust contract. They must not retain the runtime or append through observer
145/// context handles during descent. Errors terminate the traversal.
146pub fn structural_mutate_with_constraint_context<R, H, M, D, O>(
147    root: R,
148    dispatcher: H,
149) -> Result<Any>
150where
151    R: Into<Any>,
152    O: BorrowMut<D>,
153    for<'v, 'c, 'r> H: IntoDispatcher<
154        (MutationView<'v>, &'c mut StructuralMutatorWithConstraintContextView<'r, D>),
155        Option<Result<UnchangedOr<Any>>>,
156        M,
157        Target = D,
158        Output = O,
159    >,
160    for<'s, 'v, 'c, 'r> &'s mut D: Dispatcher<
161        (MutationView<'v>, &'c mut StructuralMutatorWithConstraintContextView<'r, D>),
162        Option<Result<UnchangedOr<Any>>>,
163    >,
164{
165    StructuralMutateEngine::<StructuralMutatorWithConstraintContextBase>::mutate_root(
166        root, dispatcher,
167    )
168}
