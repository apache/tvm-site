20//! Constraint-aware structural mutation. [`structural_mutate_with_constraint_context`] rewrites a
21//! value with a dispatcher whose handlers read the facts in scope from the
22//! [`StructuralMutatorWithConstraintContextView`] they are given.

40/// The base of a constraint-aware mutator, holding the constraint context of the current value.
41///
42/// Its layout mirrors the native one, so it requires the matching native build.
43#[repr(C)]
44pub struct StructuralMutatorWithConstraintContextBase {
45    base_: StructuralMutatorObj,
46    constraints: UnsafeCell<ConstraintContext>,
47    needs_child_context: UnsafeCell<bool>,
48}
49
50impl StructuralMutatorWithConstraintContextBase {
51    /// Return the constraint context in effect at the current value; `snapshot()` it to keep its
52    /// current prefix.
53    #[inline]
54    pub fn constraints(&self) -> ConstraintContext {
55        // SAFETY: the borrow ends with this clone, before anything can replace the field.
56        unsafe { (&*self.constraints.get()).clone() }
57    }
58
59    /// Borrow only while a mapper cannot recurse through this mutator.
60    /// End this borrow before default descent can replace the context field.
61    #[inline]
62    pub(super) unsafe fn constraints_for_observer(&self) -> &ConstraintContext {
63        &*self.constraints.get()
64    }
65}
66
67// SAFETY: repr(C) matches the complete native state-only semantic base. The
68// shared engine installs its vtable, owns remaps and publishes the dispatcher's
69// actual recursive reborrow. Native hooks borrow this runtime synchronously and
70// cannot retain it. No shared reference freezes the replaceable context/flag.
71// Each default call transfers exactly one owning replacement/Unchanged/Error.
72unsafe impl StructuralMutatorBase for StructuralMutatorWithConstraintContextBase {
73    fn new() -> Result<Self> {
74        Ok(Self {
75            base_: StructuralMutatorObj::default(),
76            constraints: UnsafeCell::new(ConstraintContext::new()),
77            needs_child_context: UnsafeCell::new(false),
78        })
79    }
80
81    fn type_index() -> i32 {
82        StructuralMutatorWithConstraintContextObj::type_index()
83    }
84
85    #[inline(always)]
86    unsafe fn default_mutate_raw(runtime: *mut Self, value: MutationView<'_>) -> TVMFFIAny {
87        static COPYING: ColdInitLazyLock<Option<TypeAttrColumn>> =
88            ColdInitLazyLock::new(|| TypeAttrColumn::new("__s_mutate_with_constraint_context__"));
89        static INPLACE: ColdInitLazyLock<Option<TypeAttrColumn>> = ColdInitLazyLock::new(|| {
90            TypeAttrColumn::new("__s_maybe_inplace_mutate_with_constraint_context__")
91        });
92        let borrowed = value.as_any_view();
93        let lookup = |column: Option<TypeAttrColumn>| {
94            column
95                .and_then(|column| column.get_raw_ffi_any(borrowed.type_index()))
96                .filter(|hook| hook.type_index != TVMFFITypeIndex::kTVMFFINone as i32)
97        };
98        let hook = if value.inplace_mode() == InplaceMode::Allow {
99            lookup(*INPLACE).filter(|_| {
100                // Permission alone is insufficient after a callback retains an
101                // alias. Borrow in place so the count is the slot's own reference.
102                borrowed
103                    .downcast_ref::<ObjectRef>()
104                    .is_some_and(|owner| ObjectArc::strong_count(ObjectRef::data(owner)) == 1)
105            })
106        } else {
107            None
108        }
109        .or_else(|| lookup(*COPYING));
110        if let Some(hook) = hook {
111            // Copy the borrowed registry token; the shared helper acquires
112            // Function ownership before packed reentry and handles error context.
113            StructuralMutatorObj::call_structural_mutate_hook_raw(
114                runtime.cast(),
115                *borrowed.as_raw_ffi_any(),
116                hook,
117            )
118        } else {
119            // Prefix-only fallback uses installed remap/callback thunks, never
120            // an ordinary Rust engine's dispatcher offset.
121            StructuralMutatorObj::default_mutate_raw(runtime.cast(), value)
122        }
123    }
124}

126/// The running constraint-aware mutation, passed to each handler; `view.base().constraints()` gives
127/// the facts in scope.
128pub type StructuralMutatorWithConstraintContextView<'a, D> =
129    StructuralMutatorView<'a, D, StructuralMutatorWithConstraintContextBase>;
130
131/// Mutate `root` with `dispatcher`, which sees the facts of the bindings, guards and ranges in
132/// scope.
133///
134/// The first matching handler owns descent: `view.default_mutate(self, value)` rewrites the
135/// children, `view.mutate(self, child)` one child. Handlers must not append to the context or
136/// panic; an error stops the traversal.
137///
138/// # Examples
139///
140/// ```
141/// use tvm_tirx::ir::{
142///     Bind, Evaluate, Expr, PrimExpr, PrimType, PrimVar, SeqStmt, Stmt, Var, UNKNOWN_LOC,
143/// };
144/// use tvm_tirx::script::script;
145/// use tvm_tirx::sym::Analyzer;
146/// use tvm_tirx::tirx::structural_mutate_with_constraint_context::{
147///     structural_mutate_with_constraint_context as mutate,
148///     StructuralMutatorWithConstraintContextView as View,
149/// };
150/// use tvm_tirx::tvm_ffi::{dispatch, Any, MutationView, Result, UnchangedOr};
151///
152/// struct Simplify;
153///
154/// #[dispatch(mutate, (MutationView<'_>, &mut View<'_, Self>), Result<UnchangedOr<Any>>)]
155/// impl Simplify {
156///     fn mutate_evaluate(
157///         &mut self,
158///         node: MutationView<'_, Evaluate>,
159///         mutator: &mut View<'_, Self>,
160///     ) -> Result<UnchangedOr<Any>> {
161///         let analyzer = Analyzer::new();
162///         analyzer.sync_constraint_context(&mutator.base().constraints())?;
163///         let value = PrimExpr::try_from(&node.value)?;
164///         Ok(Evaluate::new(analyzer.simplify(value, 2)?, UNKNOWN_LOC).into())
165///     }
166/// }
167///
168/// let x = PrimVar::new("x", PrimType::int(32), UNKNOWN_LOC);
169/// let bind = Stmt::from(Bind::new(&x, 4, UNKNOWN_LOC)?);
170/// let use_x = Evaluate::new(Expr::try_from(&x * 2)?, UNKNOWN_LOC);
171/// let root = SeqStmt::new([bind, use_x.into()], UNKNOWN_LOC);
172/// let result = SeqStmt::try_from(mutate(root, Simplify)?)?;
173/// assert_eq!(script(&result)?, "x: T.let[T.int32] = 4\nT.evaluate(8)");
174/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
175/// ```
176pub fn structural_mutate_with_constraint_context<R, H, M, D, O>(
177    root: R,
178    dispatcher: H,
179) -> Result<Any>
180where
181    R: Into<Any>,
182    O: BorrowMut<D>,
183    for<'v, 'c, 'r> H: IntoDispatcher<
184        (MutationView<'v>, &'c mut StructuralMutatorWithConstraintContextView<'r, D>),
185        Option<Result<UnchangedOr<Any>>>,
186        M,
187        Target = D,
188        Output = O,
189    >,
190    for<'s, 'v, 'c, 'r> &'s mut D: Dispatcher<
191        (MutationView<'v>, &'c mut StructuralMutatorWithConstraintContextView<'r, D>),
192        Option<Result<UnchangedOr<Any>>>,
193    >,
194{
195    StructuralMutateEngine::<StructuralMutatorWithConstraintContextBase>::mutate_root(
196        root, dispatcher,
197    )
198}
