20//! Index maps.
21//!
22//! An [`IndexMap`] maps its `initial_indices` to its `final_indices`, as `(i, j) -> (j, i)` does.
23//!
24//! # Examples
25//!
26//! ```
27//! use tvm_tirx::ir::{PrimExpr, PrimType, PrimVar, UNKNOWN_LOC};
28//! use tvm_tirx::script::script;
29//! use tvm_tirx::tirx::IndexMap;
30//! use tvm_tirx::tvm_ffi::{Array, Result};
31//!
32//! let i = PrimVar::new("i", PrimType::int(32), UNKNOWN_LOC);
33//! let j = PrimVar::new("j", PrimType::int(32), UNKNOWN_LOC);
34//! let transpose = IndexMap::new([&i, &j], [&j, &i], None)?;
35//! assert_eq!(script(&transpose)?, "T.index_map(lambda i, j: (j, i))");
36//!
37//! let text =
38//!     |exprs: Array<PrimExpr>| exprs.iter().map(script).collect::<Result<Vec<_>>>();
39//! assert_eq!(text(transpose.map_indices([1, 2], None)?)?, ["2", "1"]);
40//!
41//! # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
42//! ```

138impl IndexMap {
139    /// Create the map from `initial_indices` to `final_indices`, with the inverse
140    /// `inverse_index_map` if it is known (C++ `IndexMap(initial_indices, final_indices,
141    /// inverse_index_map)`).
142    pub fn new(
143        initial_indices: impl IntoIterator<Item = impl Into<PrimVar>>,
144        final_indices: impl IntoIterator<Item = impl IntoExpr<PrimExpr>>,
145        inverse_index_map: Option<&IndexMap>,
146    ) -> Result<Self> {
147        let initial_indices =
148            initial_indices.into_iter().map(Into::into).collect::<Array<PrimVar>>();
149        let final_indices = final_indices
150            .into_iter()
151            .map(IntoExpr::into_owned_expr)
152            .collect::<Result<Array<PrimExpr>>>()?;
153        let inverse_index_map = inverse_index_map.map(|map| ObjectRef::from(map.clone()));
154        Ok(Self::from_fields(initial_indices, final_indices, inverse_index_map))
155    }
156
157    /// Return the final indices that `indices` map to (`tirx.IndexMapMapIndices`).
158    ///
159    /// # Errors
160    ///
161    /// Returns an error if the number of indices differs from that of `initial_indices`.
162    pub fn map_indices(
163        &self,
164        indices: impl IntoIterator<Item = impl IntoExpr<PrimExpr>>,
165        analyzer: Option<&Analyzer>,
166    ) -> Result<Array<PrimExpr>> {
167        let indices = indices
168            .into_iter()
169            .map(IntoExpr::into_owned_expr)
170            .collect::<Result<Array<PrimExpr>>>()?;
171        tvm_ffi::cached_global_func!("tirx.IndexMapMapIndices")
172            .call_tuple((self, &indices, analyzer.cloned()))?
173            .try_into()
174    }
175
176    /// Return the shape a buffer of `shape` has after the mapping (`tirx.IndexMapMapShape`).
177    ///
178    /// # Errors
179    ///
180    /// Returns an error if the number of extents differs from that of `initial_indices`.
181    pub fn map_shape(
182        &self,
183        shape: impl IntoIterator<Item = impl IntoExpr<PrimExpr>>,
184        analyzer: Option<&Analyzer>,
185    ) -> Result<Array<PrimExpr>> {
186        let shape = shape
187            .into_iter()
188            .map(IntoExpr::into_owned_expr)
189            .collect::<Result<Array<PrimExpr>>>()?;
190        tvm_ffi::cached_global_func!("tirx.IndexMapMapShape")
191            .call_tuple((self, &shape, analyzer.cloned()))?
192            .try_into()
193    }
194
195    /// Return a copy of `tensor` with its elements moved to the mapped indices
196    /// (`tirx.IndexMapMapTensor`).
197    ///
198    /// # Errors
199    ///
200    /// Returns an error if the rank of `tensor` differs from the number of `initial_indices`.
201    pub fn map_tensor(&self, tensor: &Tensor) -> Result<Tensor> {
202        tvm_ffi::cached_global_func!("tirx.IndexMapMapTensor")
203            .call_tuple((self, tensor))?
204            .try_into()
205    }
206
207    /// Return the inverse map, which must be bijective over `ranges` (`tirx.IndexMapInverse`).
208    ///
209    /// # Errors
210    ///
211    /// Returns an error if the final indices are not affine in the initial ones, or the map is not
212    /// provably bijective over `ranges`.
213    pub fn inverse(
214        &self,
215        ranges: impl IntoIterator<Item = impl Into<Range>>,
216        analyzer: Option<&Analyzer>,
217    ) -> Result<IndexMap> {
218        let ranges = ranges.into_iter().map(Into::into).collect::<Array<Range>>();
219        tvm_ffi::cached_global_func!("tirx.IndexMapInverse")
220            .call_tuple((self, &ranges, analyzer.cloned()))?
221            .try_into()
222    }
223
224    /// Return the inverse of a map that may pad `ranges`, and the predicate under which the inverse
225    /// maps to an index in `ranges` (`tirx.IndexMapNonSurjectiveInverse`).
226    pub fn non_surjective_inverse(
227        &self,
228        ranges: impl IntoIterator<Item = impl Into<Range>>,
229        analyzer: Option<&Analyzer>,
230    ) -> Result<(IndexMap, PrimExpr)> {
231        let ranges = ranges.into_iter().map(Into::into).collect::<Array<Range>>();
232        let pair: Array<Any> = tvm_ffi::cached_global_func!("tirx.IndexMapNonSurjectiveInverse")
233            .call_tuple((self, &ranges, analyzer.cloned()))?
234            .try_into()?;
235        Ok((pair.get(0)?.try_into()?, pair.get(1)?.try_into()?))
236    }
237}
