37/// The base of a constraint-aware visitor, holding the constraint context of the current value.
38///
39/// Unlike the opaque [`StructuralVisitorWithConstraintContextObj`], its layout mirrors the native
40/// one, so it requires the matching native build.
41#[repr(C)]
42pub struct StructuralVisitorWithConstraintContextBase {
43    base: StructuralVisitorObj,
44    constraints: UnsafeCell<ConstraintContext>,
45    needs_child_context: UnsafeCell<bool>,
46}
47
48impl StructuralVisitorWithConstraintContextBase {
49    /// Return the constraint context in effect at the current value; use
50    /// [`ConstraintContextObj::snapshot`] to keep its current prefix.
51    pub fn constraints(&self) -> ConstraintContext {
52        // No reference to the replaceable field survives this short clone.
53        unsafe { (&*self.constraints.get()).clone() }
54    }
55
56    /// Borrow only while an observer has no access to this visitor's descent.
57    /// The caller must end the borrow before any operation can replace the context.
58    #[inline]
59    pub(super) unsafe fn constraints_for_observer(&self) -> &ConstraintContext {
60        &*self.constraints.get()
61    }
62}
63
64// SAFETY: repr(C) mirrors the complete base in the matching native header.
65// Stub generation cannot certify its private tail; the binding requires the
66// matching native build. All fields begin with the same state as the native
67// constructor. Native-mutated fields use UnsafeCell. The type identity is the
68// registered specialized type, while the shared engine installs its vtable and
69// owns the dispatcher. Default hooks receive that live runtime synchronously and
70// return one owning None/interrupt/error token under the ordinary hook contract.
71unsafe impl StructuralVisitorBase for StructuralVisitorWithConstraintContextBase {
72    fn new() -> Result<Self> {
73        Ok(Self {
74            base: StructuralVisitorObj::default(),
75            constraints: UnsafeCell::new(ConstraintContext::new()?),
76            needs_child_context: UnsafeCell::new(false),
77        })
78    }
79
80    fn type_index() -> i32 {
81        StructuralVisitorWithConstraintContextObj::type_index()
82    }
83
84    #[inline(always)]
85    unsafe fn default_visit_raw(runtime: *mut Self, value: AnyView<'_>) -> TVMFFIAny {
86        static HOOKS: ColdInitLazyLock<Option<TypeAttrColumn>> =
87            ColdInitLazyLock::new(|| TypeAttrColumn::new("__s_visit_with_constraint_context__"));
88        // Copy the borrowed token before reentry. Packed dispatch acquires an
89        // owning Function before invoking a callback that could register hooks.
90        let hook = HOOKS
91            .and_then(|column| column.get_raw_ffi_any(value.type_index()))
92            .filter(|hook| hook.type_index != TVMFFITypeIndex::kTVMFFINone as i32);
93        if let Some(hook) = hook {
94            StructuralVisitorObj::call_structural_visit_hook_raw(
95                runtime.cast(),
96                *value.as_raw_ffi_any(),
97                hook,
98            )
99        } else {
100            // Prefix-only default descent calls the installed vtable for each
101            // child; it must not project an ordinary engine's dispatcher offset.
102            StructuralVisitorObj::default_visit_raw(runtime.cast(), value)
103        }
104    }
105}

107/// The running constraint-aware visit, passed to each handler; `view.base().constraints()` gives
108/// the facts in scope.
109///
110/// ```compile_fail
111/// use tvm_tirx::tirx::structural_visit_with_constraint_context::StructuralVisitorWithConstraintContextView;
112/// fn escape<'a, D>(v: StructuralVisitorWithConstraintContextView<'a, D>)
113///     -> StructuralVisitorWithConstraintContextView<'static, D> { v }
114/// ```
115///
116/// ```compile_fail
117/// use tvm_tirx::tirx::structural_visit_with_constraint_context::StructuralVisitorWithConstraintContextView;
118/// use tvm_tirx::tvm_ffi::Any;
119/// fn retain<D>(v: &StructuralVisitorWithConstraintContextView<'_, D>) -> Any {
120///     v.base().into()
121/// }
122/// ```
123pub type StructuralVisitorWithConstraintContextView<'a, D> =
124    StructuralVisitorView<'a, D, StructuralVisitorWithConstraintContextBase>;
125
126/// Visit `root` with `visitor`, which sees the facts of the bindings, ranges and guards in scope.
127///
128/// A matched handler owns descent; other values are visited by default. Handlers return failures
129/// through `Result` and must not panic: a panic at the native callback boundary aborts the process.
130///
131/// # Examples
132///
133/// ```
134/// use tvm_tirx::ir::prim::op::PrimOps;
135/// use tvm_tirx::ir::{
136///     Evaluate, EvaluateObj, For, ForKind, PrimExpr, PrimType, PrimVar, UNKNOWN_LOC,
137/// };
138/// use tvm_tirx::sym::{Analyzer, ProofStrength};
139/// use tvm_tirx::tirx::structural_visit_with_constraint_context::{
140///     structural_visit_with_constraint_context as visit,
141///     StructuralVisitorWithConstraintContextView as View,
142/// };
143/// use tvm_tirx::tvm_ffi::{dispatch, AnyView, Result, VisitInterrupt};
144///
145/// #[derive(Default)]
146/// struct BelowSixteen(Vec<bool>);
147///
148/// #[dispatch(visit, (AnyView<'_>, &mut View<'_, Self>), Result<Option<VisitInterrupt>>)]
149/// impl BelowSixteen {
150///     fn visit_evaluate(
151///         &mut self,
152///         node: &EvaluateObj,
153///         visitor: &mut View<'_, Self>,
154///     ) -> Result<Option<VisitInterrupt>> {
155///         let analyzer = Analyzer::new();
156///         analyzer.sync_constraint_context(&visitor.base().constraints())?;
157///         let value = PrimExpr::try_from(&node.value)?;
158///         self.0.push(analyzer.can_prove(value.less(16), ProofStrength::Default)?);
159///         visitor.default_visit(self, node)
160///     }
161/// }
162///
163/// let i = PrimVar::new("i", PrimType::int(32), UNKNOWN_LOC);
164/// let body = Evaluate::new(&i, UNKNOWN_LOC);
165/// let root = For::new(&i, 0, 16, 1, ForKind::Default, body, None, UNKNOWN_LOC)?;
166/// let mut proofs = BelowSixteen::default();
167/// visit(&root, &mut proofs)?;
168/// assert_eq!(proofs.0, [true]);
169/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
170/// ```
171pub fn structural_visit_with_constraint_context<'input, R, H, M, D, O>(
172    root: R,
173    visitor: H,
174) -> Result<Option<VisitInterrupt>>
175where
176    R: Into<AnyView<'input>>,
177    O: BorrowMut<D>,
178    for<'v, 'c, 'r> H: IntoDispatcher<
179        (AnyView<'v>, &'c mut StructuralVisitorWithConstraintContextView<'r, D>),
180        Option<Result<Option<VisitInterrupt>>>,
181        M,
182        Target = D,
183        Output = O,
184    >,
185    for<'s, 'v, 'c, 'r> &'s mut D: Dispatcher<
186        (AnyView<'v>, &'c mut StructuralVisitorWithConstraintContextView<'r, D>),
187        Option<Result<Option<VisitInterrupt>>>,
188    >,
189{
190    StructuralVisitEngine::<StructuralVisitorWithConstraintContextBase>::visit_root(root, visitor)
191}
