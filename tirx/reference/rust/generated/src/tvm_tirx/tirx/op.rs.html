21//! Attributes of TIRx operator calls.
22//!
23//! The functions build TIRx operator calls; a [`CallFFIKernelAttr`] holds the launch parameters of
24//! a `call_ffi_kernel`.
25//!
26//! # Examples
27//!
28//! ```
29//! use tvm_tirx::ir::{PrimType, PrimVar, Var, UNKNOWN_LOC};
30//! use tvm_tirx::script::script;
31//! use tvm_tirx::tirx::{logaddexp, reinterpret};
32//!
33//! let x = PrimVar::new("x", PrimType::float(32), UNKNOWN_LOC);
34//! let y = PrimVar::new("y", PrimType::float(32), UNKNOWN_LOC);
35//! assert_eq!(script(&logaddexp(&x, &y)?)?, "T.log(T.exp(x) + T.exp(y))");
36//! assert_eq!(script(&reinterpret(PrimType::int(32), &x)?)?, "T.reinterpret(x, ty=T.int32)");
37//! # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
38//! ```

178/// Build `log(exp(a) + exp(b))` of the floating-point `a` and `b`, matched to one type
179/// (C++ `tirx::logaddexp`, `tirx._OpLogAddExp`).
180///
181/// # Errors
182///
183/// Returns an error if an operand is not floating-point.
184pub fn logaddexp(a: impl IntoExpr<PrimExpr>, b: impl IntoExpr<PrimExpr>) -> Result<PrimExpr> {
185    logaddexp_at(a, b, UNKNOWN_LOC)
186}
187
188/// Build [`logaddexp`] with the casts at `loc`.
189pub fn logaddexp_at(
190    a: impl IntoExpr<PrimExpr>,
191    b: impl IntoExpr<PrimExpr>,
192    loc: impl Into<Location>,
193) -> Result<PrimExpr> {
194    let (a, b) = (a.into_owned_expr()?, b.into_owned_expr()?);
195    for (operand, side) in [(&a, "a"), (&b, "b")] {
196        if operand.dtype().code != support::FLOAT {
197            let condition = format!("{side}.ty().MatchesCode(DLDataTypeCode::kDLFloat)");
198            let message = support::repr(operand);
199            return Err(support::check_failed(INTERNAL_ERROR, &condition, &message));
200        }
201    }
202    let (a, b) = match_types_at(a, b, loc)?;
203    log(add(exp(a)?, exp(b)?)?)
204}

206/// Whether `dtype` is a scalar 64-bit integer, which a pointer reinterprets as.
207fn is_scalar_int64(dtype: DLDataType) -> bool {
208    support::is_scalar(dtype)
209        && dtype.bits == 64
210        && matches!(dtype.code, support::INT | support::UINT)
211}

213/// C++ `PrimType::StorageBytes()`.
214fn storage_bytes(dtype: DLDataType) -> Result<u64> {
215    Ok((u64::from(dtype.bits) * support::lanes(dtype)? as u64).div_ceil(8))
216}

218/// Reinterpret `value` bit for bit as the primitive or pointer type `ty`
219/// (C++ `tirx::reinterpret(target_ty, value, loc)`, `tirx.reinterpret`).
220///
221/// Besides primitive types of the same size, a string reinterprets as a pointer and a pointer as a
222/// scalar `int64`, and back.
223///
224/// # Errors
225///
226/// Returns an error for any other pair of types, or two primitive types of different sizes.
227pub fn reinterpret(ty: impl Into<Type>, value: impl IntoExpr) -> Result<Expr> {
228    reinterpret_at(ty, value, UNKNOWN_LOC)
229}
230
231/// Build [`reinterpret`] at `loc`.
232pub fn reinterpret_at(
233    ty: impl Into<Type>,
234    value: impl IntoExpr,
235    loc: impl Into<Location>,
236) -> Result<Expr> {
237    let (ty, value, loc) = (ty.into(), value.into_owned_expr()?, loc.into());
238    let target_is_pointer = ty.downcast_ref::<PtrTypeObj>().is_some();
239    let source_is_pointer = value.ty.downcast_ref::<PtrTypeObj>().is_some();
240    let type_error = |condition: &str, message: std::string::String| {
241        Err(support::check_failed(TYPE_ERROR, condition, &message))
242    };
243    if value.downcast_ref::<StringImmObj>().is_some() {
244        if !target_is_pointer {
245            let message = format!(
246                "String reinterpret requires a pointer target, but got {}",
247                support::repr(&ty)
248            );
249            return type_error("target_ty.as<PtrTypeObj>()", message);
250        }
251    } else if let Some(target) = ty.downcast_ref::<PrimTypeObj>() {
252        let target_dtype = target.dtype;
253        if let Some(source) = value.ty.downcast_ref::<PrimTypeObj>() {
254            let source_dtype = source.dtype;
255            if source_dtype == target_dtype {
256                return Ok(value);
257            }
258            if !support::is_scalable_vector(target_dtype)
259                && !support::is_scalable_vector(source_dtype)
260            {
261                let bits = |dtype: DLDataType| {
262                    Ok::<i64, tvm_ffi::Error>(
263                        i64::from(dtype.bits) * i64::from(support::lanes(dtype)?),
264                    )
265                };
266                let float4 = DLDataTypeCode::kDLFloat4_e2m1fn as u8;
267                let same_size = bits(source_dtype)? == bits(target_dtype)?
268                    || ((source_dtype.code == float4 || target_dtype.code == float4)
269                        && storage_bytes(source_dtype)? == storage_bytes(target_dtype)?);
270                if !same_size {
271                    let condition = "value_bits == target_bits || \
272                                     ((value_dtype.MatchesCode(DLDataTypeCode::kDLFloat4_e2m1fn) || \
273                                     target_prim.MatchesCode(DLDataTypeCode::kDLFloat4_e2m1fn)) && \
274                                     value_dtype.StorageBytes() == target_prim.StorageBytes())";
275                    let message = format!(
276                        "Reinterpret requires size match {} vs {}",
277                        support::repr(&ty),
278                        support::repr(&value.ty)
279                    );
280                    return Err(support::check_failed(INTERNAL_ERROR, condition, &message));
281                }
282            }
283        } else {
284            if !source_is_pointer {
285                let message = format!(
286                    "Reinterpret source must be PrimType or PtrType, but got {}",
287                    support::repr(&value.ty)
288                );
289                return type_error("value->ty.as<PtrTypeObj>()", message);
290            }
291            if !is_scalar_int64(target_dtype) {
292                let condition = "target_dtype.value().IsScalar() && target_dtype.value().bits() == 64 && \
293                                 target_dtype.value().MatchesCode(DLDataTypeCode::kDLInt, DLDataTypeCode::kDLUInt)";
294                let message = format!(
295                    "Pointer reinterpret requires a scalar 64-bit integer target, but got {}",
296                    support::repr(&ty)
297                );
298                return type_error(condition, message);
299            }
300        }
301    } else {
302        if !target_is_pointer {
303            let message = format!(
304                "Reinterpret target must be PrimType or PtrType, but got {}",
305                support::repr(&ty)
306            );
307            return type_error("target_ty.as<PtrTypeObj>()", message);
308        }
309        if let Some(source) = value.ty.downcast_ref::<PrimTypeObj>() {
310            if !is_scalar_int64(source.dtype) {
311                let condition = "source_dtype.value().IsScalar() && source_dtype.value().bits() == 64 && \
312                                 source_dtype.value().MatchesCode(DLDataTypeCode::kDLInt, DLDataTypeCode::kDLUInt)";
313                let message = format!(
314                    "Pointer reinterpret requires a scalar 64-bit integer source, but got {}",
315                    support::repr(&value.ty)
316                );
317                return type_error(condition, message);
318            }
319        } else if !source_is_pointer {
320            let message = format!(
321                "Reinterpret source must be PrimType or PtrType, but got {}",
322                support::repr(&value.ty)
323            );
324            return type_error("value->ty.as<PtrTypeObj>()", message);
325        }
326    }
327    let op = support::cached_op("tirx.reinterpret")?;
328    Ok(Call::new(Some(&ty), op, vec![value], None, Vec::<Type>::new(), loc)?.into())
329}

331/// Build a rational approximation of `erf(arg)` on `[-4, 4]` in `bits`-bit floats, as the Metal and
332/// WebGPU backends lower `erf` (C++ `tirx::fast_erf_float_expr(arg, bits)`).
333pub fn fast_erf_float_expr(arg: impl IntoExpr<PrimExpr>, bits: u8) -> Result<PrimExpr> {
334    let dtype = PrimType::float(bits).dtype;
335    // The C++ coefficients are `float` literals.
336    let constant = |value: f32| -> ExprResult {
337        FloatImm::from_dtype(dtype, f64::from(value), UNKNOWN_LOC).into()
338    };
339    // alpha_1, alpha_3, ..., alpha_13: the odd coefficients of the numerator.
340    #[allow(clippy::excessive_precision)]
341    let alpha: [f32; 7] = [
342        -1.60960333262415e-02,
343        -2.95459980854025e-03,
344        -7.34990630326855e-04,
345        -5.69250639462346e-05,
346        -2.10102402082508e-06,
347        2.77068142495902e-08,
348        -2.72614225801306e-10,
349    ];
350    // beta_0, beta_2, ..., beta_8: the even coefficients of the denominator.
351    #[allow(clippy::excessive_precision)]
352    let beta: [f32; 5] = [
353        -1.42647390514189e-02,
354        -7.37332916720468e-03,
355        -1.68282697438203e-03,
356        -2.13374055278905e-04,
357        -1.45660718464996e-05,
358    ];
359    let x = arg.into_owned_expr()?.min(constant(4.0)).max(constant(-4.0));
360    let x2 = &x * &x;
361    // Horner's rule from the highest coefficient down, as C++ writes it out.
362    let mut p = &x2 * constant(alpha[6]) + constant(alpha[5]);
363    for coefficient in alpha[..5].iter().rev() {
364        p = &x2 * p + constant(*coefficient);
365    }
366    p = &x * p;
367    let mut q = &x2 * constant(beta[4]) + constant(beta[3]);
368    for coefficient in beta[..3].iter().rev() {
369        q = &x2 * q + constant(*coefficient);
370    }
371    p.div(q).try_into()
372}

374/// Build a return from the current GPU thread, without a value (`tirx.gpu_thread_return`).
375pub fn gpu_thread_return() -> Result<PrimExpr> {
376    gpu_thread_return_at(UNKNOWN_LOC)
377}

379/// Build [`gpu_thread_return`] at `loc`.
380pub fn gpu_thread_return_at(loc: impl Into<Location>) -> Result<PrimExpr> {

382    tvm_ffi::cached_global_func!("tirx.gpu_thread_return").call_tuple((&loc,))?.try_into()
