21//! Thread scopes: the levels of a backend's thread hierarchy.
22//!
23//! A [`ThreadScope`], such as `cuda.warp`, is looked up by name with [`ThreadScope::get`].
24//!
25//! # Examples
26//!
27//! ```
28//! use tvm_tirx::tirx::ThreadScope;
29//!
30//! let warp = ThreadScope::get("cuda.warp")?;
31//! assert_eq!((warp.backend()?, warp.local_name()?), ("cuda".into(), "warp".into()));
32//!
33//! let cta = ThreadScope::get("cuda.cta")?;
34//! assert!(cta.is_higher_than(&warp)?);
35//! # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
36//! ```
37

65/// Opaque: parent 'ffi.Enum' is opaque (no-mirror). Fields are read through the C ABI getters.
66#[repr(C)]
67#[derive(tvm_ffi::derive::Object)]
68#[type_key = "tirx.ThreadScope"]
69#[type_final]
70pub struct ThreadScopeObj {
71    base: FfiEnumObj,
72}
73
74#[repr(C)]
75#[derive(tvm_ffi::derive::ObjectRef, Clone)]
76pub struct ThreadScope {
77    base: ObjectArc<ThreadScopeObj>,
78}
79
80impl Deref for ThreadScope {
81    type Target = ThreadScopeObj;
82    fn deref(&self) -> &ThreadScopeObj {
83        &self.base
84    }
85}
86
87unsafe impl tvm_ffi::object::ImmutableObject for ThreadScopeObj {}
88// tvm-ffi-stubgen(end)
89
90impl ThreadScope {
91    /// Look up the thread scope named `name`, such as `cuda.warp` (`tirx.ThreadScopeGet`).
92    ///
93    /// # Errors
94    ///
95    /// Returns an error if no thread scope is registered as `name`.
96    pub fn get(name: &str) -> Result<Self> {
97        tvm_ffi::cached_global_func!("tirx.ThreadScopeGet")
98            .call_tuple((String::from(name),))?
99            .try_into()
100    }
101
102    /// Return the backend that registered the scope, such as `cuda` (`tirx.ThreadScopeGetBackend`).
103    pub fn backend(&self) -> Result<String> {
104        tvm_ffi::cached_global_func!("tirx.ThreadScopeGetBackend").call_tuple((self,))?.try_into()
105    }
106
107    /// Return the name of the scope within its backend, such as `warp`
108    /// (`tirx.ThreadScopeGetLocalName`).
109    pub fn local_name(&self) -> Result<String> {
110        tvm_ffi::cached_global_func!("tirx.ThreadScopeGetLocalName").call_tuple((self,))?.try_into()
111    }
112
113    /// Return the level of the scope in its backend's hierarchy; outer scopes rank lower
114    /// (`tirx.ThreadScopeGetRank`).
115    pub fn rank(&self) -> Result<i64> {
116        tvm_ffi::cached_global_func!("tirx.ThreadScopeGetRank").call_tuple((self,))?.try_into()
117    }
118
119    /// Return whether this scope is above `other` in the hierarchy of their backend
120    /// (`tirx.ScopeHigher`).
121    ///
122    /// # Errors
123    ///
124    /// Returns an error if the two scopes are of different backends.
125    pub fn is_higher_than(&self, other: impl IntoObjectRef<ThreadScope>) -> Result<bool> {
126        tvm_ffi::cached_global_func!("tirx.ScopeHigher")
127            .call_tuple((self, other.into_object_ref().borrow()))?
128            .try_into()
129    }
130}
