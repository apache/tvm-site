20//! Types of TIRx operations.
21//!
22//! [`TensorMapType`] is the type of a tensor map.
23//!
24//! # Examples
25//!
26//! ```
27//! use tvm_tirx::ir::UNKNOWN_LOC;
28//! use tvm_tirx::script::script;
29//! use tvm_tirx::tirx::TensorMapType;
30//!
31//! let ty = TensorMapType::new(UNKNOWN_LOC);
32//! assert_eq!(script(&ty)?, "T.TensorMap()");
33//!
34//! # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
35//! ```
