20//! Constraint-aware structural mapping. [`structural_map_with_constraint_context`] calls a closure
21//! on every value with the facts in scope there and replaces the value with what it returns,
22//! unless that is `Unchanged`.
23
24use std::borrow::BorrowMut;
25
26use tvm_ffi::{
27    Any, AnyView, DefRegionKind, Dispatcher, IntoDispatcher, MutationView, Object, Result,
28    UnchangedOr, WalkOrder,
29};
30
31use super::structural_mutate_with_constraint_context::{
32    structural_mutate_with_constraint_context, StructuralMutatorWithConstraintContextView,
33};
34use crate::sym::ConstraintContext;
35
36struct MapMutator<'a, D, const PRE: bool> {
37    mapper: &'a mut D,
38}
39
40impl<'v, 'c, 'r, 'a, D, const PRE: bool>
41    Dispatcher<
42        (
43            MutationView<'v>,
44            &'c mut StructuralMutatorWithConstraintContextView<'r, MapMutator<'a, D, PRE>>,
45        ),
46        Option<Result<UnchangedOr<Any>>>,
47    > for &mut MapMutator<'a, D, PRE>
48where
49    for<'s, 'b, 'ctx> &'s mut D: Dispatcher<
50            (AnyView<'b>, &'ctx ConstraintContext, DefRegionKind),
51            Option<Result<UnchangedOr<Any>>>,
52        >,
53{
54    #[inline(always)]
55    fn dispatch_tuple(
56        self,
57        (value, mutator): (
58            MutationView<'v>,
59            &'c mut StructuralMutatorWithConstraintContextView<'r, MapMutator<'a, D, PRE>>,
60        ),
61    ) -> Option<Result<UnchangedOr<Any>>> {
62        Some(self.map(value, mutator))
63    }
64}
65
66impl<D, const PRE: bool> MapMutator<'_, D, PRE>
67where
68    for<'s, 'v, 'c> &'s mut D: Dispatcher<
69            (AnyView<'v>, &'c ConstraintContext, DefRegionKind),
70            Option<Result<UnchangedOr<Any>>>,
71        >,
72{
73    #[inline(always)]
74    fn map(
75        &mut self,
76        value: MutationView<'_>,
77        mutator: &mut StructuralMutatorWithConstraintContextView<'_, Self>,
78    ) -> Result<UnchangedOr<Any>> {
79        let kind = mutator.def_region_kind();
80        if PRE {
81            let mode = value.inplace_mode();
82            // SAFETY: the mapper cannot reach this mutator, and the borrow ends with the
83            // callback, before descent can replace the field.
84            let constraints = unsafe { mutator.base().constraints_for_observer() };
85            let result = self.mapper.dispatch_tuple((value.as_any_view(), constraints, kind));
86            let Some(result) = result else {
87                return mutator.default_mutate(self, value);
88            };
89            let mapped = result?;
90            if mapped.is_unchanged() {
91                return mutator.default_mutate(self, value);
92            }
93            let mut mapped: Any = mapped.into();
94            let same_object = value
95                .downcast_ref::<Object>()
96                .zip(mapped.downcast_ref::<Object>())
97                .is_some_and(|(original, replacement)| std::ptr::eq(original, replacement));
98            if same_object {
99                // An identity callback must not make a unique root appear shared.
100                drop(mapped);
101                return mutator.default_mutate(self, value);
102            }
103            // An owned replacement can lend a capability, restricted by the
104            // original path's permission. The shared engine checks live aliases.
105            let replacement = MutationView::from(&mut mapped).narrow_inplace_mode(mode);
106            let result = mutator.default_mutate(self, replacement)?;
107            Ok(UnchangedOr::changed(result.value_or_unchanged(mapped)))
108        } else {
109            // SAFETY: this private adapter runs only inside the shared engine
110            // callback. The engine owns the root; recursive callers keep each
111            // input field owned until this synchronous callback returns. Default
112            // descent can mutate fields but cannot destroy or retype that input.
113            // This snapshot holds no typed node reference across descent. Only
114            // reborrow it afterward when Unchanged preserves the input, and never
115            // let the reborrow escape this callback.
116            let original: AnyView<'_> = unsafe { std::mem::transmute(value.as_any_view()) };
117            let mapped = mutator.default_mutate(self, value)?;
118            let current = if mapped.is_unchanged() { original } else { AnyView::from(&mapped) };
119            // Scoped native hooks have restored the incoming context here.
120            // SAFETY: as above; descent has ended.
121            let constraints = unsafe { mutator.base().constraints_for_observer() };
122            let result = self.mapper.dispatch_tuple((current, constraints, kind));
123            match result {
124                Some(result) => {
125                    let result = result?;
126                    Ok(if result.is_unchanged() { mapped } else { result })
127                }
128                None => Ok(mapped),
129            }
130        }
131    }
132}

134/// Map every value of `root` with `mapper`, in `order`, with the facts in scope at each value.
135///
136/// A callback takes a leading prefix of `(value, &ConstraintContext, DefRegionKind)` and returns a
137/// replacement, `Unchanged` or a `Result` of one; `Unchanged` still descends in pre-order.
138/// Callbacks must not append to the context or panic, and an error stops the traversal without
139/// undoing in-place changes.
140///
141/// # Examples
142///
143/// ```
144/// use tvm_tirx::ir::{Bind, Evaluate, PrimType, PrimVar, SeqStmt, Stmt, Var, UNKNOWN_LOC};
145/// use tvm_tirx::script::script;
146/// use tvm_tirx::sym::{BindingConstraintObj, ConstraintContext};
147/// use tvm_tirx::tirx::structural_map_with_constraint_context as map;
148/// use tvm_tirx::tvm_ffi::{Any, ObjectRefCore, Unchanged, WalkOrder};
149///
150/// let substitute = |var: Var, context: &ConstraintContext| -> Any {
151///     let mut value: Any = Unchanged.into();
152///     context.for_each(|fact| match fact.downcast_ref::<BindingConstraintObj>() {
153///         Some(binding) if binding.var.same_as(&var) => value = binding.value.clone().into(),
154///         _ => {}
155///     });
156///     value
157/// };
158///
159/// let x = PrimVar::new("x", PrimType::int(32), UNKNOWN_LOC);
160/// let bind = Stmt::from(Bind::new(&x, 4, UNKNOWN_LOC)?);
161/// let root = SeqStmt::new([bind, Evaluate::new(&x, UNKNOWN_LOC).into()], UNKNOWN_LOC);
162/// let result =
163///     map::structural_map_with_constraint_context(root, substitute, WalkOrder::PreOrder)?;
164/// assert_eq!(script(&SeqStmt::try_from(result)?)?, "x: T.let[T.int32] = 4\nT.evaluate(4)");
165/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
166/// ```
167pub fn structural_map_with_constraint_context<R, H, M, D, O>(
168    root: R,
169    mapper: H,
170    order: WalkOrder,
171) -> Result<Any>
172where
173    R: Into<Any>,
174    O: BorrowMut<D>,
175    for<'v, 'c> H: IntoDispatcher<
176            (AnyView<'v>, &'c ConstraintContext, DefRegionKind),
177            Option<Result<UnchangedOr<Any>>>,
178            M,
179            Target = D,
180            Output = O,
181        >,
182    for<'s, 'v, 'c> &'s mut D: Dispatcher<
183            (AnyView<'v>, &'c ConstraintContext, DefRegionKind),
184            Option<Result<UnchangedOr<Any>>>,
185        >,
186{
187    let mut storage = mapper.into_dispatcher();
188    let mapper = storage.borrow_mut();
189    match order {
190        WalkOrder::PreOrder => {
191            structural_mutate_with_constraint_context(root, MapMutator::<D, true> { mapper })
192        }
193        WalkOrder::PostOrder => {
194            structural_mutate_with_constraint_context(root, MapMutator::<D, false> { mapper })
195        }
196    }
197}
