62/// The type of a tensor of `shape` and element type `dtype` in the memory `scope`, with its data
63/// pointer aligned to `data_alignment` bytes and an optional `layout` from indices to storage.
64///
65/// The shape is the one loads and stores index. The data pointer is not part of the type; the
66/// definition of the tensor variable binds it.
67///
68/// # Examples
69///
70/// ```
71/// use tvm_tirx::ir::{Expr, PrimType, PrimVar, Subscript, Var, UNKNOWN_LOC};
72/// use tvm_tirx::runtime::MemoryScope;
73/// use tvm_tirx::script::script;
74/// use tvm_tirx::tirx::TensorType;
75/// use tvm_tirx::tvm_ffi::reflection::EnumDef;
76///
77/// let scratch = EnumDef::<MemoryScope>::new("my_ext.scratch")?.value();
78/// let ty = TensorType::new([16, 16], PrimType::float(32), &scratch, 0, None, UNKNOWN_LOC)?;
79/// assert_eq!(script(&ty)?, r#"T.Tensor((16, 16), T.float32, scope="my_ext.scratch")"#);
80///
81/// let a = Expr::from(Var::new("A", &ty, UNKNOWN_LOC));
82/// let i = PrimVar::new("i", PrimType::int(32), UNKNOWN_LOC);
83/// let point = [Subscript::Index(i.into()), Subscript::Index(0.into())];
84/// let element = a.subscript(point, UNKNOWN_LOC)?;
85/// assert_eq!(script(&Expr::try_from(element)?)?, "A[i, 0]");
86///
87/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
88/// ```
89// tvm-ffi-stubgen(begin): object/tirx.TensorType
90#[allow(missing_docs)]
91#[repr(C)]
92#[derive(tvm_ffi::derive::ObjectRef, Clone)]
93pub struct TensorType {
94    data_: ObjectArc<TensorTypeObj>,
95}

107    pub dtype: PrimType,
108    pub scope: MemoryScope,
109    pub shape: Array<PrimExpr>,

169impl TensorType {
170    /// Create the type of a tensor of `shape` and `dtype` in `scope`, aligned to `data_alignment`
171    /// bytes (the runtime's allocation alignment when 0 or less), with `layout` if given.
172    pub fn new(
173        shape: impl IntoIterator<Item = impl IntoExpr<PrimExpr>>,
174        dtype: impl IntoObjectRef<PrimType>,
175        scope: impl IntoObjectRef<MemoryScope>,
176        data_alignment: i32,
177        layout: Option<&Layout>,
178        loc: impl IntoObjectRef<Location>,
179    ) -> Result<Self> {
180        let shape = shape
181            .into_iter()
182            .map(IntoExpr::into_owned_expr)
183            .collect::<Result<Array<PrimExpr>>>()?;
184        let data_alignment = match data_alignment <= 0 {
185            true => alloc_alignment()?,
186            false => data_alignment,
187        };
188        Ok(Self::from_fields(
189            loc.into_owned_object_ref(),
190            dtype.into_owned_object_ref(),
191            scope.into_owned_object_ref(),
192            shape,
193            data_alignment,
194            layout.cloned(),
195        ))
196    }
197
198    /// Return whether the layout is absent or equals the default row-major layout of the shape
199    /// (`tirx.TensorTypeHasDefaultLayout`).
200    pub fn has_default_layout(&self, analyzer: Option<&Analyzer>) -> Result<bool> {
201        tvm_ffi::cached_global_func!("tirx.TensorTypeHasDefaultLayout")
202            .call_tuple((self, analyzer.cloned()))?
203            .try_into()
204    }
205}

207/// A signature descriptor ([`sig::ArgType`]) that matches a [`Var`](crate::ir::Var) whose type is
208/// a [`TensorType`]; it is not a type of its own.
209pub struct TensorVar;

211impl sig::ArgType for TensorVar {
212    fn check(value: AnyView<'_>) -> bool {
213        sig::var_of_type::<TensorTypeObj>(value)
214    }
215
216    fn type_str() -> std::string::String {
217        crate::ir::expr::VarObj::TYPE_KEY.into()
218    }
219}

221/// The runtime's allocation alignment, which a build may configure, read from the default the
222/// reflection registers for `TensorType::data_alignment`.
223fn alloc_alignment() -> Result<i32> {
224    static ALLOC_ALIGNMENT: std::sync::OnceLock<i32> = std::sync::OnceLock::new();
225    if let Some(alignment) = ALLOC_ALIGNMENT.get() {
226        return Ok(*alignment);
227    }
228    let Some(default) = tvm_ffi::reflection::field_default::<TensorTypeObj>("data_alignment")
229    else {
230        let message = "tirx.TensorType registers no default data_alignment";
231        return Err(tvm_ffi::Error::new(tvm_ffi::INTERNAL_ERROR, message, ""));
232    };
233    let alignment = i32::try_from(i64::try_from(default)?).map_err(|_| {
234        tvm_ffi::Error::new(tvm_ffi::INTERNAL_ERROR, "the default data_alignment exceeds i32", "")
235    })?;
236    Ok(*ALLOC_ALIGNMENT.get_or_init(|| alignment))
237}
