19//! The typed expression views, [`PrimExpr`](crate::ir::PrimExpr),
20//! [`IntExpr`](crate::ir::IntExpr) and [`PrimVar`](crate::ir::PrimVar) (C++ `TypedExpr`).
21
22/// Define a view: a reference to a `$obj`, the object of the reference `$base`, that `$check`
23/// accepts, named `$type_str` and describing a rejected value by `$mismatch` like the C++ type
24/// traits, which an `Any` holding no such value converts into by `$fallback`.
25macro_rules! typed_expr_view {
26    (
27        $(#[$meta:meta])*
28        $name:ident, $type_str:literal, base = $base:ident($obj:ident), check = $check:path,
29        mismatch = $mismatch:path, fallback = $fallback:path
30    ) => {
31        $(#[$meta])*
32        #[repr(transparent)]
33        #[derive(Clone)]
34        pub struct $name {
35            base: ObjectArc<$obj>,
36        }
37
38        // SAFETY: `base` is the reference; `from_data` requires its caller to have checked it.
39        unsafe impl ObjectRefCore for $name {
40            type ContainerType = $obj;
41
42            #[inline]
43            fn data(this: &Self) -> &ObjectArc<$obj> {
44                &this.base
45            }
46
47            #[inline]
48            fn into_data(this: Self) -> ObjectArc<$obj> {
49                this.base
50            }
51
52            /// The view of `data`.
53            ///
54            /// # Safety
55            ///
56            /// The type of `data` is one the view accepts.
57            #[inline]
58            unsafe fn from_data(data: ObjectArc<$obj>) -> Self {
59                Self { base: data }
60            }
61        }
62
63        // SAFETY: a view is an `Expr` whose `check_any_strict` accepts fewer values; the other
64        // conversions are the `Expr` ones.
65        unsafe impl AnyCompatible for $name {
66            #[inline]
67            unsafe fn is_undefined(field: *const Self) -> bool {
68                unsafe { ObjectArc::is_null(::core::ptr::addr_of!((*field).base)) }
69            }
70
71            fn type_str() -> std::string::String {
72                $type_str.into()
73            }
74
75            fn get_mismatch_type_info(data: &TVMFFIAny) -> std::string::String {
76                $mismatch(data)
77            }
78
79            #[inline(always)]
80            unsafe fn copy_to_any_view(src: &Self, data: &mut TVMFFIAny) {
81                unsafe { <Expr as AnyCompatible>::copy_to_any_view(src.as_expr(), data) }
82            }
83
84            #[inline(always)]
85            unsafe fn move_to_any(src: Self, data: &mut TVMFFIAny) {
86                unsafe { <Expr as AnyCompatible>::move_to_any(src.into(), data) }
87            }
88
89            #[inline(always)]
90            unsafe fn check_any_strict(data: &TVMFFIAny) -> bool {
91                unsafe { $check(data) }
92            }
93
94            unsafe fn copy_from_any_view_after_check(data: &TVMFFIAny) -> Self {
95                let expr =
96                    unsafe { <$base as AnyCompatible>::copy_from_any_view_after_check(data) };
97                Self { base: ObjectRefCore::into_data(expr) }
98            }
99
100            unsafe fn move_from_any_after_check(data: &mut TVMFFIAny) -> Self {
101                let expr = unsafe { <$base as AnyCompatible>::move_from_any_after_check(data) };
102                Self { base: ObjectRefCore::into_data(expr) }
103            }
104
105            unsafe fn try_cast_from_any_view(data: &TVMFFIAny) -> std::result::Result<Self, ()> {
106                unsafe {
107                    if Self::check_any_strict(data) {
108                        Ok(Self::copy_from_any_view_after_check(data))
109                    } else {
110                        $fallback(data).ok_or(())
111                    }
112                }
113            }
114        }
115
116        // SAFETY: the view is `repr(transparent)` over one object pointer, and
117        // `check_any_strict` is its type check.
118        tvm_ffi::unsafe_impl_object_ref_downcast_ref!($name);
119
120        impl AsRef<$name> for $name {
121            #[inline(always)]
122            fn as_ref(&self) -> &Self {
123                self
124            }
125        }
126
127        // As `#[derive(ObjectRef)]` generates: the view converts into itself, owned or borrowed,
128        // so `$name` and `&$name` are an `IntoExpr<$name>`.
129        impl tvm_ffi::object::IntoObjectRef<$name> for $name {
130            type Output = $name;
131
132            #[inline(always)]
133            fn into_object_ref(self) -> $name {
134                self
135            }
136        }
137
138        impl<'a> tvm_ffi::object::IntoObjectRef<$name> for &'a $name {
139            type Output = &'a $name;
140
141            #[inline(always)]
142            fn into_object_ref(self) -> &'a $name {
143                self
144            }
145        }
146
147        tvm_ffi::impl_any_compatible_conversions!($name);
148
149        // The type key and address, as `#[derive(ObjectRef)]` prints a reference, so a
150        // `downcast` result can be unwrapped.
151        impl ::core::fmt::Debug for $name {
152            fn fmt(&self, f: &mut ::core::fmt::Formatter<'_>) -> ::core::fmt::Result {
153                tvm_ffi::object::fmt_object_ref(self, f)
154            }
155        }
156        crate::def_expr_op_over_loads!($name);
157
158        impl From<&$name> for $name {
159            fn from(value: &$name) -> Self {
160                value.clone()
161            }
162        }
163
164        /// Dereference to the object, to read, as the reference it views does.
165        impl Deref for $name {
166            type Target = $obj;
167
168            fn deref(&self) -> &$obj {
169                &self.base
170            }
171        }
172
173        impl $name {
174            /// Borrow this expression as an [`Expr`], without a reference-count increment.
175            #[inline]
176            pub fn as_expr(&self) -> &Expr {
177                // SAFETY: `Expr` is a `repr(C)` struct of one object pointer, as the view is, and
178                // every value of the view is an `Expr`.
179                unsafe { &*(self as *const Self as *const Expr) }
180            }
181
182            /// Return the type of this expression, the [`PrimType`] the view guarantees (C++
183            /// `PrimExpr::ty()`).
184            pub fn ty(&self) -> PrimType {
185                self.base.ty.downcast_ref::<PrimType>().cloned().expect("a view is typed by a `PrimType`")
186            }
187
188            /// Return the dtype of this expression's type.
189            pub fn dtype(&self) -> DLDataType {
190                let ty = self.base.ty.downcast_ref::<PrimTypeObj>();
191                ty.expect("a view is typed by a `PrimType`").dtype
192            }
193        }
194    };
195}
