21//! Compilation targets.
22//!
23//! A [`Target`] has a [`TargetKind`], keys, attributes and an optional host. It is created from a
24//! tag, a kind name or a config, and made the current target with [`Target::scope`].
25//!
26//! # Examples
27//!
28//! ```
29//! use tvm_tirx::target::Target;
30//! use tvm_tirx::tvm_ffi::ObjectRefCore;
31//!
32//! let cpu = Target::new(r#"{"kind": "llvm", "mcpu": "skylake"}"#)?;
33//! assert!(cpu.as_json()?.as_str().contains(r#""keys":["cpu"]"#));
34//!
35//! let scope = cpu.scope()?;
36//! assert!(Target::current(false)?.unwrap().same_as(&cpu));
37//! drop(scope);
38//! assert!(Target::current(true)?.is_none());
39//! # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
40//! ```

108    /// Create the target that a tag (`"nvidia/nvidia-a100"`), a kind name (`"llvm"`) or a JSON
109    /// config describes (`target.Target`).
110    ///
111    /// # Errors
112    ///
113    /// Returns an error if the tag or the kind is not registered, or the JSON config is invalid.
114    pub fn new(target: &str) -> Result<Self> {
115        tvm_ffi::cached_global_func!("target.Target")
116            .call_tuple((String::from(target),))?
117            .try_into()
118    }
119
120    /// Create the target that a config map describes, such as `{"kind": "cuda", "arch": "sm_80"}`
121    /// or a `tag` with attribute overrides (`target.Target`).
122    ///
123    /// The two-target form of `target.Target`, a target with a host, is [`Target::with_host`].
124    ///
125    /// # Errors
126    ///
127    /// Returns an error if the config is invalid, as [`new`](Self::new) does.
128    pub fn from_config(config: &Map<String, Any>) -> Result<Self> {
129        tvm_ffi::cached_global_func!("target.Target").call_tuple((config,))?.try_into()
130    }
131
132    /// Make this the current target until the returned [`TargetScope`] is dropped
133    /// (`target.TargetEnterScope`, `target.TargetExitScope`).

135        tvm_ffi::cached_global_func!("target.TargetEnterScope").call_tuple((self,))?;
136        Ok(TargetScope(self.clone(), crate::ThreadBound::default()))
137    }

146        tvm_ffi::cached_global_func!("target.TargetCurrent").call_tuple((allow_none,))?.try_into()
147    }

184/// A scope in which a target is the current one, returned by [`Target::scope`].
185///
186/// The current targets are a stack of the thread, so a scope stays on the thread that opened it,
187/// and the scopes end in the reverse order they were opened: drop the innermost one first.
188///
189/// ```compile_fail
190/// fn send<T: Send>() {}
191/// send::<tvm_tirx::target::TargetScope>();
192/// ```
193#[must_use = "the target stops being the current one when the scope is dropped"]
194pub struct TargetScope(Target, crate::ThreadBound);
195
196impl Drop for TargetScope {
197    fn drop(&mut self) {
198        // A destructor cannot report the error; the scopes end in the order they were opened.
199        let _ = tvm_ffi::cached_global_func!("target.TargetExitScope").call_tuple((&self.0,));
200    }
201}

261/// Register the tag `name` for the target `config` describes, replacing a previous one only if
262/// `override_`, and return that target (`target.TargetTagAddTag`).
263///
264/// # Errors
265///
266/// Returns an error if `name` is already registered and `override_` is `false`.
267///
268/// # Examples
269///
270/// ```
271/// use tvm_tirx::target::{self, Target};
272/// use tvm_tirx::tvm_ffi::{Any, Map, String};
273///
274/// let config: Map<String, Any> =
275///     [("kind".into(), Any::from(String::from("llvm")))].into_iter().collect();
276/// target::register_tag("my_ext/cpu", &config, false)?;
277/// assert_eq!(Target::new("my_ext/cpu")?.kind()?.name()?, "llvm");
278/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
279/// ```
280pub fn register_tag(name: &str, config: &Map<String, Any>, override_: bool) -> Result<Target> {
281    tvm_ffi::cached_global_func!("target.TargetTagAddTag")
282        .call_tuple((String::from(name), config, override_))?
283        .try_into()
284}

286/// Generate the code of `module` with the code generator `target.build.<kind>`, such as `llvm`,
287/// `c`, `cuda`, `cuda_host` or `nvptx`.
288///
289/// [`crate::codegen::build_module`] picks the code generator of the target kind.
290///
291/// # Errors
292///
293/// Returns an error if the code generator is not built in, or the error it raises.
294pub fn build(kind: &str, module: impl Into<IRModule>, target: impl Into<Target>) -> Result<Module> {
295    // The code generators are looked up by name: which ones exist depends on the build.
296    tvm_ffi::Function::get_global(&format!("target.build.{kind}"))?
297        .call_tuple((&module.into(), &target.into()))?
298        .try_into()
