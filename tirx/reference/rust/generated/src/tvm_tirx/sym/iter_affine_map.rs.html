109/// The result of [`detect_iter_map`]: the detected `indices`, or empty `indices` and the `errors`
110/// that say why detection failed.
111///
112/// `padding_predicate`, over the input iterators, holds for the indices that the returned
113/// iterators cover beyond the bounds of the input ones; it is `None` if detection failed before
114/// computing it.
115///
116/// # Examples
117///
118/// ```
119/// use tvm_tirx::ir::{PrimType, PrimVar, Range, Var, UNKNOWN_LOC};
120/// use tvm_tirx::script::script;
121/// use tvm_tirx::sym::{self, IterMapLevel};
122/// use tvm_tirx::tvm_ffi::Map;
123///
124/// let x = PrimVar::new("x", PrimType::int(32), UNKNOWN_LOC);
125/// let y = PrimVar::new("y", PrimType::int(32), UNKNOWN_LOC);
126/// let four = Range::from_min_extent(0, 4, UNKNOWN_LOC)?;
127/// let iters = Map::from_iter([(x.clone(), four.clone()), (y.clone(), four)]);
128///
129/// let bijective = IterMapLevel::Bijective;
130/// let map = sym::detect_iter_map([&x * 4 + &y], iters, true, bijective, None, true)?;
131/// assert!(map.errors.is_empty());
132///
133/// let o = PrimVar::new("o", PrimType::int(32), UNKNOWN_LOC);
134/// let inverse = sym::inverse_affine_iter_map(map.indices.iter(), [&o])?;
135/// assert_eq!(script(&inverse.get(&Var::from(&x))?.unwrap())?, "o // 4");
136/// assert_eq!(script(&inverse.get(&Var::from(&y))?.unwrap())?, "o % 4");
137/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
138/// ```
139// tvm-ffi-stubgen(begin): object/sym.IterMapResult
140#[allow(missing_docs)]
141#[repr(C)]

347/// How strictly an iterator map is checked.
348#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
349#[repr(i32)]
350pub enum IterMapLevel {
351    /// The map must be bijective.
352    Bijective = 0,
353    /// The map must be surjective.
354    #[default]
355    Surjective = 1,
356    /// The map is not checked.
357    NoCheck = 3,
358}

360/// Detect `indices` as iterator maps of the iterators `input_iters` under `predicate`
361/// (`sym.DetectIterMap`).
362///
363/// When they are not, the `indices` of the result are empty and its `errors` say why.
364pub fn detect_iter_map(
365    indices: impl IntoIterator<Item = impl IntoExpr<PrimExpr>>,
366    input_iters: impl Into<Map<PrimVar, Range>>,
367    predicate: impl IntoExpr<PrimExpr>,
368    check_level: IterMapLevel,
369    analyzer: Option<&Analyzer>,
370    simplify_trivial_iterators: bool,
371) -> Result<IterMapResult> {
372    let indices =
373        indices.into_iter().map(IntoExpr::into_owned_expr).collect::<Result<Array<PrimExpr>>>()?;
374    let predicate = predicate.into_expr()?;
375    tvm_ffi::cached_global_func!("sym.DetectIterMap")
376        .call_packed(&[
377            AnyView::from(&indices),
378            AnyView::from(&input_iters.into()),
379            AnyView::from(predicate.borrow()),
380            AnyView::from(&(check_level as i32)),
381            analyzer.map_or_else(AnyView::new, AnyView::from),
382            AnyView::from(&simplify_trivial_iterators),
383        ])?
384        .try_into()
385}

431/// Divide each binding into an outer map of the iterators not in `sub_iters` and an inner map of
432/// those in it (`sym.SubspaceDivide`).
433///
434/// The result is one `[outer, inner]` pair of marks per binding, then a pair whose `extent`s are
435/// the outer and inner predicates, or empty when the bindings cannot be divided.
436pub fn subspace_divide(
437    bindings: impl IntoIterator<Item = impl IntoExpr<PrimExpr>>,
438    input_iters: impl Into<Map<PrimVar, Range>>,
439    sub_iters: impl IntoIterator<Item = impl IntoArg<PrimVar>>,
440    predicate: impl IntoExpr<PrimExpr>,
441    check_level: IterMapLevel,
442    analyzer: Option<&Analyzer>,
443    simplify_trivial_iterators: bool,
444) -> Result<Array<Array<IterMark>>> {
445    let bindings =
446        bindings.into_iter().map(IntoExpr::into_owned_expr).collect::<Result<Array<PrimExpr>>>()?;
447    let sub_iters = sub_iters.into_iter().map(IntoArg::into_owned).collect::<Array<PrimVar>>();
448    let predicate = predicate.into_expr()?;
449    tvm_ffi::cached_global_func!("sym.SubspaceDivide")
450        .call_packed(&[
451            AnyView::from(&bindings),
452            AnyView::from(&input_iters.into()),
453            AnyView::from(&sub_iters),
454            AnyView::from(predicate.borrow()),
455            AnyView::from(&(check_level as i32)),
456            analyzer.map_or_else(AnyView::new, AnyView::from),
457            AnyView::from(&simplify_trivial_iterators),
458        ])?
459        .try_into()
460}

462/// Return the inputs of the bijective iterator map `iter_map` as expressions of its `outputs`
463/// (`sym.InverseAffineIterMap`).
464pub fn inverse_affine_iter_map(
465    iter_map: impl IntoIterator<Item = impl IntoArg<IterSumExpr>>,
466    outputs: impl IntoIterator<Item = impl IntoExpr<PrimExpr>>,
467) -> Result<Map<Var, PrimExpr>> {
468    let iter_map = iter_map.into_iter().map(IntoArg::into_owned).collect::<Array<IterSumExpr>>();
469    let outputs =
470        outputs.into_iter().map(IntoExpr::into_owned_expr).collect::<Result<Array<PrimExpr>>>()?;
471    tvm_ffi::cached_global_func!("sym.InverseAffineIterMap")
472        .call_tuple((&iter_map, &outputs))?
473        .try_into()
