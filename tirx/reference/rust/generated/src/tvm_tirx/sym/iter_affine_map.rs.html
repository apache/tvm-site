21//! Iterator maps: index expressions decomposed into splits of iterators.
22//!
23//! The objects `python/tvm_tirx/sym/iter_affine_map.py` defines. An [`IterMark`] is an iterator
24//! `source` with its `extent`; an [`IterSplitExpr`] is `(source // lower_factor) % extent * scale`;
25//! an [`IterSumExpr`] is a sum of splits plus `base_`. An [`IterMapResult`] holds the detected
26//! `indices`, the `errors` and the `padding_predicate`. The functions of `iter_affine_map.py`,
27//! which detect, simplify, divide and invert iterator maps, are here too.

322/// How strictly an iterator map is checked: `sym::IterMapLevel`.
323#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
324#[repr(i32)]
325pub enum IterMapLevel {
326    /// The map must be bijective.
327    Bijective = 0,
328    /// The map must be surjective.
329    #[default]
330    Surjective = 1,
331    /// The map is not checked.
332    NoCheck = 3,
333}

335/// Detect `indices` as iterator maps of the iterators `input_iters` under `predicate`; the
336/// `indices` of the result are empty when they are not (`sym.DetectIterMap`).
337///
338/// With `simplify_trivial_iterators`, the iterators of extent 1 become constants.
339pub fn detect_iter_map(
340    indices: impl IntoIterator<Item = impl IntoExpr>,
341    input_iters: &Map<Var, Range>,
342    predicate: impl IntoExpr,
343    check_level: IterMapLevel,
344    analyzer: Option<&Analyzer>,
345    simplify_trivial_iterators: bool,
346) -> Result<IterMapResult> {
347    let indices = indices.into_iter().map(IntoExpr::into_expr).collect::<Result<Array<Expr>>>()?;
348    let (predicate, analyzer) = (predicate.into_expr()?, analyzer.cloned());
349    tvm_ffi::cached_global_func!("sym.DetectIterMap")
350        .call_packed(&[
351            AnyView::from(&indices),
352            AnyView::from(input_iters),
353            AnyView::from(&predicate),
354            AnyView::from(&(check_level as i32)),
355            AnyView::from(&analyzer),
356            AnyView::from(&simplify_trivial_iterators),
357        ])?
358        .try_into()
359}

362/// `base` (`sym.NormalizeToIterSum`).
363pub fn normalize_to_iter_sum(

366    analyzer: Option<&Analyzer>,

371}

404/// Divide each binding into an outer map of the iterators not in `sub_iters` and an inner map of
405/// those in it: one `[outer, inner]` pair of marks per binding, then a pair whose `extent`s are
406/// the outer and inner predicates, or empty when the bindings cannot be divided
407/// (`sym.SubspaceDivide`).
408pub fn subspace_divide(
409    bindings: impl IntoIterator<Item = impl IntoExpr>,
410    input_iters: &Map<Var, Range>,
411    sub_iters: impl IntoIterator<Item = impl Into<Var>>,
412    predicate: impl IntoExpr,
413    check_level: IterMapLevel,
414    analyzer: Option<&Analyzer>,
415    simplify_trivial_iterators: bool,
416) -> Result<Array<Array<IterMark>>> {
417    let bindings =
418        bindings.into_iter().map(IntoExpr::into_expr).collect::<Result<Array<Expr>>>()?;
419    let sub_iters = sub_iters.into_iter().map(Into::into).collect::<Array<Var>>();
420    let (predicate, analyzer) = (predicate.into_expr()?, analyzer.cloned());
421    tvm_ffi::cached_global_func!("sym.SubspaceDivide")
422        .call_packed(&[
423            AnyView::from(&bindings),
424            AnyView::from(input_iters),
425            AnyView::from(&sub_iters),
426            AnyView::from(&predicate),
427            AnyView::from(&(check_level as i32)),
428            AnyView::from(&analyzer),
429            AnyView::from(&simplify_trivial_iterators),
430        ])?
431        .try_into()
432}

434/// The inputs of the bijective iterator map `iter_map` as expressions of its `outputs`
435/// (`sym.InverseAffineIterMap`).
436pub fn inverse_affine_iter_map(
437    iter_map: impl IntoIterator<Item = impl Into<IterSumExpr>>,
438    outputs: impl IntoIterator<Item = impl IntoExpr>,
439) -> Result<Map<Var, Expr>> {
440    let iter_map = iter_map.into_iter().map(Into::into).collect::<Array<IterSumExpr>>();
441    let outputs = outputs.into_iter().map(IntoExpr::into_expr).collect::<Result<Array<Expr>>>()?;
442    tvm_ffi::cached_global_func!("sym.InverseAffineIterMap")
443        .call_tuple((&iter_map, &outputs))?
444        .try_into()
