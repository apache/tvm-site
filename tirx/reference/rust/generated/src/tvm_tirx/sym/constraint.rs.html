23//! Constraints: the facts in scope at a point of a traversal.
24//!
25//! A [`ConstraintContext`] holds [`BindingConstraint`]s, [`RangeConstraint`]s and
26//! [`PredicateConstraint`]s, ancestors' first. Its handles are neither `Send` nor `Sync`; do not
27//! share a context across threads, also not through an erased `ObjectRef`.
28//!
29//! # Examples
30//!
31//! ```
32//! use tvm_tirx::ir::prim::op::{ge, lt};
33//! use tvm_tirx::ir::{PrimType, PrimVar, Var, UNKNOWN_LOC};
34//! use tvm_tirx::sym::{Analyzer, ConstraintContext, PredicateConstraint, ProofStrength};
35//!
36//! let x = PrimVar::new("x", PrimType::int(32), UNKNOWN_LOC);
37//! let outer = ConstraintContext::new();
38//! outer.append(&PredicateConstraint::new(ge(&x, 0)?, false).into())?;
39//! let inner = outer.snapshot()?;
40//! inner.append(&PredicateConstraint::new(lt(&x, 16)?, false).into())?;
41//!
42//! let analyzer = Analyzer::new();
43//! analyzer.sync_constraint_context(&inner)?;
44//! assert!(analyzer.can_prove(lt(&x, 16)?, ProofStrength::Default)?);
45//! analyzer.sync_constraint_context(&outer)?;
46//! assert!(!analyzer.can_prove(lt(&x, 16)?, ProofStrength::Default)?);
47//! # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
48//! ```

301/// The object of a [`ConstraintContext`], with the layout of the native one and a stable identity.
302///
303/// Only the local array changes after construction. Its handle is replaced by native append, so
304/// shared context inspection must use interior mutability. Parent links and their captured local
305/// prefix are immutable and acyclic.
306#[repr(C)]
307#[derive(tvm_ffi::derive::Object)]
308#[type_key = "sym.ConstraintContext"]
309#[type_final]
310pub struct ConstraintContextObj {
311    base: Object,
312    parent: Option<ObjectArc<ConstraintContextObj>>,
313    parent_prefix_size: i64,
314    constraints: UnsafeCell<Array<Constraint>>,
315}
316
317/// A context of constraints: the premises in effect at a point of a traversal, ancestors' first.
318///
319/// Clones share the context's identity and later appends; use
320/// [`snapshot`](ConstraintContextObj::snapshot) to capture the current prefix.
321#[repr(C)]
322#[derive(tvm_ffi::derive::ObjectRef, Clone)]
323pub struct ConstraintContext {
324    base: ObjectArc<ConstraintContextObj>,
325}
326
327impl Deref for ConstraintContext {
328    type Target = ConstraintContextObj;
329    fn deref(&self) -> &Self::Target {
330        &self.base
331    }
332}
333
334impl ConstraintContext {
335    /// Create an empty root context, with the complete native-compatible layout (C++
336    /// `ConstraintContext()`).
337    pub fn new() -> Self {
338        Self::allocate(None, 0)
339    }
340
341    /// Create an empty child that captures the first `parent_prefix_size` local entries of
342    /// `parent`.
343    ///
344    /// Earlier ancestor prefixes are inherited unchanged; later appends to the parent are excluded.
345    ///
346    /// # Errors
347    ///
348    /// Returns an error if `parent_prefix_size` is negative or larger than the parent's local size.
349    pub fn with_parent_prefix_size(parent: Self, parent_prefix_size: i64) -> Result<Self> {
350        if parent_prefix_size < 0 || parent_prefix_size > parent.local_size() {
351            return Err(Error::new(
352                VALUE_ERROR,
353                "Constraint context parent local prefix is out of bounds",
354                "",
355            ));
356        }
357        Ok(Self::allocate(Some(parent.base), parent_prefix_size))
358    }
359
360    fn allocate(parent: Option<ObjectArc<ConstraintContextObj>>, prefix: i64) -> Self {
361        Self {
362            base: ObjectArc::new(ConstraintContextObj {
363                base: Object::new(),
364                parent,
365                parent_prefix_size: prefix,
366                // Start with an empty Rust array. Native Append owns all growth;
367                // never install a nonempty Rust allocation in this mutable field.
368                constraints: UnsafeCell::new(Array::default()),
369            }),
370        }
371    }
372}
373
374impl ConstraintContextObj {
375    /// Append `constraint` to the local entries (`sym.ConstraintContextAppend`).
376    ///
377    /// Do not append to the active context of a running native visitor.
378    pub fn append(&self, constraint: &Constraint) -> Result<()> {
379        tvm_ffi::cached_global_func!("sym.ConstraintContextAppend")
380            .call_packed(&[AnyView::from(self), AnyView::from(constraint)])?;
381        Ok(())
382    }
383
384    /// Return the number of local entries, excluding captured ancestors.
385    pub fn local_size(&self) -> i64 {
386        // No callback or mutation occurs during this short borrow.
387        unsafe { (&*self.constraints.get()).len() as i64 }
388    }
389
390    /// Capture the visible prefix in an independently extensible child.
391    pub fn snapshot(&self) -> Result<ConstraintContext> {
392        let parent = ConstraintContext::try_from(AnyView::from(self))
393            .expect("context has its registered native type");
394        ConstraintContext::with_parent_prefix_size(parent, self.local_size())
395    }
396
397    fn local_array(&self) -> Array<Constraint> {
398        // Hold an owning clone before any callback; native append then uses COW.
399        unsafe { (&*self.constraints.get()).clone() }
400    }
401
402    /// Call `callback` on each visible constraint, in inherited and then local order.
403    ///
404    /// The callback may append without entering this iteration; each borrowed constraint is valid
405    /// only during its callback.
406    pub fn for_each(&self, mut callback: impl FnMut(&Constraint)) {
407        let retained = ConstraintContext::try_from(AnyView::from(self))
408            .expect("context has its registered native type");
409        retained.for_each_prefix(retained.local_size(), &mut callback);
410    }
411
412    fn for_each_prefix(&self, prefix: i64, callback: &mut impl FnMut(&Constraint)) {
413        let local = self.local_array();
414        if let Some(parent) = &self.parent {
415            parent.for_each_prefix(self.parent_prefix_size, callback);
416        }
417        for constraint in local.iter().take(prefix as usize) {
418            callback(constraint);
419        }
420    }
421
422    /// Return the visible constraints in inherited and then local order, as an independent array.
423    ///
424    /// Later appends do not change it; constraint nodes and expressions remain shared.
425    pub fn to_array(&self) -> Array<Constraint> {
426        if self.parent.is_none() {
427            return self.local_array();
428        }
429        let mut facts = Vec::new();
430        self.for_each(|fact| facts.push(fact.clone()));
431        Array::from_vec(facts)
432    }
433}
