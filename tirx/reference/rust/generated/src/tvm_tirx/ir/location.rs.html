21//! Source locations of IR nodes, for diagnostics.
22//!
23//! The objects `python/tvm_tirx/ir/location.py` defines. A [`Location`] is an [`UnknownLoc`], a
24//! [`SourceLoc`] (a span in a [`SourceName`]), or a [`CallSiteLoc`] (a callee location seen from a
25//! caller). [`Location::unknown`] and [`SourceName::get`] wrap the native constructors.

165impl UnknownLoc {
166    /// The shared unknown location (`ir.UnknownLoc`), which IR constructors default to, fetched
167    /// once per thread.
168    pub fn new() -> Self {
169        thread_local! {
170            static UNKNOWN_LOC: std::cell::OnceCell<UnknownLoc> = const { std::cell::OnceCell::new() };
171        }
172        support::shared(&UNKNOWN_LOC, || {
173            tvm_ffi::cached_global_func!("ir.UnknownLoc").call_tuple(())?.try_into()
174        })
175    }
176}

178/// The unknown location as the `loc` argument of a constructor, which takes any
179/// `impl Into<Location>`: `Var::new("x", ty, UNKNOWN_LOC)`. [`Location::unknown`] is the same
180/// location as a value.
181#[derive(Clone, Copy, Debug)]
182pub struct Unknown;
183
184/// The unknown location, as Python's `UNKNOWN_LOC`.
185pub const UNKNOWN_LOC: Unknown = Unknown;
186
187impl From<Unknown> for Location {
188    fn from(_: Unknown) -> Self {
189        Location::unknown()
190    }
191}
