54/// A transformation of an [`IRModule`], with a name and an optimization level ([`PassInfo`]).
55///
56/// [`run`](Self::run) applies it under the current [`PassContext`].
57///
58/// # Examples
59///
60/// ```
61/// use tvm_tirx::ir::transform::{ModulePass, Pass, PassContext, PassInfo, PassInstrument};
62/// use tvm_tirx::ir::transform::Sequential;
63/// use tvm_tirx::ir::{DictAttrs, IRModule, IntImm};
64/// use tvm_tirx::tvm_ffi::Map;
65///
66/// let mark = |name: &'static str| {
67///     let pass = move |module: IRModule, _ctx| module.with_attr(name, IntImm::bool(true));
68///     ModulePass::new(pass, PassInfo::new(0, name))
69/// };
70///
71/// let info = PassInfo::new(0, "pipeline");
72/// let pipeline = Pass::from(Sequential::new([mark("a"), mark("b")], info));
73/// let ctx = PassContext::new(2, [""; 0], ["b"], Vec::<PassInstrument>::new(), None)?;
74/// let _scope = ctx.enter()?;
75/// let empty = IRModule::new(Map::new(), DictAttrs::new(Map::new()), Map::new())?;
76/// let module = pipeline.run(empty)?;
77/// assert!(module.get_attr("a")?.is_some() && module.get_attr("b")?.is_none());
78/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
79/// ```
80// tvm-ffi-stubgen(begin): object/transform.Pass
81#[allow(missing_docs)]
82#[repr(C)]
83#[derive(tvm_ffi::derive::ObjectRef, Clone)]

85    data_: ObjectArc<PassObj>,
86}

100impl Pass {
101    /// Return the name and optimization level of the pass (`transform.Info`).
102    pub fn info(&self) -> Result<PassInfo> {
103        tvm_ffi::cached_global_func!("transform.Info").call_tuple((self,))?.try_into()
104    }
105
106    /// Run the pass on `module`, which it consumes, under the current pass context
107    /// (`transform.RunPass`).
108    ///
109    /// An instrument's `should_run` hook may skip the pass, unless the context requires it.
110    pub fn run(&self, module: impl IntoObjectRef<IRModule>) -> Result<IRModule> {
111        let module = tvm_ffi::rvalue_ref::RValueRef::new(module.into_owned_object_ref());
112        tvm_ffi::cached_global_func!("transform.RunPass")
113            .call_packed(&[tvm_ffi::AnyView::from(self), tvm_ffi::AnyView::from(&module)])?
114            .try_into()
115    }
116}

208impl ModulePass {
209    /// Create a pass that applies `pass_func` to the module it runs on, which `pass_func` receives
210    /// by value with the pass context (`transform.MakeModulePass`).
211    ///
212    /// The pass may run on any thread, so `pass_func` is `Send + Sync`.
213    pub fn new(
214        pass_func: impl Fn(IRModule, PassContext) -> Result<IRModule> + Send + Sync + 'static,
215        pass_info: impl IntoObjectRef<PassInfo>,
216    ) -> Self {
217        let pass_func = tvm_ffi::Function::from_typed(
218            move |module: tvm_ffi::rvalue_ref::RValueRef<IRModule>, ctx: PassContext| {
219                pass_func(module.into_inner(), ctx)
220            },
221        );
222        support::infallible(
223            tvm_ffi::cached_global_func!("transform.MakeModulePass")
224                .call_tuple((&pass_func, pass_info.into_object_ref().borrow()))
225                .and_then(|any| any.try_into()),
226        )
227    }

274impl Sequential {
275    /// Create a pass named by `pass_info` that runs `passes` in order; a pipeline with only a name
276    /// takes `PassInfo::new(0, name)`.
277    ///
278    /// It skips the passes the current pass context disables, and those whose level exceeds the
279    /// context's `opt_level` unless the context requires them.
280    pub fn new(
281        passes: impl IntoIterator<Item = impl IntoObjectRef<Pass>>,
282        pass_info: impl IntoObjectRef<PassInfo>,
283    ) -> Self {
284        let passes =
285            passes.into_iter().map(IntoObjectRef::into_owned_object_ref).collect::<Array<Pass>>();
286        let pass_info = pass_info.into_object_ref();
287        let pass_info: &PassInfo = pass_info.borrow();
288        support::infallible(
289            tvm_ffi::cached_global_func!("transform.Sequential")
290                .call_tuple((&passes, pass_info.opt_level, &pass_info.name))
291                .and_then(|any| any.try_into()),
292        )
293    }

367impl PassContext {
368    /// Create a pass context with `config` checked and converted to the registered option types;
369    /// `None` is the empty configuration (`transform.PassContext`).
370    ///
371    /// A pass in `disabled_pass`, or above `opt_level` and not in `required_pass`, is skipped.
372    ///
373    /// # Errors
374    ///
375    /// Returns an `AttributeError` if `config` has an option that is not registered or a `None`
376    /// value, or the error of converting a value to its option type.
377    pub fn new(
378        opt_level: i32,
379        required_pass: impl IntoIterator<Item = impl AsRef<str>>,
380        disabled_pass: impl IntoIterator<Item = impl AsRef<str>>,
381        instruments: impl IntoIterator<Item = impl IntoObjectRef<PassInstrument>>,
382        config: impl Into<Option<Map<String, Any>>>,
383    ) -> Result<Self> {
384        let required_pass =
385            required_pass.into_iter().map(|name| String::from(name.as_ref())).collect::<Array<_>>();
386        let disabled_pass =
387            disabled_pass.into_iter().map(|name| String::from(name.as_ref())).collect::<Array<_>>();
388        let instruments = instruments
389            .into_iter()
390            .map(IntoObjectRef::into_owned_object_ref)
391            .collect::<Array<PassInstrument>>();
392        tvm_ffi::cached_global_func!("transform.PassContext")
393            .call_tuple((opt_level, &required_pass, &disabled_pass, &instruments, config.into()))?
394            .try_into()
395    }
396
397    /// Return the innermost pass context in effect on this thread, or the default one when there is
398    /// none (`transform.GetCurrentPassContext`).
399    pub fn current() -> Result<Self> {
400        tvm_ffi::cached_global_func!("transform.GetCurrentPassContext").call_tuple(())?.try_into()
401    }
402
403    /// Make this the current pass context until the returned [`PassContextScope`] is dropped
404    /// (`transform.EnterPassContext`, `transform.ExitPassContext`).
405    ///
406    /// # Errors
407    ///
408    /// Returns the error of an instrument's `enter_pass_ctx` hook; the instruments entered before
409    /// it are exited, and the context's instruments are cleared.
410    pub fn enter(&self) -> Result<PassContextScope> {
411        tvm_ffi::cached_global_func!("transform.EnterPassContext").call_tuple((self,))?;
412        Ok(PassContextScope(self.clone(), crate::ThreadBound::default()))
413    }
414
415    /// Replace the instruments with `instruments`, exiting the old ones and entering the new ones
416    /// (`transform.OverrideInstruments`).
417    ///
418    /// # Errors
419    ///
420    /// Returns the error of an old instrument's `exit_pass_ctx` hook or a new one's
421    /// `enter_pass_ctx` hook, after which the context has no instruments.
422    pub fn override_instruments(
423        &self,
424        instruments: impl IntoIterator<Item = impl IntoObjectRef<PassInstrument>>,
425    ) -> Result<()> {
426        let instruments = instruments
427            .into_iter()
428            .map(IntoObjectRef::into_owned_object_ref)
429            .collect::<Array<PassInstrument>>();
430        tvm_ffi::cached_global_func!("transform.OverrideInstruments")
431            .call_tuple((self, &instruments))?;
432        Ok(())
433    }
434
435    /// Return the registered configuration options, each with its metadata such as its `type`
436    /// (`transform.ListConfigs`).
437    pub fn list_configs() -> Result<Map<String, Map<String, String>>> {
438        tvm_ffi::cached_global_func!("transform.ListConfigs").call_tuple(())?.try_into()
439    }
440}

442/// A scope in which a pass context is the current one, returned by [`PassContext::enter`].
443///
444/// The entered contexts are a stack of the thread: a scope stays on its thread, and the innermost
445/// scope is dropped first.
446///
447/// # Examples
448///
449/// ```
450/// use tvm_tirx::ir::transform::{PassContext, PassInstrument};
451///
452/// let outer = PassContext::new(3, [""; 0], [""; 0], Vec::<PassInstrument>::new(), None)?;
453/// let inner = PassContext::new(1, [""; 0], [""; 0], Vec::<PassInstrument>::new(), None)?;
454/// let outer_scope = outer.enter()?;
455/// let inner_scope = inner.enter()?;
456/// assert_eq!(PassContext::current()?.opt_level, 1);
457///
458/// drop(inner_scope);
459/// assert_eq!(PassContext::current()?.opt_level, 3);
460/// drop(outer_scope);
461/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
462/// ```
463///
464/// A scope cannot move to another thread:
465///
466/// ```compile_fail
467/// fn send<T: Send>() {}
468/// send::<tvm_tirx::ir::transform::PassContextScope>();
469/// ```
470#[must_use = "the pass context is exited when the scope is dropped"]
471pub struct PassContextScope(PassContext, crate::ThreadBound);
472
473impl Drop for PassContextScope {
474    fn drop(&mut self) {
475        // A destructor cannot report the error, and exiting the innermost context does not fail.
476        let _ = tvm_ffi::cached_global_func!("transform.ExitPassContext").call_tuple((&self.0,));
477    }
478}
