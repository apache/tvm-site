20//! Pass instrumentation.
21//!
22//! A [`BasePassInstrument`] is a [`PassInstrument`] whose hooks are callbacks, such as Rust
23//! closures ([`BasePassInstrument::new`]).
24//!
25//! # Examples
26//!
27//! ```
28//! use std::sync::{Arc, Mutex};
29//! use tvm_tirx::ir::transform::{BasePassInstrument, ModulePass, Pass, PassContext, PassInfo};
30//! use tvm_tirx::ir::transform::PassInstrumentHooks;
31//! use tvm_tirx::ir::{DictAttrs, IRModule};
32//! use tvm_tirx::tvm_ffi::Map;
33//!
34//! let log = Arc::new(Mutex::new(Vec::new()));
35//! let sink = log.clone();
36//! let hooks = PassInstrumentHooks {
37//!     run_before_pass: Some(Box::new(move |_module, info| {
38//!         Ok(sink.lock().unwrap().push(info.name.as_str().to_owned()))
39//!     })),
40//!     ..Default::default()
41//! };
42//! let tracer = BasePassInstrument::new("tracer", hooks);
43//!
44//! let _scope = PassContext::new(2, [""; 0], [""; 0], [tracer], None)?.enter()?;
45//! let noop = Pass::from(ModulePass::new(|module, _ctx| Ok(module), PassInfo::new(0, "noop")));
46//! noop.run(IRModule::new(&Map::new(), DictAttrs::new(Map::new()), &Map::new())?)?;
47//! assert_eq!(*log.lock().unwrap(), ["noop"]);
48//! # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
49//! ```

105/// A hook of a [`BasePassInstrument`], called when a pass context is entered or exited.
106///
107/// It may be called from any thread, so it is `Send + Sync`; an error fails the entry or exit.
108pub type PassContextHook = Box<dyn Fn() -> Result<()> + Send + Sync>;
109
110/// A hook of a [`BasePassInstrument`], called with a module and the information of the pass to run
111/// on it; `Send + Sync` like a [`PassContextHook`].
112///
113/// An error fails the pass run.
114pub type PassHook<R> = Box<dyn Fn(IRModule, PassInfo) -> Result<R> + Send + Sync>;
115
116/// The hooks of a [`BasePassInstrument`].
117///
118/// A hook left `None` does nothing, so `..Default::default()` fills in the ones not needed.
119#[derive(Default)]
120pub struct PassInstrumentHooks {
121    /// Called when a pass context with the instrument is entered.
122    pub enter_pass_ctx: Option<PassContextHook>,
123    /// Called when a pass context with the instrument is exited.
124    pub exit_pass_ctx: Option<PassContextHook>,
125    /// Called before a pass runs on the module, to decide whether to run it; a pass the context
126    /// requires runs without asking.
127    pub should_run: Option<PassHook<bool>>,
128    /// Called before a pass runs on the module.
129    pub run_before_pass: Option<PassHook<()>>,
130    /// Called after a pass ran, with the module it returned.
131    pub run_after_pass: Option<PassHook<()>>,
132}
133
134impl BasePassInstrument {
135    /// Create the instrument `name` that calls `hooks` (`transform.PassInstrument`).
136    pub fn new(name: &str, hooks: PassInstrumentHooks) -> Self {
137        let PassInstrumentHooks {
138            enter_pass_ctx,
139            exit_pass_ctx,
140            should_run,
141            run_before_pass,
142            run_after_pass,
143        } = hooks;
144        let enter_pass_ctx = enter_pass_ctx.map(Function::from_typed);
145        let exit_pass_ctx = exit_pass_ctx.map(Function::from_typed);
146        let should_run = should_run.map(Function::from_typed);
147        let run_before_pass = run_before_pass.map(Function::from_typed);
148        let run_after_pass = run_after_pass.map(Function::from_typed);
149        support::infallible(
150            tvm_ffi::cached_global_func!("transform.PassInstrument")
151                .call_packed(&[
152                    AnyView::from(&String::from(name)),
153                    AnyView::from(&enter_pass_ctx),
154                    AnyView::from(&exit_pass_ctx),
155                    AnyView::from(&should_run),
156                    AnyView::from(&run_before_pass),
157                    AnyView::from(&run_after_pass),
158                ])
159                .and_then(|any| any.try_into()),
160        )
161    }
162}

164/// Create an instrument that times the passes, for [`render_pass_timing`]: Python's
165/// `PassTimingInstrument()` (`transform.instrument.MakePassTimingInstrument`).
166///
167pub fn pass_timing_instrument() -> Result<BasePassInstrument> {
168    tvm_ffi::cached_global_func!("transform.instrument.MakePassTimingInstrument")
169        .call_tuple(())?
170        .try_into()
171}

173/// Render the pass times that a [`pass_timing_instrument`] recorded, as Python's
174/// `PassTimingInstrument.render()` does (`transform.instrument.RenderTimePassProfiles`).
175///
176/// The result has a line per pass timed on this thread, and is empty when none was.
177///
178/// # Errors
179///
180/// Returns an `InternalError` if a pass is still running, as when a pass calls it.
181pub fn render_pass_timing() -> Result<String> {
182    tvm_ffi::cached_global_func!("transform.instrument.RenderTimePassProfiles")
183        .call_tuple(())?
184        .try_into()
