21//! Statements: bindings, assertions, loops, conditionals, tensor stores and sequences.
22//!
23//! The objects `python/tvm_tirx/ir/stmt.py` defines. [`Stmt`] is the base of every statement; a
24//! body given as one statement is wrapped in a [`SeqStmt`].
25//!
26//! # Examples
27//!
28//! ```
29//! use tvm_tirx::ir::prim::op::PrimOps;
30//! use tvm_tirx::ir::{
31//!     AssertStmt, Bind, For, ForKind, PrimType, PrimVar, SeqStmt, Stmt, StringImm, Var,
32//! };
33//! use tvm_tirx::ir::UNKNOWN_LOC;
34//! use tvm_tirx::script::script;
35//!
36//! let i = PrimVar::new("i", PrimType::int(32), UNKNOWN_LOC);
37//! let x = PrimVar::new("x", PrimType::int(32), UNKNOWN_LOC);
38//! let bind = Bind::new(&x, &i + 1, UNKNOWN_LOC)?;
39//! let kind = StringImm::new("ValueError", UNKNOWN_LOC);
40//! let check = AssertStmt::new(x.less(17), kind, [StringImm::new("x", UNKNOWN_LOC)], UNKNOWN_LOC)?;
41//!
42//! let body = SeqStmt::new([Stmt::from(bind), check.into()], UNKNOWN_LOC);
43//! let loop_ = For::new(&i, 0, 16, 1, ForKind::Default, body, None, UNKNOWN_LOC)?;
44//! let expected = "\
45//! for i in range(16):
46//!     x: T.let[T.int32] = i + 1
47//!     assert x < 17, (\"ValueError\", [\"x\"])";
48//! assert_eq!(script(&loop_)?, expected);
49//! # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
50//! ```

254    /// enclosing scope (C++ `Bind(var, value, loc)`).

345impl AssertStmt {
346    /// Create the statement asserting the boolean `condition`, raising an `error_kind` error with
347    /// the message `message_parts` when it fails (C++ `AssertStmt`).
348    ///
349    /// # Errors
350    ///
351    /// Returns an `InternalError` if `condition` is not boolean.
352    pub fn new(
353        condition: impl IntoExpr<PrimExpr>,
354        error_kind: impl Into<StringImm>,
355        message_parts: impl IntoIterator<Item = impl Into<StringImm>>,
356        loc: impl Into<Location>,
357    ) -> Result<Self> {
358        let (condition, error_kind) = (condition.into_owned_expr()?, error_kind.into());
359        let message_parts = message_parts.into_iter().map(Into::into).collect::<Array<StringImm>>();
360        let condition_ty = condition.ty();
361        if condition_ty.dtype.code != support::BOOL {
362            let message = format!(
363                "AssertStmt should have boolean condition, but received {} with dtype {}",
364                support::repr(&condition),
365                support::repr(&condition_ty)
366            );
367            let check = "condition_ty.MatchesCode(DLDataTypeCode::kDLBool)";
368            return Err(support::check_failed(tvm_ffi::INTERNAL_ERROR, check, &message));
369        }
370        Ok(Self::from_fields(loc.into(), condition, error_kind, message_parts))
371    }

461/// How a [`For`] loop runs its iterations, as C++ `ir::ForKind`.
462#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
463#[repr(i32)]
464pub enum ForKind {
465    /// Sequential iterations.
466    #[default]
467    Default = 0,
468    /// Iterations may run in parallel.
469    Parallel = 1,
470    /// The body is vectorized.
471    Vectorized = 2,
472    /// The loop is unrolled.
473    Unrolled = 3,
474}
475
476impl For {
477    /// Create the loop of the scalar integer `var` from `begin` to `end` by `step`, running `body`
478    /// with the `kind` semantics (C++ `For(var, begin, end, step, kind, body, annotations, loc)`).
479    ///
480    /// A constant bound or step of another integer type is converted to the type of `var`.
481    ///
482    /// # Errors
483    ///
484    /// Returns an `InternalError` if `var`, a bound or the step is not a scalar integer, or a bound
485    /// or the step does not fit or match the type of `var`.
486    #[allow(clippy::too_many_arguments)]
487    pub fn new(
488        var: impl Into<PrimVar>,
489        begin: impl IntoExpr<PrimExpr>,
490        end: impl IntoExpr<PrimExpr>,
491        step: impl IntoExpr<PrimExpr>,
492        kind: ForKind,
493        body: impl Into<Stmt>,
494        annotations: Option<&Map<String, Any>>,
495        loc: impl Into<Location>,
496    ) -> Result<Self> {
497        let (var, begin, end, step) =
498            (var.into(), begin.into_owned_expr()?, end.into_owned_expr()?, step.into_owned_expr()?);
499        let var_expr: PrimExpr = var.clone().into();
500        let var_ty = var.ty();
501        let body = into_body(body.into())?;
502        let annotations = annotations.cloned().unwrap_or_else(Map::new);
503        for (expr, field) in [(&var_expr, "var"), (&begin, "begin"), (&end, "end"), (&step, "step")]
504        {
505            let ty = expr.ty();
506            let dtype = ty.dtype;
507            if !support::is_scalar(dtype) || !matches!(dtype.code, support::INT | support::UINT) {
508                let message = format!(
509                    "TIR For nodes require a scalar integer as the {field}, but received {} with \
510                     dtype {}",
511                    support::repr(expr),
512                    support::repr(&ty)
513                );
514                let condition = "dtype.IsScalar() && (dtype.MatchesCode(DLDataTypeCode::kDLUInt, \
515                                 DLDataTypeCode::kDLInt))";
516                return Err(support::check_failed(tvm_ffi::INTERNAL_ERROR, condition, &message));
517            }
518        }
519        // A constant of another integer type takes the type of `var` if its value fits.
520        let promote = |expr: PrimExpr| -> Result<PrimExpr> {
521            let ty = expr.ty();
522            if ty.dtype == var_ty.dtype {
523                return Ok(expr);
524            }
525            if let Some(imm) = expr.downcast_ref::<IntImmObj>() {
526                if !int_imm_value_fits(&imm.value, var_ty.dtype)? {
527                    let message = format!(
528                        "Literal value {} is not representable in the loop variable's dtype ({})",
529                        imm.value,
530                        support::repr(&var_ty)
531                    );
532                    let condition = "IntImmValueFits(a->value, var_ty)";
533                    return Err(support::check_failed(
534                        tvm_ffi::INTERNAL_ERROR,
535                        condition,
536                        &message,
537                    ));
538                }
539                return Ok(IntImm::new(var_ty.clone(), imm.value.clone(), UNKNOWN_LOC)?.into());
540            }
541            if ty.dtype.bits > var_ty.dtype.bits {
542                let message = format!(
543                    " Loop variable's dtype ({}) is narrower than that of `begin` or `end` ({})",
544                    support::repr(&var_ty),
545                    support::repr(&ty)
546                );
547                let condition = "e_ty.bits() <= var_ty.bits()";
548                return Err(support::check_failed(tvm_ffi::INTERNAL_ERROR, condition, &message));
549            }
550            Ok(expr)
551        };
552        let same_type = |expr: &PrimExpr, condition: &str| -> Result<()> {
553            let ty = expr.ty();
554            if ty.dtype == var_ty.dtype {
555                return Ok(());
556            }
557            let message = format!("{} vs {}", support::repr(&var_ty), support::repr(&ty));
558            Err(support::check_failed(tvm_ffi::INTERNAL_ERROR, condition, &message))
559        };
560        let (begin, end) = (promote(begin)?, promote(end)?);
561        same_type(&begin, "var.ty() == begin.ty()")?;
562        same_type(&end, "var.ty() == end.ty()")?;
563        let step = promote(step)?;
564        same_type(&step, "var.ty() == step.ty()")?;
565        Ok(Self::from_fields(loc.into(), var, begin, end, step, kind as i32, body, annotations))
566    }
567}

569/// Whether `value` is in the range of the integer type `dtype` (C++ `IntImmValueFits`).
570fn int_imm_value_fits(value: &BigInt, dtype: DLDataType) -> Result<bool> {
571    let bits = dtype.bits as u32;
572    if dtype.code == support::UINT {
573        let limit = BigInt::from(1i64).try_shl(bits as i64)?;
574        return Ok(!value.is_negative() && *value < limit);
575    }
576    let limit = BigInt::from(1i64).try_shl(bits as i64 - 1)?;
577    Ok(*value >= -limit.clone() && *value < limit)
578}

912/// The parameters the body of a region of `op` on `args` binds, which the operator's
913/// `FRegionGetBodyParams` gives once its validator accepts the arguments
914/// (C++ `GetRegionBodyParams`).
915fn region_body_params(op: &Op, args: &Array<Expr>, attrs: &DictAttrs) -> Result<Array<Var>> {
916    let name = op.name.as_str();
917    let Some(get_body_params) = support::NativeHook::of_op(op, "FRegionGetBodyParams")? else {
918        let message = format!(
919            "{name} does not support region construction: FRegionGetBodyParams is required"
920        );
921        return Err(support::check_failed(tvm_ffi::VALUE_ERROR, "IsRegionOp(op)", &message));
922    };
923    let signature = Call::from_fields(
924        UNKNOWN_LOC,
925        MissingType::new(),
926        op.clone(),
927        args.clone(),
928        Some(attrs.clone().into()),
929        Array::from_vec(vec![]),
930    );
931    signature.validate()?;
932    let params: Array<Var> = get_body_params.call_as(&signature)?;
933    let mut definitions = std::collections::HashSet::new();
934    for param in params.iter() {
935        if param.ty.downcast_ref::<MissingTypeObj>().is_some() {
936            let condition = "!param->ty.as<MissingType>().has_value()";
937            let message = format!("FRegionGetBodyParams for {name} must return typed variables");
938            return Err(support::check_failed(tvm_ffi::VALUE_ERROR, condition, &message));
939        }
940        if !definitions.insert(object_address(&param)) {
941            let condition = "definitions.insert(param.get()).second";
942            let message =
943                format!("FRegionGetBodyParams for {name} must return distinct definitions");
944            return Err(support::check_failed(tvm_ffi::VALUE_ERROR, condition, &message));
945        }
946    }
947    Ok(params)

1019/// `seq` with each nested sequence replaced by its statements and the no-ops removed.
1020fn flatten(seq: Array<Stmt>) -> Array<Stmt> {
1021    let nested_or_no_op =
1022        |stmt: &Stmt| stmt.downcast_ref::<SeqStmtObj>().is_some() || is_no_op(stmt);
1023    if !seq.iter().any(|stmt| nested_or_no_op(&stmt)) {
1024        return seq;
1025    }
1026    fn append(stmt: Stmt, flattened: &mut Vec<Stmt>) {
1027        match stmt.downcast_ref::<SeqStmtObj>() {
1028            Some(nested) => nested.seq.iter().for_each(|child| append(child, flattened)),
1029            None if is_no_op(&stmt) => {}
1030            None => flattened.push(stmt),
1031        }
1032    }
1033    let mut flattened = Vec::new();
1034    seq.iter().for_each(|stmt| append(stmt, &mut flattened));
1035    Array::from_vec(flattened)
1036}
