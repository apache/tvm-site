52/// The largest finite value of each float8 type code, and whether the type is unsigned
53/// (`support::kMax*` in `src/support/limits.h`).
54pub(crate) fn float8_bound(code: u8) -> Option<(f64, bool)> {
55    use tvm_ffi::DLDataTypeCode::*;
56    let bound = match code {
57        c if c == kDLFloat8_e3m4 as u8 => (31.0, false),
58        c if c == kDLFloat8_e4m3 as u8 => (448.0, false),
59        c if c == kDLFloat8_e4m3b11fnuz as u8 => (30.0, true),
60        c if c == kDLFloat8_e4m3fn as u8 => (448.0, false),
61        c if c == kDLFloat8_e4m3fnuz as u8 => (448.0, true),
62        c if c == kDLFloat8_e5m2 as u8 => (57344.0, false),
63        c if c == kDLFloat8_e5m2fnuz as u8 => (57344.0, true),
64        // `support::kMaxE8M0FNU`, 2^128.
65        c if c == kDLFloat8_e8m0fnu as u8 => (3.402_823_669_209_385e38, true),
66        _ => return None,
67    };
68    Some(bound)
69}

76        c if c == kDLFloat6_e3m2fn as u8 => Some(28.0),

84/// The value of a native call that cannot fail: the crate links `libtvm_tirx_compiler`, so the
85/// function is registered, and the arguments are ones its C++ counterpart accepts without a
86/// check. A failure means the linked library does not match the sources the crate was built
87/// from, which is an error of the build, so it panics.
88pub(crate) fn infallible<T>(result: Result<T>) -> T {
89    result.expect("a native call that cannot fail with a matching libtvm_tirx_compiler failed")
90}

110/// A value as C++ `operator<<` prints it into an error message (`ffi.ReprPrint`).
111pub(crate) fn repr<T>(value: &T) -> std::string::String
112where
113    for<'a> AnyView<'a>: From<&'a T>,
114{
115    tvm_ffi::cached_global_func!("ffi.ReprPrint")

127/// The registered key of a type index (C++ `TypeIndexToTypeKey`).
128pub(crate) fn type_key(type_index: i32) -> std::string::String {
129    // SAFETY: the type index of a live value is registered, and its information is static.
130    unsafe {
131        let info = TVMFFIGetTypeInfo(type_index);
132        (*info).type_key.as_str().to_string()
133    }

136/// The default value the reflection of the type of `type_index` registers for its own field
137/// `name` (C++ `refl::DefaultValue`), or `None` when it registers none or a factory.
138pub(crate) fn field_default(type_index: i32, name: &str) -> Option<Any> {
139    use TVMFFIFieldFlagBitMask::*;
140    // SAFETY: the type information of a registered type is static, and its `num_fields` fields
141    // are initialized.
142    let field = unsafe {
143        let info = &*TVMFFIGetTypeInfo(type_index);
144        let fields = std::slice::from_raw_parts(info.fields, info.num_fields as usize);
145        fields.iter().find(|field| field.name.as_str() == name)?
146    };
147    let has_default = field.flags & kTVMFFIFieldFlagBitMaskHasDefault as i64 != 0;
148    let from_factory = field.flags & kTVMFFIFieldFlagBitMaskDefaultFromFactory as i64 != 0;
149    if !has_default || from_factory {
150        return None;
151    }
152    // SAFETY: the registry owns the default value; it is copied, never released.
153    let value = std::mem::ManuallyDrop::new(unsafe {
154        Any::from_raw_ffi_any(field.default_value_or_factory)
155    });
156    Some((*value).clone())
157}

159/// A double as C++ `std::ostream` prints it by default: `%g`, six significant digits.
160#[derive(Clone, Copy)]
161pub(crate) struct CppDouble(pub f64);
162
163impl Display for CppDouble {
164    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
165        let value = self.0;
166        if value.is_nan() {
167            return f.write_str(if value.is_sign_negative() { "-nan" } else { "nan" });
168        }
169        if value.is_infinite() {
170            return f.write_str(if value < 0.0 { "-inf" } else { "inf" });
171        }
172        if value == 0.0 {
173            return f.write_str(if value.is_sign_negative() { "-0" } else { "0" });
174        }
175        let trim = |text: &str| -> std::string::String {
176            match text.contains('.') {
177                true => text.trim_end_matches('0').trim_end_matches('.').to_string(),
178                false => text.to_string(),
179            }
180        };
181        let scientific = format!("{value:.5e}");
182        let (mantissa, exponent) = scientific.split_once('e').expect("an exponent");
183        let exponent: i32 = exponent.parse().expect("an integer exponent");
184        if !(-4..6).contains(&exponent) {
185            let sign = if exponent < 0 { '-' } else { '+' };
186            return write!(f, "{}e{sign}{:02}", trim(mantissa), exponent.abs());
187        }
188        let decimals = (5 - exponent) as usize;
189        f.write_str(&trim(&format!("{value:.decimals$}")))
190    }
191}

196/// A node C++ shares, such as the `UnknownLoc()` singleton, fetched once per thread so that every
197/// node Rust builds refers to the same object as the nodes C++ builds. The fetch is a native call
198/// that cannot fail (see [`infallible`]).
199pub(crate) fn shared<T: Clone + 'static>(
200    cell: &'static LocalKey<OnceCell<T>>,
201    fetch: impl FnOnce() -> Result<T>,
202) -> T {
203    if let Some(value) = cell.with(|cell| cell.get().cloned()) {
204        return value;
205    }
206    let value = infallible(fetch());
207    cell.with(|cell| cell.get_or_init(|| value.clone()).clone());
208    value
209}

214/// The type of a primitive expression. C++ takes such an argument as a `PrimExpr`, whose
215/// conversion accepts only an expression of a `PrimType`.
216pub(crate) fn prim_type(expr: &(impl AsRef<Expr> + ?Sized)) -> Result<PrimType> {
217    let expr = expr.as_ref();
218    match expr.ty.downcast_ref::<PrimTypeObj>() {
219        Some(_) => Ok(expr.ty.downcast_ref::<PrimType>().cloned().expect("a PrimType")),
220        None => Err(not_a_prim_expr(expr)),
221    }
222}
223
224/// The dtype of a primitive expression ([`prim_type`]).
225pub(crate) fn prim_dtype(expr: &(impl AsRef<Expr> + ?Sized)) -> Result<DLDataType> {
226    let expr = expr.as_ref();
227    match expr.ty.downcast_ref::<PrimTypeObj>() {
228        Some(ty) => Ok(ty.dtype),
229        None => Err(not_a_prim_expr(expr)),
230    }
231}
232
233fn not_a_prim_expr(expr: &Expr) -> Error {
234    let message = format!(
235        "Expected `ir.PrimExpr` but got `{}[ty={}]`",
236        type_key(AnyView::from(expr).type_index()),
237        type_key(AnyView::from(&expr.ty).type_index()),
238    );
239    Error::new(TYPE_ERROR, &message, "")
240}

242/// The lanes of a dtype as C++ decodes them: negative for a scalable vector.
243pub(crate) fn encoded_lanes(dtype: DLDataType) -> i16 {
244    dtype.lanes as i16
245}
246
247/// C++ `PrimType::IsScalar()`.
248pub(crate) fn is_scalar(dtype: DLDataType) -> bool {
249    encoded_lanes(dtype) == 1
250}
251
252/// C++ `PrimType::IsScalableVector()`.
253pub(crate) fn is_scalable_vector(dtype: DLDataType) -> bool {
254    encoded_lanes(dtype) < -1
255}
256
257/// C++ `PrimType::IsFixedLengthVector()`.
258pub(crate) fn is_fixed_length_vector(dtype: DLDataType) -> bool {
259    encoded_lanes(dtype) > 1
260}
261
262/// C++ `PrimType::lanes()`, which a scalable vector does not have.
263pub(crate) fn lanes(dtype: DLDataType) -> Result<i32> {
264    let lanes = encoded_lanes(dtype);
265    if lanes < 0 {
266        let message = "Can't fetch the lanes of a scalable vector at a compile time.";
267        return Err(Error::new(INTERNAL_ERROR, message, ""));
268    }
269    Ok(lanes as i32)
270}
271
272/// C++ `PrimType::ScalableVector(code, bits, factor)`: `factor` times `vscale` lanes.
273pub(crate) fn scalable_vector(dtype: DLDataType, factor: i32) -> Result<DLDataType> {
274    if factor <= 1 {
275        let message = format!("Invalid value for vscale factor {factor}");
276        return Err(compare_failed(INTERNAL_ERROR, "lanes > 1", factor, 1, &message));
277    }
278    if factor >= 32768 {
279        return Err(compare_failed(INTERNAL_ERROR, "lanes < 32768", factor, 32768, ""));
280    }
281    Ok(DLDataType { code: dtype.code, bits: dtype.bits, lanes: (-factor) as u16 })
282}

289#[repr(C)]
290pub(crate) struct SeqCell {
291    pub(crate) object: tvm_ffi::Object,
292    pub(crate) data: *const TVMFFIAny,
293    pub(crate) size: i64,
294    pub(crate) capacity: i64,
295    pub(crate) data_deleter: Option<unsafe extern "C" fn(*mut c_void)>,

298/// The elements of an `ffi.Array`, read in place.
299pub(crate) fn array_cells<T: tvm_ffi::type_traits::ContainerElement + Clone>(
300    array: &Array<T>,
301) -> &[TVMFFIAny] {
302    // SAFETY: an `ffi.Array` is the object header followed by `TVMFFISeqCell`, and `array` keeps
303    // its `size` elements alive.
304    unsafe {
305        let cell = &*(ObjectArc::as_ptr(ObjectRefCore::data(array)) as *const SeqCell);
306        if cell.size == 0 {
307            return &[];
308        }
309        std::slice::from_raw_parts(cell.data, cell.size as usize)
310    }
311}

316/// A C++ `NativeFunctionView<R(const Node*)>` hook that a registry holds: the C ABI adapter its
317/// opaque pointer names, or a packed function a frontend registered in its place.
318pub(crate) enum NativeHook {
319    Adapter(extern "C" fn(*const c_void) -> TVMFFIAny),
320    Packed(Function),
321}
322
323impl NativeHook {
324    /// The hook `value` holds.
325    pub(crate) fn from_any(value: Any) -> Result<Self> {
326        if value.type_index() != TVMFFITypeIndex::kTVMFFIOpaquePtr as i32 {
327            return Ok(Self::Packed(value.try_into()?));
328        }
329        let address: *mut c_void = value.try_into()?;
330        // SAFETY: the opaque pointer of a `NativeFunctionView` is its C ABI adapter.
331        let adapter = unsafe {
332            std::mem::transmute::<*mut c_void, extern "C" fn(*const c_void) -> TVMFFIAny>(address)
333        };
334        Ok(Self::Adapter(adapter))
335    }
336
337    /// The hook `op` registers in the attribute column `column`, if any.
338    pub(crate) fn of_op(op: &OpObj, column: &'static str) -> Result<Option<Self>> {
339        op_attr(op, column)?.map(Self::from_any).transpose()
340    }
341
342    /// C++ `CallExpected(node)`: the hook's result on `node`, or the error it reports.
343    pub(crate) fn call<T: ObjectRefCore>(&self, node: &T) -> Result<Any>
344    where
345        for<'a> AnyView<'a>: From<&'a T>,
346    {
347        let result = match self {
348            // SAFETY: `node` keeps the object alive for the call, and the adapter returns an
349            // owned value.
350            Self::Adapter(adapter) => unsafe {
351                let node = ObjectArc::as_ptr(ObjectRefCore::data(node)) as *const c_void;
352                Any::from_raw_ffi_any(adapter(node))
353            },
354            Self::Packed(function) => function.call_packed(&[AnyView::from(node)])?,
355        };
356        expected(result)
357    }
358
359    /// [`NativeHook::call`] of a hook that returns an `R`; a packed hook's result of another type
360    /// is the error C++ raises for it.
361    pub(crate) fn call_as<T: ObjectRefCore, R: TryFrom<Any>>(&self, node: &T) -> Result<R>
362    where
363        for<'a> AnyView<'a>: From<&'a T>,
364    {
365        R::try_from(self.call(node)?).map_err(|_| hook_result_mismatch())
366    }
367
368    /// [`NativeHook::call`] of a hook that returns nothing.
369    pub(crate) fn call_void<T: ObjectRefCore>(&self, node: &T) -> Result<()>
370    where
371        for<'a> AnyView<'a>: From<&'a T>,
372    {
373        match self.call(node)?.type_index() == TVMFFITypeIndex::kTVMFFINone as i32 {
374            true => Ok(()),
375            false => Err(hook_result_mismatch()),
376        }
377    }
378}

380/// C++ `CallExpected(dest, indices, value)` of a hook that takes three object references, such as
381/// `NativeFunctionView<void(const Expr&, const Array<PrimExpr>&, const PrimExpr&)>`. The C ABI
382/// adapter takes each reference as the address of an object pointer; a packed function takes the
383/// values.
384pub(crate) fn call_with_refs<A, B, C>(hook: Any, a: &A, b: &B, c: &C) -> Result<Any>
385where
386    A: ObjectRefCore,
387    B: ObjectRefCore,
388    C: ObjectRefCore,
389    for<'x> AnyView<'x>: From<&'x A> + From<&'x B> + From<&'x C>,
390{
391    if hook.type_index() != TVMFFITypeIndex::kTVMFFIOpaquePtr as i32 {
392        let function: Function = hook.try_into()?;
393        return expected(function.call_packed(&[
394            AnyView::from(a),
395            AnyView::from(b),
396            AnyView::from(c),
397        ])?);
398    }
399    type Adapter = extern "C" fn(
400        *const *const TVMFFIObject,
401        *const *const TVMFFIObject,
402        *const *const TVMFFIObject,
403    ) -> TVMFFIAny;
404    fn object<T: ObjectRefCore>(value: &T) -> *const TVMFFIObject {
405        // The pointer is only read while `value` keeps the object alive.
406        ObjectArc::as_ptr(ObjectRefCore::data(value)) as *const TVMFFIObject
407    }
408    let address: *mut c_void = hook.try_into()?;
409    let (a, b, c) = (object(a), object(b), object(c));
410    // SAFETY: the opaque pointer of a `NativeFunctionView` is its C ABI adapter; a C++ object
411    // reference is the address of its single object pointer, which outlives the call; and the
412    // adapter returns an owned value.
413    let result = unsafe {
414        let adapter = std::mem::transmute::<*mut c_void, Adapter>(address);
415        Any::from_raw_ffi_any(adapter(&a, &b, &c))
416    };
417    expected(result)
418}
419
420/// The value of an `Expected` result, or its error.
421fn expected(result: Any) -> Result<Any> {
422    match result.type_index() == TVMFFITypeIndex::kTVMFFIError as i32 {
423        true => Err(result.try_into()?),
424        false => Ok(result),
425    }
426}
427
428fn hook_result_mismatch() -> Error {
429    Error::new(TYPE_ERROR, "Packed hook result type mismatch", "")
430}

432thread_local! {
433    /// The operator attribute columns, fetched once per thread as C++ keeps them in `static`s.
434    /// A column no operator registered yet is looked up again.
435    static OP_ATTR_COLUMNS: RefCell<HashMap<&'static str, Rc<OpAttrMap<Any>>>> =
436        RefCell::new(HashMap::new());
437}
438
439/// The value `op` registers in the attribute column `column` (C++ `Op::GetAttrMap(column)[op]`),
440/// of whatever type, or `None`.
441pub(crate) fn op_attr(op: &OpObj, column: &'static str) -> Result<Option<Any>> {
442    let cached = OP_ATTR_COLUMNS.with(|columns| columns.borrow().get(column).cloned());
443    let map = match cached {
444        Some(map) => map,
445        None => {
446            let Some(map) = Op::attr_column::<Any>(column)? else { return Ok(None) };
447            let map = Rc::new(map);
448            OP_ATTR_COLUMNS.with(|columns| columns.borrow_mut().insert(column, map.clone()));
449            map
450        }
451    };
452    Ok(map.get_any(op))
453}
454
455/// The registered operator `name` (C++ `Op::Get(name)`), fetched once per thread as C++ keeps an
456/// operator in a `static`.
457pub(crate) fn cached_op(name: &'static str) -> Result<Op> {
458    thread_local! {
459        static OPS: RefCell<HashMap<&'static str, Op>> = RefCell::new(HashMap::new());
460    }
461    if let Some(op) = OPS.with(|ops| ops.borrow().get(name).cloned()) {
462        return Ok(op);
463    }
464    let op = Op::get(name)?;
465    OPS.with(|ops| ops.borrow_mut().insert(name, op.clone()));
466    Ok(op)
467}

469/// A type attribute column, looked up once it exists, as C++ keeps a `TypeAttrColumn` in a
470/// `static`.
471pub(crate) struct TypeAttr {
472    name: &'static str,
473    column: OnceLock<TypeAttrColumn>,
474}
475
476impl TypeAttr {
477    pub(crate) const fn new(name: &'static str) -> Self {
478        Self { name, column: OnceLock::new() }
479    }
480
481    /// The value the type of `type_index` registers in this column, if any.
482    pub(crate) fn get(&self, type_index: i32) -> Option<Any> {
483        let column = match self.column.get() {
484            Some(column) => *column,
485            None => {
486                let column = TypeAttrColumn::new(self.name)?;
487                *self.column.get_or_init(|| column)
488            }
489        };
490        column.get(type_index)
491    }
492}

494/// `ffi::StructuralEqual()(lhs, rhs)`, which does not map free variables.
495pub(crate) fn structural_equal<L, R>(lhs: &L, rhs: &R) -> Result<bool>
496where
497    for<'a> AnyView<'a>: From<&'a L> + From<&'a R>,
498{
499    let (map_free_vars, skip_tensor_content) = (false, false);
500    tvm_ffi::cached_global_func!("ffi.StructuralEqual")
501        .call_packed(&[
502            AnyView::from(lhs),
503            AnyView::from(rhs),
504            AnyView::from(&map_free_vars),
505            AnyView::from(&skip_tensor_content),
506        ])?
507        .try_into()
