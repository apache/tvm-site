20//! Operator registration from Rust: C++ `OpDef`.
21//!
22//! [`OpDef`] registers an operator with its signature ([`sig`]) and attributes ([`op_attr`]) in a
23//! [`tvm_ffi::static_init_block!`], as C++ does in a `TVM_FFI_STATIC_INIT_BLOCK`. Its hooks are
24//! native functions that C++ calls without packing arguments.

42/// A chain of steps that registers a new operator from Rust, C++ `OpDef`.
43///
44/// Each step updates the registry at once, and a failed step skips the rest. The chain completes
45/// when it is dropped, panicking with the error of a failed step, or by [`OpDef::op`], which
46/// returns it. Register an operator before building IR with it, as a
47/// [`tvm_ffi::static_init_block!`] does.
48///
49/// # Examples
50///
51/// ```
52/// use tvm_tirx::ir::op_attr::CALL_EFFECT_KIND;
53/// use tvm_tirx::ir::sig::{self, IntExpr};
54/// use tvm_tirx::ir::{Call, CallEffectKind, CallObj, Op, OpDef, PrimType, Type, Var, UNKNOWN_LOC};
55/// use tvm_tirx::script::script;
56/// use tvm_tirx::tvm_ffi::{self, Result};
57///
58/// fn first_arg_type(call: &CallObj) -> Result<Type> {
59///     Ok(call.args.get(0)?.ty.clone())
60/// }
61///
62/// tvm_ffi::static_init_block! {
63///     OpDef::new("my_ext.clamp", "Clamp an index to a bound.")
64///         .signature((sig::arg::<IntExpr>("index", ""), sig::arg::<IntExpr>("bound", "")))
65///         .set_infer_type(first_arg_type)
66///         .set_attr(CALL_EFFECT_KIND, CallEffectKind::Pure);
67/// }
68///
69/// let clamp = Op::get("my_ext.clamp")?;
70/// let i = Var::new("i", PrimType::int(32), UNKNOWN_LOC);
71/// let call = Call::new(None, clamp, [&i, &i], None, Vec::<Type>::new(), UNKNOWN_LOC)?;
72/// assert_eq!(script(&call.ty)?, "T.int32");
73/// call.validate()?;
74/// # Ok::<(), tvm_ffi::Error>(())
75/// ```
76pub struct OpDef(Option<Result<Op>>);
77
78impl OpDef {
79    /// Register a new operator `name` and start its chain (`ir.RegisterOp`).
80    ///
81    /// The indentation `doc` shares is removed. A name already registered fails the chain with an
82    /// `AttributeError`.
83    pub fn new(name: &str, doc: &str) -> Self {
84        let registered = tvm_ffi::cached_global_func!("ir.RegisterOp")
85            .call_tuple((String::from(name), String::from(doc_text(doc))));
86        Self(Some(registered.and_then(|_| Op::get(name))))
87    }
88
89    /// Complete the chain now, instead of when it is dropped, and return the operator.
90    ///
91    /// # Errors
92    ///
93    /// Returns the error of the step that failed, which dropping the chain would panic with, or of
94    /// the `ir.expose_op` receiver.
95    pub fn op(mut self) -> Result<Op> {
96        complete(self.take())
97    }
98
99    /// Set the signature of the operator to `specs`, a descriptor of [`sig`] or a tuple of them in
100    /// order, as C++ `OpDef::signature` does.
101    ///
102    /// The validator generated from it checks a Call's argument counts, argument types and attrs,
103    /// unless a custom validator is set.
104    pub fn signature<S: Specs>(self, specs: S) -> Self {
105        let mut description = sig::Description::default();
106        specs.describe(&mut description);
107        let lists = description.into_lists().map(Array::new);
108        let args: Vec<AnyView> = lists.iter().map(AnyView::from).collect();
109        let def = self.invoke("_set_signature", &args).then(|op| {
110            if !op.validator_is_custom() {
111                let hook: Hook = validate_signature::<S::Checks>;
112                op.write_validator(Any::from(hook as *mut c_void), false);
113            }
114            Ok(())
115        });
116        match <S::Checks as Checks>::ATTRS_TYPE_KEY {
117            Some(key) => def.attrs_type_key(key),
118            None => def,
119        }
120    }
121
122    /// Append the description of the value argument `name`, without changing what the validator
123    /// checks.
124    pub fn add_arg(self, name: &str, doc: &str) -> Self {
125        let (name, doc) = (String::from(name), String::from(doc));
126        self.invoke("add_arg", &[AnyView::from(&name), AnyView::from(&doc)])
127    }
128
129    /// Append the description of the type argument `name`, without changing what the validator
130    /// checks.
131    pub fn add_ty_arg(self, name: &str, doc: &str) -> Self {
132        let (name, doc) = (String::from(name), String::from(doc));
133        self.invoke("add_ty_arg", &[AnyView::from(&name), AnyView::from(&doc)])
134    }
135
136    /// Set the type of `Call.attrs` to the type registered as `key`; an unregistered `key` fails
137    /// the chain with an `InternalError`.
138    pub fn attrs_type_key(self, key: &str) -> Self {
139        let key = String::from(key);
140        self.invoke("set_attrs_type_key", &[AnyView::from(&key)])
141    }
142
143    /// Register the value of the attribute `key` for the operator, as C++ `set_attr(key, value)`
144    /// does; a value already registered fails the chain with a `ValueError`.
145    pub fn set_attr<T: Into<Any>>(self, key: AttrKey<T>, value: impl Into<T>) -> Self {
146        let value: T = value.into();
147        self.attr(key.name, value.into(), false)
148    }
149
150    /// Register the value of the attribute `key` for the operator, replacing a value already
151    /// registered, as C++ `set_attr(key, value, true)` does.
152    pub fn replace_attr<T: Into<Any>>(self, key: AttrKey<T>, value: impl Into<T>) -> Self {
153        let value: T = value.into();
154        self.attr(key.name, value.into(), true)
155    }
156
157    /// Remove the operator's value of the attribute `key`; a missing value is ignored, and live
158    /// [`OpAttrMap`](crate::ir::OpAttrMap)s see the removal.
159    pub fn reset_attr<T>(self, key: AttrKey<T>) -> Self {
160        let key = String::from(key.name);
161        self.invoke("reset_attr", &[AnyView::from(&key)])
162    }
163
164    /// Set the type inference of the operator, its `FInferType`, to `infer`, a function or a
165    /// closure that captures nothing.
166    ///
167    /// It gives the result type of a Call that has none; [`op_attr::FIXED_RETURN_TYPE`] takes
168    /// precedence. A second one fails the chain with a `ValueError`.
169    pub fn set_infer_type<F>(self, infer: F) -> Self
170    where
171        F: Fn(&CallObj) -> Result<Type> + 'static,
172    {
173        let _ = infer;
174        let hook: Hook = infer_type_abi::<F>;
175        self.attr("FInferType", Any::from(hook as *mut c_void), false)
176    }
177
178    /// Set a custom validator of the operator, `validate`, a function or a closure that captures
179    /// nothing.
180    ///
181    /// [`Call::validate`](crate::ir::Call::validate) runs it instead of the generated one. A second
182    /// one fails the chain with a `ValueError`.
183    pub fn set_validator<F>(self, validate: F) -> Self
184    where
185        F: Fn(&CallObj) -> Result<()> + 'static,
186    {
187        let _ = validate;
188        self.then(|op| {
189            if op.validator_is_custom() {
190                let message = format!("Validator of {} is already registered", op.name.as_str());
191                return Err(Error::new(VALUE_ERROR, &message, ""));
192            }
193            let hook: Hook = validate_abi::<F>;
194            op.write_validator(Any::from(hook as *mut c_void), true);
195            Ok(())
196        })
197    }
198
199    /// Set the body parameters of the operator's regions, its `FRegionGetBodyParams`, to
200    /// `body_params`, a function or a closure that captures nothing; this makes the operator a
201    /// region operation.
202    ///
203    /// The `RegionStmt` constructor checks a region's body parameters against it. A second hook
204    /// fails the chain with a `ValueError`.
205    pub fn set_region_body_params<F>(self, body_params: F) -> Self
206    where
207        F: Fn(&CallObj) -> Result<Array<Var>> + 'static,
208    {
209        let _ = body_params;
210        let hook: Hook = region_body_params_abi::<F>;
211        self.attr("FRegionGetBodyParams", Any::from(hook as *mut c_void), false)
212    }
213
214    /// Set the check of the operator's regions, its `FRegionValidate`, to `validate`, a function or
215    /// a closure that captures nothing.
216    ///
217    /// The `RegionStmt` constructor runs it on a complete region statement. A second check fails
218    /// the chain with a `ValueError`.
219    pub fn set_region_validate<F>(self, validate: F) -> Self
220    where
221        F: Fn(&RegionStmtObj) -> Result<()> + 'static,
222    {
223        let _ = validate;
224        let hook: RegionHook = region_validate_abi::<F>;
225        self.attr("FRegionValidate", Any::from(hook as *mut c_void), false)
226    }
227
228    fn attr(self, key: &str, value: Any, replace: bool) -> Self {
229        let key = String::from(key);
230        self.invoke(
231            "_set_attr",
232            &[AnyView::from(&key), AnyView::from(&value), AnyView::from(&replace)],
233        )
234    }
235
236    /// Call the `ir.Op` method `method` on the operator with `args`.
237    fn invoke(self, method: &str, args: &[AnyView]) -> Self {
238        self.then(|op| {
239            let packed: Vec<AnyView> =
240                std::iter::once(AnyView::from(op)).chain(args.iter().copied()).collect();
241            tvm_ffi::Function::from_type_key_method("ir.Op", method)?.call_packed(&packed)?;
242            Ok(())
243        })
244    }
245
246    fn then(mut self, step: impl FnOnce(&Op) -> Result<()>) -> Self {
247        Self(Some(self.take().and_then(|op| step(&op).map(|_| op))))
248    }
249
250    /// The state of the chain, which leaves nothing for `drop` to complete.
251    fn take(&mut self) -> Result<Op> {
252        self.0.take().expect("an `OpDef` is completed once")
253    }
254}
255
256impl Drop for OpDef {
257    fn drop(&mut self) {
258        // Dropped while a panic unwinds, the chain did not finish, and a second panic would abort.
259        if std::thread::panicking() {
260            return;
261        }
262        if let Some(Err(error)) = self.0.take().map(complete) {
263            panic!("{}: {}", error.kind(), error.message());
264        }
265    }
266}

268/// Request the APIs of a defined operator in registered destinations through `ir.expose_op`,
269/// after its signature and attributes are ready. An absent receiver is a no-op; an installed
270/// receiver's error is returned.
271fn complete(state: Result<Op>) -> Result<Op> {
272    let op = state?;
273    if let Ok(expose) = tvm_ffi::Function::get_global("ir.expose_op") {
274        expose.call_tuple((op.name.clone(),))?;
275    }
276    Ok(op)
277}

279/// The key of an operator attribute, C++ `ffi::reflection::AttrKey<T>`: its name and the type `T`
280/// of its values.
281///
282/// [`op_attr`] and [`tirx::op_attr`](crate::tirx::op_attr) have the keys C++ declares;
283/// [`AttrKey::new`] names another.
284pub struct AttrKey<T> {
285    name: &'static str,
286    value: PhantomData<fn() -> T>,
287}
288
289impl<T> AttrKey<T> {
290    /// Create the key of the attribute `name`, whose values are `T`.
291    pub const fn new(name: &'static str) -> Self {
292        Self { name, value: PhantomData }
293    }
294
295    /// Return the name of the attribute, which [`Op::attr_map`] reads the column of.
296    pub const fn name(&self) -> &'static str {
297        self.name
298    }
299}
300
301impl<T> Clone for AttrKey<T> {
302    fn clone(&self) -> Self {
303        *self
304    }
305}

307impl<T> Copy for AttrKey<T> {}

321/// The runtime effect of an operator call, C++ `CallEffectKind`: the value of
322/// [`op_attr::CALL_EFFECT_KIND`].
323#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
324#[repr(i64)]
325pub enum CallEffectKind {
326    /// An annotation, such as `likely`, which can be translated to the identity.
327    ExprAnnotation = 0,
328    /// A pure function, which does not interact with any external state.
329    Pure = 1,
330    /// A function that may read from states, such as memory.
331    ReadState = 2,
332    /// A function that may read from and write to states, such as memory; also
333    /// [`CallEffectKind::Opaque`].
334    UpdateState = 3,
335    /// A special intrinsic that annotates call arguments, valid only as a direct argument to a
336    /// call.
337    SpecialCallArg = 4,
338    /// Opaque information embedded in the expression, which cannot be generated as code.
339    EmbedInfo = 5,
340    /// A function that changes the control flow.
341    ControlJump = 6,
342}
343
344impl CallEffectKind {
345    /// An opaque function, about which nothing can be assumed: C++ `kOpaque`, which is
346    /// [`CallEffectKind::UpdateState`].
347    #[allow(non_upper_case_globals)]
348    pub const Opaque: Self = Self::UpdateState;
349}
350
351impl From<CallEffectKind> for Any {
352    fn from(kind: CallEffectKind) -> Self {
353        Any::from(kind as i64)
354    }
355}

357/// The text of `doc` without the indentation its lines share and without the blank lines around
358/// it. Text that is not indented is unchanged.
359fn doc_text(doc: &str) -> std::string::String {
360    let indent = doc
361        .lines()
362        .filter(|line| !line.trim().is_empty())
363        .map(|line| line.len() - line.trim_start().len())
364        .min()
365        .unwrap_or(0);
366    let lines: Vec<&str> =
367        doc.lines().map(|line| line.get(indent..).unwrap_or("").trim_end()).collect();
368    lines.join("\n").trim().to_string()
369}

371/// The C ABI of a native operator hook, `NativeFunctionView<R(const CallNode*)>`: it returns
372/// the result, or the error, in an owned `TVMFFIAny`.
373type Hook = extern "C" fn(*const CallObj) -> TVMFFIAny;
374
375/// The C ABI of a native hook on a region statement, `NativeFunctionView<R(const
376/// RegionStmtNode*)>`.
377type RegionHook = extern "C" fn(*const RegionStmtObj) -> TVMFFIAny;
378
379extern "C" fn validate_abi<F: Fn(&CallObj) -> Result<()>>(call: *const CallObj) -> TVMFFIAny {
380    call_hook(call, zero_sized::<F>())
381}
382
383extern "C" fn infer_type_abi<F: Fn(&CallObj) -> Result<Type>>(call: *const CallObj) -> TVMFFIAny {
384    call_hook(call, zero_sized::<F>())
385}
386
387extern "C" fn validate_signature<L: Checks>(call: *const CallObj) -> TVMFFIAny {
388    call_hook(call, sig::validate::<L>)
389}
390
391extern "C" fn region_body_params_abi<F: Fn(&CallObj) -> Result<Array<Var>>>(
392    call: *const CallObj,
393) -> TVMFFIAny {
394    call_hook(call, zero_sized::<F>())
395}
396
397extern "C" fn region_validate_abi<F: Fn(&RegionStmtObj) -> Result<()>>(
398    region: *const RegionStmtObj,
399) -> TVMFFIAny {
400    call_hook(region, zero_sized::<F>())
401}

403/// The only value of `F`, a function or a closure that captures nothing.
404fn zero_sized<F>() -> F {
405    const {
406        assert!(
407            std::mem::size_of::<F>() == 0,
408            "a native operator hook is a function or a closure that captures nothing"
409        )
410    };
411    // SAFETY: a zero-sized type has no bytes to initialize.
412    unsafe { std::mem::zeroed() }
413}
414
415/// Run `hook` on the node C++ passes (the Call or region being typed or checked) and return its
416/// result as the hook ABI does. A panic, which must not unwind into C++, becomes an error.
417fn call_hook<N, R: Into<Any>>(node: *const N, hook: impl Fn(&N) -> Result<R>) -> TVMFFIAny {
418    let result = catch_unwind(AssertUnwindSafe(|| {
419        // SAFETY: C++ passes the node being typed or checked, alive for the call.
420        match unsafe { node.as_ref() } {
421            Some(node) => hook(node).map(Into::into),
422            None => Err(Error::new(TYPE_ERROR, "Invalid Op Call", "")),
423        }
424    }));
425    let value = match result {
426        Ok(Ok(value)) => value,
427        Ok(Err(error)) => Any::from(error),
428        Err(payload) => Any::from(tvm_ffi::function_internal::panic_to_error(payload)),
429    };
430    // SAFETY: the C++ caller takes ownership of the value.
431    unsafe { Any::into_raw_ffi_any(value) }
432}

434/// Signature descriptors for [`OpDef::signature`], C++ `sig::arg<T>` and friends.
435///
436/// [`arg`](sig::arg) and [`ty_arg`](sig::ty_arg) are fixed value and type arguments,
437/// [`var_args`](sig::var_args) and [`var_ty_args`](sig::var_ty_args) variadic tails, and
438/// [`call_attrs`](sig::call_attrs) the type of `Call.attrs`. A node type accepts an instance of it,
439/// and the typed views such as [`PrimExpr`](super::PrimExpr) and [`PrimVar`](sig::PrimVar) also
440/// check its type; implement [`ArgType`](sig::ArgType) for another check.
441pub mod sig {
442    use super::*;
443    use crate::ir::expr::ExprObj;
444    use crate::ir::r#type::PrimTypeObj;
445    use tvm_ffi::type_traits::AnyCompatible;
446    use tvm_ffi::ObjectCore;
447
448    /// A check of an argument against a type, C++ `TypeTraits<T>`.
449    ///
450    /// Every managed reference that is [`AnyCompatible`] implements it with its strict check;
451    /// implement it for another check, as [`PrimVar`] does.
452    pub trait ArgType: 'static {
453        /// Return whether `value`, which is not None, is accepted, as C++ `CheckAnyStrict` does.
454        fn check(value: &TVMFFIAny) -> bool;
455        /// Return the expected type, for an error, as C++ `TypeStr` does.
456        fn type_str() -> std::string::String;
457        /// Describe a rejected value, for an error, as C++ `GetMismatchTypeInfo` does.
458        ///
459        /// By default, the description is the type key of `value`.
460        fn mismatch(value: &TVMFFIAny) -> std::string::String {
461            support::type_key(value.type_index)
462        }
463    }
464
465    impl<T: ObjectRefCore + AnyCompatible + 'static> ArgType for T {
466        fn check(value: &TVMFFIAny) -> bool {
467            // SAFETY: `value` is a live FFI value.
468            unsafe { T::check_any_strict(value) }
469        }
470
471        fn type_str() -> std::string::String {
472            T::type_str()
473        }
474
475        fn mismatch(value: &TVMFFIAny) -> std::string::String {
476            T::get_mismatch_type_info(value)
477        }
478    }
479
480    /// The typed views, which are argument types as they are.
481    pub use crate::ir::expr::{IntExpr, PrimExpr};
482
483    /// An argument type that accepts a [`Var`], not a subclass, whose type is a `PrimType`, C++
484    /// `PrimVar`.
485    pub struct PrimVar;
486
487    impl ArgType for PrimVar {
488        fn check(value: &TVMFFIAny) -> bool {
489            var_of_type::<PrimTypeObj>(value)
490        }
491
492        fn type_str() -> std::string::String {
493            crate::ir::expr::VarObj::TYPE_KEY.into()
494        }
495    }
496
497    /// Return whether `value` is a [`Var`], not a subclass, whose type is an `N`, such as
498    /// `PrimTypeObj`: the check of [`PrimVar`] and of other variable argument types.
499    pub fn var_of_type<N: ObjectCore>(value: &TVMFFIAny) -> bool {
500        value.type_index == crate::ir::expr::VarObj::type_index()
501            && expr(value).is_some_and(|var| var.ty.downcast_ref::<N>().is_some())
502    }
503
504    /// The expression `value` holds, if it holds one.
505    fn expr(value: &TVMFFIAny) -> Option<&ExprObj> {
506        let is_object = value.type_index >= TVMFFITypeIndex::kTVMFFIStaticObjectBegin as i32;
507        // SAFETY: an object value of an `ir.Expr` type points to an `ExprObj`.
508        unsafe {
509            (is_object && Expr::check_any_strict(value))
510                .then(|| &*(value.data_union.v_obj as *const ExprObj))
511        }
512    }

514    /// Describe a required value argument `name` of type `T`, C++ `sig::arg<T>`; [`Expr`] accepts

602    /// One descriptor of a signature: what it adds to the description, and what it checks.
603    #[doc(hidden)]
604    pub trait Descriptor: 'static {
605        const ARGS: usize = 0;
606        const TY_ARGS: usize = 0;
607        const VAR_ARGS: usize = 0;
608        const VAR_TY_ARGS: usize = 0;
609        const CALL_ATTRS: usize = 0;
610        /// The type key of the attrs type, if the descriptor names one.
611        const ATTRS_TYPE_KEY: Option<&'static str> = None;
612
613        fn describe(self, description: &mut Description);
614
615        /// Check what this describes of `call`. `fixed` counts the signature's fixed value and
616        /// type arguments, where its tails start, and `at` those declared before this one.
617        fn check(
618            call: &CallObj,
619            op: &OpObj,
620            fixed: (usize, usize),
621            at: (usize, usize),
622        ) -> Result<()>;
623    }

687    impl<T: ObjectRefCore + 'static> Descriptor for CallAttrs<T> {
688        const CALL_ATTRS: usize = 1;
689        const ATTRS_TYPE_KEY: Option<&'static str> = Some(T::ContainerType::TYPE_KEY);
690
691        fn describe(self, _: &mut Description) {}
692
693        fn check(call: &CallObj, op: &OpObj, _: (usize, usize), _: (usize, usize)) -> Result<()> {
694            let actual = match &call.attrs {
695                Some(attrs) if attrs.downcast_ref::<T::ContainerType>().is_some() => return Ok(()),
696                Some(attrs) => {
697                    format!("`{}`", support::type_key(AnyView::from(attrs).type_index()))
698                }
699                None => "None".into(),
700            };
701            let message = format!(
702                "Op `{}`: Call.attrs expected `{}`, got {actual}",
703                op.name.as_str(),
704                T::ContainerType::TYPE_KEY,
705            );
706            Err(Error::new(TYPE_ERROR, &message, ""))
707        }

743    impl<D: Descriptor, L: Checks> Checks for (D, L) {
744        const ARGS: usize = D::ARGS + L::ARGS;
745        const TY_ARGS: usize = D::TY_ARGS + L::TY_ARGS;
746        const VAR_ARGS: usize = D::VAR_ARGS + L::VAR_ARGS;
747        const VAR_TY_ARGS: usize = D::VAR_TY_ARGS + L::VAR_TY_ARGS;
748        const CALL_ATTRS: usize = D::CALL_ATTRS + L::CALL_ATTRS;
749        const ATTRS_TYPE_KEY: Option<&'static str> = match D::ATTRS_TYPE_KEY {
750            Some(key) => Some(key),
751            None => L::ATTRS_TYPE_KEY,
752        };
753
754        fn check(
755            call: &CallObj,
756            op: &OpObj,
757            fixed: (usize, usize),
758            at: (usize, usize),
759        ) -> Result<()> {
760            D::check(call, op, fixed, at)?;
761            L::check(call, op, fixed, (at.0 + D::ARGS, at.1 + D::TY_ARGS))
762        }

765    /// The descriptors that [`OpDef::signature`] takes: a descriptor, or a tuple of them in order.
766    ///
767    /// `()` declares no argument.
768    pub trait Specs {
769        #[doc(hidden)]
770        type Checks: Checks;
771
772        #[doc(hidden)]
773        fn describe(self, description: &mut Description);
774    }
775
776    impl<D: Descriptor> Specs for D {
777        type Checks = (D, ());
778
779        fn describe(self, description: &mut Description) {
780            Descriptor::describe(self, description);
781        }
782    }

784    /// `Specs` of the tuples of descriptors, whose checks nest in declaration order.
785    macro_rules! tuple_specs {
786        (@nest) => { () };
787        (@nest $head:ident $(, $rest:ident)*) => { ($head, tuple_specs!(@nest $($rest),*)) };
788        ($($name:ident),*) => {
789            impl<$($name: Descriptor),*> Specs for ($($name,)*) {
790                type Checks = tuple_specs!(@nest $($name),*);
791
792                #[allow(non_snake_case, unused_variables)]
793                fn describe(self, description: &mut Description) {
794                    let ($($name,)*) = self;
795                    $(Descriptor::describe($name, description);)*
796                }
797            }
798        };
799    }

815    /// C++ `ValidateSignature<Specs...>`: the argument counts, then each argument's type and the
816    /// attrs, in declaration order.
817    pub(super) fn validate<L: Checks>(call: &CallObj) -> Result<()> {
818        const {
819            assert!(
820                L::VAR_ARGS <= 1 && L::VAR_TY_ARGS <= 1 && L::CALL_ATTRS <= 1,
821                "a signature has at most one variadic tail of each kind and one call_attrs"
822            )
823        };
824        let Some(op) = call.op.downcast_ref::<OpObj>() else {
825            return Err(Error::new(TYPE_ERROR, "Invalid Op Call", ""));
826        };
827        let count = |label: &str, expected: usize, tail: bool, unit: &str, got: usize| {
828            let short = if tail { got < expected } else { got != expected };
829            if !short {
830                return Ok(());
831            }
832            let at_least = if tail { "at least " } else { "" };
833            let plural = if expected == 1 { "" } else { "s" };
834            let message = format!(
835                "Op `{}`: {label} expected {at_least}{expected} {unit}{plural}, got {got}",
836                op.name.as_str()
837            );
838            Err(Error::new(TYPE_ERROR, &message, ""))
839        };
840        count("Call.args", L::ARGS, L::VAR_ARGS == 1, "argument", call.args.len())?;
841        count(
842            "Call.ty_args",
843            L::TY_ARGS,
844            L::VAR_TY_ARGS == 1,
845            "type argument",
846            call.ty_args.len(),
847        )?;
848        L::check(call, op, (L::ARGS, L::TY_ARGS), (0, 0))
849    }

851    /// Check argument `index` against `T`. `Base` is the category's base type, which C++ does
852    /// not check.
853    fn check_value<T: ArgType, Base: 'static>(
854        op: &OpObj,
855        type_arg: bool,
856        values: &[TVMFFIAny],
857        index: usize,
858    ) -> Result<()> {
859        if TypeId::of::<T>() == TypeId::of::<Base>() {
860            return Ok(());
861        }
862        let value = &values[index];
863        if value.type_index != TVMFFITypeIndex::kTVMFFINone as i32 && T::check(value) {
864            return Ok(());
865        }
866        let (fixed, tail) = match type_arg {
867            false => (&op.args_info, &op.var_args_info),
868            true => (&op.ty_args_info, &op.var_ty_args_info),
869        };
870        let name = match fixed.get(index) {
871            Ok(info) => Some(info.name.clone()),
872            Err(_) => tail.as_ref().map(|info| info.name.clone()),
873        };
874        let name = match name {
875            Some(name) if !name.as_str().is_empty() => format!(" (`{}`)", name.as_str()),
876            _ => std::string::String::new(),
877        };
878        let message = format!(
879            "Op `{}`: `{}[{index}]`{name} expected `{}`, got `{}`.",
880            op.name.as_str(),
881            if type_arg { "Call.ty_args" } else { "Call.args" },
882            T::type_str(),
883            T::mismatch(value),
884        );
885        Err(Error::new(TYPE_ERROR, &message, ""))
886    }
