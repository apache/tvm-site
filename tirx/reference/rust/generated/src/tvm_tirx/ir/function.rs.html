52/// The base of every function: an expression, typed by the function's type, with the function
53/// attributes `attrs`, such as its `global_symbol`.
54///
55/// # Examples
56///
57/// ```
58/// use tvm_tirx::ir::{BaseFunc, DictAttrs, PrimType, Return, SeqStmt, Var, UNKNOWN_LOC};
59/// use tvm_tirx::script::script;
60/// use tvm_tirx::tirx::Function;
61/// use tvm_tirx::tvm_ffi::{Map, String};
62///
63/// let body = SeqStmt::new([Return::new(0, UNKNOWN_LOC)], UNKNOWN_LOC);
64/// let attrs = DictAttrs::new(Map::new());
65/// let func = Function::new(Vec::<Var>::new(), &body, PrimType::int(32), attrs, UNKNOWN_LOC);
66/// let func = BaseFunc::from(func);
67/// assert!(script(&func)?.contains("@T.function(private=True)"));
68///
69/// let exported = func.with_attr("global_symbol", String::from("main"))?;
70/// assert!(script(&exported)?.contains("@T.function\ndef main()"));
71/// # Ok::<(), tvm_tirx::tvm_ffi::Error>(())
72/// ```
73// tvm-ffi-stubgen(begin): object/ir.BaseFunc
74#[allow(missing_docs)]
75#[repr(C)]
76#[derive(tvm_ffi::derive::ObjectRef, Clone)]
77pub struct BaseFunc {
78    data_: ObjectArc<BaseFuncObj>,
79}

98impl Deref for BaseFuncObj {
99    type Target = ExprObj;
100    fn deref(&self) -> &ExprObj {
101        &self.base_
102    }
103}
104
105#[allow(missing_docs)]
106impl BaseFuncObj {
107    pub fn new(loc: Location, ty: Type, attrs: DictAttrs) -> Self {
108        let base_ = ExprObj::new(loc, ty);
109        Self { base_, attrs }
110    }
111}
112
113#[allow(missing_docs)]
114impl BaseFunc {
115    /// Lossless complete-field allocation.
116    pub fn new(loc: impl Into<Location>, ty: impl Into<Type>, attrs: impl Into<DictAttrs>) -> Self {
117        let obj = BaseFuncObj::new(loc.into(), ty.into(), attrs.into());
118        Self { data_: ObjectArc::new(obj) }
119    }
120}
121
122impl ShallowCopy for BaseFuncObj {
123    fn shallow_copy(&self) -> Self {
124        Self { base_: ShallowCopy::shallow_copy(&self.base_), attrs: Clone::clone(&self.attrs) }
125    }
126}

128unsafe impl UnsafeInit for BaseFuncObj {
129    unsafe fn drop_defined_fields(this: *mut Self) {
130        unsafe {
131            UnsafeInit::drop_defined_fields(addr_of_mut!((*this).base_));
132            drop_if_defined(addr_of_mut!((*this).attrs));
133        }
134    }
135}

142impl BaseFunc {
143    /// Return a new handle to the same function, as the native side returns it (`ir.BaseFuncCopy`).
144    pub fn copy(&self) -> Result<BaseFunc> {
145        tvm_ffi::cached_global_func!("ir.BaseFuncCopy").call_tuple((self,))?.try_into()
146    }
147
148    /// Return the function with the attribute `key` set to `value` (`ir.BaseFuncWithAttr`).
149    ///
150    /// Consumes `self`: the function is updated in place when this was its only handle, and copied
151    /// otherwise.
152    pub fn with_attr(self, key: &str, value: impl Into<Any>) -> Result<BaseFunc> {
153        tvm_ffi::cached_global_func!("ir.BaseFuncWithAttr")
154            .call_tuple((RValueRef::new(self), String::from(key), value.into()))?
155            .try_into()
156    }
157
158    /// Return the function with the attributes of `attrs` set (`ir.BaseFuncWithAttrs`).
159    ///
160    /// Consumes `self`, as [`with_attr`](Self::with_attr) does.
161    pub fn with_attrs(self, attrs: &Map<String, Any>) -> Result<BaseFunc> {
162        tvm_ffi::cached_global_func!("ir.BaseFuncWithAttrs")
163            .call_tuple((RValueRef::new(self), attrs))?
164            .try_into()
165    }
166
167    /// Return the function without the attribute `key` (`ir.BaseFuncWithoutAttr`).
168    ///
169    /// Consumes `self`, as [`with_attr`](Self::with_attr) does.
170    pub fn without_attr(self, key: &str) -> Result<BaseFunc> {
171        tvm_ffi::cached_global_func!("ir.BaseFuncWithoutAttr")
172            .call_tuple((RValueRef::new(self), String::from(key)))?
173            .try_into()
174    }
175}
