21//! Code generation: IR modules compiled into runtime modules.

23//! Laid out like `python/tvm_tirx/codegen/`: the functions of its `__init__.py` are here, and the

25//! [`crate::target::build`].

45/// Compile `module` into a runtime module with the code generator of the kind of `target`
46/// (`codegen.Build`).
47pub fn build_module(module: impl Into<IRModule>, target: impl Into<Target>) -> Result<Module> {
48    tvm_ffi::cached_global_func!("codegen.Build")
49        .call_tuple((&module.into(), &target.into()))?
50        .try_into()
51}
52
53/// Whether the LLVM target `target`, or the current target, has the CPU feature `feature`;
54/// `false` for a target of another kind; an error when TIRx is built without LLVM, whose backend
55/// registers it (`codegen.target_has_feature`).
56pub fn target_has_feature(feature: &str, target: Option<&Target>) -> Result<bool> {
57    tvm_ffi::Function::get_global("codegen.target_has_feature")?
58        .call_tuple((String::from(feature), target.cloned()))?
59        .try_into()
60}
