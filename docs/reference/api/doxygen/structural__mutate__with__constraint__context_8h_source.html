72 template <typename F>
73 auto WithConstraintScope(F&& f) -> decltype(f()) {
74 sym::ConstraintContext saved_context = constraints;
75 int64_t saved_prefix = constraints->LocalSize();
76 bool saved_needs_child = needs_child_context_;
77 needs_child_context_ = true;
78 auto result = std::forward<F>(f)();
79 if (!result.is_err()) {
80 constraints = saved_context->LocalSize() == saved_prefix
81 ? saved_context
82 : sym::ConstraintContext(saved_context, saved_prefix);
83 needs_child_context_ = saved_needs_child;
84 }

229 static ffi::reflection::TypeAttrColumn copying(type_attr::kStructuralMutateWithConstraintContext);
230 static ffi::reflection::TypeAttrColumn inplace(
231 type_attr::kStructuralMaybeInplaceMutateWithConstraintContext);
232 auto attr = mode == ffi::InplaceMode::kAllow ? inplace[value.type_index()] : ffi::AnyView();
233 if (attr == nullptr) attr = copying[value.type_index()];
234 if (attr == nullptr) return ffi::StructuralMutatorObj::DefaultMutateExpected(value, mode);
235 if (!attr.try_cast<FStructuralMutateWithConstraintContext>()) {
236 return ffi::Unexpected(ffi::Error(
237 "TypeError",
238 "Constraint-aware mutation hook must be an opaque function pointer or ffi.Function", ""));
239 }
240 using ABI = TVMFFIAny (*)(StructuralMutatorWithConstraintContextObj*, ffi::AnyView) noexcept;
241 TVMFFIAny raw;
242 if (attr.type_index() == ffi::TypeIndex::kTVMFFIOpaquePtr) {
243 raw = reinterpret_cast<ABI>(attr.cast<void*>())(this, value);
244 } else {
245 raw = ffi::details::ExpectedUnsafe::MoveToTVMFFIAny(
246 attr.cast<ffi::Function>().CallExpected<ffi::Any>(this, value));
247 }
248 auto result = ffi::details::ExpectedUnsafe::MoveFromTVMFFIAny<ffi::UnchangedOr<ffi::Any>>(raw);
249 if (result.is_err()) {
250 auto error = result.error();
251 ffi::details::UpdateVisitErrorContext(error, value);
252 }
253 return result;
