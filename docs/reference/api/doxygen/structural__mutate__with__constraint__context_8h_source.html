55 void AppendConstraint(const sym::Constraint& constraint) {

69 template <typename F>
70 auto WithConstraintScope(F&& f) -> decltype(f()) {
71 sym::ConstraintContext saved_context = constraints;
72 int64_t saved_prefix = constraints->LocalSize();
73 bool saved_needs_child = needs_child_context_;
74 needs_child_context_ = true;
75 auto result = std::forward<F>(f)();
76 if (!result.is_err()) {
77 constraints = saved_context->LocalSize() == saved_prefix
78 ? saved_context
79 : sym::ConstraintContext(saved_context, saved_prefix);
80 needs_child_context_ = saved_needs_child;
81 }
82 return result;
83 }

208 ffi::InplaceMode mode) noexcept {
209 static ffi::reflection::TypeAttrColumn copying(type_attr::kStructuralMutateWithConstraintContext);
210 static ffi::reflection::TypeAttrColumn inplace(
211 type_attr::kStructuralMaybeInplaceMutateWithConstraintContext);
212 auto attr = mode == ffi::InplaceMode::kAllow ? inplace[value.type_index()] : ffi::AnyView();
213 if (attr == nullptr) attr = copying[value.type_index()];
214 if (attr == nullptr) return ffi::StructuralMutatorObj::DefaultMutateExpected(value, mode);
215 if (!attr.try_cast<FStructuralMutateWithConstraintContext>()) {
216 return ffi::Unexpected(ffi::Error(
217 "TypeError",
218 "Constraint-aware mutation hook must be an opaque function pointer or ffi.Function", ""));
219 }
220 using ABI = TVMFFIAny (*)(StructuralMutatorWithConstraintContextObj*, ffi::AnyView) noexcept;
221 TVMFFIAny raw;
222 if (attr.type_index() == ffi::TypeIndex::kTVMFFIOpaquePtr) {
223 raw = reinterpret_cast<ABI>(attr.cast<void*>())(this, value);
224 } else {
225 raw = ffi::details::ExpectedUnsafe::MoveToTVMFFIAny(
226 attr.cast<ffi::Function>().CallExpected<ffi::Any>(this, value));
227 }
228 auto result = ffi::details::ExpectedUnsafe::MoveFromTVMFFIAny<ffi::UnchangedOr<ffi::Any>>(raw);
229 if (result.is_err()) {
230 ffi::detail::UpdateVisitErrorContext(&result, value);
231 }
232 return result;
