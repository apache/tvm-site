286 // Hold the patterns by const&. This follows the same usage as both
287 // the `PVar`, which occurs as `const PVar<T>&` when it appears
288 // inside other patterns. Because the `PVar<T>::value_` field is
289 // mutable, it can still be updated through these const references.
290 // So long as the call to `Match()` occurs within the same
291 // expression as created the patterns, this avoids accidental copies
292 // without creating dangling references. This may be improved in
293 // the future by use of `constexpr` constructors/operators, allowing
294 // more typical value semantics.
295 std::tuple<const TPattern&...> patterns_;
296};

298/* \brief Return a proxy object that returns true after the first match
