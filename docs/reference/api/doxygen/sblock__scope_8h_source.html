149class SRefTreeCreator : public s_tir::StmtExprVisitor {
150 public:
151 using s_tir::StmtExprVisitor::Visit_;
152
153 ffi::Optional<VisitInterrupt> Visit(ffi::AnyView value) override {
154 if (value.as<ExprNode>()) return std::nullopt;
155 return s_tir::StmtExprVisitor::Visit(value);
156 }
157
163 static std::unordered_map<const StmtNode*, StmtSRef> Create(IRModule mod,
164 bool include_loops = true) {
165 auto creator = ffi::make_object<SRefTreeCreator>(include_loops);
166 for (const auto& kv : mod->functions) {
167 const BaseFunc& base_func = kv.second;
168 if (auto opt = base_func.as<Function>()) {
169 auto func = opt.value();
170 creator->Visit(func->body);
171 }
172 }
173 return std::move(creator->stmt2ref_);
174 }
175
176 explicit SRefTreeCreator(bool include_loops) : include_loops_(include_loops) {}
177
178 private:
183 void PushSRef(const StmtNode* stmt);
184
186 void PopAndRecordSRef();
187
188 ffi::Optional<VisitInterrupt> Visit_(const ForNode* loop) final;
189
190 ffi::Optional<VisitInterrupt> Visit_(const s_tir::SBlockRealizeNode* realize) final;
191
192 ffi::Optional<VisitInterrupt> Visit_(const SeqStmtNode* seq_stmt) final;
193
194 bool include_loops_;
196 std::unordered_map<const StmtNode*, StmtSRef> stmt2ref_;
198 std::vector<StmtSRef> srefs_;
199};
