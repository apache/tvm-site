147class SRefTreeCreator : public StmtExprVisitor {
148 public:
149 using StmtExprVisitor::Visit_;
150
151 ffi::Optional<VisitInterrupt> Visit(ffi::AnyView value) override {
152 if (value.as<ExprNode>()) return std::nullopt;
153 return StmtExprVisitor::Visit(value);
154 }
155
161 static std::unordered_map<const StmtNode*, StmtSRef> Create(IRModule mod,
162 bool include_loops = true) {
163 auto creator = ffi::make_object<SRefTreeCreator>(include_loops);
164 for (const auto& kv : mod->functions) {
165 const BaseFunc& base_func = kv.second;
166 if (auto opt = base_func.as<PrimFunc>()) {
167 auto func = opt.value();
168 creator->Visit(func->body);
169 }
170 }
171 return std::move(creator->stmt2ref_);
172 }
173
174 explicit SRefTreeCreator(bool include_loops) : include_loops_(include_loops) {}
175
176 private:
181 void PushSRef(const StmtNode* stmt);
182
184 void PopAndRecordSRef();
185
186 ffi::Optional<VisitInterrupt> Visit_(const ForNode* loop) final;
187
188 ffi::Optional<VisitInterrupt> Visit_(const SBlockRealizeNode* realize) final;
189
190 ffi::Optional<VisitInterrupt> Visit_(const SeqStmtNode* seq_stmt) final;
191
192 bool include_loops_;
194 std::unordered_map<const StmtNode*, StmtSRef> stmt2ref_;
196 std::vector<StmtSRef> srefs_;
197};
