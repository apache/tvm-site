148class SRefTreeCreator : public s_tir::StmtExprVisitor {
149 public:
150 using s_tir::StmtExprVisitor::Visit_;
151
152 ffi::Optional<VisitInterrupt> Visit(ffi::AnyView value) override {
153 if (value.as<ExprNode>()) return std::nullopt;
154 return s_tir::StmtExprVisitor::Visit(value);
155 }
156
162 static std::unordered_map<const StmtNode*, StmtSRef> Create(IRModule mod,
163 bool include_loops = true) {
164 auto creator = ffi::make_object<SRefTreeCreator>(include_loops);
165 for (const auto& kv : mod->functions) {
166 const BaseFunc& base_func = kv.second;
167 if (auto opt = base_func.as<PrimFunc>()) {
168 auto func = opt.value();
169 creator->Visit(func->body);
170 }
171 }
172 return std::move(creator->stmt2ref_);
173 }
174
175 explicit SRefTreeCreator(bool include_loops) : include_loops_(include_loops) {}
176
177 private:
182 void PushSRef(const StmtNode* stmt);
183
185 void PopAndRecordSRef();
186
187 ffi::Optional<VisitInterrupt> Visit_(const ForNode* loop) final;
188
189 ffi::Optional<VisitInterrupt> Visit_(const s_tir::SBlockRealizeNode* realize) final;
190
191 ffi::Optional<VisitInterrupt> Visit_(const SeqStmtNode* seq_stmt) final;
192
193 bool include_loops_;
195 std::unordered_map<const StmtNode*, StmtSRef> stmt2ref_;
197 std::vector<StmtSRef> srefs_;
198};
