293 /* \brief Allow tirx.Var as key in STL tables
294  *
295  * For most TIR expressions, it would be ambiguous whether the
296  * expression should follow reference equality or structural equality.
297  * This is not the case for variables, which do not contain nested
298  * internal structure, and are frequently used as keys in lookup
299  * tables.
300  *
301  * Providing `std::hash` and `std::equal_to` specializations for
302  * `tirx::Var` allows it to be used as a key in STL tables. For
303  * `PrimExpr`, the user must specify the type of equality used
304  * (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>` or
305  * `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
306  */
307 template <>
308 struct std::hash<tvm::tirx::Var> {
309  std::size_t operator()(const tvm::tirx::Var& var) const { return tvm::ffi::ObjectPtrHash()(var); }
310 };
311 
312 template <>
313 struct std::equal_to<tvm::tirx::Var> {
314  bool operator()(const tvm::tirx::Var& var_a, const tvm::tirx::Var& var_b) const {
315  return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
316  }
317 };
