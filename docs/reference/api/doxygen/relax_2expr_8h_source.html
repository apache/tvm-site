405class SeqExprNode : public ExprNode {
406 public:
407 ffi::Array<BindingBlock> blocks;
408 Expr body;
409
410 static void RegisterReflection() {
411 namespace refl = tvm::ffi::reflection;
412 refl::ObjectDef<SeqExprNode>()
413 .def_ro("blocks", &SeqExprNode::blocks)
414 .def_ro("body", &SeqExprNode::body);
415 refl::TypeAttrDef<SeqExprNode>()
416 .def("__s_equal__", &SeqExprNode::SEqual)
417 .def("__s_hash__", &SeqExprNode::SHash);
418 }
419
420 bool SEqual(const SeqExprNode* other,
421 ffi::TypedFunction<bool(AnyView, AnyView, bool, AnyView)> equal) const {
422 // Establish mappings for symbolic variables defined by bindings before
423 // comparing their uses in the SeqExpr result type and body.
424 return equal(blocks, other->blocks, false, "blocks") && equal(ty, other->ty, false, "ty") &&
425 equal(body, other->body, false, "body");
426 }
427
428 int64_t SHash(int64_t init_hash, ffi::TypedFunction<int64_t(AnyView, int64_t, bool)> hash) const {
429 int64_t hash_value = init_hash;
430 hash_value = hash(blocks, hash_value, false);
431 hash_value = hash(ty, hash_value, false);
432 hash_value = hash(body, hash_value, false);
433 return hash_value;
434 }
435 TVM_FFI_DECLARE_OBJECT_INFO_FINAL("relax.expr.SeqExpr", SeqExprNode, ExprNode);
436};

438class SeqExpr : public Expr {
439 public:
440 /* \brief Implicit conversion constructor
441 *
442 * Relax nodes that introduce a new scope (e.g. `relax::Function`)
443 * are required to be held as SeqExpr. This implicit conversion
444 * provides allows callsites to use these member variables when the
445 * C++ compile-time type is a `relax::Expr`. For example,
446 * a transform may use `func.CopyOnWrite()->body = expr;`.

448 * If the expression is already a `relax::SeqExpr`, the same
