272
273 bool SEqual(const SeqExprNode* other,
274 ffi::TypedFunction<bool(AnyView, AnyView, bool, AnyView)> equal) const {
275 // Establish mappings for symbolic variables defined by bindings before
276 // comparing their uses in the SeqExpr result type and body.
277 return equal(blocks, other->blocks, false, "blocks") && equal(ty, other->ty, false, "ty") &&
278 equal(body, other->body, false, "body");
279 }
280
281 int64_t SHash(int64_t init_hash, ffi::TypedFunction<int64_t(AnyView, int64_t, bool)> hash) const {
282 int64_t hash_value = init_hash;
283 hash_value = hash(blocks, hash_value, false);
284 hash_value = hash(ty, hash_value, false);
285 hash_value = hash(body, hash_value, false);
286 return hash_value;
287 }
288 TVM_FFI_DECLARE_OBJECT_INFO_FINAL("relax.expr.SeqExpr", SeqExprNode, ExprNode);
289};

291class SeqExpr : public Expr {
292 public:
293 /* \brief Implicit conversion constructor
294 *
295 * Relax nodes that introduce a new scope (e.g. `relax::Function`)
296 * are required to be held as SeqExpr. This implicit conversion
297 * provides allows callsites to use these member variables when the
298 * C++ compile-time type is a `relax::Expr`. For example,
299 * a transform may use `func.CopyOnWrite()->body = expr;`.
300 *
301 * If the expression is already a `relax::SeqExpr`, the same
302 * underlying `relax::SeqExprNode` is used, and no copies are made.
