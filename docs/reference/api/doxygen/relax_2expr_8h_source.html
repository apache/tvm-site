735 /* \brief Allow relax.Var as key in STL tables
736  *
737  * For most Relax expressions, it would be ambiguous whether the
738  * expression should follow reference equality or structural equality.
739  * This is not the case for variables, which do not contain nested
740  * internal structure, and are frequently used as keys in lookup
741  * tables.
742  *
743  * Providing `std::hash` and `std::equal_to` specializations for
744  * `relax::Var` allows it to be used as a key in STL tables. For
745  * `relax::Expr`, the user must specify the type of equality used
746  * (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>` or
747  * `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
748  */
749 template <>
750 struct std::hash<tvm::relax::Var> {
751  std::size_t operator()(const tvm::relax::Var& var) const {
752  return tvm::ffi::ObjectPtrHash()(var);
753  }
754 };
755 
756 template <>
757 struct std::equal_to<tvm::relax::Var> {
758  bool operator()(const tvm::relax::Var& var_a, const tvm::relax::Var& var_b) const {
759  return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
760  }
761 };
