747 typename = typename std::enable_if<(std::is_standard_layout<ValueType>::value &&

752template <typename ValueType>
753inline PrimExpr MakeConstScalar(PrimType dtype, ValueType value, Location loc = UnknownLoc()) {
754 if constexpr (std::is_enum_v<ValueType>) {
755 return MakeConstScalar(dtype, static_cast<std::underlying_type_t<ValueType>>(value), loc);
756 } else {
757 DLDataTypeCode code = dtype.code();
758 if (code == DLDataTypeCode::kDLInt || code == DLDataTypeCode::kDLBool) {
759 return IntImm(dtype, ffi::BigInt(value), loc);
760 }
761 if (code == DLDataTypeCode::kDLUInt) {
762 TVM_FFI_ICHECK(value >= 0) << "cannot make uint from negative value " << value;
763 return IntImm(dtype, ffi::BigInt(value), loc);
764 }
765 if (dtype.MatchesCode(DLDataTypeCode::kDLFloat, DLDataTypeCode::kDLFloat8_e3m4,
766 DLDataTypeCode::kDLFloat8_e4m3, DLDataTypeCode::kDLFloat8_e4m3b11fnuz,
767 DLDataTypeCode::kDLFloat8_e4m3fn, DLDataTypeCode::kDLFloat8_e4m3fnuz,
768 DLDataTypeCode::kDLFloat8_e5m2, DLDataTypeCode::kDLFloat8_e5m2fnuz,
769 DLDataTypeCode::kDLFloat8_e8m0fnu, DLDataTypeCode::kDLFloat6_e2m3fn,
770 DLDataTypeCode::kDLFloat6_e3m2fn, DLDataTypeCode::kDLFloat4_e2m1fn) ||
771 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) {
772 return FloatImm(dtype, static_cast<double>(value), loc);
773 }
774 TVM_FFI_THROW(InternalError) << "cannot make const for type " << dtype;
775 throw;
776 }

784template <typename ValueType, typename>
785inline PrimExpr MakeConst(PrimType dtype, ValueType value, Location loc) {
786 if (!dtype.IsScalableVector() && !dtype.IsFixedLengthVector()) {
787 return MakeConstScalar(dtype, value, loc);
788 }
789 PrimType elem_ty = dtype.WithLanes(1);
790 if (dtype.IsFixedLengthVector()) {
791 return prim::Broadcast(MakeConstScalar(elem_ty, value, loc), dtype.lanes(), loc);
792 }
793 PrimExpr lanes = prim::Mul(Call(PrimType::Int(32), prim::vscale_op(), {}).as_or_throw<PrimExpr>(),
794 dtype.VScaleFactor());
795 return prim::Broadcast(MakeConstScalar(elem_ty, value, loc), lanes, loc);
796}

904 "Checkout these functions in ir/prim/op.h.");

1200// Intrinsic operators
1201#define TVM_DECLARE_INTRIN_UNARY_WITH_CHECK(OpName, CheckInputDType) \
1202 inline PrimExpr OpName(PrimExpr x, Location loc = UnknownLoc()) { \
1203 static const Op op = Op::Get("prim." #OpName); \
1204 PrimType x_ty = x.ty(); \
1205 CheckInputDType(#OpName, x_ty); \
1206 if (x_ty.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) { \
1207 PrimType bf16_ty = x_ty; \
1208 PrimType f32_ty = \
1209 x_ty.IsScalableVector() \
1210 ? PrimType::ScalableVector(DLDataTypeCode::kDLFloat, 32, x_ty.VScaleFactor()) \
1211 : PrimType::Float(32, x_ty.lanes()); \
1212 PrimExpr x_fp32 = prim::Cast(f32_ty, x, loc); \
1213 PrimExpr result_fp32 = Call(f32_ty, op, {x_fp32}, {}, {}, loc).as_or_throw<PrimExpr>(); \
1214 return prim::Cast(bf16_ty, result_fp32, loc); \
1215 } else { \
1216 return Call(x_ty, op, {x}, {}, {}, loc).as_or_throw<PrimExpr>(); \
1217 } \
1218 }
