75 virtual R VisitType(const Type& n, Args... args) {
76 TVM_FFI_ICHECK(n.defined());
77 TVM_FFI_ICHECK(!n.as<MissingType>().has_value()) << "TypeFunctor cannot visit Type::Missing()";
