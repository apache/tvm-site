416inline ffi::Optional<ffi::String> GetThreadBinding(const ForObj* loop) {
417 if (auto tag = loop->annotations.Get(tvm::s_tir::attr::kThreadBinding.name)) {
418 TVM_FFI_CHECK(loop->kind == ForKind::kParallel, ValueError)
419 << "thread_binding requires a parallel loop";
420 TVM_FFI_CHECK(tag->as<ffi::String>().has_value(), TypeError)
421 << "thread_binding annotation must be a string";
422 TVM_FFI_CHECK(loop->HasTrivialStep(), ValueError) << "Thread binding loops require a unit step";
423 return tag->cast<ffi::String>();
424 }
425 return std::nullopt;
426}
427inline ffi::Optional<ffi::String> GetThreadBinding(const For& loop) {
428 return GetThreadBinding(loop.get());
429}
