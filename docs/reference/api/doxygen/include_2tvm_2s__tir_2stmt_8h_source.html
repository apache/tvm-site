417inline ffi::Optional<ffi::String> GetThreadBinding(const ForNode* loop) {
418 if (auto tag = loop->annotations.Get(tvm::s_tir::attr::kThreadBinding.name)) {
419 TVM_FFI_CHECK(loop->kind == ForKind::kParallel, ValueError)
420 << "thread_binding requires a parallel loop";
421 TVM_FFI_CHECK(tag->as<ffi::String>().has_value(), TypeError)
422 << "thread_binding annotation must be a string";
423 TVM_FFI_CHECK(loop->HasTrivialStep(), ValueError) << "Thread binding loops require a unit step";
424 return tag->cast<ffi::String>();
425 }
426 return std::nullopt;
427}
428inline ffi::Optional<ffi::String> GetThreadBinding(const For& loop) {
429 return GetThreadBinding(loop.get());
430}
