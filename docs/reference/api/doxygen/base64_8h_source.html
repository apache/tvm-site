104class Base64InStream : public tvm::support::Stream {
105 public:
106 explicit Base64InStream(tvm::support::Stream* fs) : reader_(256) { reader_.set_stream(fs); }
111 void InitPosition(void) {
112 // get a character
113 do {
114 temp_ch_ = reader_.GetChar();
115 } while (isspace(temp_ch_));
116 }
118 bool IsEOF(void) const { return num_prev_ == 0 && (temp_ch_ == EOF || isspace(temp_ch_)); }
119
120 using tvm::support::Stream::Read;
121 // override read function.
122 size_t Read(void* ptr, size_t size) final {
123 using base64::DecodeTable;
124 if (size == 0) return 0;
125 // use tlen to record left size
126 size_t tlen = size;
127 unsigned char* cptr = static_cast<unsigned char*>(ptr);
128 // if anything left, load from previous buffered result
129 if (num_prev_ != 0) {
130 if (num_prev_ == 2) {
131 if (tlen >= 2) {
132 *cptr++ = buf_prev[0];
133 *cptr++ = buf_prev[1];
134 tlen -= 2;
135 num_prev_ = 0;
136 } else {
137 // assert tlen == 1
138 *cptr++ = buf_prev[0];
139 --tlen;
140 buf_prev[0] = buf_prev[1];
141 num_prev_ = 1;
142 }
143 } else {
144 // assert num_prev_ == 1
145 *cptr++ = buf_prev[0];
146 --tlen;
147 num_prev_ = 0;
148 }
149 }
150 if (tlen == 0) return size;
151 int nvalue;
152 // note: everything goes with 4 bytes in Base64
153 // so we process 4 bytes a unit
154 while (tlen && temp_ch_ != EOF && !isspace(temp_ch_)) {
155 // first byte
156 nvalue = DecodeTable[temp_ch_] << 18;
157 {
158 // second byte
159 temp_ch_ = reader_.GetChar();
160 TVM_FFI_ICHECK(temp_ch_ != EOF && !isspace(temp_ch_)) << "invalid base64 format";
161 nvalue |= DecodeTable[temp_ch_] << 12;
162 *cptr++ = (nvalue >> 16) & 0xFF;
163 --tlen;
164 }
165 {
166 // third byte
167 temp_ch_ = reader_.GetChar();
168 TVM_FFI_ICHECK(temp_ch_ != EOF && !isspace(temp_ch_)) << "invalid base64 format";
169 // handle termination
170 if (temp_ch_ == '=') {
171 temp_ch_ = reader_.GetChar();
172 TVM_FFI_ICHECK(temp_ch_ == '=') << "invalid base64 format";
173 temp_ch_ = reader_.GetChar();
174 TVM_FFI_ICHECK(temp_ch_ == EOF || isspace(temp_ch_)) << "invalid base64 format";
175 break;
176 }
177 nvalue |= DecodeTable[temp_ch_] << 6;
178 if (tlen) {
179 *cptr++ = (nvalue >> 8) & 0xFF;
180 --tlen;
181 } else {
182 buf_prev[num_prev_++] = (nvalue >> 8) & 0xFF;
183 }
184 }
185 {
186 // fourth byte
187 temp_ch_ = reader_.GetChar();
188 TVM_FFI_ICHECK(temp_ch_ != EOF && !isspace(temp_ch_)) << "invalid base64 format";
189 if (temp_ch_ == '=') {
190 temp_ch_ = reader_.GetChar();
191 TVM_FFI_ICHECK(temp_ch_ == EOF || isspace(temp_ch_)) << "invalid base64 format";
192 break;
193 }
194 nvalue |= DecodeTable[temp_ch_];
195 if (tlen) {
196 *cptr++ = nvalue & 0xFF;
197 --tlen;
198 } else {
199 buf_prev[num_prev_++] = nvalue & 0xFF;
200 }
201 }
202 // get next char
203 temp_ch_ = reader_.GetChar();
204 }
205 if (kStrictCheck) {
206 TVM_FFI_ICHECK_EQ(tlen, 0) << "Base64InStream: read incomplete";
207 }
208 return size - tlen;
209 }
210 size_t Write(const void* ptr, size_t size) final {
211 TVM_FFI_THROW(InternalError) << "Base64InStream do not support write";
212 return 0;
213 }
214
215 private:
216 // internal reader
217 StreamBufferReader reader_;
218 int temp_ch_{0};
219 int num_prev_{0};
220 unsigned char buf_prev[2];
221 // whether we need to do strict check
222 static const bool kStrictCheck = false;
223};

228class Base64OutStream : public tvm::support::Stream {
229 public:
230 explicit Base64OutStream(tvm::support::Stream* fp) : fp_(fp) {}
231
232 using tvm::support::Stream::Write;
233
234 size_t Write(const void* ptr, size_t size) final {
235 using base64::EncodeTable;
236 size_t tlen = size;
237 const unsigned char* cptr = static_cast<const unsigned char*>(ptr);
238 while (tlen) {
239 while (buf__top_ < 3 && tlen != 0) {
240 buf_[++buf__top_] = *cptr++;
241 --tlen;
242 }
243 if (buf__top_ == 3) {
244 // flush 4 bytes out
245 PutChar(EncodeTable[buf_[1] >> 2]);
246 PutChar(EncodeTable[((buf_[1] << 4) | (buf_[2] >> 4)) & 0x3F]);
247 PutChar(EncodeTable[((buf_[2] << 2) | (buf_[3] >> 6)) & 0x3F]);
248 PutChar(EncodeTable[buf_[3] & 0x3F]);
249 buf__top_ = 0;
250 }
251 }
252 return size;
253 }
254 virtual size_t Read(void* ptr, size_t size) {
255 TVM_FFI_THROW(InternalError) << "Base64OutStream do not support read";
256 return 0;
257 }
262 void Finish(int endch = EOF) {
263 using base64::EncodeTable;
264 if (buf__top_ == 1) {
265 PutChar(EncodeTable[buf_[1] >> 2]);
266 PutChar(EncodeTable[(buf_[1] << 4) & 0x3F]);
267 PutChar('=');
268 PutChar('=');
269 }
270 if (buf__top_ == 2) {
271 PutChar(EncodeTable[buf_[1] >> 2]);
272 PutChar(EncodeTable[((buf_[1] << 4) | (buf_[2] >> 4)) & 0x3F]);
273 PutChar(EncodeTable[(buf_[2] << 2) & 0x3F]);
274 PutChar('=');
275 }
276 buf__top_ = 0;
277 if (endch != EOF) PutChar(endch);
278 this->Flush();
279 }
280
281 private:
282 static constexpr size_t kBufferSize = 256;
283
284 tvm::support::Stream* fp_{nullptr};
285 int buf__top_{0};
286 unsigned char buf_[4];
287 std::string out_buf_;
288
289 void PutChar(char ch) {
290 out_buf_ += ch;
291 if (out_buf_.length() >= kBufferSize) Flush();
292 }
293 void Flush(void) {
294 if (out_buf_.length() != 0) {
295 fp_->Write(&out_buf_[0], out_buf_.length());
296 out_buf_.clear();
297 }
298 }
