71template <typename T, bool dtype_from_arg = false>
72inline PrimExpr DispatchPureExtern(const PrimExpr& e) {
73 const CallNode* call = e.as<CallNode>();
74 TVM_FFI_ICHECK(call != nullptr);
75 // Both primitive and TIRx intrinsic names use a five-character namespace prefix.
76 const OpNode* op = call->op.as<OpNode>();
77 TVM_FFI_ICHECK(op != nullptr);
78 std::string name = op->name;
79 TVM_FFI_ICHECK(name.substr(0, 5) == "tirx." || name.substr(0, 5) == "prim.")
80 << "Unexpected intrinsic name: " << name;
81 if (dtype_from_arg) {
82 TVM_FFI_ICHECK_EQ(call->args.size(), 1U);
83 }
84 PrimType type = dtype_from_arg ? call->args[0].as_or_throw<PrimExpr>().ty()
85 : call->ty.as_or_throw<PrimType>();
86 name = T()(type, name.substr(5));
87
88 if (!name.empty()) {
89 ffi::Array<Expr> new_args = {StringImm(name)};
90 for (const PrimExpr& arg : call->args.as_or_throw<ffi::Array<PrimExpr>>()) {
91 new_args.push_back(arg);
92 }
93 return Call(e.ty(), tirx::call_pure_extern_op(), new_args).as_or_throw<PrimExpr>();
94 } else {
95 return e;
96 }
97}
