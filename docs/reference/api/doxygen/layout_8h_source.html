196 public:
197 explicit TileLayout(ffi::Array<Iter> shard, ffi::Array<Iter> replica,
198 ffi::Map<Axis, PrimExpr> offset)
199 : Layout(ffi::UnsafeInit{}) {
200 auto n = ffi::make_object<TileLayoutObj>();
201 n->shard = std::move(shard);
202 n->replica = std::move(replica);
203 n->offset = std::move(offset);
204 data_ = std::move(n);
205 }
206
208 ffi::Array<PrimExpr> GetShardShape() const {
209 return get()->shard.Map([](const Iter& iter) { return iter->extent; });
210 }
211
219 TVM_TIRX_DLL bool IsContiguous(ffi::Optional<sym::Analyzer> analyzer = std::nullopt) const;
220
222 bool IsTrivial() const {
223 if (get()->shard.size() > 1) return false;
224 if (get()->shard.size() == 1) {
225 if (!get()->shard[0]->axis->IsMemoryAxis() || !prim::IsOne(get()->shard[0]->stride))
226 return false;
227 }
228 return get()->replica.size() == 0 && get()->offset.size() == 0;
229 }
230
232 bool HasThreadAxis() const {
233 return std::any_of(get()->shard.begin(), get()->shard.end(),
234 [](const Iter& iter) { return iter->axis->IsThreadAxis(); });
235 }
236
238 TVM_TIRX_DLL ffi::Optional<Tuple<ThreadScope, ThreadScope>> GetScope() const;
239
241 TVM_TIRX_DLL static TileLayout DefaultLayout(ffi::Array<PrimExpr> shape);
