186 public:

189 : Layout(ffi::UnsafeInit{}) {
190 auto n = ffi::make_object<TileLayoutObj>();
191 n->shard = std::move(shard);
192 n->replica = std::move(replica);
193 n->offset = std::move(offset);
194 data_ = std::move(n);
195 }
196
198 ffi::Array<PrimExpr> GetShardShape() const {
199 return get()->shard.Map([](const Iter& iter) { return iter->extent; });
200 }
201
209 TVM_TIRX_DLL bool IsContiguous(ffi::Optional<sym::Analyzer> analyzer = std::nullopt) const;
210
212 bool IsTrivial() const {
213 if (get()->shard.size() > 1) return false;
214 if (get()->shard.size() == 1) {
215 if (!get()->shard[0]->axis->IsMemoryAxis() || !prim::IsOne(get()->shard[0]->stride))
216 return false;
217 }
218 return get()->replica.size() == 0 && get()->offset.size() == 0;
219 }
220
222 bool HasThreadAxis() const {
223 return std::any_of(get()->shard.begin(), get()->shard.end(),
224 [](const Iter& iter) { return iter->axis->IsThreadAxis(); });
225 }
226
228 TVM_TIRX_DLL ffi::Optional<Tuple<ThreadScope, ThreadScope>> GetScope() const;
229
231 TVM_TIRX_DLL static TileLayout DefaultLayout(ffi::Array<PrimExpr> shape);
