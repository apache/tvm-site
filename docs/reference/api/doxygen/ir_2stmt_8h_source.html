663inline const char* ForKind2String(ForKind t) {
664 switch (t) {
665 case ForKind::kDefault:
666 return "serial";
667 case ForKind::kParallel:
668 return "parallel";
669 case ForKind::kVectorized:
670 return "vectorized";
671 case ForKind::kUnrolled:
672 return "unroll";
673 }
674 TVM_FFI_THROW(InternalError) << "Unknown ForKind" << t;
675 TVM_FFI_UNREACHABLE();
676}
