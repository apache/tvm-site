659inline const char* ForKind2String(ForKind t) {
660 switch (t) {
661 case ForKind::kDefault:
662 return "serial";
663 case ForKind::kParallel:
664 return "parallel";
665 case ForKind::kVectorized:
666 return "vectorized";
667 case ForKind::kUnrolled:
668 return "unroll";

671 TVM_FFI_UNREACHABLE();
672}
