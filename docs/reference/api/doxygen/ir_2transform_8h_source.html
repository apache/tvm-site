150 template <typename ValueType>
151 static void RegisterConfigOption(const char* key) {
152 if constexpr (std::is_base_of_v<ffi::ObjectRef, ValueType>) {
153 // Static-init blocks may run before the cached runtime type index is initialized.
154 int32_t tindex = ValueType::ContainerType::_GetOrAllocRuntimeTypeIndex();
155 auto type_key = ffi::TypeIndexToTypeKey(tindex);
156 auto legalization = [=](ffi::Any value) -> ffi::Any {
157 if (auto opt_map = value.try_cast<ffi::Map<ffi::String, ffi::Any>>()) {
158 return ffi::reflection::ObjectCreator(type_key)(opt_map.value());
159 } else {
160 auto opt_val = value.try_cast<ValueType>();
161 if (!opt_val.has_value()) {
162 TVM_FFI_THROW(AttributeError)
163 << "Expect config " << key << " to have type " << type_key << ", but instead get "
164 << ffi::details::AnyUnsafe::GetMismatchTypeInfo<ValueType>(value);
165 }
166 return *opt_val;
167 }
168 };
169 RegisterConfigOption(key, type_key, legalization);
170 } else {
171 // non-object type, do not support implicit conversion from map
172 std::string type_str = ffi::TypeTraits<ValueType>::TypeStr();
173 auto legalization = [=](ffi::Any value) -> ffi::Any {
174 auto opt_val = value.try_cast<ValueType>();
175 if (!opt_val.has_value()) {
176 TVM_FFI_THROW(AttributeError)
177 << "Expect config " << key << " to have type " << type_str << ", but instead get "
178 << ffi::details::AnyUnsafe::GetMismatchTypeInfo<ValueType>(value);
179 } else {
180 return *opt_val;
181 }
182 };
183 RegisterConfigOption(key, type_str, legalization);
184 }
185 }

211template <typename TConfig>
212inline TConfig PassConfigWithDefaults() {
213 static_assert(std::is_base_of_v<ffi::ObjectRef, TConfig>,
214 "Can only create ObjectRef-derived types");
215 using ContainerType = typename TConfig::ContainerType;
216 static auto finit_object = ffi::Function::GetGlobalRequired("ffi.MakeObjectFromPackedArgs");
217 ffi::AnyView packed_args[1];
218 packed_args[0] = ContainerType::RuntimeTypeIndex();
219 ffi::Any rv;
220 finit_object.CallPacked(ffi::PackedArgs(packed_args, 1), &rv);
221 return rv.cast<TConfig>();
222}
