178 template <typename ValueType>
179 static int32_t RegisterConfigOption(const char* key) {
180 // NOTE: we could further update the function later.
181 if constexpr (std::is_base_of_v<ffi::ObjectRef, ValueType>) {
182 int32_t tindex = ffi::TypeToRuntimeTypeIndex<ValueType>::v();
183 auto type_key = ffi::TypeIndexToTypeKey(tindex);
184 auto legalization = [=](ffi::Any value) -> ffi::Any {
185 if (auto opt_map = value.try_cast<ffi::Map<ffi::String, ffi::Any>>()) {
186 return ffi::reflection::ObjectCreator(type_key)(opt_map.value());
187 } else {
188 auto opt_val = value.try_cast<ValueType>();
189 if (!opt_val.has_value()) {
190 TVM_FFI_THROW(AttributeError)
191 << "Expect config " << key << " to have type " << type_key << ", but instead get "
192 << ffi::details::AnyUnsafe::GetMismatchTypeInfo<ValueType>(value);
193 }
194 return *opt_val;
195 }
196 };
197 RegisterConfigOption(key, type_key, legalization);
198 } else {
199 // non-object type, do not support implicit conversion from map
200 std::string type_str = ffi::TypeTraits<ValueType>::TypeStr();
201 auto legalization = [=](ffi::Any value) -> ffi::Any {
202 auto opt_val = value.try_cast<ValueType>();
203 if (!opt_val.has_value()) {
204 TVM_FFI_THROW(AttributeError)
205 << "Expect config " << key << " to have type " << type_str << ", but instead get "
206 << ffi::details::AnyUnsafe::GetMismatchTypeInfo<ValueType>(value);
207 } else {
208 return *opt_val;
209 }
210 };
211 RegisterConfigOption(key, type_str, legalization);
212 }
213 return 0;
214 }

240template <typename TConfig>
241inline TConfig PassConfigWithDefaults() {
242 static_assert(std::is_base_of_v<ffi::ObjectRef, TConfig>,
243 "Can only create ObjectRef-derived types");
244 using ContainerType = typename TConfig::ContainerType;
245 static auto finit_object = ffi::Function::GetGlobalRequired("ffi.MakeObjectFromPackedArgs");
246 ffi::AnyView packed_args[1];
247 packed_args[0] = ContainerType::RuntimeTypeIndex();
248 ffi::Any rv;
249 finit_object.CallPacked(ffi::PackedArgs(packed_args, 1), &rv);
250 return rv.cast<TConfig>();
251}
