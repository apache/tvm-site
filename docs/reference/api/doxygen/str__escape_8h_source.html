50inline std::string StrEscape(const char* data, size_t size, bool use_octal_escape = false,
51 bool escape_whitespace_special_chars = true) {
52 std::ostringstream stream;
53 for (size_t i = 0; i < size; ++i) {
54 unsigned char c = data[i];
55 if (c >= ' ' && c <= '~' && c != '\\' && c != '"') {
56 stream << c;
57 } else {
58 switch (c) {
59 case '"':
60 stream << '\\' << '"';
61 break;
62 case '\\':
63 stream << '\\' << '\\';
64 break;
65 case '\t':
66 if (escape_whitespace_special_chars) {
67 stream << '\\' << 't';
68 } else {
69 stream << c;
70 }
71 break;
72 case '\r':
73 if (escape_whitespace_special_chars) {
74 stream << '\\' << 'r';
75 } else {
76 stream << c;
77 }
78 break;
79 case '\n':
80 if (escape_whitespace_special_chars) {
81 stream << '\\' << 'n';
82 } else {
83 stream << c;
84 }
85 break;
86 default:
87 if (use_octal_escape) {
88 stream << '\\' << static_cast<unsigned char>('0' + ((c >> 6) & 0x03))
89 << static_cast<unsigned char>('0' + ((c >> 3) & 0x07))
90 << static_cast<unsigned char>('0' + (c & 0x07));
91 } else {
92 const char* hex_digits = "0123456789ABCDEF";
93 stream << '\\' << 'x' << hex_digits[c >> 4] << hex_digits[c & 0xf];
94 }
95 }
96 }
97 }
98 return stream.str();
99}
