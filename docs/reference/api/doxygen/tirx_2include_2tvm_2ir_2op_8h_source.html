248template <typename T>
249inline bool ReportTypeMismatch(ffi::Expected<void>* out, const ValidateState* state, bool type_arg,
250 size_t index, ffi::AnyView actual) {
251 const auto& fixed = type_arg ? state->op->ty_args_info : state->op->args_info;
252 const auto& tail = type_arg ? state->op->var_ty_args_info : state->op->var_args_info;
253 std::string name;
254 if (index < fixed.size()) {
255 name = fixed[index]->name;
256 } else if (tail.has_value()) {
257 name = tail.value()->name;
258 }
259 std::string name_clause = name.empty() ? "" : " (`" + name + "`)";
260 TVMFFIAny raw = actual.CopyToTVMFFIAny();
261 *out = TVM_FFI_UNEXPECTED(TypeError)
262 << "Op `" << state->op->name << "`: `" << (type_arg ? "Call.ty_args[" : "Call.args[")
263 << index << "]`" << name_clause << " expected `" << ffi::TypeTraits<T>::TypeStr()
264 << "`, got `" << ffi::TypeTraits<T>::GetMismatchTypeInfo(&raw) << "`.";
265 return false;
266}

637 template <typename... Specs>
638 TVM_FFI_INLINE static ffi::Expected<void> ValidateSignature(const CallNode* call) noexcept {
639 static_assert((std::is_base_of_v<sig::SignatureTrait, Specs> && ...),
640 "Unknown signature descriptor");
641 constexpr size_t n_args = (size_t{0} + ... + Specs::kArgsCount);
642 constexpr size_t n_ty_args = (size_t{0} + ... + Specs::kTyArgsCount);
643 constexpr size_t n_var_args = (size_t{0} + ... + Specs::kVarArgsCount);
644 constexpr size_t n_var_ty_args = (size_t{0} + ... + Specs::kVarTyArgsCount);
645 constexpr size_t n_attrs = (size_t{0} + ... + Specs::kCallAttrsCount);
646 static_assert(n_var_args <= 1 && n_var_ty_args <= 1 && n_attrs <= 1,
647 "Duplicate signature tail or call_attrs");
648 const OpNode* op = call ? call->op.as<OpNode>() : nullptr;
649 if (TVM_FFI_PREDICT_FALSE(!op)) {
650 return TVM_FFI_UNEXPECTED(TypeError) << "Invalid Op Call";
651 }
652 if (TVM_FFI_PREDICT_FALSE(!call->args.defined() || !call->ty_args.defined())) {
653 return TVM_FFI_UNEXPECTED(TypeError) << "Op `" << op->name << "`: invalid Call";
654 }
655 ffi::Expected<void> result;
656 auto* out = &result;
657 if (TVM_FFI_PREDICT_FALSE(n_var_args ? call->args.size() < n_args
658 : call->args.size() != n_args)) {
659 *out = TVM_FFI_UNEXPECTED(TypeError)
660 << "Op `" << op->name << "`: Call.args expected " << (n_var_args ? "at least " : "")
661 << n_args << (n_args == 1 ? " argument" : " arguments") << ", got "
662 << call->args.size();
663 return result;
664 }
665 if (TVM_FFI_PREDICT_FALSE(n_var_ty_args ? call->ty_args.size() < n_ty_args
666 : call->ty_args.size() != n_ty_args)) {
667 *out = TVM_FFI_UNEXPECTED(TypeError)
668 << "Op `" << op->name << "`: Call.ty_args expected "
669 << (n_var_ty_args ? "at least " : "") << n_ty_args
670 << (n_ty_args == 1 ? " type argument" : " type arguments") << ", got "
671 << call->ty_args.size();
672 return result;
673 }
674 if constexpr (sizeof...(Specs) != 0) {
675 sig::ValidateState state{call, op};
676 if (TVM_FFI_PREDICT_FALSE(!(Specs::Validate(&state, out) && ...))) {
677 return result;
678 }
679 }
680 return result;
681 }
