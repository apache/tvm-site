385 static void RegisterReflection() {
386 // ffi::Functions are all not registered, because the reflection system doesn't take care of
387 // them, so it cannot be accessible on the python side. If there is such need from the future,
388 // we can then add corresponding accessor methods to help access on python.
389 // `f_has_workload` is not registered
390 // `f_commit_workload` is not registered
391 // `f_commit_tuning_record` is not registered
392 // `f_get_top_k` is not registered
393 // `f_get_all_tuning_records` is not registered
394 // `f_query_tuning_record` is not registered
395 // `f_query_schedule` is not registered
396 // `f_query_ir_module` is not registered
397 // `f_size` is not registered
398 namespace refl = tvm::ffi::reflection;
399 refl::ObjectDef<PyDatabaseObj>();
400 }
