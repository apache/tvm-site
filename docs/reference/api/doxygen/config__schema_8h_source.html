121 ConfigMap Resolve(ConfigMap config) const {
122 ConfigMap result;
123
124 // Step 1: validate/coerce and materialize options in declaration order
125 for (const auto& e : options_) {
126 auto it = config.find(e.key);
127 if (it != config.end()) {
128 result.Set(e.key, e.validator((*it).second));
129 } else if (e.has_default) {
130 if (e.default_from_factory) {
131 result.Set(e.key, e.default_value.cast<ffi::Function>()());
132 } else {
133 result.Set(e.key, e.default_value);
134 }
135 }
136 // else: missing non-required option, stays absent
137 }
138
139 // Step 2: unknown-key policy
140 if (error_on_unknown_) {
141 for (const auto& kv : config) {
142 if (!key_to_index_.count(std::string(kv.first))) {
143 std::ostringstream os;
144 os << "Unknown config option '" << kv.first << "'. Known options: ";
145 bool first = true;
146 for (const auto& e : options_) {
147 if (!first) os << ", ";
148 os << "'" << e.key << "'";
149 first = false;
150 }
151 TVM_FFI_THROW(ValueError) << os.str();
152 }
153 }
154 }
155
156 // Step 3: whole-object canonicalization (last)
157 if (canonicalizer_ != nullptr) {
158 result = canonicalizer_(result);
159 }
160
161 return result;
162 }

174 template <typename T>
175 static ffi::TypedFunction<ffi::Any(ffi::Any)> MakeValidator(const ffi::String& key) {
176 return ffi::TypedFunction<ffi::Any(ffi::Any)>([key](ffi::Any val) -> ffi::Any {
177 auto opt = val.try_cast<T>();
178 if (!opt.has_value()) {
179 TVM_FFI_THROW(TypeError) << "Option '" << key << "': expected type '"
180 << ffi::TypeTraits<T>::TypeStr() << "' but got '"
181 << val.GetTypeKey() << "'";
182 }
183 return ffi::Any(opt.value());
184 });

187 template <typename Trait>
188 static void ApplyTrait(OptionEntry* entry, ffi::reflection::FieldInfoBuilder* info,
189 Trait&& trait) {
190 using T = std::decay_t<Trait>;
191 if constexpr (std::is_same_v<T, AttrValidator>) {
192 entry->validator = std::move(trait.func);
193 } else if constexpr (std::is_base_of_v<ffi::reflection::InfoTrait, T>) {
194 trait.Apply(info);
195 } else if constexpr (std::is_same_v<T, const char*> || std::is_same_v<T, char*>) {
196 const char* doc = trait;
197 if (doc != nullptr && doc[0] != '\0') {
198 info->doc = TVMFFIByteArray{doc, std::char_traits<char>::length(doc)};
199 }
200 }
201 }

203 template <typename T, typename... Traits>
204 OptionEntry MakeEntry(const ffi::String& key, Traits&&... traits) {
205 OptionEntry e;
206 e.key = key;
207 e.type_str = ffi::String(ffi::TypeTraits<T>::TypeStr());
208 e.validator = MakeValidator<T>(key);
209 // Apply traits through a temporary FieldInfoBuilder so existing
210 // reflection traits (notably refl::DefaultValue) are reused unchanged.
211 ffi::reflection::FieldInfoBuilder info{};
212 info.flags = 0;
213 info.default_value_or_factory = ffi::AnyView(nullptr).CopyToTVMFFIAny();
214 info.doc = TVMFFIByteArray{nullptr, 0};
215 (ApplyTrait(&e, &info, std::forward<Traits>(traits)), ...);
216 if (info.flags & kTVMFFIFieldFlagBitMaskHasDefault) {
217 e.has_default = true;
218 e.default_from_factory = (info.flags & kTVMFFIFieldFlagBitMaskDefaultFromFactory) != 0;
219 e.default_value = ffi::AnyView::CopyFromTVMFFIAny(info.default_value_or_factory);
220 // Release the extra ref created by CopyToTVMFFIAny in Apply
221 if (info.default_value_or_factory.type_index >= TVMFFITypeIndex::kTVMFFIStaticObjectBegin) {
222 ffi::details::ObjectUnsafe::DecRefObjectHandle(info.default_value_or_factory.v_obj);
223 }
224 }
225 return e;
226 }
