734template <>
735struct TypeTraits<PrimVar> : public ObjectRefTypeTraitsBase<PrimVar> {
736 using Base = ObjectRefTypeTraitsBase<PrimVar>;
737 using Base::CopyFromAnyViewAfterCheck;
738 using Base::CopyToAnyView;
739 using Base::GetMismatchTypeInfo;
740 using Base::MoveFromAnyAfterCheck;
741 using Base::MoveToAny;
742 using Base::TypeSchema;
743 using Base::TypeStr;
744
745 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
746 if (src->type_index == TypeIndex::kTVMFFINone) {
747 return PrimVar::_type_is_nullable;
748 }
749 if (src->type_index != VarNode::RuntimeTypeIndex()) {
750 return false;
751 }
752 const auto* var = static_cast<const VarNode*>(
753 details::ObjectUnsafe::ObjectPtrFromUnowned<Object>(src->v_obj).get());
754 return details::AnyUnsafe::CheckAnyStrict<PrimType>(var->ExprNode::ty);
755 }
756
757 TVM_FFI_INLINE static std::optional<PrimVar> TryCastFromAnyView(const TVMFFIAny* src) {
758 if (CheckAnyStrict(src)) {
759 if (src->type_index == TypeIndex::kTVMFFINone) {
760 return details::ObjectUnsafe::ObjectRefFromObjectPtr<PrimVar>(nullptr);
761 }
762 return details::ObjectUnsafe::ObjectRefFromObjectPtr<PrimVar>(
763 details::ObjectUnsafe::ObjectPtrFromUnowned<VarNode>(src->v_obj));
764 }
765 return std::nullopt;
766 }
767};

806/* \brief Allow tvm.Var and tvm.GlobalVar as keys in STL tables
807 *
808 * For most IR expressions, it would be ambiguous whether the
809 * expression should follow reference equality or structural equality.
810 * This is not the case for variables, which do not contain nested
811 * internal structure, and are frequently used as keys in lookup
812 * tables.
813 *
814 * Providing `std::hash` and `std::equal_to` specializations for
815 * `tvm::Var` and `tvm::GlobalVar` allows them to be used as keys in STL tables. For
816 * other IR expressions, the user must specify the type of equality
817 * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
818 * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
819 */
820template <>
821struct std::hash<tvm::Var> {
822 std::size_t operator()(const tvm::Var& var) const { return tvm::ffi::ObjectPtrHash()(var); }
823};
824
825template <>
826struct std::equal_to<tvm::Var> {
827 bool operator()(const tvm::Var& var_a, const tvm::Var& var_b) const {
828 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
829 }
830};
831
832template <>
833struct std::hash<tvm::GlobalVar> {
834 std::size_t operator()(const tvm::GlobalVar& var) const { return tvm::ffi::ObjectPtrHash()(var); }
835};
836
837template <>
838struct std::equal_to<tvm::GlobalVar> {
839 bool operator()(const tvm::GlobalVar& var_a, const tvm::GlobalVar& var_b) const {
840 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
841 }
842};
