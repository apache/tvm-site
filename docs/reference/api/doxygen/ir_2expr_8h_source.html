312 return TypeTraits<TypedExpr<PrimType>>::GetMismatchTypeInfo(src);

316 return IntImm::Bool(value);

395// Use ordinary inline: array elements recursively invoke this conversion.
396inline std::optional<Expr> ffi::TypeTraits<Expr>::TryCastFromAnyView(const TVMFFIAny* src) {
397 if (auto value =
398 ObjectRefWithFallbackTraitsBase<Expr, PrimExpr, ffi::String>::TryCastFromAnyView(src)) {
399 return value;
400 }
401 if (auto fields = ffi::TypeTraits<ffi::Array<Expr>>::TryCastFromAnyView(src)) {
402 return tvm::Tuple(std::move(*fields));
403 }
404 return std::nullopt;
405}

1199/* \brief Allow tvm.Var and tvm.GlobalVar as keys in STL tables
1200 *
1201 * For most IR expressions, it would be ambiguous whether the
1202 * expression should follow reference equality or structural equality.
1203 * This is not the case for variables, which do not contain nested
1204 * internal structure, and are frequently used as keys in lookup
1205 * tables.
1206 *
1207 * Providing `std::hash` and `std::equal_to` specializations for
1208 * `tvm::Var` and `tvm::GlobalVar` allows them to be used as keys in STL tables. For
1209 * other IR expressions, the user must specify the type of equality
1210 * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
1211 * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
1212 */

1216};

1223};
