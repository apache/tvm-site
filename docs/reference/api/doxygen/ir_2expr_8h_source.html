674template <>
675struct TypeTraits<PrimVar> : public ObjectRefTypeTraitsBase<PrimVar> {
676 using Base = ObjectRefTypeTraitsBase<PrimVar>;
677 using Base::CopyFromAnyViewAfterCheck;
678 using Base::CopyToAnyView;
679 using Base::GetMismatchTypeInfo;
680 using Base::MoveFromAnyAfterCheck;
681 using Base::MoveToAny;
682 using Base::TypeSchema;
683 using Base::TypeStr;
684
685 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
686 if (src->type_index == TypeIndex::kTVMFFINone) {
687 return PrimVar::_type_is_nullable;
688 }
689 if (src->type_index != VarNode::RuntimeTypeIndex()) {
690 return false;
691 }
692 const auto* var = static_cast<const VarNode*>(
693 details::ObjectUnsafe::ObjectPtrFromUnowned<Object>(src->v_obj).get());
694 return details::AnyUnsafe::CheckAnyStrict<PrimType>(var->ExprNode::ty);
695 }
696
697 TVM_FFI_INLINE static std::optional<PrimVar> TryCastFromAnyView(const TVMFFIAny* src) {
698 if (CheckAnyStrict(src)) {
699 if (src->type_index == TypeIndex::kTVMFFINone) {
700 return details::ObjectUnsafe::ObjectRefFromObjectPtr<PrimVar>(nullptr);
701 }
702 return details::ObjectUnsafe::ObjectRefFromObjectPtr<PrimVar>(
703 details::ObjectUnsafe::ObjectPtrFromUnowned<VarNode>(src->v_obj));
704 }
705 return std::nullopt;
706 }
707};

745/* \brief Allow tvm.Var and tvm.GlobalVar as keys in STL tables
746 *
747 * For most IR expressions, it would be ambiguous whether the
748 * expression should follow reference equality or structural equality.
749 * This is not the case for variables, which do not contain nested
750 * internal structure, and are frequently used as keys in lookup
751 * tables.
752 *
753 * Providing `std::hash` and `std::equal_to` specializations for
754 * `tvm::Var` and `tvm::GlobalVar` allows them to be used as keys in STL tables. For
755 * other IR expressions, the user must specify the type of equality
756 * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
757 * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
758 */
759template <>
760struct std::hash<tvm::Var> {
761 std::size_t operator()(const tvm::Var& var) const { return tvm::ffi::ObjectPtrHash()(var); }
762};
763
764template <>
765struct std::equal_to<tvm::Var> {
766 bool operator()(const tvm::Var& var_a, const tvm::Var& var_b) const {
767 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
768 }
769};
770
771template <>
772struct std::hash<tvm::GlobalVar> {
773 std::size_t operator()(const tvm::GlobalVar& var) const { return tvm::ffi::ObjectPtrHash()(var); }
774};
775
776template <>
777struct std::equal_to<tvm::GlobalVar> {
778 bool operator()(const tvm::GlobalVar& var_a, const tvm::GlobalVar& var_b) const {
779 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
780 }
781};
