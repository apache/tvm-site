394inline std::optional<Expr> ffi::TypeTraits<Expr>::TryCastFromAnyView(const TVMFFIAny* src) {
395 if (auto value =
396 ObjectRefWithFallbackTraitsBase<Expr, PrimExpr, ffi::String>::TryCastFromAnyView(src)) {
397 return value;
398 }
399 if (auto fields = ffi::TypeTraits<ffi::Array<Expr>>::TryCastFromAnyView(src)) {
400 return tvm::Tuple(std::move(*fields));
401 }
402 return std::nullopt;
403}

1196/* \brief Allow tvm.Var and tvm.GlobalVar as keys in STL tables
1197 *
1198 * For most IR expressions, it would be ambiguous whether the
1199 * expression should follow reference equality or structural equality.
1200 * This is not the case for variables, which do not contain nested
1201 * internal structure, and are frequently used as keys in lookup
1202 * tables.
1203 *
1204 * Providing `std::hash` and `std::equal_to` specializations for
1205 * `tvm::Var` and `tvm::GlobalVar` allows them to be used as keys in STL tables. For
1206 * other IR expressions, the user must specify the type of equality
1207 * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
1208 * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
1209 */

1213};

1217 bool operator()(const tvm::Var& var_a, const tvm::Var& var_b) const {
1218 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
1219 }
1220};

1222template <>
1223struct std::hash<tvm::GlobalVar> {
