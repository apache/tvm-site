719struct TypeTraits<PrimVar> : public ObjectRefTypeTraitsBase<PrimVar> {
720 using Base = ObjectRefTypeTraitsBase<PrimVar>;
721 using Base::CopyFromAnyViewAfterCheck;
722 using Base::CopyToAnyView;
723 using Base::GetMismatchTypeInfo;
724 using Base::MoveFromAnyAfterCheck;
725 using Base::MoveToAny;
726 using Base::TypeSchema;
727 using Base::TypeStr;
728
729 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
730 if (src->type_index == TypeIndex::kTVMFFINone) {
731 return PrimVar::_type_is_nullable;
732 }
733 if (src->type_index != VarNode::RuntimeTypeIndex()) {
734 return false;
735 }
736 const auto* var = static_cast<const VarNode*>(
737 details::ObjectUnsafe::ObjectPtrFromUnowned<Object>(src->v_obj).get());
738 return details::AnyUnsafe::CheckAnyStrict<PrimType>(var->ExprNode::ty);
739 }
740
741 TVM_FFI_INLINE static std::optional<PrimVar> TryCastFromAnyView(const TVMFFIAny* src) {
742 if (CheckAnyStrict(src)) {
743 if (src->type_index == TypeIndex::kTVMFFINone) {
744 return details::ObjectUnsafe::ObjectRefFromObjectPtr<PrimVar>(nullptr);
745 }
746 return details::ObjectUnsafe::ObjectRefFromObjectPtr<PrimVar>(
747 details::ObjectUnsafe::ObjectPtrFromUnowned<VarNode>(src->v_obj));
748 }
749 return std::nullopt;
750 }
751};

790/* \brief Allow tvm.Var and tvm.GlobalVar as keys in STL tables
791 *
792 * For most IR expressions, it would be ambiguous whether the
793 * expression should follow reference equality or structural equality.
794 * This is not the case for variables, which do not contain nested
795 * internal structure, and are frequently used as keys in lookup
796 * tables.
797 *
798 * Providing `std::hash` and `std::equal_to` specializations for
799 * `tvm::Var` and `tvm::GlobalVar` allows them to be used as keys in STL tables. For
800 * other IR expressions, the user must specify the type of equality
801 * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
802 * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
803 */
804template <>
805struct std::hash<tvm::Var> {
806 std::size_t operator()(const tvm::Var& var) const { return tvm::ffi::ObjectPtrHash()(var); }
807};
808
809template <>
810struct std::equal_to<tvm::Var> {
811 bool operator()(const tvm::Var& var_a, const tvm::Var& var_b) const {
812 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
813 }
814};
815
816template <>
817struct std::hash<tvm::GlobalVar> {
818 std::size_t operator()(const tvm::GlobalVar& var) const { return tvm::ffi::ObjectPtrHash()(var); }
819};
820
821template <>
822struct std::equal_to<tvm::GlobalVar> {
823 bool operator()(const tvm::GlobalVar& var_a, const tvm::GlobalVar& var_b) const {
824 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
825 }
826};
