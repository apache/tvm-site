539enum class CallEffectKind : int {

546 kPure = 1,

563 kSpecialCallArg = 4,

574inline std::ostream& operator<<(std::ostream& os, CallEffectKind side_effect) {
575 switch (side_effect) {
576 case CallEffectKind::kExprAnnotation:
577 return os << "kExprAnnotation";
578
579 case CallEffectKind::kPure:
580 return os << "kPure";
581
582 case CallEffectKind::kReadState:
583 return os << "kReadState";
584
585 case CallEffectKind::kUpdateState:
586 return os << "kUpdateState";
587
588 case CallEffectKind::kSpecialCallArg:
589 return os << "kSpecialCallArg";
590
591 case CallEffectKind::kEmbedInfo:
592 return os << "kEmbedInfo";
593
594 case CallEffectKind::kControlJump:
595 return os << "kControlJump";
596
597 default:
598 TVM_FFI_THROW(InternalError) << "Unknown CallEffectKind: " << static_cast<int>(side_effect);
599 }

980struct TypeTraits<PrimVar> : public ObjectRefTypeTraitsBase<PrimVar> {
981 using Base = ObjectRefTypeTraitsBase<PrimVar>;
982 using Base::CopyFromAnyViewAfterCheck;
983 using Base::CopyToAnyView;
984 using Base::GetMismatchTypeInfo;
985 using Base::MoveFromAnyAfterCheck;
986 using Base::MoveToAny;
987 using Base::TypeSchema;

989
990 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
991 if (src->type_index == TypeIndex::kTVMFFINone) {
992 return PrimVar::_type_is_nullable;
993 }
994 if (src->type_index != VarNode::RuntimeTypeIndex()) {
995 return false;
996 }
997 const auto* var = static_cast<const VarNode*>(
998 details::ObjectUnsafe::ObjectPtrFromUnowned<Object>(src->v_obj).get());
999 return details::AnyUnsafe::CheckAnyStrict<PrimType>(var->ExprNode::ty);
1000 }
1001
1002 TVM_FFI_INLINE static std::optional<PrimVar> TryCastFromAnyView(const TVMFFIAny* src) {
1003 if (CheckAnyStrict(src)) {
1004 if (src->type_index == TypeIndex::kTVMFFINone) {
1005 return details::ObjectUnsafe::ObjectRefFromObjectPtr<PrimVar>(nullptr);
1006 }
1007 return details::ObjectUnsafe::ObjectRefFromObjectPtr<PrimVar>(
1008 details::ObjectUnsafe::ObjectPtrFromUnowned<VarNode>(src->v_obj));
1009 }
1010 return std::nullopt;
1011 }

1061/* \brief Allow tvm.Var and tvm.GlobalVar as keys in STL tables
1062 *
1063 * For most IR expressions, it would be ambiguous whether the
1064 * expression should follow reference equality or structural equality.
1065 * This is not the case for variables, which do not contain nested
1066 * internal structure, and are frequently used as keys in lookup
1067 * tables.
1068 *
1069 * Providing `std::hash` and `std::equal_to` specializations for
1070 * `tvm::Var` and `tvm::GlobalVar` allows them to be used as keys in STL tables. For
1071 * other IR expressions, the user must specify the type of equality
1072 * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
1073 * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
1074 */
1075template <>
1076struct std::hash<tvm::Var> {
1077 std::size_t operator()(const tvm::Var& var) const { return tvm::ffi::ObjectPtrHash()(var); }
1078};
1079
1080template <>
1081struct std::equal_to<tvm::Var> {
1082 bool operator()(const tvm::Var& var_a, const tvm::Var& var_b) const {
1083 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
1084 }
1085};
1086
1087template <>
1088struct std::hash<tvm::GlobalVar> {
1089 std::size_t operator()(const tvm::GlobalVar& var) const { return tvm::ffi::ObjectPtrHash()(var); }
1090};
1091
1092template <>
1093struct std::equal_to<tvm::GlobalVar> {
1094 bool operator()(const tvm::GlobalVar& var_a, const tvm::GlobalVar& var_b) const {
1095 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
1096 }
1097};
