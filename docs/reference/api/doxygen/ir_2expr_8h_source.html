291template <>
292struct TypeTraits<PrimExpr>
293 : public ObjectRefWithFallbackTraitsBase<PrimExpr, StrictBool, int64_t, double> {

295 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
296 if (src->type_index == TypeIndex::kTVMFFINone) return PrimExpr::_type_is_nullable;
297 return TypeTraits<TypedExpr<PrimType>>::CheckAnyStrict(src);
298 }

301 // A required expression cannot accept None.

303 return Base::TryCastFromAnyView(src);
304 }

306 TVM_FFI_INLINE static std::string TypeStr() { return "ir.PrimExpr"; }
307
308 TVM_FFI_INLINE static std::string GetMismatchTypeInfo(const TVMFFIAny* src) {
309 return TypeTraits<TypedExpr<PrimType>>::GetMismatchTypeInfo(src);
310 }
311
312 TVM_FFI_INLINE static PrimExpr ConvertFallbackValue(StrictBool value) {
313 return IntImm::Bool(value);
314 }
315 TVM_FFI_INLINE static PrimExpr ConvertFallbackValue(int64_t value) {
316 return TypeTraits<IntImm>::ConvertFallbackValue(value);

326template <>
327struct TypeTraits<IntExpr> : public ObjectRefWithFallbackTraitsBase<IntExpr, int64_t> {
328 using Base = ObjectRefWithFallbackTraitsBase<IntExpr, int64_t>;
329
330 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
331 if (!Base::CheckAnyStrict(src)) return false;
332 if (src->type_index == TypeIndex::kTVMFFINone) return true;

336 (ty->dtype.code == kDLInt || ty->dtype.code == kDLUInt);
337 }

340 return TypeTraits<PrimExpr>::ConvertFallbackValue(value).as_or_throw<IntExpr>();
341 }

392// Use ordinary inline: array elements recursively invoke this conversion.
393inline std::optional<Expr> ffi::TypeTraits<Expr>::TryCastFromAnyView(const TVMFFIAny* src) {
394 if (auto value =
395 ObjectRefWithFallbackTraitsBase<Expr, PrimExpr, ffi::String>::TryCastFromAnyView(src)) {
396 return value;
397 }
398 if (auto fields = ffi::TypeTraits<ffi::Array<Expr>>::TryCastFromAnyView(src)) {
399 return tvm::Tuple(std::move(*fields));
400 }
401 return std::nullopt;
402}

1148/* \brief Allow tvm.Var and tvm.GlobalVar as keys in STL tables
1149 *
1150 * For most IR expressions, it would be ambiguous whether the
1151 * expression should follow reference equality or structural equality.
1152 * This is not the case for variables, which do not contain nested
1153 * internal structure, and are frequently used as keys in lookup
1154 * tables.
1155 *
1156 * Providing `std::hash` and `std::equal_to` specializations for
1157 * `tvm::Var` and `tvm::GlobalVar` allows them to be used as keys in STL tables. For
1158 * other IR expressions, the user must specify the type of equality
1159 * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
1160 * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
1161 */

1165};

1171 }
1172};
