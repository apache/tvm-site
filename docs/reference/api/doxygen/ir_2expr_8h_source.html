538 /* \brief Allow tvm.GLobalVar as key in STL tables
539  *
540  * For most IR expressions, it would be ambiguous whether the
541  * expression should follow reference equality or structural equality.
542  * This is not the case for variables, which do not contain nested
543  * internal structure, and are frequently used as keys in lookup
544  * tables.
545  *
546  * Providing `std::hash` and `std::equal_to` specializations for
547  * `tvm::GlobalVar` allows it to be used as a key in STL tables. For
548  * other IR expressions, the user must specify the type of equality
549  * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
550  * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
551  */
552 template <>
553 struct std::hash<tvm::GlobalVar> {
554  std::size_t operator()(const tvm::GlobalVar& var) const { return tvm::ffi::ObjectPtrHash()(var); }
555 };
556 
557 template <>
558 struct std::equal_to<tvm::GlobalVar> {
559  bool operator()(const tvm::GlobalVar& var_a, const tvm::GlobalVar& var_b) const {
560  return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
561  }
562 };
