616 /* \brief Allow tvm.GLobalVar as key in STL tables
617  *
618  * For most IR expressions, it would be ambiguous whether the
619  * expression should follow reference equality or structural equality.
620  * This is not the case for variables, which do not contain nested
621  * internal structure, and are frequently used as keys in lookup
622  * tables.
623  *
624  * Providing `std::hash` and `std::equal_to` specializations for
625  * `tvm::GlobalVar` allows it to be used as a key in STL tables. For
626  * other IR expressions, the user must specify the type of equality
627  * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
628  * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
629  */
630 template <>
631 struct std::hash<tvm::GlobalVar> {
632  std::size_t operator()(const tvm::GlobalVar& var) const { return tvm::ffi::ObjectPtrHash()(var); }
633 };
634 
635 template <>
636 struct std::equal_to<tvm::GlobalVar> {
637  bool operator()(const tvm::GlobalVar& var_a, const tvm::GlobalVar& var_b) const {
638  return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
639  }
640 };
