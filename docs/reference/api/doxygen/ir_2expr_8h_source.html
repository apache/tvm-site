584/* \brief Allow tvm.Var and tvm.GlobalVar as keys in STL tables
585 *
586 * For most IR expressions, it would be ambiguous whether the
587 * expression should follow reference equality or structural equality.
588 * This is not the case for variables, which do not contain nested
589 * internal structure, and are frequently used as keys in lookup
590 * tables.
591 *
592 * Providing `std::hash` and `std::equal_to` specializations for
593 * `tvm::Var` and `tvm::GlobalVar` allows them to be used as keys in STL tables. For
594 * other IR expressions, the user must specify the type of equality
595 * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
596 * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
597 */
598template <>
599struct std::hash<tvm::Var> {
600 std::size_t operator()(const tvm::Var& var) const { return tvm::ffi::ObjectPtrHash()(var); }
601};
602
603template <>
604struct std::equal_to<tvm::Var> {
605 bool operator()(const tvm::Var& var_a, const tvm::Var& var_b) const {
606 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
607 }
608};
609
610template <>
611struct std::hash<tvm::GlobalVar> {
612 std::size_t operator()(const tvm::GlobalVar& var) const { return tvm::ffi::ObjectPtrHash()(var); }
613};
614
615template <>
616struct std::equal_to<tvm::GlobalVar> {
617 bool operator()(const tvm::GlobalVar& var_a, const tvm::GlobalVar& var_b) const {
618 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
619 }
620};
