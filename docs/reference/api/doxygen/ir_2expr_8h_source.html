643/* \brief Allow tvm.Var and tvm.GlobalVar as keys in STL tables
644 *
645 * For most IR expressions, it would be ambiguous whether the
646 * expression should follow reference equality or structural equality.
647 * This is not the case for variables, which do not contain nested
648 * internal structure, and are frequently used as keys in lookup
649 * tables.
650 *
651 * Providing `std::hash` and `std::equal_to` specializations for
652 * `tvm::Var` and `tvm::GlobalVar` allows them to be used as keys in STL tables. For
653 * other IR expressions, the user must specify the type of equality
654 * used (e.g. `std::unordered_set<T, StructuralHash, StructuralEqual>`
655 * or `std::unordered_set<T, ffi::ObjectPtrHash, ffi::ObjectPtrEqual>`).
656 */
657template <>
658struct std::hash<tvm::Var> {
659 std::size_t operator()(const tvm::Var& var) const { return tvm::ffi::ObjectPtrHash()(var); }
660};
661
662template <>
663struct std::equal_to<tvm::Var> {
664 bool operator()(const tvm::Var& var_a, const tvm::Var& var_b) const {
665 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
666 }
667};
668
669template <>
670struct std::hash<tvm::GlobalVar> {
671 std::size_t operator()(const tvm::GlobalVar& var) const { return tvm::ffi::ObjectPtrHash()(var); }
672};
673
674template <>
675struct std::equal_to<tvm::GlobalVar> {
676 bool operator()(const tvm::GlobalVar& var_a, const tvm::GlobalVar& var_b) const {
677 return tvm::ffi::ObjectPtrEqual()(var_a, var_b);
678 }
679};
