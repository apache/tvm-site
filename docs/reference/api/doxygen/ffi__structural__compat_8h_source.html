68template <typename T>
69auto VisitReturn(T&& result) {
70 if constexpr (std::is_same_v<std::remove_cv_t<std::remove_reference_t<T>>,
71 ffi::Optional<ffi::VisitInterrupt>>) {
72 return std::forward<T>(result);
73 } else {
74 return ExpectedReturn(std::forward<T>(result));
75 }
76}
