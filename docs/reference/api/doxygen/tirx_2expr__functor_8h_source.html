49template <typename R, typename... Args>
50class ExprFunctor<R(const Expr&, Args...)> : public tvm::ExprFunctor<R(const Expr&, Args...)> {
51 private:
52 using TSelf = ExprFunctor<R(const Expr&, Args...)>;
53 using Parent = tvm::ExprFunctor<R(const Expr&, Args...)>;
54
55 public:
56 using Parent::Dispatch_;
57
59 ExprFunctor() : Parent(GlobalVTable()) {}
61 virtual ~ExprFunctor() = default;
62
63 virtual R Dispatch_(const BufferRegionNode* node, Args... args) {
64 return this->DispatchDefault_(node, std::forward<Args>(args)...);
65 }
66
67 protected:
68 using Parent::SetDispatch;
69 using typename Parent::VTable;
70
72 explicit ExprFunctor(const VTable* vtable) : Parent(vtable) {}
73
75 static void InitVTable(VTable* vtable) {
76 Parent::InitVTable(vtable);
77 Parent::template SetDispatch<TSelf, BufferRegionNode>(vtable);
78 }
79
80 private:
81 static const VTable* GlobalVTable() {
82 static const VTable table = [] {
83 VTable table;
84 InitVTable(&table);
85 table.Finalize();
86 return table;
87 }();
88 return &table;
89 }
90};
