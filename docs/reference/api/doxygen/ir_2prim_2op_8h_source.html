520template <typename ValueType>
521inline PrimExpr MakeConstScalar(PrimType dtype, ValueType value, Span span = Span()) {
522 if constexpr (std::is_enum_v<ValueType>) {
523 return MakeConstScalar(dtype, static_cast<std::underlying_type_t<ValueType>>(value), span);
524 } else {
525 DLDataTypeCode code = dtype.code();
526 if (code == DLDataTypeCode::kDLInt || code == DLDataTypeCode::kDLBool) {
527 return IntImm(dtype, ffi::BigInt(value), span);
528 }
529 if (code == DLDataTypeCode::kDLUInt) {
530 TVM_FFI_ICHECK(value >= 0) << "cannot make uint from negative value " << value;
531 return IntImm(dtype, ffi::BigInt(value), span);
532 }
533 if (dtype.MatchesCode(DLDataTypeCode::kDLFloat, DLDataTypeCode::kDLFloat8_e3m4,
534 DLDataTypeCode::kDLFloat8_e4m3, DLDataTypeCode::kDLFloat8_e4m3b11fnuz,
535 DLDataTypeCode::kDLFloat8_e4m3fn, DLDataTypeCode::kDLFloat8_e4m3fnuz,
536 DLDataTypeCode::kDLFloat8_e5m2, DLDataTypeCode::kDLFloat8_e5m2fnuz,
537 DLDataTypeCode::kDLFloat8_e8m0fnu, DLDataTypeCode::kDLFloat6_e2m3fn,
538 DLDataTypeCode::kDLFloat6_e3m2fn, DLDataTypeCode::kDLFloat4_e2m1fn) ||
539 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) {
540 return FloatImm(dtype, static_cast<double>(value), span);
541 }
542 TVM_FFI_THROW(InternalError) << "cannot make const for type " << dtype;
543 throw;
544 }
545}

552template <typename ValueType, typename>
553inline PrimExpr MakeConst(PrimType dtype, ValueType value, Span span) {
554 if (!dtype.IsScalableVector() && !dtype.IsFixedLengthVector()) {
555 return MakeConstScalar(dtype, value, span);
556 }
557 PrimType elem_ty = dtype.WithLanes(1);
558 if (dtype.IsFixedLengthVector()) {
559 return prim::Broadcast(MakeConstScalar(elem_ty, value, span), dtype.lanes(), span);
560 }
561 PrimExpr lanes = prim::Mul(Call(PrimType::Int(32), prim::vscale_op(), {}).as_or_throw<PrimExpr>(),
562 dtype.VScaleFactor());
563 return prim::Broadcast(MakeConstScalar(elem_ty, value, span), lanes, span);
564}

978#define TVM_DECLARE_INTRIN_UNARY_WITH_CHECK(OpName, CheckInputDType) \
979 inline PrimExpr OpName(PrimExpr x, Span span = Span()) { \
980 static const Op op = Op::Get("prim." #OpName); \
981 PrimType x_ty = x.ty(); \
982 CheckInputDType(#OpName, x_ty); \
983 if (x_ty.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) { \
984 PrimType bf16_ty = x_ty; \
985 PrimType f32_ty = \
986 x_ty.IsScalableVector() \
987 ? PrimType::ScalableVector(DLDataTypeCode::kDLFloat, 32, x_ty.VScaleFactor()) \
988 : PrimType::Float(32, x_ty.lanes()); \
989 PrimExpr x_fp32 = prim::Cast(f32_ty, x, span); \
990 PrimExpr result_fp32 = Call(f32_ty, op, {x_fp32}, {}, {}, span).as_or_throw<PrimExpr>(); \
991 return prim::Cast(bf16_ty, result_fp32, span); \
992 } else { \
993 return Call(x_ty, op, {x}, {}, {}, span).as_or_throw<PrimExpr>(); \
994 } \
995 }
