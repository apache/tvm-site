554inline bool is_const_number(const PrimExpr& x) {
555 if (x.as<IntImmNode>()) {
556 return true;
557 } else if (x.as<FloatImmNode>()) {
558 return true;
559 } else if (const auto* op = x.as<prim::BroadcastNode>()) {
560 return (op->value->IsInstance<IntImmNode>() || op->value->IsInstance<FloatImmNode>());
561 }
562 return false;

590template <typename ValueType>
591inline PrimExpr MakeConstScalar(PrimType dtype, ValueType value, Span span = Span()) {
592 DLDataTypeCode code = dtype.code();
593 if (code == DLDataTypeCode::kDLInt || code == DLDataTypeCode::kDLBool) {
594 return IntImm(dtype, static_cast<int64_t>(value), span);
595 }
596 if (code == DLDataTypeCode::kDLUInt) {
597 // Use IntImm if it is a small integer
598 uint64_t uval = static_cast<uint64_t>(value);
599 if (value < static_cast<ValueType>(0)) {
600 TVM_FFI_THROW(InternalError) << "cannot make uint from negative value " << value;
601 } else if (uval <= static_cast<uint64_t>(std::numeric_limits<int64_t>::max())) {
602 return IntImm(dtype, static_cast<int64_t>(value), span);
603 } else {
604 return LargeUIntImm(dtype, static_cast<int64_t>(uval & 0xffffffffULL),
605 static_cast<int64_t>(uval >> 32U), span);
606 }
607 }
608 if (dtype.MatchesCode(DLDataTypeCode::kDLFloat, DLDataTypeCode::kDLFloat8_e3m4,
609 DLDataTypeCode::kDLFloat8_e4m3, DLDataTypeCode::kDLFloat8_e4m3b11fnuz,
610 DLDataTypeCode::kDLFloat8_e4m3fn, DLDataTypeCode::kDLFloat8_e4m3fnuz,
611 DLDataTypeCode::kDLFloat8_e5m2, DLDataTypeCode::kDLFloat8_e5m2fnuz,
612 DLDataTypeCode::kDLFloat8_e8m0fnu, DLDataTypeCode::kDLFloat6_e2m3fn,
613 DLDataTypeCode::kDLFloat6_e3m2fn, DLDataTypeCode::kDLFloat4_e2m1fn) ||
614 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) {
615 return FloatImm(dtype, static_cast<double>(value), span);
616 }
617 TVM_FFI_THROW(InternalError) << "cannot make const for type " << dtype;
618 throw;
619}

622inline PrimExpr MakeConstScalar(PrimType dtype, bool value, Span span) {
623 return MakeConstScalar(dtype, static_cast<int>(value), span);

626template <typename ValueType, typename>
627inline PrimExpr MakeConst(PrimType dtype, ValueType value, Span span) {
628 if (!dtype.IsScalableVector() && !dtype.IsFixedLengthVector()) {
629 return MakeConstScalar(dtype, value, span);
630 }
631 PrimType elem_ty = dtype.WithLanes(1);
632 if (dtype.IsFixedLengthVector()) {
633 return prim::Broadcast(MakeConstScalar(elem_ty, value, span), dtype.lanes(), span);
634 }
635 PrimExpr lanes =
636 prim::Mul(Call(PrimType::Int(32), prim::builtin::vscale(), {}).as_or_throw<PrimExpr>(),
637 dtype.VScaleFactor());
638 return prim::Broadcast(MakeConstScalar(elem_ty, value, span), lanes, span);
639}
