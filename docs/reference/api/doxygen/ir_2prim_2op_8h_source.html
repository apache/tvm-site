540 if (x.as<IntImmNode>()) {

565template <typename ValueType>
566inline PrimExpr MakeConstScalar(PrimType dtype, ValueType value, Span span = Span()) {
567 if constexpr (std::is_enum_v<ValueType>) {
568 return MakeConstScalar(dtype, static_cast<std::underlying_type_t<ValueType>>(value), span);
569 } else {
570 DLDataTypeCode code = dtype.code();
571 if (code == DLDataTypeCode::kDLInt || code == DLDataTypeCode::kDLBool) {
572 return IntImm(dtype, ffi::BigInt(value), span);
573 }
574 if (code == DLDataTypeCode::kDLUInt) {
575 TVM_FFI_ICHECK(value >= 0) << "cannot make uint from negative value " << value;
576 return IntImm(dtype, ffi::BigInt(value), span);
577 }
578 if (dtype.MatchesCode(DLDataTypeCode::kDLFloat, DLDataTypeCode::kDLFloat8_e3m4,
579 DLDataTypeCode::kDLFloat8_e4m3, DLDataTypeCode::kDLFloat8_e4m3b11fnuz,
580 DLDataTypeCode::kDLFloat8_e4m3fn, DLDataTypeCode::kDLFloat8_e4m3fnuz,
581 DLDataTypeCode::kDLFloat8_e5m2, DLDataTypeCode::kDLFloat8_e5m2fnuz,
582 DLDataTypeCode::kDLFloat8_e8m0fnu, DLDataTypeCode::kDLFloat6_e2m3fn,
583 DLDataTypeCode::kDLFloat6_e3m2fn, DLDataTypeCode::kDLFloat4_e2m1fn) ||
584 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) {
585 return FloatImm(dtype, static_cast<double>(value), span);
586 }
587 TVM_FFI_THROW(InternalError) << "cannot make const for type " << dtype;
588 throw;
589 }
590}

597template <typename ValueType, typename>
598inline PrimExpr MakeConst(PrimType dtype, ValueType value, Span span) {
599 if (!dtype.IsScalableVector() && !dtype.IsFixedLengthVector()) {
600 return MakeConstScalar(dtype, value, span);
601 }
602 PrimType elem_ty = dtype.WithLanes(1);
603 if (dtype.IsFixedLengthVector()) {
604 return prim::Broadcast(MakeConstScalar(elem_ty, value, span), dtype.lanes(), span);
605 }
606 PrimExpr lanes =
607 prim::Mul(Call(PrimType::Int(32), prim::builtin::vscale(), {}).as_or_throw<PrimExpr>(),
608 dtype.VScaleFactor());
609 return prim::Broadcast(MakeConstScalar(elem_ty, value, span), lanes, span);
610}
