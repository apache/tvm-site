288template <typename Node, typename = std::enable_if_t<std::is_base_of_v<StmtNode, Node>>>
289bool ContainsNode(const Stmt& stmt) {
290 struct Visitor : StmtExprVisitor {
291 // Early bail-out, if we already found the node. Skip expression operands.
292 ffi::Optional<VisitInterrupt> Visit(ffi::AnyView value) final {
293 if (contains_node || value.as<ExprNode>()) {
294 return std::nullopt;
295 }
296 return StmtExprVisitor::Visit(value);
297 }
298
299 ffi::Optional<VisitInterrupt> Visit_(const Node* block) override {
300 contains_node = true;
301 return std::nullopt;
302 }
303
304 bool contains_node{false};
305 };
306
307 auto visitor = ffi::make_object<Visitor>();
308 visitor->Visit(stmt);
309 return visitor->contains_node;
310}
