64 StmtFunctor() : StmtFunctor(GlobalVTable()) {}
66 virtual ~StmtFunctor() = default;
68 TVM_FFI_INLINE R operator()(const Stmt& node, Args... args) {
69 return Dispatch(node, std::forward<Args>(args)...);

72 TVM_FFI_INLINE virtual R Dispatch(const Stmt& node, Args... args) {
73 TVM_FFI_ICHECK(node.defined()) << "Cannot dispatch a null statement";
74 return (*vtable_)(node, this, std::forward<Args>(args)...);

144 SetDispatch<TSelf, BindNode>(vtable);
145 SetDispatch<TSelf, AttrStmtNode>(vtable);

161 SetDispatch<TSelf, tirx::TilePrimitiveCallNode>(vtable);
162 }
164 template <typename Self, typename Node>
165 static void SetDispatch(VTable* vtable) {
166 vtable->template SetDispatch<Node>(
167 [](const ffi::ObjectRef& node, TSelf* self, Args... args) -> R {
168 return static_cast<Self*>(self)->Dispatch_(static_cast<const Node*>(node.get()),
169 std::forward<Args>(args)...);
170 });
171 }
172
173 private:
174 static const VTable* GlobalVTable() {
175 static const VTable table = [] {
176 VTable table;
177 InitVTable(&table);
178 table.Finalize();
179 return table;
180 }();
181 return &table;
182 }
183 const VTable* vtable_;

329template <typename Node, typename = std::enable_if_t<std::is_base_of_v<StmtNode, Node>>>
330bool ContainsNode(const Stmt& stmt) {
331 struct Visitor : StmtExprVisitor {
332 // Early bail-out, if we already found the node. Skip expression operands.
333 ffi::Optional<VisitInterrupt> Visit(ffi::AnyView value) final {
334 if (contains_node || value.as<ExprNode>()) {
335 return std::nullopt;
336 }
337 return StmtExprVisitor::Visit(value);
338 }
339
340 ffi::Optional<VisitInterrupt> Visit_(const Node* block) override {
341 contains_node = true;
342 return std::nullopt;
343 }
344
345 bool contains_node{false};
346 };
347
348 auto visitor = ffi::make_object<Visitor>();
349 visitor->Visit(stmt);
350 return visitor->contains_node;
351}
