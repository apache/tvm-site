204class TVM_DLL StmtMutator : protected StmtFunctor<Stmt(const Stmt&)> {
205 public:
214 Stmt operator()(Stmt stmt) {
215 allow_copy_on_write_ = true;
216 return VisitStmt(stmt);
217 }
218
219 protected:
221 ffi::Map<BufferVar, BufferVar> buffer_remap_;
222 // We perform copy on write optimizations on the StmtMutator
223 // so that an unique copy of parent can be mutated inplace
224 // when some of its children changed.
225 // We only do such optimization for Stmt nests(instead of Exprs) for now
226 // as Stmt's parent state is more likely remain unchanged when one of
227 // its child block changes.
232 bool allow_copy_on_write_{false};
242 template <typename TNode>
243 ffi::ObjectPtr<TNode> CopyOnWrite(const TNode* node) {
244 static_assert(std::is_base_of<StmtNode, TNode>::value,
245 "StmtMutator:: CopyOnWrite requires us to track uniqueness of all parent "
246 "nodes during the recursion. Because the child classes do not necessarily "
247 "check the Array, Expr and other structures during the visit, it is only safe to "
248 "call this function with StmtNodes for now. "
249 "Please create a new node directly in other cases.");
250 if (allow_copy_on_write_) {
251 // return the old node.
252 return ffi::GetObjectPtr<TNode>(const_cast<TNode*>(node));
253 } else {
254 // Make a new copy of the node.
255 // need to rely on the default copy constructor
256 return ffi::make_object<TNode>(*node);
257 }
258 }
265 Stmt VisitStmt(const Stmt& stmt) override {
266 if (allow_copy_on_write_ && !stmt.unique()) {
267 allow_copy_on_write_ = false;
268 Stmt ret = StmtFunctor::VisitStmt(stmt);
269 allow_copy_on_write_ = true;
270 return ret;
271 } else {
272 return StmtFunctor::VisitStmt(stmt);
273 }
274 }

399 return std::nullopt;
400 }

402 }
