383 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
384 if (src->type_index == TypeIndex::kTVMFFINone) {
385 return false;
386 }
387 if (src->type_index != tirx::VarNode::RuntimeTypeIndex()) {
388 return false;
389 }
390 const auto* var = static_cast<const tirx::VarNode*>(
391 details::ObjectUnsafe::ObjectPtrFromUnowned<Object>(src->v_obj).get());
392 return details::AnyUnsafe::CheckAnyStrict<tirx::BufferType>(var->ExprNode::ty);
393 }

395 TVM_FFI_INLINE static std::optional<tirx::BufferVar> TryCastFromAnyView(const TVMFFIAny* src) {
396 if (CheckAnyStrict(src)) {
397 return details::ObjectUnsafe::ObjectRefFromObjectPtr<tirx::BufferVar>(
398 details::ObjectUnsafe::ObjectPtrFromUnowned<tirx::VarNode>(src->v_obj));
