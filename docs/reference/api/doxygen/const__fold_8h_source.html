90inline ffi::BigInt GetFoldResult(ffi::BigInt value, const PrimType& dtype) {
91 bool is_signed = dtype.MatchesCode(DLDataTypeCode::kDLInt);
92 if (auto small = value.as<int64_t>(); small.has_value()) {
93 if (dtype.bits() >= 64 && (is_signed || *small >= 0)) return value;
94 if (dtype.bits() < 64) {
95 uint64_t bits = static_cast<uint64_t>(*small) & ((uint64_t{1} << dtype.bits()) - 1);
96 int64_t result = static_cast<int64_t>(bits);
97 if (is_signed) {
98 int64_t sign_bit = int64_t{1} << (dtype.bits() - 1);
99 result = (result ^ sign_bit) - sign_bit;
100 }
101 return ffi::BigInt(result);
102 }
103 }
104 ffi::BigInt modulus = ffi::BigInt(1) << dtype.bits();
105 value &= modulus - 1;
106 if (is_signed && value >= (modulus >> 1)) value -= modulus;
107 return value;
108}

111inline double GetFoldResultDoubleRepr(float x) {
112 double res = static_cast<double>(x);
113 if (std::isinf(res) || std::isnan(res)) {
114 return res;
115 }
116 // certain platform (eg, on gcc7-i386) do the folding arithmetic
117 // on float and write back to double is optimized to double
118 // precision arithmetic, this is legal and we check the output
119 // range thus to ensure consistency when the float result is inf.
120 if (res < std::numeric_limits<float>::lowest()) {
121 LOG(WARNING) << "underlying float value overflow";
122 return -std::numeric_limits<double>::infinity();
123 } else if (res > std::numeric_limits<float>::max()) {
124 LOG(WARNING) << "underlying float value overflow";
125 return std::numeric_limits<double>::infinity();
126 }
127 return res;
128}

149inline ffi::Optional<PrimExpr> TryConstFold<prim::Add>(PrimExpr a, PrimExpr b) {
150 TVM_PRIM_CONST_PROPAGATION({
151 PrimType result_ty = a.ty();
152 if (pa && pb) {
153 ffi::BigInt res = pa->value + pb->value;
154 return IntImm(result_ty, details::GetFoldResult(res, result_ty));
155 }
156 if (pa && pa->value == 0) return b;
157 if (pb && pb->value == 0) return a;
158 if (fa && fb) {
159 if (result_ty.bits() == 32) {
160 return FloatImm(result_ty, details::GetFoldResultDoubleRepr(static_cast<float>(fa->value) +
161 static_cast<float>(fb->value)));
162 } else if (result_ty.bits() == 64) {
163 return FloatImm(result_ty, fa->value + fb->value);
164 }
165 }
166 if (fa && fa->value == 0) return b;
167 if (fb && fb->value == 0) return a;
168 });
169 return std::nullopt;
170}

173inline ffi::Optional<PrimExpr> TryConstFold<prim::Sub>(PrimExpr a, PrimExpr b) {
174 TVM_PRIM_CONST_PROPAGATION({
175 TVM_FFI_ICHECK(!((pa && pa->ty.as_or_throw<PrimType>().MatchesCode(DLDataTypeCode::kDLUInt) &&
176 pa->value == 0U) &&
177 (pb && pb->ty.as_or_throw<PrimType>().MatchesCode(DLDataTypeCode::kDLUInt) &&
178 pb->value > 0U)))
179 << "Checked failed. Minuend 's value is 0U and it's dtype is uint "
180 << "while Subtrahend's dtype is uint; which will cause a negative uint";
181 PrimType result_ty = a.ty();
182 if (pa && pb) {
183 ffi::BigInt res = pa->value - pb->value;
184 return IntImm(result_ty, details::GetFoldResult(res, result_ty));
185 }
186 if (pb && pb->value == 0) return a;
187 if (fa && fb) {
188 if (result_ty.bits() == 32) {
189 return FloatImm(result_ty, details::GetFoldResultDoubleRepr(static_cast<float>(fa->value) -
190 static_cast<float>(fb->value)));
191 } else if (result_ty.bits() == 64) {
192 return FloatImm(result_ty, fa->value - fb->value);
193 }
194 }
195 if (fb && fb->value == 0) return a;
196 });
197 return std::nullopt;

236template <>
237inline ffi::Optional<PrimExpr> TryConstFold<prim::Div>(PrimExpr a, PrimExpr b) {
238 TVM_PRIM_CONST_PROPAGATION({
239 PrimType result_ty = a.ty();
240 if (pa && pb) {
241 // due to division and mod can have different modes
242 // NOTE: this will assumes truc div.
243 TVM_FFI_ICHECK_NE(pb->value, 0) << "Divide by zero";
244 ffi::BigInt res = pa->value / pb->value;
245 return IntImm(result_ty, details::GetFoldResult(res, result_ty));
246 }
247 if (pa) {
248 if (pa->value == 0) return a;
249 }
250 if (pb) {
251 if (pb->value == 1) return a;
252 TVM_FFI_ICHECK_NE(pb->value, 0) << "Divide by zero";
253 }
254 if (fa && fb) {
255 TVM_FFI_ICHECK_NE(fb->value, 0) << "Divide by zero";
256 if (result_ty.bits() == 32) {
257 return FloatImm(result_ty, details::GetFoldResultDoubleRepr(static_cast<float>(fa->value) /
258 static_cast<float>(fb->value)));
259 } else if (result_ty.bits() == 64) {
260 return FloatImm(result_ty, fa->value / fb->value);
261 }
262 }
263 if (fa && fa->value == 0) return a;
264 if (fb) {
265 if (fb->value == 1) return a;
266 TVM_FFI_ICHECK_NE(fb->value, 0) << "Divide by zero";
267 }
268 });
269 return std::nullopt;
270}

272template <>
273inline ffi::Optional<PrimExpr> TryConstFold<prim::Mod>(PrimExpr a, PrimExpr b) {
274 TVM_PRIM_INDEX_CONST_PROPAGATION({
275 PrimType result_ty = a.ty();
276 if (pa && pb) {
277 TVM_FFI_ICHECK_NE(pb->value, 0) << "Divide by zero";
278 ffi::BigInt res = pa->value % pb->value;
279 return IntImm(result_ty, details::GetFoldResult(res, result_ty));
280 }
281 if (pa) {
282 if (pa->value == 0) return a;
283 }
284 if (pb) {
285 if (pb->value == 1) return IntImm(result_ty, 0);
286 TVM_FFI_ICHECK_NE(pb->value, 0) << "Divide by zero";
287 }
288 });
289 // Fold wide literals without enabling symbolic index rewrites for their type.
290 if (pa && pb && a.ty().bits() >= 64) {
291 TVM_FFI_ICHECK_NE(pb->value, 0) << "Divide by zero";
292 return IntImm(a.ty(), details::GetFoldResult(pa->value % pb->value, a.ty()));
293 }
294 return std::nullopt;
295}

297template <>
298inline ffi::Optional<PrimExpr> TryConstFold<prim::FloorDiv>(PrimExpr a, PrimExpr b) {
299 TVM_PRIM_CONST_PROPAGATION({
300 PrimType result_ty = a.ty();
301 if (pa && pb) {
302 TVM_FFI_ICHECK_NE(pb->value, 0) << "Divide by zero";
303 ffi::BigInt res = ffi::floordiv(pa->value, pb->value);
304 return IntImm(result_ty, details::GetFoldResult(res, result_ty));
305 }
306 if (pa) {
307 if (pa->value == 0) return a;
308 }
309 if (pb) {
310 if (pb->value == 1) return a;
311 TVM_FFI_ICHECK_NE(pb->value, 0) << "Divide by zero";
312 }
313 if (fa && fb && fb->value != 0) {
314 if (result_ty.bits() == 32) {
315 return FloatImm(
316 result_ty, details::GetFoldResultDoubleRepr(std::floor(static_cast<float>(fa->value) /
317 static_cast<float>(fb->value))));
318 } else if (result_ty.bits() == 64) {
319 return FloatImm(result_ty, std::floor(fa->value / fb->value));
320 } else {
321 return std::nullopt;
322 }
323 }
324 if (fa && fa->value == 0) return a;
325 if (fb) {
326 if (fb->value == 1) return a;
327 TVM_FFI_ICHECK_NE(fb->value, 0) << "Divide by zero";
328 }
329 });
330 return std::nullopt;
331}
