215 TVM_TIRX_DLL static PrimType ScalableVector(DLDataTypeCode code, int bits, int lanes);
216
218 TVM_FFI_INLINE DLDataTypeCode code() const {
219 return static_cast<DLDataTypeCode>(static_cast<int>(get()->dtype.code));
220 }
221
223 TVM_FFI_INLINE int32_t bits() const { return get()->dtype.bits; }
224
229 TVM_FFI_INLINE int32_t lanes() const {
230 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
231 if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
232 TVM_FFI_THROW(InternalError)
233 << "Can't fetch the lanes of a scalable vector at a compile time.";
234 }
235 return encoded_lanes;
236 }
237
242 TVM_FFI_INLINE bool MatchesElementType(DLDataTypeCode code, int bits) const {
243 DLDataType dtype = get()->dtype;
244 return dtype.code == static_cast<uint8_t>(code) && dtype.bits == bits;
245 }
246
251 template <typename... Codes>
252 TVM_FFI_INLINE bool MatchesCode(Codes... codes) const {
253 uint8_t dtype_code = get()->dtype.code;
254 return ((dtype_code == static_cast<uint8_t>(codes)) || ...);
255 }
256
258 TVM_FFI_INLINE bool IsScalar() const {
259 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
260 return encoded_lanes == 1;
261 }
262
264 TVM_FFI_INLINE bool IsVoid() const {
265 DLDataType dtype = get()->dtype;
266 return dtype.code == static_cast<uint8_t>(DLDataTypeCode::kDLOpaqueHandle) && dtype.bits == 0 &&
267 static_cast<int16_t>(dtype.lanes) == 0;
268 }
269
271 TVM_FFI_INLINE bool IsScalableVector() const {
272 return static_cast<int16_t>(get()->dtype.lanes) < -1;
273 }
274
276 TVM_FFI_INLINE bool IsFixedLengthVector() const {
277 return static_cast<int16_t>(get()->dtype.lanes) > 1;
278 }
279
286 TVM_FFI_INLINE size_t StorageBytes() const {
287 DLDataType dtype = get()->dtype;
288 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
289 if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
290 TVM_FFI_THROW(InternalError)
291 << "Cannot compute compile-time storage bytes for non-fixed vector type " << dtype;
292 }
293 return static_cast<size_t>(
294 (static_cast<uint64_t>(dtype.bits) * static_cast<uint64_t>(dtype.lanes) + 7) / 8);
295 }
296
298 TVM_FFI_INLINE PrimType WithCode(DLDataTypeCode code) const {
299 DLDataType dtype = get()->dtype;
300 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
301 if (encoded_lanes < -1) {
302 return ScalableVector(code, dtype.bits, -encoded_lanes);
303 }
304 return PrimType(code, dtype.bits, encoded_lanes);
305 }
306
308 TVM_FFI_INLINE PrimType WithBits(int bits) const {
309 DLDataType dtype = get()->dtype;
310 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
311 if (encoded_lanes < -1) {
312 return ScalableVector(this->code(), bits, -encoded_lanes);
313 }
314 return PrimType(this->code(), bits, encoded_lanes);
315 }
316
318 TVM_FFI_INLINE PrimType WithLanes(int lanes) const {
319 return PrimType(this->code(), this->bits(), lanes);
320 }
321
323 TVM_FFI_INLINE int32_t VScaleFactor() const {
324 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
325 if (encoded_lanes >= -1) {
326 TVM_FFI_THROW(InternalError) << "A fixed length vector doesn't have a vscale factor.";
327 }
328 return -encoded_lanes;
329 }
