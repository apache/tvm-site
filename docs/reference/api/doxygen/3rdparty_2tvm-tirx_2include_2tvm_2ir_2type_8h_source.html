182class PrimType final : public Type {
183 public:
188 TVM_TIRX_DLL explicit PrimType(DLDataType dtype);
189
196 TVM_TIRX_DLL PrimType(DLDataTypeCode code, int bits, int lanes = 1);
197
199 TVM_TIRX_DLL static PrimType Int(int bits, int lanes = 1);
201 TVM_TIRX_DLL static PrimType UInt(int bits, int lanes = 1);
203 TVM_TIRX_DLL static PrimType Float(int bits, int lanes = 1);
205 TVM_TIRX_DLL static PrimType BFloat(int bits, int lanes = 1);
207 TVM_TIRX_DLL static PrimType Bool(int lanes = 1);
209 TVM_TIRX_DLL static PrimType Void();
216 TVM_TIRX_DLL static PrimType ScalableVector(DLDataTypeCode code, int bits, int lanes);
217
219 TVM_FFI_INLINE DLDataTypeCode code() const {
220 return static_cast<DLDataTypeCode>(static_cast<int>(get()->dtype.code));
221 }
222
224 TVM_FFI_INLINE int32_t bits() const { return get()->dtype.bits; }
225
230 TVM_FFI_INLINE int32_t lanes() const {
231 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
232 if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
233 TVM_FFI_THROW(InternalError)
234 << "Can't fetch the lanes of a scalable vector at a compile time.";
235 }
236 return encoded_lanes;
237 }
238
243 TVM_FFI_INLINE bool MatchesElementType(DLDataTypeCode code, int bits) const {
244 DLDataType dtype = get()->dtype;
245 return dtype.code == static_cast<uint8_t>(code) && dtype.bits == bits;
246 }
247
252 template <typename... Codes>
253 TVM_FFI_INLINE bool MatchesCode(Codes... codes) const {
254 uint8_t dtype_code = get()->dtype.code;
255 return ((dtype_code == static_cast<uint8_t>(codes)) || ...);
256 }
257
259 TVM_FFI_INLINE bool IsScalar() const {
260 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
261 return encoded_lanes == 1;
262 }
263
265 TVM_FFI_INLINE bool IsVoid() const {
266 DLDataType dtype = get()->dtype;
267 return dtype.code == static_cast<uint8_t>(DLDataTypeCode::kDLOpaqueHandle) && dtype.bits == 0 &&
268 static_cast<int16_t>(dtype.lanes) == 0;
269 }
270
272 TVM_FFI_INLINE bool IsScalableVector() const {
273 return static_cast<int16_t>(get()->dtype.lanes) < -1;
274 }
275
277 TVM_FFI_INLINE bool IsFixedLengthVector() const {
278 return static_cast<int16_t>(get()->dtype.lanes) > 1;
279 }
280
287 TVM_FFI_INLINE size_t StorageBytes() const {
288 DLDataType dtype = get()->dtype;
289 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
290 if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
291 TVM_FFI_THROW(InternalError)
292 << "Cannot compute compile-time storage bytes for non-fixed vector type " << dtype;
293 }
294 return static_cast<size_t>(
295 (static_cast<uint64_t>(dtype.bits) * static_cast<uint64_t>(dtype.lanes) + 7) / 8);
296 }
297
299 TVM_FFI_INLINE PrimType WithCode(DLDataTypeCode code) const {
300 DLDataType dtype = get()->dtype;
301 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
302 if (encoded_lanes < -1) {
303 return ScalableVector(code, dtype.bits, -encoded_lanes);
304 }
305 return PrimType(code, dtype.bits, encoded_lanes);
306 }
307
309 TVM_FFI_INLINE PrimType WithBits(int bits) const {
310 DLDataType dtype = get()->dtype;
311 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
312 if (encoded_lanes < -1) {
313 return ScalableVector(this->code(), bits, -encoded_lanes);
314 }
315 return PrimType(this->code(), bits, encoded_lanes);
316 }
317
319 TVM_FFI_INLINE PrimType WithLanes(int lanes) const {
320 return PrimType(this->code(), this->bits(), lanes);
321 }
322
324 TVM_FFI_INLINE int32_t VScaleFactor() const {
325 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
326 if (encoded_lanes >= -1) {
327 TVM_FFI_THROW(InternalError) << "A fixed length vector doesn't have a vscale factor.";
328 }
329 return -encoded_lanes;
330 }

332 TVM_FFI_DEFINE_OBJECT_REF_METHODS_NOTNULLABLE(PrimType, Type, PrimTypeObj);
333};
