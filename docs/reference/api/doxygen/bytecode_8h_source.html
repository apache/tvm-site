177 TVM_FFI_ICHECK_LE(value, kValueMaxLimit);
178 TVM_FFI_ICHECK_GE(value, kValueMinLimit);
179 data_ = (static_cast<ExecWord>(kind) << kValueBit) | (value & kValueMask);
