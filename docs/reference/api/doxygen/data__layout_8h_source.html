46class SLayoutAxis {
47 public:
48 static const SLayoutAxis& Get(const char name);
49
50 // Get the singleton SLayoutAxis using itvar->var->name
51 static const SLayoutAxis& Get(const s_tir::IterVar& itvar);
52
53 // Get the singleton SLayoutAxis using name[0] (size of name must be 1).
54 static const SLayoutAxis& Get(const std::string& name);
55
56 inline bool IsPrimal() const { return name_ >= 'A' && name_ <= 'Z'; }
57 inline std::string name() const { return std::string(1, name_); }
58
59 // if current axis is primal, switch the axis to its subordinate one,
60 // else switch to the primal.
61 inline const SLayoutAxis& ToDual() const {
62 if (name_ >= 'A' && name_ <= 'Z') {
63 return SLayoutAxis::Get(name_ - 'A' + 'a');
64 } else {
65 return SLayoutAxis::Get(name_ - 'a' + 'A');
66 }
67 }
68
69 // return the primal axis. If it is already primal, return itself.
70 const SLayoutAxis& ToPrimal() const { return IsPrimal() ? *this : ToDual(); }
71
72 // return the subordinate axis. If it is already subordinate, return itself.
73 const SLayoutAxis& ToSubordinate() const { return IsPrimal() ? ToDual() : *this; }
74
75 inline bool operator==(const SLayoutAxis& rhs) const { return name_ == rhs.name_; }
76
77 friend std::ostream& operator<<(std::ostream& os, const SLayoutAxis& l) {
78 os << l.name();
79 return os;
80 }
81
82 private:
83 static const SLayoutAxis UPPER_CASE[];
84 static const SLayoutAxis LOWER_CASE[];
85 SLayoutAxis(const SLayoutAxis&);
86 SLayoutAxis& operator=(const SLayoutAxis&);
87 explicit SLayoutAxis(const char name) : name_(name) {}
88
89 const char name_;
90};

206 inline size_t ndim_primal() const {
207 if (!defined()) return 0;
208 size_t ct = 0;
209 for (auto px : operator->()->axes) {
210 auto iter_vars = UnpackIterVar(px);
211 for (auto x : iter_vars) {
212 if (SLayoutAxis::Get(x).IsPrimal()) {
213 ct++;
214 }

225 inline SLayout ExpandPrimal(const SLayout& dst_layout) {
226 SLayout new_src_layout;
227 // 1) Find the axis which are missing in the current layout. Make them the prefix.
228 std::string new_src_layout_str = "";
229 for (auto packed_axis : dst_layout->axes) {
230 auto iter_vars = UnpackIterVar(packed_axis);
231 for (auto dst_axis : iter_vars) {
232 if (SLayoutAxis::Get(dst_axis).IsPrimal()) {
233 if (!this->Contains(SLayoutAxis::Get(dst_axis))) {
234 new_src_layout_str += dst_axis->var->name;
235 }
236 }
237 }
238 }
239 // 2) Now, add the primal axis of the current layout.
240 new_src_layout_str += this->name();
241 new_src_layout = SLayout(new_src_layout_str);
242 return new_src_layout;
243 }
244
252 inline int32_t IndexOf(const std::string& axis) const {
253 if (!this->defined()) return -1;
254 const auto axes = operator->()->axes;
255 for (size_t i = 0; i < axes.size(); ++i) {
256 if (axes[i]->var->name == axis) return static_cast<int32_t>(i);
257 }
258 return -1;
259 }
