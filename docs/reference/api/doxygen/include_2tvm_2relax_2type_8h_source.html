176 ffi::Optional<ffi::Array<PrimExpr>> GetShape() const {
177 if (!shape.has_value()) return {};
178 const Expr& shape_expr = this->shape.value();
179 if (shape_expr->ty.as<MissingType>().has_value()) return {};
180 if (const auto* shape_ty = shape_expr->ty.as<ShapeTypeObj>()) {

350template <typename T>
351inline ffi::Optional<T> MatchType(const Expr& expr) {
352 if (!expr.defined()) {
353 return std::nullopt;
354 }
355 using TNode = typename T::ContainerType;
356 if (const TNode* ptr = expr->ty.as<TNode>()) {
357 return ffi::GetRef<T>(ptr);
358 } else {
359 return std::nullopt;
360 }
361}
