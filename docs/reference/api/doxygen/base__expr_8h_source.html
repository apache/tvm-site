161class PrimType final : public Type {
162 public:
167 TVM_DLL explicit PrimType(DLDataType dtype);
168
175 TVM_DLL PrimType(DLDataTypeCode code, int bits, int lanes = 1);
176
178 TVM_DLL static PrimType Int(int bits, int lanes = 1);
180 TVM_DLL static PrimType UInt(int bits, int lanes = 1);
182 TVM_DLL static PrimType Float(int bits, int lanes = 1);
184 TVM_DLL static PrimType BFloat(int bits, int lanes = 1);
186 TVM_DLL static PrimType Bool(int lanes = 1);
188 TVM_DLL static PrimType Void();
195 TVM_DLL static PrimType ScalableVector(DLDataTypeCode code, int bits, int lanes);
196
198 TVM_FFI_INLINE DLDataTypeCode code() const {
199 return static_cast<DLDataTypeCode>(static_cast<int>(get()->dtype.code));
200 }
201
203 TVM_FFI_INLINE int32_t bits() const { return get()->dtype.bits; }
204
209 TVM_FFI_INLINE int32_t lanes() const {
210 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
211 if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
212 TVM_FFI_THROW(InternalError)
213 << "Can't fetch the lanes of a scalable vector at a compile time.";
214 }
215 return encoded_lanes;
216 }
217
222 TVM_FFI_INLINE bool MatchesElementType(DLDataTypeCode code, int bits) const {
223 DLDataType dtype = get()->dtype;
224 return dtype.code == static_cast<uint8_t>(code) && dtype.bits == bits;
225 }
226
231 template <typename... Codes>
232 TVM_FFI_INLINE bool MatchesCode(Codes... codes) const {
233 uint8_t dtype_code = get()->dtype.code;
234 return ((dtype_code == static_cast<uint8_t>(codes)) || ...);
235 }
236
238 TVM_FFI_INLINE bool IsScalar() const {
239 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
240 return encoded_lanes == 1;
241 }
242
244 TVM_FFI_INLINE bool IsVoid() const {
245 DLDataType dtype = get()->dtype;
246 return dtype.code == static_cast<uint8_t>(DLDataTypeCode::kDLOpaqueHandle) && dtype.bits == 0 &&
247 static_cast<int16_t>(dtype.lanes) == 0;
248 }
249
251 TVM_FFI_INLINE bool IsScalableVector() const {
252 return static_cast<int16_t>(get()->dtype.lanes) < -1;
253 }
254
256 TVM_FFI_INLINE bool IsFixedLengthVector() const {
257 return static_cast<int16_t>(get()->dtype.lanes) > 1;
258 }
259
266 TVM_FFI_INLINE size_t StorageBytes() const {
267 DLDataType dtype = get()->dtype;
268 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
269 if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
270 TVM_FFI_THROW(InternalError)
271 << "Cannot compute compile-time storage bytes for non-fixed vector type " << dtype;
272 }
273 return static_cast<size_t>(
274 (static_cast<uint64_t>(dtype.bits) * static_cast<uint64_t>(dtype.lanes) + 7) / 8);
275 }
276
278 TVM_FFI_INLINE PrimType WithCode(DLDataTypeCode code) const {
279 DLDataType dtype = get()->dtype;
280 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
281 if (encoded_lanes < -1) {
282 return ScalableVector(code, dtype.bits, -encoded_lanes);
283 }
284 return PrimType(code, dtype.bits, encoded_lanes);
285 }
286
288 TVM_FFI_INLINE PrimType WithBits(int bits) const {
289 DLDataType dtype = get()->dtype;
290 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
291 if (encoded_lanes < -1) {
292 return ScalableVector(this->code(), bits, -encoded_lanes);
293 }
294 return PrimType(this->code(), bits, encoded_lanes);
295 }
296
298 TVM_FFI_INLINE PrimType WithLanes(int lanes) const {
299 return PrimType(this->code(), this->bits(), lanes);
300 }
301
303 TVM_FFI_INLINE int32_t VScaleFactor() const {
304 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
305 if (encoded_lanes >= -1) {
306 TVM_FFI_THROW(InternalError) << "A fixed length vector doesn't have a vscale factor.";
307 }
308 return -encoded_lanes;
309 }
310
311 TVM_FFI_DEFINE_OBJECT_REF_METHODS_NOTNULLABLE(PrimType, Type, PrimTypeNode);
312};

496template <typename ExpectedType>
497inline constexpr bool use_default_type_traits_v<TypedExpr<ExpectedType>> = false;
498
499template <typename ExpectedType>
500struct TypeTraits<TypedExpr<ExpectedType>>
501 : public ObjectRefTypeTraitsBase<TypedExpr<ExpectedType>> {
502 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
503 if (src->type_index == TypeIndex::kTVMFFINone) {
504 return TypedExpr<ExpectedType>::_type_is_nullable;
505 }
506 if (src->type_index < TypeIndex::kTVMFFIStaticObjectBegin ||
507 !details::IsObjectInstance<ExprNode>(src->type_index)) {
508 return false;
509 }
510 // Non-owning: this only reads `ty`, and the owning form's incref/decref pair costs two
511 // atomics per check on a path every typed field assignment takes.
512 const auto* expr = details::ObjectUnsafe::RawObjectPtrFromUnowned<ExprNode>(src->v_obj);
513 return details::AnyUnsafe::CheckAnyViewStrict<ExpectedType>(expr->ty);
514 }
515
516 TVM_FFI_INLINE static std::string GetMismatchTypeInfo(const TVMFFIAny* src) {
517 if (src->type_index >= TypeIndex::kTVMFFIStaticObjectBegin && src->v_obj != nullptr &&
518 details::IsObjectInstance<ExprNode>(src->type_index)) {
519 const auto* expr = details::ObjectUnsafe::RawObjectPtrFromUnowned<ExprNode>(src->v_obj);
520 if (expr->ty.defined()) {
521 const auto* prim_ty = expr->ty.as<PrimTypeNode>();
522 std::string ty =
523 prim_ty ? std::string(ffi::DLDataTypeToString(prim_ty->dtype)) : expr->ty.GetTypeKey();
524 return TypeIndexToTypeKey(src->type_index) + "[ty=" + ty + "]";
525 }
526 }
527 return TypeTraitsBase::GetMismatchTypeInfo(src);
528 }
