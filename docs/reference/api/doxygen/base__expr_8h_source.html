203 TVM_FFI_INLINE DLDataTypeCode code() const {
204 return static_cast<DLDataTypeCode>(static_cast<int>(get()->dtype.code));
205 }
206
208 TVM_FFI_INLINE int32_t bits() const { return get()->dtype.bits; }
209
214 TVM_FFI_INLINE int32_t lanes() const {
215 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
216 if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
217 TVM_FFI_THROW(InternalError)
218 << "Can't fetch the lanes of a scalable vector at a compile time.";
219 }
220 return encoded_lanes;
221 }

227 TVM_FFI_INLINE bool MatchesElementType(DLDataTypeCode code, int bits) const {
228 DLDataType dtype = get()->dtype;
229 return dtype.code == static_cast<uint8_t>(code) && dtype.bits == bits;
230 }
231
236 template <typename... Codes>
237 TVM_FFI_INLINE bool MatchesCode(Codes... codes) const {
238 uint8_t dtype_code = get()->dtype.code;
239 return ((dtype_code == static_cast<uint8_t>(codes)) || ...);
240 }
241
243 TVM_FFI_INLINE bool IsScalar() const {
244 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
245 return encoded_lanes == 1;
246 }
247
249 TVM_FFI_INLINE bool IsVoid() const {
250 DLDataType dtype = get()->dtype;
251 return dtype.code == static_cast<uint8_t>(DLDataTypeCode::kDLOpaqueHandle) && dtype.bits == 0 &&
252 static_cast<int16_t>(dtype.lanes) == 0;
253 }
254
256 TVM_FFI_INLINE bool IsScalableVector() const {
257 return static_cast<int16_t>(get()->dtype.lanes) < -1;
258 }
259
261 TVM_FFI_INLINE bool IsFixedLengthVector() const {
262 return static_cast<int16_t>(get()->dtype.lanes) > 1;
263 }
264
271 TVM_FFI_INLINE size_t StorageBytes() const {
272 DLDataType dtype = get()->dtype;
273 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
274 if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
275 TVM_FFI_THROW(InternalError)
276 << "Cannot compute compile-time storage bytes for non-fixed vector type " << dtype;
277 }
278 return static_cast<size_t>(
279 (static_cast<uint64_t>(dtype.bits) * static_cast<uint64_t>(dtype.lanes) + 7) / 8);
280 }
281
283 TVM_FFI_INLINE PrimType WithCode(DLDataTypeCode code) const {
284 DLDataType dtype = get()->dtype;
285 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
286 if (encoded_lanes < -1) {
287 return ScalableVector(code, dtype.bits, -encoded_lanes);
288 }
289 return PrimType(code, dtype.bits, encoded_lanes);
290 }
291
293 TVM_FFI_INLINE PrimType WithBits(int bits) const {
294 DLDataType dtype = get()->dtype;
295 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
296 if (encoded_lanes < -1) {
297 return ScalableVector(this->code(), bits, -encoded_lanes);
298 }
299 return PrimType(this->code(), bits, encoded_lanes);
300 }
301
303 TVM_FFI_INLINE PrimType WithLanes(int lanes) const {
304 return PrimType(this->code(), this->bits(), lanes);
305 }
306
308 TVM_FFI_INLINE int32_t VScaleFactor() const {
309 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
310 if (encoded_lanes >= -1) {
311 TVM_FFI_THROW(InternalError) << "A fixed length vector doesn't have a vscale factor.";
312 }
313 return -encoded_lanes;
314 }

501template <typename ExpectedType>
502inline constexpr bool use_default_type_traits_v<TypedExpr<ExpectedType>> = false;
503
504template <typename ExpectedType>
505struct TypeTraits<TypedExpr<ExpectedType>>
506 : public ObjectRefTypeTraitsBase<TypedExpr<ExpectedType>> {
507 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
508 if (src->type_index == TypeIndex::kTVMFFINone) {
509 return TypedExpr<ExpectedType>::_type_is_nullable;
510 }
511 if (src->type_index < TypeIndex::kTVMFFIStaticObjectBegin ||
512 !details::IsObjectInstance<ExprNode>(src->type_index)) {
513 return false;
514 }
515 // Non-owning: this only reads `ty`, and the owning form's incref/decref pair costs two
516 // atomics per check on a path every typed field assignment takes.
517 const auto* expr = details::ObjectUnsafe::RawObjectPtrFromUnowned<ExprNode>(src->v_obj);
518 return details::AnyUnsafe::CheckAnyViewStrict<ExpectedType>(expr->ty);
519 }
520
521 TVM_FFI_INLINE static std::string GetMismatchTypeInfo(const TVMFFIAny* src) {
522 if (src->type_index >= TypeIndex::kTVMFFIStaticObjectBegin && src->v_obj != nullptr &&
523 details::IsObjectInstance<ExprNode>(src->type_index)) {
524 const auto* expr = details::ObjectUnsafe::RawObjectPtrFromUnowned<ExprNode>(src->v_obj);
525 if (expr->ty.defined()) {
526 const auto* prim_ty = expr->ty.as<PrimTypeNode>();
527 std::string ty =
528 prim_ty ? std::string(ffi::DLDataTypeToString(prim_ty->dtype)) : expr->ty.GetTypeKey();
529 return TypeIndexToTypeKey(src->type_index) + "[ty=" + ty + "]";
530 }
531 }
532 return TypeTraitsBase::GetMismatchTypeInfo(src);
533 }
534};
