153  TVM_FFI_INLINE DLDataTypeCode code() const {
154  return static_cast<DLDataTypeCode>(static_cast<int>(get()->dtype.code));

156 
158  TVM_FFI_INLINE int32_t bits() const { return get()->dtype.bits; }
159 
164  TVM_FFI_INLINE int32_t lanes() const {
165  int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
166  if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
167  TVM_FFI_THROW(InternalError)
168  << "Can't fetch the lanes of a scalable vector at a compile time.";
169  }
170  return encoded_lanes;
171  }

177  TVM_FFI_INLINE bool MatchesElementType(DLDataTypeCode code, int bits) const {
178  DLDataType dtype = get()->dtype;
179  return dtype.code == static_cast<uint8_t>(code) && dtype.bits == bits;
180  }
181 
186  template <typename... Codes>
187  TVM_FFI_INLINE bool MatchesCode(Codes... codes) const {
188  uint8_t dtype_code = get()->dtype.code;
189  return ((dtype_code == static_cast<uint8_t>(codes)) || ...);
190  }
191 
193  TVM_FFI_INLINE bool IsScalar() const {
194  int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
195  return encoded_lanes == 1;
196  }
197 
199  TVM_FFI_INLINE bool IsVoid() const {
200  DLDataType dtype = get()->dtype;
201  return dtype.code == static_cast<uint8_t>(DLDataTypeCode::kDLOpaqueHandle) && dtype.bits == 0 &&
202  static_cast<int16_t>(dtype.lanes) == 0;
203  }
204 
206  TVM_FFI_INLINE bool IsHandle() const {
207  return this->code() == DLDataTypeCode::kDLOpaqueHandle && !this->IsVoid();

211  TVM_FFI_INLINE bool IsScalableVector() const {
212  return static_cast<int16_t>(get()->dtype.lanes) < -1;
213  }
214 
216  TVM_FFI_INLINE bool IsFixedLengthVector() const {
217  return static_cast<int16_t>(get()->dtype.lanes) > 1;
218  }
219 
226  TVM_FFI_INLINE size_t StorageBytes() const {
227  DLDataType dtype = get()->dtype;
228  int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
229  if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
230  TVM_FFI_THROW(InternalError)
231  << "Cannot compute compile-time storage bytes for non-fixed vector type " << dtype;
232  }
233  return static_cast<size_t>(
234  (static_cast<uint64_t>(dtype.bits) * static_cast<uint64_t>(dtype.lanes) + 7) / 8);
235  }
236 
238  TVM_FFI_INLINE PrimType WithCode(DLDataTypeCode code) const {
239  DLDataType dtype = get()->dtype;
240  int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
241  if (encoded_lanes < -1) {
242  return ScalableVector(code, dtype.bits, -encoded_lanes);
243  }
244  return PrimType(code, dtype.bits, encoded_lanes);
245  }
246 
248  TVM_FFI_INLINE PrimType WithBits(int bits) const {
249  DLDataType dtype = get()->dtype;
250  int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
251  if (encoded_lanes < -1) {
252  return ScalableVector(this->code(), bits, -encoded_lanes);
253  }
254  return PrimType(this->code(), bits, encoded_lanes);
255  }
256 
258  TVM_FFI_INLINE PrimType WithLanes(int lanes) const {
259  return PrimType(this->code(), this->bits(), lanes);
260  }
261 
263  TVM_FFI_INLINE int32_t VScaleFactor() const {
264  int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
265  if (encoded_lanes >= -1) {
266  TVM_FFI_THROW(InternalError) << "A fixed length vector doesn't have a vscale factor.";
267  }
268  return -encoded_lanes;
269  }

434  TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
435  if (src->type_index == TypeIndex::kTVMFFINone) {
436  return TypedExpr<ExpectedType>::_type_is_nullable;
437  }
438  if (src->type_index < TypeIndex::kTVMFFIStaticObjectBegin ||
439  !details::IsObjectInstance<ExprNode>(src->type_index)) {
440  return false;
441  }
442  const auto* expr = static_cast<const ExprNode*>(
443  details::ObjectUnsafe::ObjectPtrFromUnowned<Object>(src->v_obj).get());
444  return details::AnyUnsafe::CheckAnyStrict<ExpectedType>(expr->ty);
445  }

447  TVM_FFI_INLINE static std::optional<TypedExpr<ExpectedType>> TryCastFromAnyView(
448  const TVMFFIAny* src) {
449  if (CheckAnyStrict(src)) {
450  if (src->type_index == TypeIndex::kTVMFFINone) {
451  return details::ObjectUnsafe::ObjectRefFromObjectPtr<TypedExpr<ExpectedType>>(nullptr);
452  }
453  return details::ObjectUnsafe::ObjectRefFromObjectPtr<TypedExpr<ExpectedType>>(
454  details::ObjectUnsafe::ObjectPtrFromUnowned<ExprNode>(src->v_obj));
455  }
456  return std::nullopt;
457  }
