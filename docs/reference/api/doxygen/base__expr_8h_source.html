140class PrimType final : public Type {
141 public:
146 TVM_DLL explicit PrimType(DLDataType dtype);
147
154 TVM_DLL PrimType(DLDataTypeCode code, int bits, int lanes = 1);
155
157 TVM_DLL static PrimType Int(int bits, int lanes = 1);
159 TVM_DLL static PrimType UInt(int bits, int lanes = 1);
161 TVM_DLL static PrimType Float(int bits, int lanes = 1);
163 TVM_DLL static PrimType BFloat(int bits, int lanes = 1);
165 TVM_DLL static PrimType Bool(int lanes = 1);
167 TVM_DLL static PrimType Void();
174 TVM_DLL static PrimType ScalableVector(DLDataTypeCode code, int bits, int lanes);
175
177 TVM_FFI_INLINE DLDataTypeCode code() const {
178 return static_cast<DLDataTypeCode>(static_cast<int>(get()->dtype.code));
179 }
180
182 TVM_FFI_INLINE int32_t bits() const { return get()->dtype.bits; }
183
188 TVM_FFI_INLINE int32_t lanes() const {
189 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
190 if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
191 TVM_FFI_THROW(InternalError)
192 << "Can't fetch the lanes of a scalable vector at a compile time.";
193 }
194 return encoded_lanes;
195 }
196
201 TVM_FFI_INLINE bool MatchesElementType(DLDataTypeCode code, int bits) const {
202 DLDataType dtype = get()->dtype;
203 return dtype.code == static_cast<uint8_t>(code) && dtype.bits == bits;
204 }
205
210 template <typename... Codes>
211 TVM_FFI_INLINE bool MatchesCode(Codes... codes) const {
212 uint8_t dtype_code = get()->dtype.code;
213 return ((dtype_code == static_cast<uint8_t>(codes)) || ...);
214 }
215
217 TVM_FFI_INLINE bool IsScalar() const {
218 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
219 return encoded_lanes == 1;
220 }
221
223 TVM_FFI_INLINE bool IsVoid() const {
224 DLDataType dtype = get()->dtype;
225 return dtype.code == static_cast<uint8_t>(DLDataTypeCode::kDLOpaqueHandle) && dtype.bits == 0 &&
226 static_cast<int16_t>(dtype.lanes) == 0;
227 }
228
230 TVM_FFI_INLINE bool IsScalableVector() const {
231 return static_cast<int16_t>(get()->dtype.lanes) < -1;
232 }
233
235 TVM_FFI_INLINE bool IsFixedLengthVector() const {
236 return static_cast<int16_t>(get()->dtype.lanes) > 1;
237 }
238
245 TVM_FFI_INLINE size_t StorageBytes() const {
246 DLDataType dtype = get()->dtype;
247 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
248 if (TVM_FFI_PREDICT_FALSE(encoded_lanes < 0)) {
249 TVM_FFI_THROW(InternalError)
250 << "Cannot compute compile-time storage bytes for non-fixed vector type " << dtype;
251 }
252 return static_cast<size_t>(
253 (static_cast<uint64_t>(dtype.bits) * static_cast<uint64_t>(dtype.lanes) + 7) / 8);
254 }
255
257 TVM_FFI_INLINE PrimType WithCode(DLDataTypeCode code) const {
258 DLDataType dtype = get()->dtype;
259 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
260 if (encoded_lanes < -1) {
261 return ScalableVector(code, dtype.bits, -encoded_lanes);
262 }
263 return PrimType(code, dtype.bits, encoded_lanes);
264 }
265
267 TVM_FFI_INLINE PrimType WithBits(int bits) const {
268 DLDataType dtype = get()->dtype;
269 int16_t encoded_lanes = static_cast<int16_t>(dtype.lanes);
270 if (encoded_lanes < -1) {
271 return ScalableVector(this->code(), bits, -encoded_lanes);
272 }
273 return PrimType(this->code(), bits, encoded_lanes);
274 }
275
277 TVM_FFI_INLINE PrimType WithLanes(int lanes) const {
278 return PrimType(this->code(), this->bits(), lanes);
279 }
280
282 TVM_FFI_INLINE int32_t VScaleFactor() const {
283 int16_t encoded_lanes = static_cast<int16_t>(get()->dtype.lanes);
284 if (encoded_lanes >= -1) {
285 TVM_FFI_THROW(InternalError) << "A fixed length vector doesn't have a vscale factor.";
286 }
287 return -encoded_lanes;
288 }
289
290 TVM_FFI_DEFINE_OBJECT_REF_METHODS_NOTNULLABLE(PrimType, Type, PrimTypeNode);
291};

483template <typename ExpectedType>
484inline constexpr bool use_default_type_traits_v<TypedExpr<ExpectedType>> = false;
485
486template <typename ExpectedType>
487struct TypeTraits<TypedExpr<ExpectedType>>
488 : public ObjectRefTypeTraitsBase<TypedExpr<ExpectedType>> {
489 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
490 if (src->type_index == TypeIndex::kTVMFFINone) {
491 return TypedExpr<ExpectedType>::_type_is_nullable;
492 }
493 if (src->type_index < TypeIndex::kTVMFFIStaticObjectBegin ||
494 !details::IsObjectInstance<ExprNode>(src->type_index)) {
495 return false;
496 }
497 // Non-owning: this only reads `ty`, and the owning form's incref/decref pair costs two
498 // atomics per check on a path every typed field assignment takes.
499 const auto* expr = details::ObjectUnsafe::RawObjectPtrFromUnowned<ExprNode>(src->v_obj);
500 return details::AnyUnsafe::CheckAnyViewStrict<ExpectedType>(expr->ty);
501 }
502
503 TVM_FFI_INLINE static std::string GetMismatchTypeInfo(const TVMFFIAny* src) {
504 if (src->type_index >= TypeIndex::kTVMFFIStaticObjectBegin && src->v_obj != nullptr &&
505 details::IsObjectInstance<ExprNode>(src->type_index)) {
506 const auto* expr = details::ObjectUnsafe::RawObjectPtrFromUnowned<ExprNode>(src->v_obj);
507 if (expr->ty.defined()) {
508 const auto* prim_ty = expr->ty.as<PrimTypeNode>();
509 std::string ty =
510 prim_ty ? std::string(ffi::DLDataTypeToString(prim_ty->dtype)) : expr->ty.GetTypeKey();
511 return TypeIndexToTypeKey(src->type_index) + "[ty=" + ty + "]";
512 }
513 }
514 return TypeTraitsBase::GetMismatchTypeInfo(src);
515 }
516};
