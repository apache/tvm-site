82inline Tensor fast_tanh_float(const Tensor& in, std::string name, std::string tag) {
83 using namespace tvm::prim;
84 // Clamp the inputs to the range [-9, 9] since anything outside
85 // this range is +/-1.0f in single-precision.
86 PrimType input_type = in->GetDataType();
87 auto x = maximum(MakeConst(input_type, -9.0), minimum(MakeConst(input_type, 9.0), in));
88
89 // The monomial coefficients of the numerator polynomial (odd).
90 auto alpha_1 = MakeConst(input_type, 4.89352455891786e-03);
91 auto alpha_3 = MakeConst(input_type, 6.37261928875436e-04);
92 auto alpha_5 = MakeConst(input_type, 1.48572235717979e-05);
93 auto alpha_7 = MakeConst(input_type, 5.12229709037114e-08);
94 auto alpha_9 = MakeConst(input_type, -8.60467152213735e-11);
95 auto alpha_11 = MakeConst(input_type, 2.00018790482477e-13);
96 auto alpha_13 = MakeConst(input_type, -2.76076847742355e-16);
97
98 // The monomial coefficients of the denominator polynomial (even).
99 auto beta_0 = MakeConst(input_type, 4.89352518554385e-03);
100 auto beta_2 = MakeConst(input_type, 2.26843463243900e-03);
101 auto beta_4 = MakeConst(input_type, 1.18534705686654e-04);
102 auto beta_6 = MakeConst(input_type, 1.19825839466702e-06);
103
104 return compute(
105 x->shape,
106 [&](const ffi::Array<PrimVar>& i) {
107 auto x2 = x(i) * x(i);
108 auto p = x2 * alpha_13 + alpha_11;
109 p = x2 * p + alpha_9;
110 p = x2 * p + alpha_7;
111 p = x2 * p + alpha_5;
112 p = x2 * p + alpha_3;
113 p = x2 * p + alpha_1;
114 p = x(i) * p;
115
116 auto q = x2 * beta_6 + beta_4;
117 q = x2 * q + beta_2;
118 q = x2 * q + beta_0;
119 return p / q;
120 },
121 name, tag);
122}

210inline Tensor sign(const Tensor& x, std::string name = "T_sign", std::string tag = kElementWise) {
211 using namespace tvm::prim;
212 return compute(
213 x->shape,
214 [&](const ffi::Array<PrimVar>& i) {
215 PrimType x_type(x->GetDataType());
216 PrimExpr zero = MakeConst(x_type, 0);
217 PrimExpr one = MakeConst(x_type, 1);
218 PrimExpr minus_one = MakeConst(x_type, -1);
219 auto s1 = tvm::prim::Select((x(i) < zero), minus_one, zero);
220 auto s2 = tvm::prim::Select((x(i) > zero), one, s1);
221 return s2;
222 },
223 name, tag);

291inline Tensor cast(const Tensor& x, PrimType type, std::string name = "T_cast",
292 std::string tag = kElementWise) {
293 return compute(
294 x->shape,
295 [&](const ffi::Array<PrimVar>& i) -> PrimExpr {
296 auto expr = x(i);
297 PrimType expr_ty = expr.ty();
298 if (expr_ty.MatchesElementType(type.code(), type.bits())) {
299 if (expr_ty.lanes() == type.lanes()) {
300 return expr;
301 } else if (expr_ty.lanes() == 1 && type.IsFixedLengthVector()) {
302 return tvm::prim::Broadcast(expr, type.lanes());
303 }
304 }
305
306 return tvm::prim::cast(type, x(i));
307 },
308 name, tag);
309}

345inline Tensor elemwise_sum(const ffi::Array<Tensor>& xs, std::string name = "T_elemwise_sum",
346 std::string tag = kElementWise) {
347 TVM_FFI_ICHECK_GT(xs.size(), 0) << "elemwise sum must have at least one input tensor.";
348 return compute(
349 xs[0]->shape,
350 [&](const ffi::Array<PrimVar>& i) {
351 auto sum_expr = xs[0](i);
352 for (size_t j = 1; j < xs.size(); j++) {

354 }
355 return sum_expr;

426inline Tensor fast_exp_float32(const Tensor& _x, std::string name, std::string tag) {
427 PrimType f32_ty = PrimType::Float(32);
428 auto x_hi = FloatImm(f32_ty, 88.3762626647950f);
429 auto x_lo = FloatImm(f32_ty, -88.3762626647949f);
430 auto log2e = FloatImm(f32_ty, 1.44269504088896341f);
431 auto ln2 = FloatImm(f32_ty, 0.6931471805599453f);
432 PrimExpr p[6] = {FloatImm(f32_ty, 1.9875691500E-4f), FloatImm(f32_ty, 1.3981999507E-3f),
433 FloatImm(f32_ty, 8.3334519073E-3f), FloatImm(f32_ty, 4.1665795894E-2f),
434 FloatImm(f32_ty, 1.6666665459E-1f), FloatImm(f32_ty, 5.0000001201E-1f)};
435 auto one = FloatImm(f32_ty, 1.0f);
436 auto one_half = FloatImm(f32_ty, 0.5f);
437 auto b = FloatImm(f32_ty, 127.0f);
438
439 return compute(
440 _x->shape,
441 [&](const ffi::Array<PrimVar>& i) {
442 // clamp x
443 auto x = ::tvm::max(::tvm::min(_x(i), x_hi), x_lo);
444 // integer part
445 auto n = ::tvm::prim::floor(x * log2e + one_half);
446 // fractional part
447 auto f = x - n * ln2;
448 auto y =
449 (((((p[0] * f + p[1]) * f + p[2]) * f + p[3]) * f + p[4]) * f + p[5]) * f * f + f + one;
450 // Return 2^m * exp(r).
451 auto ef = tvm::prim::reinterpret(PrimType::Float(32),
452 ::tvm::prim::cast(PrimType::Int(32), n + b) << 23);
453 return ::tvm::max(ef * y, _x(i)); // NOLINT(*)
454 },
455 name, tag);
456}
