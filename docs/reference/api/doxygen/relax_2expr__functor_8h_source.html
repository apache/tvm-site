106template <typename R, typename... Args>
107class ExprFunctor<R(const Expr& n, Args...)> {
108 private:
109 using TSelf = ExprFunctor<R(const Expr& n, Args...)>;
110 using FType = tvm::NodeFunctor<R(const ffi::ObjectRef& n, TSelf* self, Args...)>;
111
112 public:
114 using result_type = R;
116 virtual ~ExprFunctor() {}
123 R operator()(const Expr& n, Args... args) { return VisitExpr(n, std::forward<Args>(args)...); }
130 virtual R VisitExpr(const Expr& n, Args... args) {
131 TVM_FFI_ICHECK(n.defined())
132 << "Found null pointer node while traversing AST. The previous pass may "
133 "have generated invalid data.";
134 static FType vtable = InitVTable();
135 if (vtable.can_dispatch(n)) {
136 return vtable(n, this, std::forward<Args>(args)...);
137 }
138 return VisitExprFallback_(n.get(), std::forward<Args>(args)...);
139 }
140 // Functions that can be overriden by subclass
141 // NOTE: cross dialect calls are invoked through global var
142 // We do not expect inline PrimFunc to appear in relax IR.
143 virtual R VisitExpr_(const ConstantNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
144 virtual R VisitExpr_(const TupleNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
145 virtual R VisitExpr_(const VarNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
146 virtual R VisitExpr_(const DataflowVarNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
147 virtual R VisitExpr_(const ShapeExprNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
148 virtual R VisitExpr_(const ExternFuncNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
149 virtual R VisitExpr_(const GlobalVarNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
150 virtual R VisitExpr_(const FunctionNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
151 virtual R VisitExpr_(const CallNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
152 virtual R VisitExpr_(const SeqExprNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
153 virtual R VisitExpr_(const IfNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
154 virtual R VisitExpr_(const OpNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
155 virtual R VisitExpr_(const TupleGetItemNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
156 virtual R VisitExprFallback_(const ExprNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
157 virtual R VisitExpr_(const StringImmNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
158 virtual R VisitExpr_(const DataTypeImmNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
159 virtual R VisitExprDefault_(const ffi::Object* op, Args...) {
160 TVM_FFI_THROW(InternalError) << "Do not have a default for " << op->GetTypeKey();
161 throw;
162 }
163
164 private:
165 // initialize the vtable.
166 static FType InitVTable() {
167 FType vtable;
168 // Set dispatch
169 RELAX_EXPR_FUNCTOR_DISPATCH(ConstantNode);
170 RELAX_EXPR_FUNCTOR_DISPATCH(TupleNode);
171 RELAX_EXPR_FUNCTOR_DISPATCH(VarNode);
172 RELAX_EXPR_FUNCTOR_DISPATCH(DataflowVarNode);
173 RELAX_EXPR_FUNCTOR_DISPATCH(ShapeExprNode);
174 RELAX_EXPR_FUNCTOR_DISPATCH(ExternFuncNode);
175 RELAX_EXPR_FUNCTOR_DISPATCH(GlobalVarNode);
176 RELAX_EXPR_FUNCTOR_DISPATCH(FunctionNode);
177 RELAX_EXPR_FUNCTOR_DISPATCH(CallNode);
178 RELAX_EXPR_FUNCTOR_DISPATCH(SeqExprNode);
179 RELAX_EXPR_FUNCTOR_DISPATCH(IfNode);
180 RELAX_EXPR_FUNCTOR_DISPATCH(OpNode);
181 RELAX_EXPR_FUNCTOR_DISPATCH(TupleGetItemNode);
182 RELAX_EXPR_FUNCTOR_DISPATCH(StringImmNode);
183 RELAX_EXPR_FUNCTOR_DISPATCH(DataTypeImmNode);
184 vtable.Finalize();
185 return vtable;
186 }
