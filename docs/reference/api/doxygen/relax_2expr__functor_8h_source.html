146 class ExprFunctor<R(const Expr& n, Args...)> {
147  private:
148  using TSelf = ExprFunctor<R(const Expr& n, Args...)>;
149  using FType = tvm::NodeFunctor<R(const ffi::ObjectRef& n, TSelf* self, Args...)>;
150 
151  public:
153  using result_type = R;
155  virtual ~ExprFunctor() {}
162  R operator()(const Expr& n, Args... args) { return VisitExpr(n, std::forward<Args>(args)...); }
169  virtual R VisitExpr(const Expr& n, Args... args) {
170  TVM_FFI_ICHECK(n.defined())
171  << "Found null pointer node while traversing AST. The previous pass may "
172  "have generated invalid data.";
173  static FType vtable = InitVTable();
174  return vtable(n, this, std::forward<Args>(args)...);
175  }
176  // Functions that can be overriden by subclass
177  // NOTE: cross dialect calls are invoked through global var
178  // We do not expect inline PrimFunc to appear in relax IR.
179  virtual R VisitExpr_(const ConstantNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
180  virtual R VisitExpr_(const TupleNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
181  virtual R VisitExpr_(const VarNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
182  virtual R VisitExpr_(const DataflowVarNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
183  virtual R VisitExpr_(const ShapeExprNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
184  virtual R VisitExpr_(const ExternFuncNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
185  virtual R VisitExpr_(const GlobalVarNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
186  virtual R VisitExpr_(const FunctionNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
187  virtual R VisitExpr_(const CallNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
188  virtual R VisitExpr_(const SeqExprNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
189  virtual R VisitExpr_(const IfNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
190  virtual R VisitExpr_(const OpNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
191  virtual R VisitExpr_(const TupleGetItemNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
192  virtual R VisitExpr_(const PrimExprNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
193  virtual R VisitExpr_(const StringImmNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
194  virtual R VisitExpr_(const DataTypeImmNode* op, Args... args) EXPR_FUNCTOR_DEFAULT;
195  virtual R VisitExprDefault_(const ffi::Object* op, Args...) {
196  TVM_FFI_THROW(InternalError) << "Do not have a default for " << op->GetTypeKey();
197  throw;
198  }
199 
200  private:
201  // initialize the vtable.
202  static FType InitVTable() {
203  FType vtable;
204  // Set dispatch
205  RELAX_EXPR_FUNCTOR_DISPATCH(ConstantNode);
206  RELAX_EXPR_FUNCTOR_DISPATCH(TupleNode);
207  RELAX_EXPR_FUNCTOR_DISPATCH(VarNode);
208  RELAX_EXPR_FUNCTOR_DISPATCH(DataflowVarNode);
209  RELAX_EXPR_FUNCTOR_DISPATCH(ShapeExprNode);
210  RELAX_EXPR_FUNCTOR_DISPATCH(ExternFuncNode);
211  RELAX_EXPR_FUNCTOR_DISPATCH(GlobalVarNode);
212  RELAX_EXPR_FUNCTOR_DISPATCH(FunctionNode);
213  RELAX_EXPR_FUNCTOR_DISPATCH(CallNode);
214  RELAX_EXPR_FUNCTOR_DISPATCH(SeqExprNode);
215  RELAX_EXPR_FUNCTOR_DISPATCH(IfNode);
216  RELAX_EXPR_FUNCTOR_DISPATCH(OpNode);
217  RELAX_EXPR_FUNCTOR_DISPATCH(TupleGetItemNode);
218  RELAX_PRIM_EXPR_NODE_DISPATCH_LIST(RELAX_EXPR_FUNCTOR_DISPATCH);
219  RELAX_EXPR_FUNCTOR_DISPATCH(StringImmNode);
220  RELAX_EXPR_FUNCTOR_DISPATCH(DataTypeImmNode);
221  vtable.Finalize();
222  return vtable;
223  }
