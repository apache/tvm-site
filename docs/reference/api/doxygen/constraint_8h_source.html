194 ConstraintContextObj() = default;
201 ConstraintContextObj(ffi::ObjectPtr<ConstraintContextObj> parent, int64_t parent_local_prefix)
202 : parent_(std::move(parent)), parent_prefix_size_(parent_local_prefix) {
203 TVM_FFI_CHECK(
204 parent_ && 0 <= parent_local_prefix && parent_local_prefix <= parent_->LocalSize(),
205 ValueError)
206 << "Constraint context parent local prefix is out of bounds";
207 }
208 // Context nodes have stable identity; use Snapshot to capture their current position.
209 ConstraintContextObj(const ConstraintContextObj&) = delete;
210 ConstraintContextObj& operator=(const ConstraintContextObj&) = delete;

217 void Append(const Constraint& constraint) { constraints_.push_back(constraint); }

248 template <typename F>
249 void ForEach(F&& f) const {
250 // Keep this object alive if a callback replaces the caller's last handle.
251 auto context = ffi::GetObjectPtr<ConstraintContextObj>(const_cast<ConstraintContextObj*>(this));
252 context->ForEach(context->LocalSize(), std::forward<F>(f));
253 }

262 ffi::Array<Constraint> ToArray() const { return ToArray(LocalSize()); }
263
265 static constexpr bool _type_mutable = true;
267 static void RegisterReflection();
268 TVM_FFI_DECLARE_OBJECT_INFO_FINAL("sym.ConstraintContext", ConstraintContextObj, ffi::Object);
269
270 private:
271 friend class AnalyzerObj;
272 // Materialize the fixed inherited sequence and the requested local prefix.
273 ffi::Array<Constraint> ToArray(int64_t local_prefix) const {
274 if (!parent_) {
275 if (local_prefix == LocalSize()) return constraints_;
276 return ffi::Array<Constraint>(constraints_.begin(), constraints_.begin() + local_prefix);
277 }
278 auto result = parent_->ToArray(parent_prefix_size_);
279 if (local_prefix != 0) {
280 result.insert(result.end(), constraints_.begin(), constraints_.begin() + local_prefix);
281 }
282 return result;
283 }
284
285 // Traverse a fixed prefix even if callbacks append more local constraints.
286 template <typename F>
287 void ForEach(int64_t local_prefix, F&& f) const {
288 if (parent_) parent_->ForEach(parent_prefix_size_, f);
289 for (int64_t i = 0; i < local_prefix; ++i) {
290 // Read by value: a callback may append and reallocate the local array.
291 const Constraint constraint = constraints_[i];
292 f(constraint);
293 }
294 }
295
296 // Fixed parent context; null for a root. Parent links are acyclic.
297 const ffi::ObjectPtr<ConstraintContextObj> parent_;
298 // Fixed prefix length in the parent's LOCAL sequence, not its effective count.
299 const int64_t parent_prefix_size_{0};
300 // Append-only local sequence; captured entries cannot be replaced or mutated through aliases.
301 ffi::Array<Constraint> constraints_;
302};
