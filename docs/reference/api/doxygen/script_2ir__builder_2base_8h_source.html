277template <typename TFrame>
278inline ffi::Optional<TFrame> IRBuilderNode::FindFrame() const {
279 using TFrameNode = typename TFrame::ContainerType;
280 for (auto it = frames.rbegin(); it != frames.rend(); ++it) {
281 if (const TFrameNode* p = (*it).template as<TFrameNode>()) {
282 return ffi::GetRef<TFrame>(p);
283 }
284 }
285 return std::nullopt;
286}
287
288template <typename TFrame>
289inline ffi::Optional<TFrame> IRBuilderNode::GetLastFrame() const {
290 using TFrameNode = typename TFrame::ContainerType;
291 if (!frames.empty() && frames.back()->IsInstance<TFrameNode>()) {
292 return frames.back().template as_or_throw<TFrame>();
293 }
294 return std::nullopt;
295}

297template <typename TObjectRef>
298inline TObjectRef IRBuilderNode::Get() const {
299 using TObject = typename TObjectRef::ContainerType;
300 TVM_FFI_CHECK(result.has_value(), IndexError) << "No result exists in IRBuilder yet";
301 const auto* n = result.as<TObject>();
302 TVM_FFI_CHECK(n != nullptr, TypeError)
303 << "IRBuilder result is not of type: " << TObject::_type_key;
304 return ffi::GetRef<TObjectRef>(n);
305}
