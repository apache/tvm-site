272template <typename TFrame>
273inline ffi::Optional<TFrame> IRBuilderNode::FindFrame() const {
274 using TFrameNode = typename TFrame::ContainerType;
275 for (auto it = frames.rbegin(); it != frames.rend(); ++it) {
276 if (const TFrameNode* p = (*it).template as<TFrameNode>()) {
277 return ffi::GetRef<TFrame>(p);
278 }
279 }
280 return std::nullopt;
281}
282
283template <typename TFrame>
284inline ffi::Optional<TFrame> IRBuilderNode::GetLastFrame() const {
285 using TFrameNode = typename TFrame::ContainerType;
286 if (!frames.empty() && frames.back()->IsInstance<TFrameNode>()) {
287 return frames.back().template as_or_throw<TFrame>();
288 }
289 return std::nullopt;
290}

292template <typename TObjectRef>
293inline TObjectRef IRBuilderNode::Get() const {
294 using TObject = typename TObjectRef::ContainerType;
295 TVM_FFI_CHECK(result.has_value(), IndexError) << "No result exists in IRBuilder yet";
296 const auto* n = result.as<TObject>();
297 TVM_FFI_CHECK(n != nullptr, TypeError)
298 << "IRBuilder result is not of type: " << TObject::_type_key;
299 return ffi::GetRef<TObjectRef>(n);
300}
