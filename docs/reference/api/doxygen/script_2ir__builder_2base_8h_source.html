280template <typename TFrame>
281inline ffi::Optional<TFrame> IRBuilderObj::FindFrame() const {
282 using TFrameNode = typename TFrame::ContainerType;
283 for (auto it = frames.rbegin(); it != frames.rend(); ++it) {
284 if (const TFrameNode* p = (*it).template as<TFrameNode>()) {
285 return ffi::GetRef<TFrame>(p);
286 }
287 }
288 return std::nullopt;
289}
290
291template <typename TFrame>
292inline ffi::Optional<TFrame> IRBuilderObj::GetLastFrame() const {
293 using TFrameNode = typename TFrame::ContainerType;
294 if (!frames.empty() && frames.back()->IsInstance<TFrameNode>()) {
295 return frames.back().template as_or_throw<TFrame>();
296 }
297 return std::nullopt;
298}

300template <typename TObjectRef>
301inline TObjectRef IRBuilderObj::Get() const {
302 using TObject = typename TObjectRef::ContainerType;
303 TVM_FFI_CHECK(result.has_value(), IndexError) << "No result exists in IRBuilder yet";
304 const auto* n = result.as<TObject>();
305 TVM_FFI_CHECK(n != nullptr, TypeError)
306 << "IRBuilder result is not of type: " << TObject::_type_key;
307 return ffi::GetRef<TObjectRef>(n);
308}
