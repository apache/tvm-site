284template <typename TFrame>
285inline ffi::Optional<TFrame> IRBuilderObj::FindFrame() const {
286 using TFrameNode = typename TFrame::ContainerType;
287 for (auto it = frames.rbegin(); it != frames.rend(); ++it) {
288 if (const TFrameNode* p = (*it).template as<TFrameNode>()) {
289 return ffi::GetRef<TFrame>(p);
290 }
291 }
292 return std::nullopt;
293}
294
295template <typename TFrame>
296inline ffi::Optional<TFrame> IRBuilderObj::GetLastFrame() const {
297 using TFrameNode = typename TFrame::ContainerType;
298 if (!frames.empty() && frames.back()->IsInstance<TFrameNode>()) {
299 return frames.back().template as_or_throw<TFrame>();
300 }
301 return std::nullopt;
302}

304template <typename TObjectRef>
305inline TObjectRef IRBuilderObj::Get() const {
306 using TObject = typename TObjectRef::ContainerType;
307 TVM_FFI_CHECK(result.has_value(), IndexError) << "No result exists in IRBuilder yet";
308 const auto* n = result.as<TObject>();
309 TVM_FFI_CHECK(n != nullptr, TypeError)
310 << "IRBuilder result is not of type: " << TObject::_type_key;
311 return ffi::GetRef<TObjectRef>(n);
312}
