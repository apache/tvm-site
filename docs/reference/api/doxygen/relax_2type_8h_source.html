175 ffi::Optional<ffi::Array<PrimExpr>> GetShape() const {
176 if (!shape.has_value()) return {};
177 const Expr& shape_expr = this->shape.value();
178 if (shape_expr->ty.IsMissing()) return {};
179 if (const auto* shape_ty = shape_expr->ty.as<ShapeTypeNode>()) {

347template <typename T>
348inline ffi::Optional<T> MatchType(const Expr& expr) {
349 if (!expr.defined()) {
350 return std::nullopt;
351 }
352 using TNode = typename T::ContainerType;
353 if (const TNode* ptr = expr->ty.as<TNode>()) {
354 return ffi::GetRef<T>(ptr);
355 } else {
356 return std::nullopt;
357 }
358}
