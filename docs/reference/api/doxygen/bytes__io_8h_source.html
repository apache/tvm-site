57 size_t Read(void* ptr, size_t size) override {
58 size_t nread = std::min(size_ - pos_, size);
59 if (nread != 0) std::memcpy(ptr, data_ + pos_, nread);
60 pos_ += nread;
61 return nread;
62 }
63
64 size_t Write(const void*, size_t) override {
65 TVM_FFI_THROW(InternalError) << "write to input stream";

67 }

69 private:
