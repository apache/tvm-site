65inline ffi::Array<Tensor> make_extern(const ffi::Array<ffi::Array<PrimExpr>>& out_shapes,
66 const std::vector<PrimType>& out_types,
67 const ffi::Array<Tensor>& inputs, FExtern fextern,
68 std::string name, std::string tag,
69 ::tvm::ffi::Map<ffi::String, ffi::Any> attrs) {
70 TVM_FFI_ICHECK_EQ(out_shapes.size(), out_types.size())
71 << "make_extern: out_shapes and out_types must have equal size";
72
73 ffi::Array<TensorVar> input_placeholders;
74 for (auto t : inputs) {
75 input_placeholders.push_back(
76 tvm::tirx::TensorVar(t->op.value()->name, tvm::tirx::TensorType(t->shape, t->dtype)));
77 }
78 ffi::Array<TensorVar> output_placeholders;
79 for (size_t i = 0; i < out_shapes.size(); ++i) {
80 output_placeholders.push_back(
81 tvm::tirx::TensorVar(name, tvm::tirx::TensorType(out_shapes[i], out_types[i])));
82 }
83
84 auto body = fextern(input_placeholders, output_placeholders);
85 auto body_stmt = tvm::Evaluate(body);
86
87 auto op = ExternOp(name, tag, attrs, inputs, input_placeholders, output_placeholders, body_stmt);
88
89 ffi::Array<Tensor> outputs;
90 for (size_t i = 0; i < output_placeholders.size(); ++i) {
91 outputs.push_back(op.output(i));
92 }
93 return outputs;
94}

104inline Expr pack_buffer(TensorVar buf) {
105 using namespace tvm::prim;
106 TVM_FFI_ICHECK_GT(buf->shape.size(), 0) << "buf shape must have at least one element";
107 Expr shape = Call(PtrType(PrimType::Int(64)), tvm::tirx::stack_make_shape_op(), buf->shape);
108 Expr strides;
109 if (buf->strides.size() > 0) {
110 strides = Call(PtrType(PrimType::Int(64)), tvm::tirx::stack_make_shape_op(), buf->strides);
111 } else {
112 strides = PrimExpr(0);
113 }
114 ffi::Array<Expr> pack_args{buf->data,
115 shape,
116 strides,
117 IntImm::Int32(static_cast<int64_t>(buf->shape.size())),
118 MakeConst(PrimType(buf->dtype), 0),
119 buf->elem_offset};
120 return Call(PtrType::VoidPtrType(), tvm::tirx::stack_make_dltensor_op(), pack_args);
121}
