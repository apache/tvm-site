284 bool SEqual(const SeqExprNode* other,
285 ffi::TypedFunction<bool(AnyView, AnyView, bool, AnyView)> equal) const {
286 // Establish mappings for symbolic variables defined by bindings before
287 // comparing their uses in the SeqExpr result type and body.
288 return equal(blocks, other->blocks, false, "blocks") && equal(ty, other->ty, false, "ty") &&
289 equal(body, other->body, false, "body");
290 }
291
292 int64_t SHash(int64_t init_hash, ffi::TypedFunction<int64_t(AnyView, int64_t, bool)> hash) const {
293 int64_t hash_value = init_hash;
294 hash_value = hash(blocks, hash_value, false);
295 hash_value = hash(ty, hash_value, false);
296 hash_value = hash(body, hash_value, false);
297 return hash_value;
298 }
299 TVM_FFI_DECLARE_OBJECT_INFO_FINAL("relax.expr.SeqExpr", SeqExprNode, ExprNode);
300};

304 /* \brief Implicit conversion constructor
305 *
306 * Relax nodes that introduce a new scope (e.g. `relax::Function`)
307 * are required to be held as SeqExpr. This implicit conversion

309 * C++ compile-time type is a `relax::Expr`. For example,
310 * a transform may use `func.CopyOnWrite()->body = expr;`.
