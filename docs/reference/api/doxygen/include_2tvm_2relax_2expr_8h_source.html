283 bool SEqual(const SeqExprObj* other,
284 ffi::TypedFunction<bool(AnyView, AnyView, bool, AnyView)> equal) const {
285 // Establish mappings for symbolic variables defined by bindings before
286 // comparing their uses in the SeqExpr result type and body.
287 return equal(blocks, other->blocks, false, "blocks") && equal(ty, other->ty, false, "ty") &&
288 equal(body, other->body, false, "body");
289 }
290
291 int64_t SHash(int64_t init_hash, ffi::TypedFunction<int64_t(AnyView, int64_t, bool)> hash) const {
292 int64_t hash_value = init_hash;
293 hash_value = hash(blocks, hash_value, false);
294 hash_value = hash(ty, hash_value, false);
295 hash_value = hash(body, hash_value, false);
296 return hash_value;
297 }
298 TVM_FFI_DECLARE_OBJECT_INFO_FINAL("relax.expr.SeqExpr", SeqExprObj, ExprObj);
299};

302 public:
303 /* \brief Implicit conversion constructor
304 *
305 * Relax nodes that introduce a new scope (e.g. `relax::Function`)
306 * are required to be held as SeqExpr. This implicit conversion
307 * provides allows callsites to use these member variables when the
308 * C++ compile-time type is a `relax::Expr`. For example,
309 * a transform may use `func.CopyOnWrite()->body = expr;`.
310 *
311 * If the expression is already a `relax::SeqExpr`, the same
