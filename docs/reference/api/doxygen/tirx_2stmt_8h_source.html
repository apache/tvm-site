401 template <typename... Args>
402 static Stmt Flatten(Args&&... seq_args) {
403 ffi::Array<Stmt> seq;
404
405 ffi::details::for_each(Flattener(&seq), std::forward<Args>(seq_args)...);
406
407 if (seq.empty()) {
408 return Evaluate(0);
409 } else if (seq.size() == 1) {
410 return seq[0];
411 }
412
413 // If the argument is a single SeqStmt argument with no
414 // flattening or unwrapping required, then we may
415 // return the SeqStmt as-is.
416 if constexpr (sizeof...(seq_args) == 1) {
417 if (auto opt = Flattener::AsSeqStmt(std::forward<Args>(seq_args)...)) {
418 SeqStmt original = opt.value();
419 bool all_same = [&]() {
420 if (original->seq.size() != seq.size()) {
421 return false;
422 }
423 for (size_t i = 0; i < seq.size(); i++) {
424 if (!original->seq[i].same_as(seq[i])) {
425 return false;
426 }
427 }
428 return true;
429 }();
430 if (all_same) {
431 return original;
432 }
433 }
434 }
435
436 return SeqStmt(seq);
437 }

443 template <typename T>
444 static ffi::Optional<SeqStmt> AsSeqStmt(const T& t) {
445 if constexpr (std::is_same_v<T, SeqStmt>) {
446 return t;
447 }
448 if constexpr (!std::is_base_of_v<T, SeqStmt>) {
449 return std::nullopt;
450 }
451 if constexpr (std::is_base_of_v<Stmt, T>) {
452 if (const SeqStmtNode* ptr = t.template as<SeqStmtNode>()) {
453 return ffi::GetRef<SeqStmt>(ptr);
454 } else {
455 return std::nullopt;
456 }
457 }
458 return std::nullopt;

461 template <typename T>
462 void operator()(size_t i, const T& stmt_or_seq) const {
463 if constexpr (std::is_base_of_v<ObjectRef, T>) {
464 // Early bail-out, applicable to any ObjectRef
465 if (!stmt_or_seq.defined()) {
466 return;
467 }
468 }
469
470 if constexpr (std::is_same_v<T, SeqStmt>) {
471 // Static type-checking for a SeqStmt that could be flattened.
472 (*this)(0, stmt_or_seq->seq);
473 return;
474 }
475
476 if constexpr (std::is_base_of_v<T, SeqStmt>) {
477 // Dynamic type-checking for a SeqStmt that could be
478 // flattened.
479 if (auto* op = stmt_or_seq.template as<SeqStmtNode>()) {
480 operator()(0, op->seq);
481 return;
482 }
483 }
484
485 if constexpr (std::is_base_of_v<T, Evaluate>) {
486 // Evaluate(0) is used to represent a no-op, and may be
487 // generated by previous calls to SeqStmt::Flatten(). These
488 // should be removed to ensure that Flatten(a+b) is equivalent
489 // to Flatten(Flatten(a), Flatten(b)).
490 if (auto* op = stmt_or_seq.template as<EvaluateNode>()) {
491 if (auto* as_int = op->value.template as<IntImmNode>(); as_int && as_int->value == 0) {
492 return;
493 }
494 }
495 }
496
497 if constexpr (std::is_base_of_v<Stmt, T>) {
498 // Any other Stmt type just gets appended.
499 seq_->push_back(stmt_or_seq);
500 } else {
501 // Anything else is treated as an iterable of Stmt.
502 for (auto v : stmt_or_seq) {
503 this->operator()(0, v);
504 }
505 }
506 }
507
508 private:
509 ffi::Array<Stmt>* seq_;
510 };
