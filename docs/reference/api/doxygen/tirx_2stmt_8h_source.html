744inline bool IsNoOp(const Stmt& stmt) {
745 if (!stmt.defined()) return true;
746 if (const auto* op = stmt.as<EvaluateNode>()) return op->value.as<IntImmNode>() != nullptr;
747 if (const auto* op = stmt.as<SeqStmtNode>()) return op->seq.empty();
748 return false;
749}
