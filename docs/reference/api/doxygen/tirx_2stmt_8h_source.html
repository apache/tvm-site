363 template <typename... Args>
364 static Stmt Flatten(Args&&... seq_args) {
365 ffi::Array<Stmt> seq;
366
367 ffi::details::for_each(Flattener(&seq), std::forward<Args>(seq_args)...);
368
369 if (seq.empty()) {
370 return Evaluate(0);
371 } else if (seq.size() == 1) {
372 return seq[0];
373 }
374
375 // If the argument is a single SeqStmt argument with no
376 // flattening or unwrapping required, then we may
377 // return the SeqStmt as-is.
378 if constexpr (sizeof...(seq_args) == 1) {
379 if (auto opt = Flattener::AsSeqStmt(std::forward<Args>(seq_args)...)) {
380 SeqStmt original = opt.value();
381 bool all_same = [&]() {
382 if (original->seq.size() != seq.size()) {
383 return false;
384 }
385 for (size_t i = 0; i < seq.size(); i++) {
386 if (!original->seq[i].same_as(seq[i])) {
387 return false;
388 }
389 }
390 return true;
391 }();
392 if (all_same) {
393 return original;
394 }
395 }
396 }
397
398 return SeqStmt(seq);
399 }

405 template <typename T>
406 static ffi::Optional<SeqStmt> AsSeqStmt(const T& t) {
407 if constexpr (std::is_same_v<T, SeqStmt>) {
408 return t;
409 }
410 if constexpr (!std::is_base_of_v<T, SeqStmt>) {
411 return std::nullopt;
412 }
413 if constexpr (std::is_base_of_v<Stmt, T>) {
414 if (const SeqStmtNode* ptr = t.template as<SeqStmtNode>()) {
415 return ffi::GetRef<SeqStmt>(ptr);
416 } else {
417 return std::nullopt;
418 }
419 }
420 return std::nullopt;
421 }
422
423 void operator()(size_t i, const ffi::Optional<Stmt>& stmt) const {
424 if (stmt.has_value()) (*this)(i, stmt.value());
425 }
426
427 template <typename T>
428 void operator()(size_t i, const T& stmt_or_seq) const {
429 if constexpr (std::is_base_of_v<ObjectRef, T>) {
430 // Early bail-out, applicable to any ObjectRef
431 if (!stmt_or_seq.defined()) {
432 return;
433 }
434 }
435
436 if constexpr (std::is_same_v<T, SeqStmt>) {
437 // Static type-checking for a SeqStmt that could be flattened.
438 (*this)(0, stmt_or_seq->seq);
439 return;
440 }
441
442 if constexpr (std::is_base_of_v<T, SeqStmt>) {
443 // Dynamic type-checking for a SeqStmt that could be
444 // flattened.
445 if (auto* op = stmt_or_seq.template as<SeqStmtNode>()) {
446 operator()(0, op->seq);
447 return;
448 }
449 }
450
451 if constexpr (std::is_base_of_v<T, Evaluate>) {
452 // Evaluate(0) is used to represent a no-op, and may be
453 // generated by previous calls to SeqStmt::Flatten(). These
454 // should be removed to ensure that Flatten(a+b) is equivalent
455 // to Flatten(Flatten(a), Flatten(b)).
456 if (auto* op = stmt_or_seq.template as<EvaluateNode>()) {
457 if (auto* as_int = op->value.template as<IntImmNode>(); as_int && as_int->value == 0) {
458 return;
459 }
460 }
461 }
462
463 if constexpr (std::is_base_of_v<Stmt, T>) {
464 // Any other Stmt type just gets appended.
465 seq_->push_back(stmt_or_seq);
466 } else {
467 // Anything else is treated as an iterable of Stmt.
468 for (auto v : stmt_or_seq) {
469 this->operator()(0, v);
470 }
471 }
472 }
473
474 private:
475 ffi::Array<Stmt>* seq_;
476 };
