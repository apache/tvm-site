33inline ffi::Optional<ffi::String> GetThreadBinding(const ForNode* loop) {
34 if (auto tag = loop->annotations.Get("thread_binding")) {
35 TVM_FFI_CHECK(loop->kind == ForKind::kParallel, ValueError)
36 << "thread_binding requires a parallel loop";
37 TVM_FFI_CHECK(tag->as<ffi::String>().has_value(), TypeError)
38 << "thread_binding annotation must be a string";
39 TVM_FFI_CHECK(loop->HasTrivialStep(), ValueError) << "Thread binding loops require a unit step";
40 return tag->cast<ffi::String>();
41 }
42 return std::nullopt;
43}
44inline ffi::Optional<ffi::String> GetThreadBinding(const For& loop) {
45 return GetThreadBinding(loop.get());
46}

108inline bool IsNoOp(const Stmt& stmt) {
109 if (!stmt.defined()) return true;
110 if (const auto* op = stmt.as<EvaluateNode>()) return op->value.as<IntImmNode>() != nullptr;
111 if (const auto* op = stmt.as<SeqStmtNode>()) return op->seq.empty();
112 return false;
113}
