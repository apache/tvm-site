378class SeqStmt : public Stmt {
379 public:
385 TVM_DLL explicit SeqStmt(ffi::Array<Stmt> seq, Span span = Span());
386
388 size_t size() const { return operator->()->size(); }
392 Stmt operator[](size_t index) const { return (*(operator->()))[index]; }
413 template <typename... Args>
414 static Stmt Flatten(Args&&... seq_args) {
415 ffi::Array<Stmt> seq;
416
417 ffi::details::for_each(Flattener(&seq), std::forward<Args>(seq_args)...);
418
419 if (seq.empty()) {
420 return Evaluate(0);
421 } else if (seq.size() == 1) {
422 return seq[0];
423 }
424
425 // If the argument is a single SeqStmt argument with no
426 // flattening or unwrapping required, then we may
427 // return the SeqStmt as-is.
428 if constexpr (sizeof...(seq_args) == 1) {
429 if (auto opt = Flattener::AsSeqStmt(std::forward<Args>(seq_args)...)) {
430 SeqStmt original = opt.value();
431 bool all_same = [&]() {
432 if (original->seq.size() != seq.size()) {
433 return false;
434 }
435 for (size_t i = 0; i < seq.size(); i++) {
436 if (!original->seq[i].same_as(seq[i])) {
437 return false;
438 }
439 }
440 return true;
441 }();
442 if (all_same) {
443 return original;
444 }
445 }
446 }
447
448 return SeqStmt(seq);
449 }
451 class Flattener {
452 public:
453 explicit Flattener(ffi::Array<Stmt>* seq) : seq_(seq) {}
454
455 template <typename T>
456 static ffi::Optional<SeqStmt> AsSeqStmt(const T& t) {
457 if constexpr (std::is_same_v<T, SeqStmt>) {
458 return t;
459 }
460 if constexpr (!std::is_base_of_v<T, SeqStmt>) {
461 return std::nullopt;
462 }
463 if constexpr (std::is_base_of_v<Stmt, T>) {
464 if (const SeqStmtNode* ptr = t.template as<SeqStmtNode>()) {
465 return ffi::GetRef<SeqStmt>(ptr);
466 } else {
467 return std::nullopt;
468 }
469 }
470 return std::nullopt;
471 }
472
473 void operator()(size_t i, const ffi::Optional<Stmt>& stmt) const {
474 if (stmt.has_value()) (*this)(i, stmt.value());
475 }
476
477 template <typename T>
478 void operator()(size_t i, const T& stmt_or_seq) const {
479 if constexpr (std::is_base_of_v<ObjectRef, T>) {
480 // Early bail-out, applicable to any ObjectRef
481 if (!stmt_or_seq.defined()) {
482 return;
483 }
484 }
485
486 if constexpr (std::is_same_v<T, SeqStmt>) {
487 // Static type-checking for a SeqStmt that could be flattened.
488 (*this)(0, stmt_or_seq->seq);
489 return;
490 }
491
492 if constexpr (std::is_base_of_v<T, SeqStmt>) {
493 // Dynamic type-checking for a SeqStmt that could be
494 // flattened.
495 if (auto* op = stmt_or_seq.template as<SeqStmtNode>()) {
496 operator()(0, op->seq);
497 return;
498 }
499 }
500
501 if constexpr (std::is_base_of_v<T, Evaluate>) {
502 // Evaluate(0) is used to represent a no-op, and may be
503 // generated by previous calls to SeqStmt::Flatten(). These
504 // should be removed to ensure that Flatten(a+b) is equivalent
505 // to Flatten(Flatten(a), Flatten(b)).
506 if (auto* op = stmt_or_seq.template as<EvaluateNode>()) {
507 if (auto* as_int = op->value.template as<IntImmNode>(); as_int && as_int->value == 0) {
508 return;
509 }
510 }
511 }
512
513 if constexpr (std::is_base_of_v<Stmt, T>) {
514 // Any other Stmt type just gets appended.
515 seq_->push_back(stmt_or_seq);
516 } else {
517 // Anything else is treated as an iterable of Stmt.
518 for (auto v : stmt_or_seq) {
519 this->operator()(0, v);
520 }
521 }
522 }
523
524 private:
525 ffi::Array<Stmt>* seq_;
526 };

528 explicit SeqStmt(ffi::ObjectPtr<SeqStmtNode> node) : Stmt(std::move(node)) {}

904inline const char* ForKind2String(ForKind t) {
905 switch (t) {
906 case ForKind::kSerial:
907 return "serial";
908 case ForKind::kParallel:
909 return "parallel";
910 case ForKind::kVectorized:
911 return "vectorized";

915 return "thread_binding";

919}
