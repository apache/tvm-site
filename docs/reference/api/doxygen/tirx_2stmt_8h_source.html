33inline bool IsNoOp(const Stmt& stmt) {
34 if (!stmt.defined()) return true;
35 if (const auto* op = stmt.as<EvaluateNode>()) return op->value.as<IntImmNode>() != nullptr;
36 if (const auto* op = stmt.as<SeqStmtNode>()) return op->seq.empty();
37 return false;
38}

68inline ffi::Optional<ffi::String> GetThreadBinding(const ForNode* loop) {
69 if (auto tag = loop->annotations.Get(tvm::tirx::attr::kThreadBinding)) {
70 TVM_FFI_CHECK(loop->kind == ForKind::kParallel, ValueError)
71 << "thread_binding requires a parallel loop";
72 TVM_FFI_CHECK(tag->as<ffi::String>().has_value(), TypeError)
73 << "thread_binding annotation must be a string";
74 TVM_FFI_CHECK(loop->HasTrivialStep(), ValueError) << "Thread binding loops require a unit step";
75 return tag->cast<ffi::String>();
76 }
77 return std::nullopt;
78}
79inline ffi::Optional<ffi::String> GetThreadBinding(const For& loop) {
80 return GetThreadBinding(loop.get());
81}
