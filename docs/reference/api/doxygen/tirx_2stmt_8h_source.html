325 template <typename... Args>
326 static Stmt Flatten(Args&&... seq_args) {
327 ffi::Array<Stmt> seq;
328
329 ffi::details::for_each(Flattener(&seq), std::forward<Args>(seq_args)...);
330
331 if (seq.empty()) {
332 return Evaluate(0);
333 } else if (seq.size() == 1) {
334 return seq[0];
335 }
336
337 // If the argument is a single SeqStmt argument with no
338 // flattening or unwrapping required, then we may
339 // return the SeqStmt as-is.
340 if constexpr (sizeof...(seq_args) == 1) {
341 if (auto opt = Flattener::AsSeqStmt(std::forward<Args>(seq_args)...)) {
342 SeqStmt original = opt.value();
343 bool all_same = [&]() {
344 if (original->seq.size() != seq.size()) {
345 return false;
346 }
347 for (size_t i = 0; i < seq.size(); i++) {
348 if (!original->seq[i].same_as(seq[i])) {
349 return false;
350 }
351 }
352 return true;
353 }();
354 if (all_same) {
355 return original;
356 }
357 }
358 }
359
360 return SeqStmt(seq);
361 }

367 template <typename T>
368 static ffi::Optional<SeqStmt> AsSeqStmt(const T& t) {
369 if constexpr (std::is_same_v<T, SeqStmt>) {
370 return t;
371 }
372 if constexpr (!std::is_base_of_v<T, SeqStmt>) {
373 return std::nullopt;
374 }
375 if constexpr (std::is_base_of_v<Stmt, T>) {
376 if (const SeqStmtNode* ptr = t.template as<SeqStmtNode>()) {
377 return ffi::GetRef<SeqStmt>(ptr);
378 } else {
379 return std::nullopt;
380 }
381 }
382 return std::nullopt;
383 }
384
385 template <typename T>
386 void operator()(size_t i, const T& stmt_or_seq) const {
387 if constexpr (std::is_base_of_v<ObjectRef, T>) {
388 // Early bail-out, applicable to any ObjectRef
389 if (!stmt_or_seq.defined()) {
390 return;
391 }
392 }
393
394 if constexpr (std::is_same_v<T, SeqStmt>) {
395 // Static type-checking for a SeqStmt that could be flattened.
396 (*this)(0, stmt_or_seq->seq);
397 return;
398 }
399
400 if constexpr (std::is_base_of_v<T, SeqStmt>) {
401 // Dynamic type-checking for a SeqStmt that could be
402 // flattened.
403 if (auto* op = stmt_or_seq.template as<SeqStmtNode>()) {
404 operator()(0, op->seq);
405 return;
406 }
407 }
408
409 if constexpr (std::is_base_of_v<T, Evaluate>) {
410 // Evaluate(0) is used to represent a no-op, and may be
411 // generated by previous calls to SeqStmt::Flatten(). These
412 // should be removed to ensure that Flatten(a+b) is equivalent
413 // to Flatten(Flatten(a), Flatten(b)).
414 if (auto* op = stmt_or_seq.template as<EvaluateNode>()) {
415 if (auto* as_int = op->value.template as<IntImmNode>(); as_int && as_int->value == 0) {
416 return;
417 }
418 }
419 }
420
421 if constexpr (std::is_base_of_v<Stmt, T>) {
422 // Any other Stmt type just gets appended.
423 seq_->push_back(stmt_or_seq);
424 } else {
425 // Anything else is treated as an iterable of Stmt.
426 for (auto v : stmt_or_seq) {
427 this->operator()(0, v);
428 }
429 }
430 }
431
432 private:
433 ffi::Array<Stmt>* seq_;
434 };
