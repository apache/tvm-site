53template <typename R, typename... Args>
54class StmtFunctor<R(const Stmt&, Args...)> {
55 private:
56 using TSelf = StmtFunctor<R(const Stmt&, Args...)>;
57
58 public:
60 using result_type = R;
62 StmtFunctor() : StmtFunctor(GlobalVTable()) {}
64 virtual ~StmtFunctor() = default;
66 TVM_FFI_INLINE R operator()(const Stmt& node, Args... args) {
67 return Dispatch(node, std::forward<Args>(args)...);
68 }
70 TVM_FFI_INLINE virtual R Dispatch(const Stmt& node, Args... args) {
71 TVM_FFI_ICHECK(node.defined()) << "Cannot dispatch a null statement";
72 try {
73 return (*vtable_)(node, this, std::forward<Args>(args)...);
74 } catch (ffi::Error& error) {
75 ffi::detail::UpdateVisitErrorContext(error, node);
76 throw;
77 }
78 }
79
80 virtual R Dispatch_(const BindObj* node, Args... args) {
81 return DispatchDefault_(node, std::forward<Args>(args)...);
82 }
83 virtual R Dispatch_(const RegionStmtObj* node, Args... args) {
84 return DispatchDefault_(node, std::forward<Args>(args)...);
85 }
86 virtual R Dispatch_(const IfObj* node, Args... args) {

89 virtual R Dispatch_(const ForObj* node, Args... args) {

120 }

122 protected:
124 using VTable = ObjectFunctor<R(const ffi::ObjectRef&, TSelf*, Args...)>;
126 explicit StmtFunctor(const VTable* vtable) : vtable_(vtable) {}
128 static void InitVTable(VTable* vtable) {
129 vtable->template SetDispatch<StmtObj>(
130 [](const ffi::ObjectRef& node, TSelf* self, Args... args) -> R {
131 return self->DispatchDefault_(node.get(), std::forward<Args>(args)...);
132 });
133 SetDispatch<TSelf, BindObj>(vtable);
134 SetDispatch<TSelf, RegionStmtObj>(vtable);
135 SetDispatch<TSelf, IfObj>(vtable);
136 SetDispatch<TSelf, ForObj>(vtable);
137 SetDispatch<TSelf, WhileObj>(vtable);
138 SetDispatch<TSelf, ReturnObj>(vtable);
139 SetDispatch<TSelf, BreakObj>(vtable);
140 SetDispatch<TSelf, ContinueObj>(vtable);
141 SetDispatch<TSelf, TensorStoreObj>(vtable);
142 SetDispatch<TSelf, AssertStmtObj>(vtable);
143 SetDispatch<TSelf, SeqStmtObj>(vtable);
144 SetDispatch<TSelf, EvaluateObj>(vtable);
145 }
147 template <typename Self, typename Node>
148 static void SetDispatch(VTable* vtable) {
149 vtable->template SetDispatch<Node>(
150 [](const ffi::ObjectRef& node, TSelf* self, Args... args) -> R {
151 return static_cast<Self*>(self)->Dispatch_(static_cast<const Node*>(node.get()),
152 std::forward<Args>(args)...);
153 });
154 }
155
156 private:
157 static const VTable* GlobalVTable() {
158 static const VTable table = [] {
159 VTable table;
160 InitVTable(&table);
161 table.Finalize();
162 return table;
163 }();
164 return &table;
165 }
166 const VTable* vtable_;
167};

239 TVM_FFI_INLINE UnchangedOr<Stmt> Mutate(const Stmt& stmt,
240 InplaceMode inplace_mode = InplaceMode::kDisallow) {
241 return ffi::detail::UnchangedOrUnsafe::MoveFromTVMFFIAny<Stmt>(
242 ffi::detail::UnchangedOrUnsafe::MoveToTVMFFIAny(Mutate(ffi::AnyView(stmt), inplace_mode)));

246 TVM_FFI_INLINE UnchangedOr<SeqStmt> Mutate(const SeqStmt& stmt,
247 InplaceMode inplace_mode = InplaceMode::kDisallow) {
248 auto result = Mutate(static_cast<const Stmt&>(stmt), inplace_mode);
249 if (result.UnchangedOrSameAs(stmt)) return ffi::Unchanged();
250 return SeqStmt(std::move(result).ValueUnchecked());
251 }

253 TVM_FFI_INLINE UnchangedOr<ffi::Optional<SeqStmt>> Mutate(
254 const ffi::Optional<SeqStmt>& stmt, InplaceMode inplace_mode = InplaceMode::kDisallow) {
255 if (!stmt.has_value()) return ffi::Unchanged();
256 auto result = Mutate(stmt.value(), inplace_mode);
257 if (result.IsUnchanged()) return ffi::Unchanged();
258 return ffi::Optional<SeqStmt>(std::move(result).ValueUnchecked());
259 }

266 if (result.IsUnchanged()) return ffi::Unchanged();

326template <typename Node, typename = std::enable_if_t<std::is_base_of_v<StmtObj, Node>>>
327bool ContainsNode(const Stmt& stmt) {
328 struct Visitor : StmtExprVisitor {
329 // Early bail-out, if we already found the node. Skip expression operands.
330 ffi::Optional<VisitInterrupt> Visit(ffi::AnyView value) final {
331 if (contains_node || value.as<ExprObj>()) {
332 return std::nullopt;
333 }
334 if (value.as<Node>()) {
335 contains_node = true;
336 return std::nullopt;
337 }
338 return StmtExprVisitor::Visit(value);
339 }
340
341 bool contains_node{false};
342 };
343
344 auto visitor = ffi::make_object<Visitor>();
345 visitor->Visit(stmt);
346 return visitor->contains_node;
347}
