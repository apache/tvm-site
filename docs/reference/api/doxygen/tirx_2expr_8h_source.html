198 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
199 if (src->type_index == TypeIndex::kTVMFFINone) {
200 return false;
201 }
202 if (src->type_index != tvm::VarNode::RuntimeTypeIndex()) {
203 return false;
204 }
205 const auto* var = static_cast<const tvm::VarNode*>(
206 details::ObjectUnsafe::ObjectPtrFromUnowned<Object>(src->v_obj).get());
207 return details::AnyUnsafe::CheckAnyStrict<tirx::TensorType>(var->ExprNode::ty);
208 }
209
210 TVM_FFI_INLINE static std::optional<tirx::TensorVar> TryCastFromAnyView(const TVMFFIAny* src) {
211 if (CheckAnyStrict(src)) {
212 return details::ObjectUnsafe::ObjectRefFromObjectPtr<tirx::TensorVar>(
213 details::ObjectUnsafe::ObjectPtrFromUnowned<tvm::VarNode>(src->v_obj));
214 }
215 return std::nullopt;
216 }
