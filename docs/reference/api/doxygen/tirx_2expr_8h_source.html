271 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
272 if (src->type_index == TypeIndex::kTVMFFINone) {
273 return false;
274 }
275 if (src->type_index != tirx::VarNode::RuntimeTypeIndex()) {
276 return false;
277 }
278 const auto* var = static_cast<const tirx::VarNode*>(
279 details::ObjectUnsafe::ObjectPtrFromUnowned<Object>(src->v_obj).get());
280 return details::AnyUnsafe::CheckAnyStrict<tirx::BufferType>(var->ExprNode::ty);
281 }
282
283 TVM_FFI_INLINE static std::optional<tirx::BufferVar> TryCastFromAnyView(const TVMFFIAny* src) {
284 if (CheckAnyStrict(src)) {
285 return details::ObjectUnsafe::ObjectRefFromObjectPtr<tirx::BufferVar>(
286 details::ObjectUnsafe::ObjectPtrFromUnowned<tirx::VarNode>(src->v_obj));
287 }
288 return std::nullopt;
289 }
