121 TVM_FFI_INLINE static bool CheckAnyStrict(const TVMFFIAny* src) {
122 if (src->type_index == TypeIndex::kTVMFFINone) {
123 return false;
124 }
125 if (src->type_index != tvm::VarNode::RuntimeTypeIndex()) {
126 return false;
127 }
128 const auto* var = static_cast<const tvm::VarNode*>(
129 details::ObjectUnsafe::ObjectPtrFromUnowned<Object>(src->v_obj).get());
130 return details::AnyUnsafe::CheckAnyStrict<tirx::TensorType>(var->ExprNode::ty);
131 }

133 TVM_FFI_INLINE static std::optional<tirx::TensorVar> TryCastFromAnyView(const TVMFFIAny* src) {

136 details::ObjectUnsafe::ObjectPtrFromUnowned<tvm::VarNode>(src->v_obj));
