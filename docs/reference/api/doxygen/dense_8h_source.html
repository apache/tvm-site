48inline tvm::te::Tensor dense(const tvm::te::Tensor& data, const tvm::te::Tensor& weight,
49 const tvm::te::Tensor& bias, const PrimType& out_dtype) {
50 TVM_FFI_ICHECK_EQ(data->shape.size(), 2) << "dense requires 2-D data";
51 TVM_FFI_ICHECK_EQ(weight->shape.size(), 2) << "dense requires 2-D weight";
52 if (bias.defined()) {
53 TVM_FFI_ICHECK_EQ(bias->shape.size(), 1) << "dense requires 1-D bias";
54 }
55
56 auto batch = data->shape[0];
57 auto in_dim = data->shape[1];
58 auto out_dim = weight->shape[0];
59
60 auto k = tvm::te::reduce_axis(Range(0, in_dim), "k");
61 auto matmul = tvm::te::compute(
62 {batch, out_dim},
63 [&](PrimVar i, PrimVar j) {
64 return tvm::sum(tvm::cast(out_dtype, data(i, k)) * tvm::cast(out_dtype, weight(j, k)), {k});
65 },
66 "tensor", "dense");
67
68 if (bias.defined()) {
69 matmul = tvm::te::compute(
70 {batch, out_dim},
71 [&](PrimVar i, PrimVar j) { return matmul(i, j) + tvm::cast(out_dtype, bias(j)); },
72 "tensor", kBroadcast);
73 }
74
75 return matmul;
76}
