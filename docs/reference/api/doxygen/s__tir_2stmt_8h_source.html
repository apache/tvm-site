395inline ffi::Optional<ffi::String> GetThreadBinding(const ForNode* loop) {
396 if (auto tag = loop->annotations.Get(tvm::s_tir::attr::kThreadBinding)) {
397 TVM_FFI_CHECK(loop->kind == ForKind::kParallel, ValueError)
398 << "thread_binding requires a parallel loop";
399 TVM_FFI_CHECK(tag->as<ffi::String>().has_value(), TypeError)
400 << "thread_binding annotation must be a string";
401 TVM_FFI_CHECK(loop->HasTrivialStep(), ValueError) << "Thread binding loops require a unit step";
402 return tag->cast<ffi::String>();
403 }
404 return std::nullopt;
405}
406inline ffi::Optional<ffi::String> GetThreadBinding(const For& loop) {
407 return GetThreadBinding(loop.get());
408}
