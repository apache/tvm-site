644// inline implementations
645inline const char* ForKind2String(ForKind t) {
646 switch (t) {
647 case ForKind::kDefault:
648 return "serial";
649 case ForKind::kParallel:
650 return "parallel";
651 case ForKind::kVectorized:
652 return "vectorized";
653 case ForKind::kUnrolled:
654 return "unroll";
655 }
656 TVM_FFI_THROW(InternalError) << "Unknown ForKind" << t;
657 TVM_FFI_UNREACHABLE();
658}
