53template <typename R, typename... Args>
54class StmtFunctor<R(const Stmt&, Args...)> {
55 private:
56 using TSelf = StmtFunctor<R(const Stmt&, Args...)>;
57
58 public:
60 using result_type = R;
62 StmtFunctor() : StmtFunctor(GlobalVTable()) {}
64 virtual ~StmtFunctor() = default;
66 TVM_FFI_INLINE R operator()(const Stmt& node, Args... args) {
67 return Dispatch(node, std::forward<Args>(args)...);
68 }
70 TVM_FFI_INLINE virtual R Dispatch(const Stmt& node, Args... args) {
71 TVM_FFI_ICHECK(node.defined()) << "Cannot dispatch a null statement";
72 return (*vtable_)(node, this, std::forward<Args>(args)...);
73 }

123 protected:
125 using VTable = ObjectFunctor<R(const ffi::ObjectRef&, TSelf*, Args...)>;
127 explicit StmtFunctor(const VTable* vtable) : vtable_(vtable) {}
129 static void InitVTable(VTable* vtable) {
130 vtable->template SetDispatch<StmtNode>(
131 [](const ffi::ObjectRef& node, TSelf* self, Args... args) -> R {
132 return self->DispatchDefault_(node.get(), std::forward<Args>(args)...);
133 });
134 SetDispatch<TSelf, BindNode>(vtable);
135 SetDispatch<TSelf, RegionStmtNode>(vtable);
136 SetDispatch<TSelf, IfThenElseNode>(vtable);
137 SetDispatch<TSelf, ForNode>(vtable);
138 SetDispatch<TSelf, WhileNode>(vtable);
139 SetDispatch<TSelf, ReturnNode>(vtable);
140 SetDispatch<TSelf, BreakNode>(vtable);
141 SetDispatch<TSelf, ContinueNode>(vtable);
142 SetDispatch<TSelf, TensorStoreNode>(vtable);
143 SetDispatch<TSelf, AssertStmtNode>(vtable);
144 SetDispatch<TSelf, SeqStmtNode>(vtable);
145 SetDispatch<TSelf, EvaluateNode>(vtable);
146 SetDispatch<TSelf, ScopeIdDefStmtNode>(vtable);
147 SetDispatch<TSelf, tirx::TileOpCallNode>(vtable);
148 }
150 template <typename Self, typename Node>
151 static void SetDispatch(VTable* vtable) {
152 vtable->template SetDispatch<Node>(
153 [](const ffi::ObjectRef& node, TSelf* self, Args... args) -> R {
154 return static_cast<Self*>(self)->Dispatch_(static_cast<const Node*>(node.get()),
155 std::forward<Args>(args)...);
156 });
157 }
158
159 private:
160 static const VTable* GlobalVTable() {
161 static const VTable table = [] {
162 VTable table;
163 InitVTable(&table);
164 table.Finalize();
165 return table;
166 }();
167 return &table;
168 }
169 const VTable* vtable_;

244 TVM_FFI_INLINE UnchangedOr<Stmt> Mutate(const Stmt& stmt,
245 InplaceMode inplace_mode = InplaceMode::kDisallow) {
246 return ffi::details::UnchangedOrUnsafe::MoveFromTVMFFIAny<Stmt>(
247 ffi::details::UnchangedOrUnsafe::MoveToTVMFFIAny(Mutate(ffi::AnyView(stmt), inplace_mode)));

251 TVM_FFI_INLINE UnchangedOr<SeqStmt> Mutate(const SeqStmt& stmt,
252 InplaceMode inplace_mode = InplaceMode::kDisallow) {
253 auto result = Mutate(static_cast<const Stmt&>(stmt), inplace_mode);
254 if (result.UnchangedOrSameAs(stmt)) return ffi::Unchanged();
255 return SeqStmt(std::move(result).ValueUnchecked());
256 }
257
258 TVM_FFI_INLINE UnchangedOr<ffi::Optional<SeqStmt>> Mutate(
259 const ffi::Optional<SeqStmt>& stmt, InplaceMode inplace_mode = InplaceMode::kDisallow) {
260 if (!stmt.has_value()) return ffi::Unchanged();
261 auto result = Mutate(stmt.value(), inplace_mode);
262 if (result.IsUnchanged()) return ffi::Unchanged();
263 return ffi::Optional<SeqStmt>(std::move(result).ValueUnchecked());
264 }

268 const ffi::Optional<Stmt>& stmt, InplaceMode inplace_mode = InplaceMode::kDisallow) {
269 if (!stmt.has_value()) return ffi::Unchanged();
270 auto result = Mutate(stmt.value(), inplace_mode);
271 if (result.IsUnchanged()) return ffi::Unchanged();
272 return ffi::Optional<Stmt>(std::move(result).ValueUnchecked());
273 }

333template <typename Node, typename = std::enable_if_t<std::is_base_of_v<StmtNode, Node>>>
334bool ContainsNode(const Stmt& stmt) {
335 struct Visitor : StmtExprVisitor {
336 // Early bail-out, if we already found the node. Skip expression operands.
337 ffi::Optional<VisitInterrupt> Visit(ffi::AnyView value) final {
338 if (contains_node || value.as<ExprNode>()) {
339 return std::nullopt;
340 }
341 if (value.as<Node>()) {
342 contains_node = true;
343 return std::nullopt;
344 }
345 return StmtExprVisitor::Visit(value);
346 }
347
348 bool contains_node{false};
349 };
350
351 auto visitor = ffi::make_object<Visitor>();
352 visitor->Visit(stmt);
353 return visitor->contains_node;
354}
