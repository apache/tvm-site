55template <typename R, typename... Args>
56class StmtFunctor<R(const Stmt&, Args...)> {
57 private:
58 using TSelf = StmtFunctor<R(const Stmt&, Args...)>;
59
60 public:
62 using result_type = R;
64 StmtFunctor() : StmtFunctor(GlobalVTable()) {}
66 virtual ~StmtFunctor() = default;
68 TVM_FFI_INLINE R operator()(const Stmt& node, Args... args) {
69 return Dispatch(node, std::forward<Args>(args)...);
70 }
72 TVM_FFI_INLINE virtual R Dispatch(const Stmt& node, Args... args) {
73 TVM_FFI_ICHECK(node.defined()) << "Cannot dispatch a null statement";
74 return (*vtable_)(node, this, std::forward<Args>(args)...);
75 }

123 }
124
125 protected:
127 using VTable = ObjectFunctor<R(const ffi::ObjectRef&, TSelf*, Args...)>;
129 explicit StmtFunctor(const VTable* vtable) : vtable_(vtable) {}
131 static void InitVTable(VTable* vtable) {
132 vtable->template SetDispatch<StmtNode>(
133 [](const ffi::ObjectRef& node, TSelf* self, Args... args) -> R {
134 return self->DispatchDefault_(node.get(), std::forward<Args>(args)...);
135 });
136 SetDispatch<TSelf, BindNode>(vtable);
137 SetDispatch<TSelf, RegionStmtNode>(vtable);
138 SetDispatch<TSelf, IfThenElseNode>(vtable);
139 SetDispatch<TSelf, ForNode>(vtable);
140 SetDispatch<TSelf, WhileNode>(vtable);
141 SetDispatch<TSelf, ReturnNode>(vtable);
142 SetDispatch<TSelf, BreakNode>(vtable);
143 SetDispatch<TSelf, ContinueNode>(vtable);
144 SetDispatch<TSelf, TensorStoreNode>(vtable);
145 SetDispatch<TSelf, AssertStmtNode>(vtable);
146 SetDispatch<TSelf, SeqStmtNode>(vtable);
147 SetDispatch<TSelf, EvaluateNode>(vtable);
148 SetDispatch<TSelf, ScopeIdDefStmtNode>(vtable);
149 SetDispatch<TSelf, tirx::TilePrimitiveCallNode>(vtable);
150 }
152 template <typename Self, typename Node>
153 static void SetDispatch(VTable* vtable) {
154 vtable->template SetDispatch<Node>(
155 [](const ffi::ObjectRef& node, TSelf* self, Args... args) -> R {
156 return static_cast<Self*>(self)->Dispatch_(static_cast<const Node*>(node.get()),
157 std::forward<Args>(args)...);
158 });
159 }
160
161 private:
162 static const VTable* GlobalVTable() {
163 static const VTable table = [] {
164 VTable table;
165 InitVTable(&table);
166 table.Finalize();
167 return table;
168 }();
169 return &table;
170 }
171 const VTable* vtable_;
172};

319template <typename Node, typename = std::enable_if_t<std::is_base_of_v<StmtNode, Node>>>
320bool ContainsNode(const Stmt& stmt) {
321 struct Visitor : StmtExprVisitor {
322 // Early bail-out, if we already found the node. Skip expression operands.
323 ffi::Optional<VisitInterrupt> Visit(ffi::AnyView value) final {
324 if (contains_node || value.as<ExprNode>()) {
325 return std::nullopt;
326 }
327 if (value.as<Node>()) {
328 contains_node = true;
329 return std::nullopt;
330 }
331 return StmtExprVisitor::Visit(value);
332 }
333
334 bool contains_node{false};
335 };
336
337 auto visitor = ffi::make_object<Visitor>();
338 visitor->Visit(stmt);
339 return visitor->contains_node;
340}
