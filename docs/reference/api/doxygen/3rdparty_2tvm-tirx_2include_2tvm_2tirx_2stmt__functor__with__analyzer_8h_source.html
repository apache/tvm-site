145 template <typename FLambda>
146 void WithRecordIterPredicate(PrimExpr condition, FLambda callback) {
147 std::unordered_set<const VarNode*> iter_var_nodes;
148 for (const auto& [var, _] : iter_vars_) {
149 iter_var_nodes.insert(var.get());
150 }
151 auto f_use_itervar = [&iter_var_nodes](const tvm::VarNode* v) {
152 return iter_var_nodes.count(v);
153 };
154 auto walkfn = [&](const tvm::Var& var) -> ffi::Expected<ffi::WalkResult> {
155 return f_use_itervar(var.get()) ? ffi::WalkResult::Interrupt(ffi::VisitInterrupt(var))
156 : ffi::WalkResult::Advance();
157 };
158 // simple heuristics for detecting predicate
159 if (ffi::StructuralWalk<ffi::WalkOrder::kPreOrder>(condition, walkfn).has_value()) {
160 iter_predicates_.push_back(condition);
161 callback();
162 iter_predicates_.pop_back();
163 } else {
164 callback();
165 }
166 }
167};
