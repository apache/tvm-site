49inline Tensor pool_grad_impl(const Tensor& out_grad, const Tensor& x,
50 const ffi::Array<PrimExpr>& kernel_size,
51 const ffi::Array<PrimExpr>& stride_size,
52 const ffi::Array<PrimExpr>& padding_size, PoolType pool_type,
53 bool ceil_mode, const size_t height_axis, const size_t width_axis,
54 bool count_include_pad) {
55 using namespace tvm::prim;
56 TVM_FFI_ICHECK(out_grad->shape.size() >= 2) << "Pooling grad output must >= 2-D (H, W)";
57 TVM_FFI_ICHECK(x->shape.size() >= 2) << "Pooling input must >= 2-D (H, W)";
58 TVM_FFI_ICHECK_EQ(kernel_size.size(), 2) << "Pooling kernel_size must have 2 elements";
59 TVM_FFI_ICHECK_EQ(stride_size.size(), 2) << "Pooling stride_size must have 2 elements";
60 TVM_FFI_ICHECK_EQ(padding_size.size(), 4) << "Pooling padding_size must have 4 elements";
61
62 auto kernel_height = kernel_size[0];
63 auto kernel_width = kernel_size[1];
64 auto stride_height = stride_size[0];
65 auto stride_width = stride_size[1];
66
67 auto height = x->shape[height_axis];
68 auto width = x->shape[width_axis];
69
70 auto pad_top = padding_size[0];
71 auto pad_left = padding_size[1];
72 auto pad_bottom = padding_size[2];
73 auto pad_right = padding_size[3];
74
75 if (ceil_mode) {
76 // Additional padding to ensure we do ceil instead of floor when
77 // dividing by stride.
78 pad_bottom += stride_height - 1;
79 pad_right += stride_width - 1;
80 }
81
82 ffi::Array<PrimExpr> pad_before(std::vector<PrimExpr>(x->shape.size(), 0));
83 pad_before.Set(height_axis, pad_top);
84 pad_before.Set(width_axis, pad_left);
85
86 ffi::Array<PrimExpr> pad_after(std::vector<PrimExpr>(x->shape.size(), 0));
87 pad_after.Set(height_axis, pad_bottom);
88 pad_after.Set(width_axis, pad_right);
89 sym::Analyzer analyzer;
90 auto out_height =
91 analyzer->Simplify((height - kernel_height + pad_top + pad_bottom) / stride_height + 1);
92 auto out_width =
93 analyzer->Simplify((width - kernel_width + pad_left + pad_right) / stride_width + 1);
94
95 auto dheight = tvm::te::reduce_axis(Range(0, kernel_height), "dh");
96 auto dwidth = tvm::te::reduce_axis(Range(0, kernel_width), "dw");
97
98 ffi::Array<PrimExpr> data_shape = x->shape;
99 ffi::Array<PrimExpr> out_shape = data_shape;
100 out_shape.Set(height_axis, out_height);
101 out_shape.Set(width_axis, out_width);
102
103 const auto* padding_h0 = pad_top.as<IntImmNode>();
104 const auto* padding_w0 = pad_left.as<IntImmNode>();
105 const auto* padding_h1 = pad_bottom.as<IntImmNode>();
106 const auto* padding_w1 = pad_right.as<IntImmNode>();
107 const bool do_pad =
108 ((padding_h0 && padding_h0->value != 0) || (padding_w0 && padding_w0->value != 0)) ||
109 ((padding_h1 && padding_h1->value != 0) || (padding_w1 && padding_w1->value != 0));
110
111 if (pool_type == kMaxPool) {
112 ffi::Array<PrimExpr> ravel_shape{data_shape.begin(), data_shape.end()};
113 ravel_shape.Set(height_axis, ravel_shape[height_axis] + pad_top + pad_bottom);
114 ravel_shape.Set(width_axis, ravel_shape[width_axis] + pad_left + pad_right);
115
116 auto windowh =
117 tvm::te::reduce_axis(Range(0, (kernel_height + stride_height - 1) / stride_height), "wh");
118 auto windoww =
119 tvm::te::reduce_axis(Range(0, (kernel_width + stride_width - 1) / stride_width), "ww");
120
121 auto argmax = MakeArgmaxReducer();
122 auto pad_x =
123 do_pad ? pad(x, pad_before, pad_after, tvm::prim::min_value(PrimType(x->dtype)), "pad_temp")
124 : x;
125
126 auto mp_argmax = tvm::te::compute(
127 out_shape,
128 [&](const ffi::Array<PrimVar>& inds) {
129 ffi::Array<PrimExpr> window_inds =
130 inds.Map([](const PrimVar& var) { return var.as_or_throw<PrimExpr>(); });
131 window_inds.Set(height_axis, inds[height_axis] * stride_height + dheight);
132 window_inds.Set(width_axis, inds[width_axis] * stride_width + dwidth);
133 auto idx = detail::RavelIndex(window_inds, ravel_shape);
134 return argmax({idx, pad_x(window_inds)}, {dheight, dwidth}, nullptr);
135 },
136 "maxpool_grad_argmax", kCommReduceIdx);
137
138 auto mp_inds = mp_argmax[0];
139
140 return tvm::te::compute(
141 data_shape,
142 [&](const ffi::Array<PrimVar>& inds) {
143 ffi::Array<PrimExpr> pad_inds =
144 inds.Map([](const PrimVar& var) { return var.as_or_throw<PrimExpr>(); });
145 pad_inds.Set(height_axis, pad_inds[height_axis] + pad_top);
146 pad_inds.Set(width_axis, pad_inds[width_axis] + pad_left);
147 auto idx = detail::RavelIndex(pad_inds, ravel_shape);
148
149 ffi::Array<PrimExpr> out_idx =
150 inds.Map([](const PrimVar& var) { return var.as_or_throw<PrimExpr>(); });
151 out_idx.Set(height_axis, (inds[height_axis] + pad_top) / stride_height - windowh);
152 out_idx.Set(width_axis, (inds[width_axis] + pad_left) / stride_width - windoww);
153
154 PrimExpr out_idx_lower_h = prim::Select(
155 pad_inds[height_axis] < kernel_height, IntImm(pad_inds[height_axis].ty(), 0),
156 (pad_inds[height_axis] - kernel_height) / stride_height + 1);
157 PrimExpr out_idx_lower_w = prim::Select(
158 pad_inds[width_axis] < kernel_width, IntImm(pad_inds[width_axis].ty(), 0),
159 (pad_inds[width_axis] - kernel_width) / stride_width + 1);
160
161 return tvm::prim::sum(
162 tvm::if_then_else(prim::And(prim::And(out_idx[height_axis] >= out_idx_lower_h,
163 out_idx[width_axis] >= out_idx_lower_w),
164 mp_inds(out_idx) == idx),
165 out_grad(out_idx), MakeConst(PrimType(x->dtype), 0)),
166 {windowh, windoww});
167 },
168 "T_pool_grad", "pool_grad_max");
169 } else if (pool_type == kAvgPool) {
170 auto windowh =
171 tvm::te::reduce_axis(Range(0, (kernel_height + stride_height - 1) / stride_height), "wh");
172 auto windoww =
173 tvm::te::reduce_axis(Range(0, (kernel_width + stride_width - 1) / stride_width), "ww");
174 return tvm::te::compute(
175 data_shape,
176 [&](const ffi::Array<PrimVar>& inds) {
177 PrimExpr pad_h_idx = inds[height_axis] + pad_top;
178 PrimExpr pad_w_idx = inds[width_axis] + pad_left;
179
180 // output indices whose pooling windows cover current input element (can be out-of-bound)
181 ffi::Array<PrimExpr> out_idx =
182 inds.Map([](const PrimVar& var) { return var.as_or_throw<PrimExpr>(); });
183 out_idx.Set(height_axis, (pad_h_idx / stride_height - windowh));
184 out_idx.Set(width_axis, (pad_w_idx / stride_width - windoww));
185
186 PrimExpr out_idx_lower_h =
187 prim::Select(pad_h_idx < kernel_height, IntImm(pad_h_idx.ty(), 0),
188 (pad_h_idx - kernel_height) / stride_height + 1);
189 PrimExpr out_idx_lower_w =
190 prim::Select(pad_w_idx < kernel_width, IntImm(pad_w_idx.ty(), 0),
191 (pad_w_idx - kernel_width) / stride_width + 1);
192
193 PrimExpr divide_factor = kernel_height * kernel_width; // number of pooled elements
194 if (!count_include_pad) {
195 PrimExpr h_start = out_idx[height_axis] * stride_height - pad_top;
196 PrimExpr w_start = out_idx[width_axis] * stride_width - pad_left;
197
198 PrimExpr h_end = min(h_start + kernel_height, height);
199 PrimExpr w_end = min(w_start + kernel_width, width);
200 h_start = max(h_start, IntImm(h_start.ty(), 0));
201 w_start = max(w_start, IntImm(w_start.ty(), 0));
202 divide_factor = max((h_end - h_start) * (w_end - w_start), IntImm(h_end.ty(), 1));
203 }
204 return tvm::prim::sum(
205 tvm::if_then_else(prim::And(prim::And(out_idx[height_axis] >= out_idx_lower_h,
206 out_idx[height_axis] < out_height),
207 prim::And(out_idx[width_axis] >= out_idx_lower_w,
208 out_idx[width_axis] < out_width)),
209 out_grad(out_idx) / divide_factor,
210 MakeConst(PrimType(out_grad->dtype), 0)),
211 {windowh, windoww});
212 },
213 "T_pool_grad", "pool_grad_avg");
214 } else {
215 TVM_FFI_THROW(ValueError) << "Unrecognized pool_type: " << pool_type;
216 }
217}

230inline bool find_depth_height_width(const std::string& layout, int* depth_axis, int* height_axis,
231 int* width_axis) {
232 if (depth_axis) *depth_axis = -1;
233 if (height_axis) *height_axis = -1;
234 if (width_axis) *width_axis = -1;
235 int curr_idx = 0;
236 for (size_t i = 0; i < layout.size(); ++i) {
237 if ((layout[i] >= 'A' && layout[i] <= 'Z') || (layout[i] >= 'a' && layout[i] <= 'z')) {
238 if (layout[i] == 'D' && depth_axis) {
239 if (*depth_axis != -1) return false;
240 *depth_axis = curr_idx;
241 } else if (layout[i] == 'H' && height_axis) {
242 if (*height_axis != -1) return false;
243 *height_axis = curr_idx;
244 } else if (layout[i] == 'W' && width_axis) {
245 if (*width_axis != -1) return false;
246 *width_axis = curr_idx;
247 } else if (layout[i] == 'd' || layout[i] == 'h' || layout[i] == 'w') {
248 // do not support split on height, width or depth, e.g., NCHW16w
249 return false;
250 }
251 ++curr_idx;
252 }
253 }
254 if ((depth_axis && *depth_axis == -1) || (height_axis && *height_axis == -1) ||
255 (width_axis && *width_axis == -1))
256 return false;
257 return true;
258}
259
260inline bool find_height_width(const std::string& layout, int* height_axis, int* width_axis) {
261 return find_depth_height_width(layout, /*depth_axis=*/nullptr, height_axis, width_axis);
262}
263
264inline bool find_width(const std::string& layout, int* width_axis) {
265 return find_depth_height_width(layout, /*depth_axis=*/nullptr, /*height_axis=*/nullptr,
266 width_axis);
267}

316inline PrimExpr end_index(const PrimVar& out_index, const PrimExpr& odim, const PrimExpr& idim) {
317 PrimExpr tmp = indexdiv((out_index + 1) * idim, odim);

331inline Tensor adaptive_pool_impl(const Tensor& x, const ffi::Array<PrimExpr>& output_size,
332 PoolType pool_type, const std::vector<int>& axes) {
333 const auto n_dim = output_size.size();
334 TVM_FFI_ICHECK_EQ(axes.size(), n_dim) << "The number of axes not equal to the in/out dimension";
335
336 ffi::Array<PrimExpr> data_shape = x->shape;
337 ffi::Array<PrimExpr> out_shape = data_shape;
338 ffi::Array<PrimExpr> in_size, out_size;
339 for (size_t i = 0; i < n_dim; ++i) {
340 in_size.push_back(data_shape[axes[i]]);
341 out_size.push_back(output_size[i]);
342 out_shape.Set(axes[i], out_size[i]);
343 }
344
345 auto get_iter_vars = [=](const ffi::Array<PrimVar>& output, bool reduce_indices) {
346 ffi::Array<PrimExpr> indices;
347 for (size_t i = 0; i < output.size(); ++i) indices.push_back(output[i]);
348 ffi::Array<tirx::IterVar> reduce_axes;
349 for (size_t i = 0; i < n_dim; ++i) {
350 auto i_start = start_index(output[axes[i]], out_size[i], in_size[i]);
351 auto i_end = end_index(output[axes[i]], out_size[i], in_size[i]);
352 auto rv_name = "rv" + std::to_string(i);
353 auto rv_axis = tvm::te::reduce_axis(Range(0, i_end - i_start), rv_name);
354 reduce_axes.push_back(rv_axis);
355 if (reduce_indices) {
356 indices.Set(axes[i], i_start + rv_axis);
357 }
358 }
359 return std::make_tuple(indices, reduce_axes);
360 };
361
362 ffi::Map<ffi::String, ffi::Any> attrs;
363 if (pool_type == kMaxPool) {
364 attrs.Set("schedule_rule", tvm::ffi::String("meta_schedule.adaptive_pool_max"));
365 return tvm::te::compute(
366 out_shape,
367 [&](const ffi::Array<PrimVar>& output) {
368 ffi::Array<PrimExpr> indices;
369 ffi::Array<tirx::IterVar> reduce_axes;
370 std::tie(indices, reduce_axes) = get_iter_vars(output, true);
371 return tvm::max(x(indices), reduce_axes); // NOLINT(*)
372 },
373 "adaptive_pool_max", "adaptive_pool_max", attrs);
374 } else if (pool_type == kAvgPool) {
375 attrs.Set("schedule_rule", tvm::ffi::String("meta_schedule.adaptive_pool_avg"));
376 auto pool_sum = tvm::te::compute(
377 out_shape,
378 [&](const ffi::Array<PrimVar>& output) {
379 ffi::Array<PrimExpr> indices;
380 ffi::Array<tirx::IterVar> reduce_axes;
381 std::tie(indices, reduce_axes) = get_iter_vars(output, true);
382 return tvm::prim::sum(x(indices), reduce_axes);
383 },
384 "adaptive_pool_sum", "adaptive_pool_sum");
385
386 return tvm::te::compute(
387 out_shape,
388 [&](const ffi::Array<PrimVar>& output) {
389 ffi::Array<PrimExpr> indices;
390 ffi::Array<tirx::IterVar> reduce_axes;
391 std::tie(indices, reduce_axes) = get_iter_vars(output, false);
392
393 PrimExpr divide_factor = tvm::prim::cast(PrimType(x->dtype), 1);
394 for (size_t i = 0; i < n_dim; ++i) {
395 divide_factor *= tvm::prim::cast(PrimType::Int(32), reduce_axes[i]->dom->extent);
396 }
397
398 return div(pool_sum(indices), divide_factor);
399 },
400 "adaptive_pool_avg", kElementWise, attrs);
401 } else {
402 LOG(ERROR) << "Unrecognized pool_type: " << pool_type;
403 return x;
404 }
405}

517inline Tensor pool_impl_nd(const Tensor& x, const ffi::Array<PrimExpr>& kernel_size,
518 const ffi::Array<PrimExpr>& stride_size,
519 const ffi::Array<PrimExpr>& dilation_size,
520 const ffi::Array<PrimExpr>& padding_size, PoolType pool_type,
521 bool ceil_mode, const std::vector<int>& axis, bool count_include_pad) {
522 using namespace tvm::prim;
523 int k_size = kernel_size.size();
524 int x_size = x->shape.size();
525 TVM_FFI_ICHECK_EQ(stride_size.size(), k_size)
526 << "Pooling stride_size must have same elements as kernel";
527 TVM_FFI_ICHECK_EQ(padding_size.size(), k_size * 2)
528 << "Pooling padding_size must has double elements of"
529 " kernel";
530 TVM_FFI_ICHECK_EQ(axis.size(), k_size) << "axis must have same elements as kernel";
531
532 ffi::Array<IterVar> daxis;
533 std::vector<PrimExpr> kernel(kernel_size.begin(), kernel_size.end());
534 std::vector<PrimExpr> stride(stride_size.begin(), stride_size.end());
535 std::vector<PrimExpr> dilation(dilation_size.begin(), dilation_size.end());
536 std::vector<PrimExpr> pad_head(padding_size.begin(), padding_size.begin() + k_size);
537 std::vector<PrimExpr> pad_tail(padding_size.begin() + k_size, padding_size.end());
538 std::vector<PrimExpr> offset(k_size, 0);
539 ffi::Array<PrimExpr> pad_before(std::vector<PrimExpr>(x_size, 0));
540 ffi::Array<PrimExpr> pad_after(std::vector<PrimExpr>(x_size, 0));
541 ffi::Array<PrimExpr> data_shape = x->shape;
542 ffi::Array<PrimExpr> out_shape = data_shape;
543
544 bool do_pad = false;
545 for (int i = 0; i < k_size; i++) {
546 int ii = axis[i];
547
548 if (ceil_mode) {
549 // The offset[i] is an additional padding to ensure we do ceil instead of floor when
550 // dividing by stride.
551 // In the case of ceil_mode=True and count_include_pad=True,
552 // in order to obtain the correct boundary,
553 // we also need to use the offset[i] to eliminate this extra padding.
554 offset[i] = stride[i] - 1;
555 pad_tail[i] += offset[i];
556 }
557
558 const auto* padding0 = pad_head[i].as<IntImmNode>();
559 const auto* padding1 = pad_tail[i].as<IntImmNode>();
560 do_pad = do_pad || (padding0 && padding0->value != 0) || (padding1 && padding1->value != 0);
561
562 daxis.push_back(tvm::te::reduce_axis(Range(0, kernel[i]), "rv" + std::to_string(i)));
563
564 pad_before.Set(ii, pad_head[i]);
565 pad_after.Set(ii, pad_tail[i]);
566
567 sym::Analyzer analyzer;
568
569 PrimExpr numerator =
570 data_shape[ii] - (kernel[i] - 1) * dilation[i] - 1 + pad_head[i] + pad_tail[i];
571 auto raw_out = indexdiv(numerator, stride[i]) + 1;
572 if (ceil_mode) {
573 // In the case of ceil_mode=True, we need to check if the last pooling window is valid.
574 // If not, we skip the last window as it would start in the bottom padded region,
575 // we need to minus 1 to get the correct output shape.
576 auto invalid_last = (raw_out - 1) * stride[i] >= data_shape[ii] + pad_head[i];
577 auto out_dim = analyzer->Simplify(if_then_else(invalid_last, raw_out - 1, raw_out));
578 out_shape.Set(ii, out_dim);
579 } else {
580 auto out_dim = analyzer->Simplify(raw_out);
581 out_shape.Set(ii, out_dim);
582 }
583 }
584
585 ffi::Map<ffi::String, ffi::Any> attrs;
586 if (pool_type == kMaxPool) {
587 auto temp =
588 do_pad ? pad(x, pad_before, pad_after, tvm::prim::min_value(PrimType(x->dtype)), "pad_temp")
589 : x;
590 attrs.Set("schedule_rule", tvm::ffi::String("meta_schedule.pool_max"));
591 return tvm::te::compute(
592 out_shape,
593 [&](const ffi::Array<PrimVar>& output) {
594 ffi::Array<PrimExpr> indices;
595 for (const PrimVar& var : output) indices.push_back(var);
596
597 for (int i = 0; i < k_size; i++) {
598 int ii = axis[i];
599 indices.Set(ii, output[ii] * stride[i] + daxis[i] * dilation[i]);
600 }
601 return tvm::max(temp(indices), daxis);
602 },
603 "pool_max", "pool_max", attrs);
604 } else if (pool_type == kAvgPool) {
605 attrs.Set("schedule_rule", tvm::ffi::String("meta_schedule.pool_avg"));
606 // Pad the inputs
607 auto temp = do_pad ? pad(x, pad_before, pad_after, 0, "pad_temp") : x;
608
609 // TVM compute for summing the pooling window.
610 auto pool_sum = tvm::te::compute(
611 out_shape,
612 [&](const ffi::Array<PrimVar>& output) {
613 ffi::Array<PrimExpr> indices;
614 for (const PrimVar& var : output) indices.push_back(var);
615
616 for (int i = 0; i < k_size; i++) {
617 int ii = axis[i];
618 indices.Set(ii, output[ii] * stride[i] + daxis[i] * dilation[i]);
619 }
620 return tvm::prim::sum(temp(indices), daxis);
621 },
622 "pool_sum", "pool_sum");
623
624 // TVM compute for dividing the reduced window sum by kernel size.
625 return tvm::te::compute(
626 out_shape,
627 [&](const ffi::Array<PrimVar>& output) {
628 ffi::Array<PrimExpr> indices;
629 for (const PrimVar& var : output) indices.push_back(var);
630 if (count_include_pad) {
631 auto num_el = IntImm::Int32(1);
632 for (int i = 0; i < k_size; i++) {
633 int ii = axis[i];
634 PrimExpr start = output[ii] * stride[i] - pad_head[i];
635 // When computing the output shape in ceil_mode,
636 // we have added the extra padding of offset[i],
637 // so now in order to calculate the correct boundary ,
638 // we need to substract the offset[i].
639 PrimExpr end = start + (kernel[i] - 1) * dilation[i];
640 end = min(end, data_shape[ii] + pad_tail[i] - 1 - offset[i]);
641 num_el *= (end - start) / dilation[i] + 1;
642 }
643 return div(pool_sum(indices), num_el);
644 } else {
645 auto num_el = IntImm::Int32(1);
646 for (int i = 0; i < k_size; i++) {
647 int ii = axis[i];
648
649 // Let start and end contain the first and last index of our Tensor
650 // along the relevant dimension we use in our calculation.
651 // Assume indices -1, -2 represent the padding before (tail) and
652 // len(arr), len(arr) + 1 represent the padding after (head).
653 PrimExpr start = output[ii] * stride[i] - pad_head[i];
654 PrimExpr end = start + (kernel[i] - 1) * dilation[i];
655
656 // if start < 0, e.g. we start on a tail padded number this will be a positive
657 // number that represents the number of steps along the dilated kernel to reach a
658 // non-padded value. Otherwise this should be 0.
659 PrimExpr jumps_to_non_pad = (dilation[i] - 1 - start) / dilation[i];
660 jumps_to_non_pad = max(jumps_to_non_pad, IntImm(jumps_to_non_pad.ty(), 0));
661
662 end = min(end, data_shape[ii] - 1);
663 num_el *= (end - (start + dilation[i] * jumps_to_non_pad)) / dilation[i] + 1;
664 }
665
666 PrimExpr divide_factor = max(num_el, IntImm::Int32(1));
667 return div(pool_sum(indices), divide_factor);
668 }
669 },
670 "pool_avg", kElementWise, attrs);
671 } else {
672 LOG(ERROR) << "Unrecognized pool_type: " << pool_type;
673 return x;
674 }
675}
