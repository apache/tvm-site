98inline void SaveTensor(ffi::TensorView source, support::Stream* strm) {
99 const DLTensor* tensor = ffi::AnyView(source).cast<DLTensor*>();
100 uint64_t header = kTVMTensorMagic, reserved = 0;
101 strm->Write(header);
102 strm->Write(reserved);
103 // Always save data as CPU context
104 //
105 // Parameters that get serialized should be in CPU by default.
106 // So even the array's context is GPU, it will be stored as CPU array.
107 // This is used to prevent case when another user loads the parameters
108 // back on machine that do not have GPU or related context.
109 //
110 // We can always do array.CopyTo(target_dev) to get a corresponding
111 // array in the target context.
112 Device cpu_dev;
113 cpu_dev.device_type = kDLCPU;
114 cpu_dev.device_id = 0;
115 strm->Write(cpu_dev);
116 strm->Write(tensor->ndim);
117 strm->Write(tensor->dtype);
118 int ndim = tensor->ndim;
119 strm->WriteArray(tensor->shape, ndim);
120 int type_bytes = (tensor->dtype.bits + 7) / 8;
121 int64_t num_elems = 1;
122 for (int i = 0; i < ndim; ++i) {
123 num_elems *= tensor->shape[i];
124 }
125 int64_t data_byte_size = type_bytes * num_elems;
126 strm->Write(data_byte_size);
127
128 if (TVM_FFI_IO_NO_ENDIAN_SWAP && tensor->device.device_type == kDLCPU &&
129 ffi::IsContiguous(*tensor) && tensor->byte_offset == 0) {
130 // quick path
131 strm->Write(tensor->data, data_byte_size);
132 } else {
133 std::vector<uint8_t> bytes(data_byte_size);
134 CopyToBytes(source, bytes.data(), data_byte_size);
135 if (!TVM_FFI_IO_NO_ENDIAN_SWAP) {
136 ffi::ByteSwap(bytes.data(), type_bytes, num_elems);
137 }
138 strm->Write(bytes.data(), data_byte_size);
139 }
140}

143inline ffi::Tensor LoadTensor(support::Stream* strm) {
144 uint64_t header, reserved;
145 TVM_FFI_ICHECK(strm->Read(&header)) << "Invalid DLTensor file format";
146 TVM_FFI_ICHECK(strm->Read(&reserved)) << "Invalid DLTensor file format";
147 TVM_FFI_ICHECK(header == kTVMTensorMagic) << "Invalid DLTensor file format";
148 Device dev;
149 int ndim;
150 DLDataType dtype;
151 TVM_FFI_ICHECK(strm->Read(&dev)) << "Invalid DLTensor file format";
152 TVM_FFI_ICHECK(strm->Read(&ndim)) << "Invalid DLTensor file format";
153 TVM_FFI_ICHECK(strm->Read(&dtype)) << "Invalid DLTensor file format";
154 TVM_FFI_ICHECK_EQ(dev.device_type, kDLCPU)
155 << "Invalid DLTensor device: can only save as CPU tensor";
156 std::vector<int64_t> shape(ndim);
157 if (ndim != 0) {
158 TVM_FFI_ICHECK(strm->ReadArray(&shape[0], ndim)) << "Invalid DLTensor file format";
159 }
160 ffi::Tensor ret = Empty(ffi::Shape(shape), dtype, dev);
161 int64_t num_elems = 1;
162 int elem_bytes = (ret.GetDLTensorPtr()->dtype.bits + 7) / 8;
163 for (int i = 0; i < ret.GetDLTensorPtr()->ndim; ++i) {
164 num_elems *= ret.shape()[i];
165 }
166 int64_t data_byte_size;
167 TVM_FFI_ICHECK(strm->Read(&data_byte_size)) << "Invalid DLTensor file format";
168 TVM_FFI_ICHECK(data_byte_size == num_elems * elem_bytes) << "Invalid DLTensor file format";
169 auto read_ret = strm->Read(ret.GetDLTensorPtr()->data, data_byte_size);
170 // Only check non-empty data
171 if (ndim > 0 && shape[0] != 0) {
172 TVM_FFI_ICHECK(read_ret) << "Invalid DLTensor file format";
173 }
174 if (!TVM_FFI_IO_NO_ENDIAN_SWAP) {
175 ffi::ByteSwap(ret.GetDLTensorPtr()->data, elem_bytes, num_elems);
176 }
177 return ret;
178}

186inline Device GetPreferredHostDevice(Device device) {
187 if (device.device_type == DLDeviceType::kDLCUDA) {
188 return Device{DLDeviceType::kDLCUDAHost, 0};
189 } else if (device.device_type == DLDeviceType::kDLROCM) {
190 return Device{DLDeviceType::kDLROCMHost, 0};
191 } else {
192 // Fallback to CPU.
193 return Device{DLDeviceType::kDLCPU, 0};
194 }
195}
