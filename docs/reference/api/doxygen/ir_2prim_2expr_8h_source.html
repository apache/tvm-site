657/*
658 * \brief Template function to convert Map to unordered_map
659 * Sometimes useful for API gluing when internal uses unordered_map
660 * \param dmap The container map
661 * \return The corresponding unordered_map.
662 * \tparam K the key of the Map.
663 * \tparam V the value of the Map.
664 */
665template <typename K, typename V>
666inline std::unordered_map<K, V> as_unordered_map(const ffi::Map<K, V>& dmap) {
667 std::unordered_map<K, V> ret;
668 for (auto kv : dmap) {
669 ret[kv.first] = kv.second;
670 }
671 return ret;
672}
