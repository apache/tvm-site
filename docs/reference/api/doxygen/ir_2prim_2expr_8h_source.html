763/*
764 * \brief Template function to convert Map to unordered_map
765 * Sometimes useful for API gluing when internal uses unordered_map
766 * \param dmap The container map
767 * \return The corresponding unordered_map.
768 * \tparam K the key of the Map.
769 * \tparam V the value of the Map.
770 */
771template <typename K, typename V>
772inline std::unordered_map<K, V> as_unordered_map(const ffi::Map<K, V>& dmap) {
773 std::unordered_map<K, V> ret;
774 for (auto kv : dmap) {
775 ret[kv.first] = kv.second;
776 }
777 return ret;
778}
