543/*
544 * \brief Template function to convert Map to unordered_map
545 * Sometimes useful for API gluing when internal uses unordered_map
546 * \param dmap The container map
547 * \return The corresponding unordered_map.
548 * \tparam K the key of the Map.
549 * \tparam V the value of the Map.
550 */
551template <typename K, typename V>
552inline std::unordered_map<K, V> as_unordered_map(const ffi::Map<K, V>& dmap) {
553 std::unordered_map<K, V> ret;
554 for (auto kv : dmap) {
555 ret[kv.first] = kv.second;
556 }
557 return ret;
558}
