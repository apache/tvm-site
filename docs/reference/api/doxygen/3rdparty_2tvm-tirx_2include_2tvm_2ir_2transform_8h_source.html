151 template <typename ValueType>
152 static void RegisterConfigOption(const char* key) {
153 if constexpr (std::is_base_of_v<ffi::ObjectRef, ValueType>) {
154 // Static-init blocks may run before the cached runtime type index is initialized.
155 int32_t tindex = ValueType::ContainerType::_GetOrAllocRuntimeTypeIndex();
156 auto type_key = ffi::TypeIndexToTypeKey(tindex);
157 auto legalization = [=](ffi::Any value) -> ffi::Any {
158 if (auto opt_map = value.try_cast<ffi::Map<ffi::String, ffi::Any>>()) {
159 return ffi::reflection::ObjectCreator(type_key)(opt_map.value());
160 } else {
161 auto opt_val = value.try_cast<ValueType>();
162 if (!opt_val.has_value()) {
163 TVM_FFI_THROW(AttributeError)
164 << "Expect config " << key << " to have type " << type_key << ", but instead get "
165 << ffi::details::AnyUnsafe::GetMismatchTypeInfo<ValueType>(value);
166 }
167 return *opt_val;
168 }
169 };
170 RegisterConfigOption(key, type_key, legalization);
171 } else {
172 // non-object type, do not support implicit conversion from map
173 std::string type_str = ffi::TypeTraits<ValueType>::TypeStr();
174 auto legalization = [=](ffi::Any value) -> ffi::Any {
175 auto opt_val = value.try_cast<ValueType>();
176 if (!opt_val.has_value()) {
177 TVM_FFI_THROW(AttributeError)
178 << "Expect config " << key << " to have type " << type_str << ", but instead get "
179 << ffi::details::AnyUnsafe::GetMismatchTypeInfo<ValueType>(value);
180 TVM_FFI_UNREACHABLE();
181 } else {
182 return *opt_val;
183 }
184 };
185 RegisterConfigOption(key, type_str, legalization);
186 }
187 }

213template <typename TConfig>
214inline TConfig PassConfigWithDefaults() {
215 static_assert(std::is_base_of_v<ffi::ObjectRef, TConfig>,
216 "Can only create ObjectRef-derived types");
217 using ContainerType = typename TConfig::ContainerType;
218 static auto finit_object = ffi::Function::GetGlobalRequired("ffi.MakeObjectFromPackedArgs");
219 ffi::AnyView packed_args[1];
220 packed_args[0] = ContainerType::RuntimeTypeIndex();
221 ffi::Any rv;
222 finit_object.CallPacked(ffi::PackedArgs(packed_args, 1), &rv);
223 return rv.cast<TConfig>();
224}
