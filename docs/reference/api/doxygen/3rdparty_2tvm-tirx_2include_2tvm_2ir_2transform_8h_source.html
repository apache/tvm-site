152 template <typename ValueType>
153 static void RegisterConfigOption(const char* key) {
154 if constexpr (std::is_base_of_v<ffi::ObjectRef, ValueType>) {
155 // Static-init blocks may run before the cached runtime type index is initialized.
156 int32_t tindex = ValueType::ContainerType::_GetOrAllocRuntimeTypeIndex();
157 auto type_key = ffi::TypeIndexToTypeKey(tindex);
158 auto legalization = [=](ffi::Any value) -> ffi::Any {
159 if (auto opt_map = value.try_cast<ffi::Map<ffi::String, ffi::Any>>()) {
160 return ffi::reflection::ObjectCreator(type_key)(opt_map.value());
161 } else {
162 auto opt_val = value.try_cast<ValueType>();
163 if (!opt_val.has_value()) {
164 TVM_FFI_THROW(AttributeError)
165 << "Expect config " << key << " to have type " << type_key << ", but instead get "
166 << ffi::detail::AnyUnsafe::GetMismatchTypeInfo<ValueType>(value);
167 }
168 return *opt_val;
169 }
170 };
171 RegisterConfigOption(key, type_key, legalization);
172 } else {
173 // non-object type, do not support implicit conversion from map
174 std::string type_str = ffi::TypeTraits<ValueType>::TypeStr();
175 auto legalization = [=](ffi::Any value) -> ffi::Any {
176 auto opt_val = value.try_cast<ValueType>();
177 if (!opt_val.has_value()) {
178 TVM_FFI_THROW(AttributeError)
179 << "Expect config " << key << " to have type " << type_str << ", but instead get "
180 << ffi::detail::AnyUnsafe::GetMismatchTypeInfo<ValueType>(value);
181 TVM_FFI_UNREACHABLE();
182 } else {
183 return *opt_val;
184 }
185 };
186 RegisterConfigOption(key, type_str, legalization);
187 }
188 }

215template <typename TConfig>
216inline TConfig PassConfigWithDefaults() {
217 static_assert(std::is_base_of_v<ffi::ObjectRef, TConfig>,
218 "Can only create ObjectRef-derived types");
219 using ContainerType = typename TConfig::ContainerType;
220 static auto finit_object = ffi::Function::GetGlobalRequired("ffi.MakeObjectFromPackedArgs");
221 ffi::AnyView packed_args[1];
222 packed_args[0] = ContainerType::RuntimeTypeIndex();
223 ffi::Any rv;
224 finit_object.CallPacked(ffi::PackedArgs(packed_args, 1), &rv);
225 return rv.cast<TConfig>();
226}
