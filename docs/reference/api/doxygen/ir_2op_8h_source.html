224template <typename T>
225inline bool ReportTypeMismatch(ffi::Expected<void>* out, const ValidateState* state, bool type_arg,
226 size_t index, ffi::AnyView actual) {
227 const auto& fixed = type_arg ? state->op->ty_args_info : state->op->args_info;
228 const auto& tail = type_arg ? state->op->var_ty_args_info : state->op->var_args_info;
229 std::string name;
230 if (index < fixed.size()) {
231 name = fixed[index]->name;
232 } else if (tail.has_value()) {
233 name = tail.value()->name;
234 }
235 std::string name_clause = name.empty() ? "" : " (`" + name + "`)";
236 TVMFFIAny raw = actual.CopyToTVMFFIAny();
237 *out = TVM_FFI_UNEXPECTED(TypeError)
238 << "Op `" << state->op->name << "`: `" << (type_arg ? "Call.ty_args[" : "Call.args[")
239 << index << "]`" << name_clause << " expected `" << ffi::TypeTraits<T>::TypeStr()
240 << "`, got `" << ffi::TypeTraits<T>::GetMismatchTypeInfo(&raw) << "`.";
241 return false;
242}

580 template <typename... Specs>
581 TVM_FFI_INLINE static ffi::Expected<void> ValidateSignature(const CallNode* call) noexcept {
582 static_assert((std::is_base_of_v<sig::SignatureTrait, Specs> && ...),
583 "Unknown signature descriptor");
584 constexpr size_t n_args = (size_t{0} + ... + Specs::kArgsCount);
585 constexpr size_t n_ty_args = (size_t{0} + ... + Specs::kTyArgsCount);
586 constexpr size_t n_var_args = (size_t{0} + ... + Specs::kVarArgsCount);
587 constexpr size_t n_var_ty_args = (size_t{0} + ... + Specs::kVarTyArgsCount);
588 constexpr size_t n_attrs = (size_t{0} + ... + Specs::kCallAttrsCount);
589 static_assert(n_var_args <= 1 && n_var_ty_args <= 1 && n_attrs <= 1,
590 "Duplicate signature tail or call_attrs");
591 const OpNode* op = call ? call->op.as<OpNode>() : nullptr;
592 if (TVM_FFI_PREDICT_FALSE(!op)) {
593 return TVM_FFI_UNEXPECTED(TypeError) << "Invalid Op Call";
594 }
595 if (TVM_FFI_PREDICT_FALSE(!call->args.defined() || !call->ty_args.defined())) {
596 return TVM_FFI_UNEXPECTED(TypeError) << "Op `" << op->name << "`: invalid Call";
597 }
598 ffi::Expected<void> result;
599 auto* out = &result;
600 if (TVM_FFI_PREDICT_FALSE(n_var_args ? call->args.size() < n_args
601 : call->args.size() != n_args)) {
602 *out = TVM_FFI_UNEXPECTED(TypeError)
603 << "Op `" << op->name << "`: Call.args expected " << (n_var_args ? "at least " : "")
604 << n_args << (n_args == 1 ? " argument" : " arguments") << ", got "
605 << call->args.size();
606 return result;
607 }
608 if (TVM_FFI_PREDICT_FALSE(n_var_ty_args ? call->ty_args.size() < n_ty_args
609 : call->ty_args.size() != n_ty_args)) {
610 *out = TVM_FFI_UNEXPECTED(TypeError)
611 << "Op `" << op->name << "`: Call.ty_args expected "
612 << (n_var_ty_args ? "at least " : "") << n_ty_args
613 << (n_ty_args == 1 ? " type argument" : " type arguments") << ", got "
614 << call->ty_args.size();
615 return result;
616 }
617 if constexpr (sizeof...(Specs) != 0) {
618 sig::ValidateState state{call, op};
619 if (TVM_FFI_PREDICT_FALSE(!(Specs::Validate(&state, out) && ...))) {
620 return result;
621 }
622 }
623 return result;
624 }
