216template <typename T>
217inline bool ReportTypeMismatch(ffi::Expected<void>* out, const ValidateState* state, bool type_arg,
218 size_t index, ffi::AnyView actual) {
219 const auto& fixed = type_arg ? state->op->ty_args_info : state->op->args_info;
220 const auto& tail = type_arg ? state->op->var_ty_args_info : state->op->var_args_info;
221 std::string name;
222 if (index < fixed.size()) {
223 name = fixed[index]->name;
224 } else if (tail.has_value()) {
225 name = tail.value()->name;
226 }
227 std::string name_clause = name.empty() ? "" : " (`" + name + "`)";
228 TVMFFIAny raw = actual.CopyToTVMFFIAny();
229 *out = TVM_FFI_UNEXPECTED(TypeError)
230 << "Op `" << state->op->name << "`: `" << (type_arg ? "Call.ty_args[" : "Call.args[")
231 << index << "]`" << name_clause << " expected `" << ffi::TypeTraits<T>::TypeStr()
232 << "`, got `" << ffi::TypeTraits<T>::GetMismatchTypeInfo(&raw) << "`.";
233 return false;
234}

574 template <typename... Specs>
575 TVM_FFI_INLINE static ffi::Expected<void> ValidateSignature(const CallNode* call) noexcept {
576 static_assert((std::is_base_of_v<sig::SignatureTrait, Specs> && ...),
577 "Unknown signature descriptor");
578 constexpr size_t n_args = (size_t{0} + ... + Specs::kArgsCount);
579 constexpr size_t n_ty_args = (size_t{0} + ... + Specs::kTyArgsCount);
580 constexpr size_t n_var_args = (size_t{0} + ... + Specs::kVarArgsCount);
581 constexpr size_t n_var_ty_args = (size_t{0} + ... + Specs::kVarTyArgsCount);
582 constexpr size_t n_attrs = (size_t{0} + ... + Specs::kCallAttrsCount);
583 static_assert(n_var_args <= 1 && n_var_ty_args <= 1 && n_attrs <= 1,
584 "Duplicate signature tail or call_attrs");
585 const OpNode* op = call ? call->op.as<OpNode>() : nullptr;
586 if (TVM_FFI_PREDICT_FALSE(!op)) {
587 return TVM_FFI_UNEXPECTED(TypeError) << "Invalid Op Call";
588 }
589 if (TVM_FFI_PREDICT_FALSE(!call->args.defined() || !call->ty_args.defined())) {
590 return TVM_FFI_UNEXPECTED(TypeError) << "Op `" << op->name << "`: invalid Call";
591 }
592 ffi::Expected<void> result;
593 auto* out = &result;
594 if (TVM_FFI_PREDICT_FALSE(n_var_args ? call->args.size() < n_args
595 : call->args.size() != n_args)) {
596 *out = TVM_FFI_UNEXPECTED(TypeError)
597 << "Op `" << op->name << "`: Call.args expected " << (n_var_args ? "at least " : "")
598 << n_args << (n_args == 1 ? " argument" : " arguments") << ", got "
599 << call->args.size();
600 return result;
601 }
602 if (TVM_FFI_PREDICT_FALSE(n_var_ty_args ? call->ty_args.size() < n_ty_args
603 : call->ty_args.size() != n_ty_args)) {
604 *out = TVM_FFI_UNEXPECTED(TypeError)
605 << "Op `" << op->name << "`: Call.ty_args expected "
606 << (n_var_ty_args ? "at least " : "") << n_ty_args
607 << (n_ty_args == 1 ? " type argument" : " type arguments") << ", got "
608 << call->ty_args.size();
609 return result;
610 }
611 if constexpr (sizeof...(Specs) != 0) {
612 sig::ValidateState state{call, op};
613 if (TVM_FFI_PREDICT_FALSE(!(Specs::Validate(&state, out) && ...))) {
614 return result;
615 }
616 }
617 return result;
618 }

633template <typename ValueType>
634class OpAttrMap {
635 public:
641 int count(const Op& op) const {
642 return op->index_ < column_.size() && column_[op->index_] != nullptr;
643 }
649 ValueType operator[](const Op& op) const {
650 TVM_FFI_ICHECK(count(op)) << "Attribute " << attr_name_ << " has not been registered for "
651 << op->name;
652 if constexpr (std::is_same_v<ValueType, ffi::Any>) {
653 return column_[op->index_];
654 } else {
655 return column_[op->index_].template cast<ValueType>();
656 }
657 }
664 ValueType get(const Op& op, ValueType def_value) const {
665 return count(op) ? (*this)[op] : def_value;

674 TVM_FFI_ICHECK(expr.defined());

676 return get(ffi::GetRef<Op>(op), def_value);
677 }
678 return def_value;
679 }
