235template <typename T>
236inline bool ReportTypeMismatch(ffi::Expected<void>* out, const ValidateState* state, bool type_arg,
237 size_t index, ffi::AnyView actual) {
238 const auto& fixed = type_arg ? state->op->ty_args_info : state->op->args_info;
239 const auto& tail = type_arg ? state->op->var_ty_args_info : state->op->var_args_info;
240 std::string name;
241 if (index < fixed.size()) {
242 name = fixed[index]->name;
243 } else if (tail.has_value()) {
244 name = tail.value()->name;
245 }
246 std::string name_clause = name.empty() ? "" : " (`" + name + "`)";
247 TVMFFIAny raw = actual.CopyToTVMFFIAny();
248 *out = TVM_FFI_UNEXPECTED(TypeError)
249 << "Op `" << state->op->name << "`: `" << (type_arg ? "Call.ty_args[" : "Call.args[")
250 << index << "]`" << name_clause << " expected `" << ffi::TypeTraits<T>::TypeStr()
251 << "`, got `" << ffi::TypeTraits<T>::GetMismatchTypeInfo(&raw) << "`.";
252 return false;
253}

590 template <typename... Specs>
591 TVM_FFI_INLINE static ffi::Expected<void> ValidateSignature(const CallNode* call) noexcept {
592 static_assert((std::is_base_of_v<sig::SignatureTrait, Specs> && ...),
593 "Unknown signature descriptor");
594 constexpr size_t n_args = (size_t{0} + ... + Specs::kArgsCount);
595 constexpr size_t n_ty_args = (size_t{0} + ... + Specs::kTyArgsCount);
596 constexpr size_t n_var_args = (size_t{0} + ... + Specs::kVarArgsCount);
597 constexpr size_t n_var_ty_args = (size_t{0} + ... + Specs::kVarTyArgsCount);
598 constexpr size_t n_attrs = (size_t{0} + ... + Specs::kCallAttrsCount);
599 static_assert(n_var_args <= 1 && n_var_ty_args <= 1 && n_attrs <= 1,
600 "Duplicate signature tail or call_attrs");
601 const OpNode* op = call ? call->op.as<OpNode>() : nullptr;
602 if (TVM_FFI_PREDICT_FALSE(!op)) {
603 return TVM_FFI_UNEXPECTED(TypeError) << "Invalid Op Call";
604 }
605 if (TVM_FFI_PREDICT_FALSE(!call->args.defined() || !call->ty_args.defined())) {
606 return TVM_FFI_UNEXPECTED(TypeError) << "Op `" << op->name << "`: invalid Call";
607 }
608 ffi::Expected<void> result;
609 auto* out = &result;
610 if (TVM_FFI_PREDICT_FALSE(n_var_args ? call->args.size() < n_args
611 : call->args.size() != n_args)) {
612 *out = TVM_FFI_UNEXPECTED(TypeError)
613 << "Op `" << op->name << "`: Call.args expected " << (n_var_args ? "at least " : "")
614 << n_args << (n_args == 1 ? " argument" : " arguments") << ", got "
615 << call->args.size();
616 return result;
617 }
618 if (TVM_FFI_PREDICT_FALSE(n_var_ty_args ? call->ty_args.size() < n_ty_args
619 : call->ty_args.size() != n_ty_args)) {
620 *out = TVM_FFI_UNEXPECTED(TypeError)
621 << "Op `" << op->name << "`: Call.ty_args expected "
622 << (n_var_ty_args ? "at least " : "") << n_ty_args
623 << (n_ty_args == 1 ? " type argument" : " type arguments") << ", got "
624 << call->ty_args.size();
625 return result;
626 }
627 if constexpr (sizeof...(Specs) != 0) {
628 sig::ValidateState state{call, op};
629 if (TVM_FFI_PREDICT_FALSE(!(Specs::Validate(&state, out) && ...))) {
630 return result;
631 }
632 }
633 return result;
634 }
