213template <typename T>
214inline bool ReportTypeMismatch(ffi::Expected<void>* out, const ValidateState* state, bool type_arg,
215 size_t index, ffi::AnyView actual) {
216 const auto& fixed = type_arg ? state->op->ty_args_info : state->op->args_info;
217 const auto& tail = type_arg ? state->op->var_ty_args_info : state->op->var_args_info;
218 std::string name;
219 if (index < fixed.size()) {
220 name = fixed[index]->name;
221 } else if (tail.has_value()) {
222 name = tail.value()->name;
223 }
224 std::string name_clause = name.empty() ? "" : " (`" + name + "`)";
225 TVMFFIAny raw = actual.CopyToTVMFFIAny();
226 *out = TVM_FFI_UNEXPECTED(TypeError)
227 << "Op `" << state->op->name << "`: `" << (type_arg ? "Call.ty_args[" : "Call.args[")
228 << index << "]`" << name_clause << " expected `" << ffi::TypeTraits<T>::TypeStr()
229 << "`, got `" << ffi::TypeTraits<T>::GetMismatchTypeInfo(&raw) << "`.";
230 return false;
231}

571 template <typename... Specs>
572 TVM_FFI_INLINE static ffi::Expected<void> ValidateSignature(const CallNode* call) noexcept {
573 static_assert((std::is_base_of_v<sig::SignatureTrait, Specs> && ...),
574 "Unknown signature descriptor");
575 constexpr size_t n_args = (size_t{0} + ... + Specs::kArgsCount);
576 constexpr size_t n_ty_args = (size_t{0} + ... + Specs::kTyArgsCount);
577 constexpr size_t n_var_args = (size_t{0} + ... + Specs::kVarArgsCount);
578 constexpr size_t n_var_ty_args = (size_t{0} + ... + Specs::kVarTyArgsCount);
579 constexpr size_t n_attrs = (size_t{0} + ... + Specs::kCallAttrsCount);
580 static_assert(n_var_args <= 1 && n_var_ty_args <= 1 && n_attrs <= 1,
581 "Duplicate signature tail or call_attrs");
582 const OpNode* op = call ? call->op.as<OpNode>() : nullptr;
583 if (TVM_FFI_PREDICT_FALSE(!op)) {
584 return TVM_FFI_UNEXPECTED(TypeError) << "Invalid Op Call";
585 }
586 if (TVM_FFI_PREDICT_FALSE(!call->args.defined() || !call->ty_args.defined())) {
587 return TVM_FFI_UNEXPECTED(TypeError) << "Op `" << op->name << "`: invalid Call";
588 }
589 ffi::Expected<void> result;
590 auto* out = &result;
591 if (TVM_FFI_PREDICT_FALSE(n_var_args ? call->args.size() < n_args
592 : call->args.size() != n_args)) {
593 *out = TVM_FFI_UNEXPECTED(TypeError)
594 << "Op `" << op->name << "`: Call.args expected " << (n_var_args ? "at least " : "")
595 << n_args << (n_args == 1 ? " argument" : " arguments") << ", got "
596 << call->args.size();
597 return result;
598 }
599 if (TVM_FFI_PREDICT_FALSE(n_var_ty_args ? call->ty_args.size() < n_ty_args
600 : call->ty_args.size() != n_ty_args)) {
601 *out = TVM_FFI_UNEXPECTED(TypeError)
602 << "Op `" << op->name << "`: Call.ty_args expected "
603 << (n_var_ty_args ? "at least " : "") << n_ty_args
604 << (n_ty_args == 1 ? " type argument" : " type arguments") << ", got "
605 << call->ty_args.size();
606 return result;
607 }
608 if constexpr (sizeof...(Specs) != 0) {
609 sig::ValidateState state{call, op};
610 if (TVM_FFI_PREDICT_FALSE(!(Specs::Validate(&state, out) && ...))) {
611 return result;
612 }
613 }
614 return result;
615 }
