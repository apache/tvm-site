254template <typename ValueType>
255class OpAttrMap {
256 public:
262 int count(const Op& op) const {
263 return op->index_ < column_.size() && column_[op->index_] != nullptr;
264 }
270 ValueType operator[](const Op& op) const {
271 TVM_FFI_ICHECK(count(op)) << "Attribute " << attr_name_ << " has not been registered for "
272 << op->name;
273 if constexpr (std::is_same_v<ValueType, ffi::Any>) {
274 return column_[op->index_];
275 } else {
276 return column_[op->index_].template cast<ValueType>();
277 }
278 }
285 ValueType get(const Op& op, ValueType def_value) const {
286 return count(op) ? (*this)[op] : def_value;
287 }
294 ValueType get(const Expr& expr, ValueType def_value) const {
295 TVM_FFI_ICHECK(expr.defined());
296 if (const auto* op = expr.as<OpNode>()) {
297 return get(ffi::GetRef<Op>(op), def_value);
298 }
299 return def_value;
300 }
301
302 private:
303 friend class Op;
304 OpAttrMap(ffi::List<ffi::Any> column, ffi::String attr_name)
305 : column_(std::move(column)), attr_name_(std::move(attr_name)) {}
306 ffi::List<ffi::Any> column_;
307 ffi::String attr_name_;
308};
