314 template <typename Callback>
315 ffi::List<Doc> WithDocScope(Callback&& callback) {
316 size_t depth = doc_scopes_.size();
317 BeginDocScope();
318 try {
319 std::forward<Callback>(callback)();
320 TVM_FFI_CHECK(doc_scopes_.size() == depth + 1, ValueError)
321 << "printer unbalanced Doc collection scopes";
322 return EndDocScope();
323 } catch (...) {
324 while (doc_scopes_.size() > depth) doc_scopes_.pop_back();
325 throw;
326 }

398inline ffi::Array<StmtDoc> ToStmtDocArray(const ffi::List<Doc>& docs) {
399 ffi::Array<StmtDoc> statements;
400 for (const auto& doc : docs) {
401 if (auto block = doc.as<StmtBlockDoc>()) {
402 for (const auto& stmt : block.value()->stmts) statements.push_back(stmt);
403 } else {
404 statements.push_back(doc.as_or_throw<StmtDoc>());
405 }
406 }
407 return statements;
408}

448inline ffi::Optional<ExprDoc> DocTranslatorObj::DefaultTranslate(ffi::AnyView value,
449 const ffi::Object* destination) {
450 static ffi::reflection::TypeAttrColumn column(
451 tvm::script::printer::type_attr::kDocTranslate.name);
452 ffi::AnyView attr = column[value.type_index()];
453 if (attr.type_index() == ffi::TypeIndex::kTVMFFIOpaquePtr) {
454 return type_attr::kDocTranslate.ValueFrom(attr)
455 .value()
456 .CallExpected(this, value, destination)
457 .value();
458 }
459 if (attr.type_index() == ffi::TypeIndex::kTVMFFIFunction) {
460 ffi::Any destination_arg = nullptr;
461 if (destination) destination_arg = ffi::GetRef<ffi::ObjectRef>(destination);
462 return attr.cast<ffi::Function>()
463 .CallExpected<ffi::Optional<ExprDoc>>(this, value, destination_arg)
464 .value();
465 }
466 if (attr.type_index() == ffi::TypeIndex::kTVMFFINone) {
467 if (value == nullptr) return LiteralDoc::None(std::nullopt);
468 if (auto string = value.as<ffi::String>()) return LiteralDoc::Str(string.value(), std::nullopt);
469 if (auto boolean = value.as<bool>()) return LiteralDoc::Boolean(boolean.value(), std::nullopt);
470 if (auto integer = value.as<int64_t>()) return LiteralDoc::Int(integer.value(), std::nullopt);
471 if (auto floating = value.as<double>())
472 return LiteralDoc::Float(floating.value(), std::nullopt);
473 if (auto dtype = value.as<DLDataType>())
474 return LiteralDoc::DataType(dtype.value(), std::nullopt);
475 if (auto type = value.as<Type>()) return DefaultTypeTranslate(type.value());
476 }
477 TVM_FFI_THROW(TypeError) << (attr.type_index() == ffi::TypeIndex::kTVMFFINone
478 ? std::string("printer has no Doc hook for ") +
479 value.GetTypeKey()
480 : "printer Doc hook must be a native pointer or ffi.Function");
481 TVM_FFI_UNREACHABLE();
482}
