265 template <typename Callback>
266 ffi::List<Doc> WithDocScope(Callback&& callback) {
267 size_t depth = doc_scopes_.size();
268 BeginDocScope();
269 try {
270 std::forward<Callback>(callback)();
271 TVM_FFI_CHECK(doc_scopes_.size() == depth + 1, ValueError)
272 << "printer unbalanced Doc collection scopes";
273 return EndDocScope();
274 } catch (...) {

346inline ffi::Array<StmtDoc> ToStmtDocArray(const ffi::List<Doc>& docs) {
347 ffi::Array<StmtDoc> statements;
348 for (const auto& doc : docs) {
349 if (auto block = doc.as<StmtBlockDoc>()) {
350 for (const auto& stmt : block.value()->stmts) statements.push_back(stmt);
351 } else {
352 statements.push_back(doc.as_or_throw<StmtDoc>());
353 }
354 }
355 return statements;
356}

400inline ffi::Optional<ExprDoc> DocTranslatorObj::DefaultTranslate(ffi::AnyView value,
401 const ffi::Object* destination) {
402 static ffi::reflection::TypeAttrColumn column(tvm::script::printer::type_attr::kDocTranslate);
403 ffi::AnyView attr = column[value.type_index()];
404 if (attr.type_index() == ffi::TypeIndex::kTVMFFIOpaquePtr) {
405 return ffi::details::ExpectedUnsafe::MoveFromTVMFFIAny<ffi::Optional<ExprDoc>>(
406 reinterpret_cast<decltype(DocTranslatorVTable::translate)>(attr.cast<void*>())(
407 this, value, destination))
408 .value();
409 }
410 if (attr.type_index() == ffi::TypeIndex::kTVMFFIFunction) {
411 ffi::Any destination_arg = nullptr;
412 if (destination) destination_arg = ffi::GetRef<ffi::ObjectRef>(destination);
413 return attr.cast<ffi::Function>()
414 .CallExpected<ffi::Optional<ExprDoc>>(this, value, destination_arg)
415 .value();
416 }
417 TVM_FFI_THROW(TypeError) << (attr.type_index() == ffi::TypeIndex::kTVMFFINone
418 ? std::string("printer has no Doc hook for ") +
419 value.GetTypeKey()
420 : "printer Doc hook must be a native pointer or ffi.Function");
421}
