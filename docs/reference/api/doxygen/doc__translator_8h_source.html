278 template <typename Callback>
279 ffi::List<Doc> WithDocScope(Callback&& callback) {
280 size_t depth = doc_scopes_.size();
281 BeginDocScope();

284 TVM_FFI_CHECK(doc_scopes_.size() == depth + 1, ValueError)

310 ffi::Optional<ExprDoc> DefaultTranslate(ffi::AnyView value,
311 const ffi::Object* destination = nullptr) {
312 static ffi::reflection::TypeAttrColumn column(kDocTranslate);
313 ffi::AnyView attr = column[value.type_index()];
314 if (attr.type_index() == ffi::TypeIndex::kTVMFFIOpaquePtr) {
315 return ffi::details::ExpectedUnsafe::MoveFromTVMFFIAny<ffi::Optional<ExprDoc>>(
316 reinterpret_cast<decltype(DocTranslatorVTable::translate)>(attr.cast<void*>())(
317 this, value, destination))
318 .value();
319 }
320 if (attr.type_index() == ffi::TypeIndex::kTVMFFIFunction) {
321 ffi::Any destination_arg = nullptr;
322 if (destination) destination_arg = ffi::GetRef<ffi::ObjectRef>(destination);
323 return attr.cast<ffi::Function>()
324 .CallExpected<ffi::Optional<ExprDoc>>(this, value, destination_arg)
325 .value();
326 }
327 TVM_FFI_THROW(TypeError) << (attr.type_index() == ffi::TypeIndex::kTVMFFINone
328 ? std::string("printer has no Doc hook for ") +
329 value.GetTypeKey()
330 : "printer Doc hook must be a native pointer or ffi.Function");
331 }

379inline ffi::Array<StmtDoc> ToStmtDocArray(const ffi::List<Doc>& docs) {
380 ffi::Array<StmtDoc> statements;
381 for (const auto& doc : docs) {
382 if (auto block = doc.as<StmtBlockDoc>()) {
383 for (const auto& stmt : block.value()->stmts) statements.push_back(stmt);
384 } else {
385 statements.push_back(doc.as_or_throw<StmtDoc>());
386 }
387 }
388 return statements;
389}
