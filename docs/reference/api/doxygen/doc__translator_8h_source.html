310 template <typename Callback>
311 ffi::List<Doc> WithDocScope(Callback&& callback) {
312 size_t depth = doc_scopes_.size();
313 BeginDocScope();
314 try {
315 std::forward<Callback>(callback)();
316 TVM_FFI_CHECK(doc_scopes_.size() == depth + 1, ValueError)
317 << "printer unbalanced Doc collection scopes";
318 return EndDocScope();
319 } catch (...) {
320 while (doc_scopes_.size() > depth) doc_scopes_.pop_back();
321 throw;
322 }
323 }

394inline ffi::Array<StmtDoc> ToStmtDocArray(const ffi::List<Doc>& docs) {
395 ffi::Array<StmtDoc> statements;
396 for (const auto& doc : docs) {
397 if (auto block = doc.as<StmtBlockDoc>()) {
398 for (const auto& stmt : block.value()->stmts) statements.push_back(stmt);
399 } else {
400 statements.push_back(doc.as_or_throw<StmtDoc>());
401 }
402 }

444inline ffi::Optional<ExprDoc> DocTranslatorObj::DefaultTranslate(ffi::AnyView value,
445 const ffi::Object* destination) {
446 static ffi::reflection::TypeAttrColumn column(tvm::script::printer::type_attr::kDocTranslate);
447 ffi::AnyView attr = column[value.type_index()];
448 if (attr.type_index() == ffi::TypeIndex::kTVMFFIOpaquePtr) {
449 return ffi::details::ExpectedUnsafe::MoveFromTVMFFIAny<ffi::Optional<ExprDoc>>(
450 reinterpret_cast<decltype(DocTranslatorVTable::translate)>(attr.cast<void*>())(
451 this, value, destination))
452 .value();
453 }
454 if (attr.type_index() == ffi::TypeIndex::kTVMFFIFunction) {
455 ffi::Any destination_arg = nullptr;
456 if (destination) destination_arg = ffi::GetRef<ffi::ObjectRef>(destination);
457 return attr.cast<ffi::Function>()
458 .CallExpected<ffi::Optional<ExprDoc>>(this, value, destination_arg)
459 .value();
460 }
461 if (attr.type_index() == ffi::TypeIndex::kTVMFFINone) {
462 if (value == nullptr) return LiteralDoc::None(std::nullopt);
463 if (auto string = value.as<ffi::String>()) return LiteralDoc::Str(string.value(), std::nullopt);
464 if (auto boolean = value.as<bool>()) return LiteralDoc::Boolean(boolean.value(), std::nullopt);
465 if (auto integer = value.as<int64_t>()) return LiteralDoc::Int(integer.value(), std::nullopt);
466 if (auto floating = value.as<double>())
467 return LiteralDoc::Float(floating.value(), std::nullopt);
468 if (auto dtype = value.as<DLDataType>())
469 return LiteralDoc::DataType(dtype.value(), std::nullopt);
470 if (auto type = value.as<Type>()) return DefaultTypeTranslate(type.value());
471 }
472 TVM_FFI_THROW(TypeError) << (attr.type_index() == ffi::TypeIndex::kTVMFFINone
473 ? std::string("printer has no Doc hook for ") +
474 value.GetTypeKey()
475 : "printer Doc hook must be a native pointer or ffi.Function");
476}
