256template <typename T>
257inline bool ReportTypeMismatch(ffi::Expected<void>* out, const ValidateState* state, bool type_arg,
258 size_t index, ffi::AnyView actual) {
259 const auto& fixed = type_arg ? state->op->ty_args_info : state->op->args_info;
260 const auto& tail = type_arg ? state->op->var_ty_args_info : state->op->var_args_info;
261 std::string name;
262 if (index < fixed.size()) {
263 name = fixed[index]->name;
264 } else if (tail.has_value()) {
265 name = tail.value()->name;
266 }
267 std::string name_clause = name.empty() ? "" : " (`" + name + "`)";
268 TVMFFIAny raw = actual.CopyToTVMFFIAny();
269 *out = TVM_FFI_UNEXPECTED(TypeError)
270 << "Op `" << state->op->name << "`: `" << (type_arg ? "Call.ty_args[" : "Call.args[")
271 << index << "]`" << name_clause << " expected `" << ffi::TypeTraits<T>::TypeStr()
272 << "`, got `" << ffi::TypeTraits<T>::GetMismatchTypeInfo(&raw) << "`.";
273 return false;
274}

650 template <typename... Specs>
651 TVM_FFI_INLINE static ffi::Expected<void> ValidateSignature(const CallObj* call) noexcept {
652 static_assert((std::is_base_of_v<sig::SignatureTrait, Specs> && ...),
653 "Unknown signature descriptor");
654 constexpr size_t n_args = (size_t{0} + ... + Specs::kArgsCount);
655 constexpr size_t n_ty_args = (size_t{0} + ... + Specs::kTyArgsCount);
656 constexpr size_t n_var_args = (size_t{0} + ... + Specs::kVarArgsCount);
657 constexpr size_t n_var_ty_args = (size_t{0} + ... + Specs::kVarTyArgsCount);
658 constexpr size_t n_attrs = (size_t{0} + ... + Specs::kCallAttrsCount);
659 static_assert(n_var_args <= 1 && n_var_ty_args <= 1 && n_attrs <= 1,
660 "Duplicate signature tail or call_attrs");
661 const OpObj* op = call ? call->op.as<OpObj>() : nullptr;
662 if (TVM_FFI_PREDICT_FALSE(!op)) {
663 return TVM_FFI_UNEXPECTED(TypeError) << "Invalid Op Call";
664 }
665 if (TVM_FFI_PREDICT_FALSE(!call->args.defined() || !call->ty_args.defined())) {
666 return TVM_FFI_UNEXPECTED(TypeError) << "Op `" << op->name << "`: invalid Call";
667 }
668 ffi::Expected<void> result;
669 auto* out = &result;
670 if (TVM_FFI_PREDICT_FALSE(n_var_args ? call->args.size() < n_args
671 : call->args.size() != n_args)) {
672 *out = TVM_FFI_UNEXPECTED(TypeError)
673 << "Op `" << op->name << "`: Call.args expected " << (n_var_args ? "at least " : "")
674 << n_args << (n_args == 1 ? " argument" : " arguments") << ", got "
675 << call->args.size();
676 return result;
677 }
678 if (TVM_FFI_PREDICT_FALSE(n_var_ty_args ? call->ty_args.size() < n_ty_args
679 : call->ty_args.size() != n_ty_args)) {
680 *out = TVM_FFI_UNEXPECTED(TypeError)
681 << "Op `" << op->name << "`: Call.ty_args expected "
682 << (n_var_ty_args ? "at least " : "") << n_ty_args
683 << (n_ty_args == 1 ? " type argument" : " type arguments") << ", got "
684 << call->ty_args.size();
685 return result;
686 }
687 if constexpr (sizeof...(Specs) != 0) {
688 sig::ValidateState state{call, op};
689 if (TVM_FFI_PREDICT_FALSE(!(Specs::Validate(&state, out) && ...))) {
690 return result;
691 }
692 }
693 return result;
694 }
