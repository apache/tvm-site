718inline void CheckMathUnaryOpInputDType(const char* op_name, const PrimType& dtype) {
719 TVM_FFI_CHECK(dtype.code() == DLDataTypeCode::kDLFloat ||
720 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16),
721 TypeError)
722 << "tirx." << op_name << " only supports floating-point inputs, but got " << dtype;
723}

725// Intrinsic operators
726#define TVM_DECLARE_INTRIN_UNARY_WITH_CHECK(OpName, CheckInputDType) \
727 inline PrimExpr OpName(PrimExpr x, Span span = Span()) { \
728 static const Op op = Op::Get("tirx." #OpName); \
729 PrimType x_ty = x.ty(); \
730 CheckInputDType(#OpName, x_ty); \
731 if (x_ty.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) { \
732 PrimType bf16_ty = x_ty; \
733 PrimType f32_ty = \
734 x_ty.IsScalableVector() \
735 ? PrimType::ScalableVector(DLDataTypeCode::kDLFloat, 32, x_ty.VScaleFactor()) \
736 : PrimType::Float(32, x_ty.lanes()); \
737 PrimExpr x_fp32 = tirx::Cast(f32_ty, x, span); \
738 PrimExpr result_fp32 = Call(f32_ty, op, {x_fp32}, {}, {}, span).as_or_throw<PrimExpr>(); \
739 return tirx::Cast(bf16_ty, result_fp32, span); \
740 } else { \
741 return Call(x_ty, op, {x}, {}, {}, span).as_or_throw<PrimExpr>(); \
742 } \
743 }

797inline bool IsPointerType(const Type& type, DLDataType element_type) {
798 if (type.IsMissing()) return false;
799 if (const auto* ptr_type = type.as<PointerTypeNode>()) {
800 if (const auto* prim_type = ptr_type->element_type.as<PrimTypeNode>()) {
801 return prim_type->dtype == element_type;
802 }
803 }
804 return false;
805}

924inline bool is_const_number(const PrimExpr& x) {
925 if (x.as<tirx::IntImmNode>()) {
926 return true;
927 } else if (x.as<tirx::FloatImmNode>()) {
928 return true;
929 } else if (const auto* op = x.as<tirx::BroadcastNode>()) {
930 return (op->value->IsInstance<tirx::IntImmNode>() ||
931 op->value->IsInstance<tirx::FloatImmNode>());
932 }
933 return false;

951inline bool is_no_op(const tirx::Stmt& stmt) {
952 if (!stmt.defined()) return true;
953 if (const auto* op = stmt.as<tirx::EvaluateNode>()) {
954 auto value = op->value.as<PrimExpr>();
955 return value && is_const_int(value.value());
956 }
957 if (const auto* op = stmt.as<tirx::SeqStmtNode>()) {
958 return op->seq.size() == 0;
959 }
960 return false;
961}

963template <typename ValueType>
964inline PrimExpr MakeConstScalar(PrimType dtype, ValueType value, Span span = Span()) {
965 DLDataTypeCode code = dtype.code();
966 if (code == DLDataTypeCode::kDLInt || code == DLDataTypeCode::kDLBool) {
967 return IntImm(dtype, static_cast<int64_t>(value), span);
968 }
969 if (code == DLDataTypeCode::kDLUInt) {
970 // Use IntImm if it is a small integer
971 uint64_t uval = static_cast<uint64_t>(value);
972 if (value < static_cast<ValueType>(0)) {
973 TVM_FFI_THROW(InternalError) << "cannot make uint from negative value " << value;
974 } else if (uval <= static_cast<uint64_t>(std::numeric_limits<int64_t>::max())) {
975 return IntImm(dtype, static_cast<int64_t>(value), span);
976 } else {
977 uint64_t mask = (static_cast<uint64_t>(1) << 32U) - 1U;
978 uint64_t low = uval & mask;
979 uint64_t high = uval >> 32U;
980 return LargeUIntImm(dtype, static_cast<int64_t>(low), static_cast<int64_t>(high), span);
981 }
982 }
983 if (dtype.MatchesCode(DLDataTypeCode::kDLFloat, DLDataTypeCode::kDLFloat8_e3m4,
984 DLDataTypeCode::kDLFloat8_e4m3, DLDataTypeCode::kDLFloat8_e4m3b11fnuz,
985 DLDataTypeCode::kDLFloat8_e4m3fn, DLDataTypeCode::kDLFloat8_e4m3fnuz,
986 DLDataTypeCode::kDLFloat8_e5m2, DLDataTypeCode::kDLFloat8_e5m2fnuz,
987 DLDataTypeCode::kDLFloat8_e8m0fnu, DLDataTypeCode::kDLFloat6_e2m3fn,
988 DLDataTypeCode::kDLFloat6_e3m2fn, DLDataTypeCode::kDLFloat4_e2m1fn) ||
989 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) {
990 return FloatImm(dtype, static_cast<double>(value), span);
991 }
992 TVM_FFI_THROW(InternalError) << "cannot make const for type " << dtype;
993 throw;
994}

1001template <typename ValueType, typename>
1002inline PrimExpr MakeConst(PrimType dtype, ValueType value, Span span) {
1003 if (!dtype.IsScalableVector() && !dtype.IsFixedLengthVector()) {
1004 return MakeConstScalar(dtype, value, span);
1005 }
1006 PrimType elem_ty = dtype.WithLanes(1);
1007 if (dtype.IsFixedLengthVector()) {
1008 return tirx::Broadcast(MakeConstScalar(elem_ty, value, span), dtype.lanes(), span);
1009 }
1010 PrimExpr lanes =
1011 tirx::Mul(Call(PrimType::Int(32), tirx::builtin::vscale(), {}).as_or_throw<PrimExpr>(),
1012 dtype.VScaleFactor());
1013 return tirx::Broadcast(MakeConstScalar(elem_ty, value, span), lanes, span);
1014}
