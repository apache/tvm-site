265inline void CheckMathUnaryOpInputDType(const char* op_name, const PrimType& dtype) {
266 TVM_FFI_CHECK(dtype.code() == DLDataTypeCode::kDLFloat ||
267 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16),
268 TypeError)
269 << "tirx." << op_name << " only supports floating-point inputs, but got " << dtype;
270}

272// Intrinsic operators
273#define TVM_DECLARE_INTRIN_UNARY_WITH_CHECK(OpName, CheckInputDType) \
274 inline PrimExpr OpName(PrimExpr x, Span span = Span()) { \
275 static const Op op = Op::Get("tirx." #OpName); \
276 PrimType x_ty = x.ty(); \
277 CheckInputDType(#OpName, x_ty); \
278 if (x_ty.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) { \
279 PrimType bf16_ty = x_ty; \
280 PrimType f32_ty = \
281 x_ty.IsScalableVector() \
282 ? PrimType::ScalableVector(DLDataTypeCode::kDLFloat, 32, x_ty.VScaleFactor()) \
283 : PrimType::Float(32, x_ty.lanes()); \
284 PrimExpr x_fp32 = prim::Cast(f32_ty, x, span); \
285 PrimExpr result_fp32 = Call(f32_ty, op, {x_fp32}, {}, {}, span).as_or_throw<PrimExpr>(); \
286 return prim::Cast(bf16_ty, result_fp32, span); \
287 } else { \
288 return Call(x_ty, op, {x}, {}, {}, span).as_or_throw<PrimExpr>(); \
289 } \
290 }

340inline bool IsPointerType(const Type& type, DLDataType element_type) {
341 if (type.as<MissingType>().has_value()) return false;
342 if (const auto* ptr_type = type.as<PointerTypeNode>()) {
343 if (const auto* prim_type = ptr_type->element_type.as<PrimTypeNode>()) {
344 return prim_type->dtype == element_type;
345 }
346 }
347 return false;
348}

376 Span span = Span()) {
377 for (PrimExpr val : values) {

383inline bool is_no_op(const tirx::Stmt& stmt) {
384 if (!stmt.defined()) return true;
385 if (const auto* op = stmt.as<tirx::EvaluateNode>()) {
386 auto value = op->value.as<PrimExpr>();
387 return value && is_const_int(value.value());
388 }
389 if (const auto* op = stmt.as<tirx::SeqStmtNode>()) {
390 return op->seq.size() == 0;
391 }
392 return false;
393}
