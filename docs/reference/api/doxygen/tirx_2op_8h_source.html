286inline void CheckMathUnaryOpInputDType(const char* op_name, const PrimType& dtype) {
287 TVM_FFI_CHECK(dtype.code() == DLDataTypeCode::kDLFloat ||
288 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16),
289 TypeError)
290 << "tirx." << op_name << " only supports floating-point inputs, but got " << dtype;
291}

293// Intrinsic operators
294#define TVM_DECLARE_INTRIN_UNARY_WITH_CHECK(OpName, CheckInputDType) \
295 inline PrimExpr OpName(PrimExpr x, Span span = Span()) { \
296 static const Op op = Op::Get("tirx." #OpName); \
297 PrimType x_ty = x.ty(); \
298 CheckInputDType(#OpName, x_ty); \
299 if (x_ty.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) { \
300 PrimType bf16_ty = x_ty; \
301 PrimType f32_ty = \
302 x_ty.IsScalableVector() \
303 ? PrimType::ScalableVector(DLDataTypeCode::kDLFloat, 32, x_ty.VScaleFactor()) \
304 : PrimType::Float(32, x_ty.lanes()); \
305 PrimExpr x_fp32 = prim::Cast(f32_ty, x, span); \
306 PrimExpr result_fp32 = Call(f32_ty, op, {x_fp32}, {}, {}, span).as_or_throw<PrimExpr>(); \
307 return prim::Cast(bf16_ty, result_fp32, span); \
308 } else { \
309 return Call(x_ty, op, {x}, {}, {}, span).as_or_throw<PrimExpr>(); \
310 } \
311 }

361inline bool IsPointerType(const Type& type, DLDataType element_type) {
362 if (type.IsMissing()) return false;
363 if (const auto* ptr_type = type.as<PointerTypeNode>()) {
364 if (const auto* prim_type = ptr_type->element_type.as<PrimTypeNode>()) {
365 return prim_type->dtype == element_type;
366 }
367 }
368 return false;
369}

397 Span span = Span()) {
398 for (PrimExpr val : values) {
399 init_value = freduce(init_value, val, span);

404inline bool is_no_op(const tirx::Stmt& stmt) {
405 if (!stmt.defined()) return true;
406 if (const auto* op = stmt.as<tirx::EvaluateNode>()) {
407 auto value = op->value.as<PrimExpr>();
408 return value && is_const_int(value.value());
409 }
410 if (const auto* op = stmt.as<tirx::SeqStmtNode>()) {
411 return op->seq.size() == 0;
412 }
413 return false;
414}
