969inline void CheckMathUnaryOpInputDType(const char* op_name, const PrimType& dtype) {
970 TVM_FFI_CHECK(dtype.code() == DLDataTypeCode::kDLFloat ||
971 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16),
972 TypeError)
973 << "tirx." << op_name << " only supports floating-point inputs, but got " << dtype;
974}

976// Intrinsic operators
977#define TVM_DECLARE_INTRIN_UNARY_WITH_CHECK(OpName, CheckInputDType) \
978 inline PrimExpr OpName(PrimExpr x, Span span = Span()) { \
979 static const Op op = Op::Get("tirx." #OpName); \
980 PrimType x_ty = x.ty(); \
981 CheckInputDType(#OpName, x_ty); \
982 if (x_ty.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) { \
983 PrimType bf16_ty = x_ty; \
984 PrimType f32_ty = \
985 x_ty.IsScalableVector() \
986 ? PrimType::ScalableVector(DLDataTypeCode::kDLFloat, 32, x_ty.VScaleFactor()) \
987 : PrimType::Float(32, x_ty.lanes()); \
988 PrimExpr x_fp32 = prim::Cast(f32_ty, x, span); \
989 PrimExpr result_fp32 = Call(f32_ty, op, {x_fp32}, {}, {}, span).as_or_throw<PrimExpr>(); \
990 return prim::Cast(bf16_ty, result_fp32, span); \
991 } else { \
992 return Call(x_ty, op, {x}, {}, {}, span).as_or_throw<PrimExpr>(); \
993 } \
994 }

1044inline bool IsPointerType(const Type& type, DLDataType element_type) {
1045 if (type.as<MissingType>().has_value()) return false;
1046 if (const auto* ptr_type = type.as<PointerTypeNode>()) {
1047 if (const auto* prim_type = ptr_type->element_type.as<PrimTypeNode>()) {
1048 return prim_type->dtype == element_type;

1050 }
1051 return false;

1087inline bool is_no_op(const tirx::Stmt& stmt) {
1088 if (!stmt.defined()) return true;
1089 if (const auto* op = stmt.as<tirx::EvaluateNode>()) {
1090 auto value = op->value.as<PrimExpr>();
1091 return value && IsConstInt(value.value());
1092 }
1093 if (const auto* op = stmt.as<tirx::SeqStmtNode>()) {
1094 return op->seq.size() == 0;
1095 }
1096 return false;
1097}
