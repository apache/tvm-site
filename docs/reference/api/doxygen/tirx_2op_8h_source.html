272inline void CheckMathUnaryOpInputDType(const char* op_name, const PrimType& dtype) {
273 TVM_FFI_CHECK(dtype.code() == DLDataTypeCode::kDLFloat ||
274 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16),
275 TypeError)
276 << "tirx." << op_name << " only supports floating-point inputs, but got " << dtype;
277}

279// Intrinsic operators
280#define TVM_DECLARE_INTRIN_UNARY_WITH_CHECK(OpName, CheckInputDType) \
281 inline PrimExpr OpName(PrimExpr x, Span span = Span()) { \
282 static const Op op = Op::Get("tirx." #OpName); \
283 PrimType x_ty = x.ty(); \
284 CheckInputDType(#OpName, x_ty); \
285 if (x_ty.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) { \
286 PrimType bf16_ty = x_ty; \
287 PrimType f32_ty = \
288 x_ty.IsScalableVector() \
289 ? PrimType::ScalableVector(DLDataTypeCode::kDLFloat, 32, x_ty.VScaleFactor()) \
290 : PrimType::Float(32, x_ty.lanes()); \
291 PrimExpr x_fp32 = prim::Cast(f32_ty, x, span); \
292 PrimExpr result_fp32 = Call(f32_ty, op, {x_fp32}, {}, {}, span).as_or_throw<PrimExpr>(); \
293 return prim::Cast(bf16_ty, result_fp32, span); \
294 } else { \
295 return Call(x_ty, op, {x}, {}, {}, span).as_or_throw<PrimExpr>(); \
296 } \
297 }

347inline bool IsPointerType(const Type& type, DLDataType element_type) {
348 if (type.IsMissing()) return false;
349 if (const auto* ptr_type = type.as<PointerTypeNode>()) {
350 if (const auto* prim_type = ptr_type->element_type.as<PrimTypeNode>()) {
351 return prim_type->dtype == element_type;
352 }
353 }
354 return false;
355}

390inline bool is_no_op(const tirx::Stmt& stmt) {
391 if (!stmt.defined()) return true;
392 if (const auto* op = stmt.as<tirx::EvaluateNode>()) {
393 auto value = op->value.as<PrimExpr>();
394 return value && is_const_int(value.value());
395 }
396 if (const auto* op = stmt.as<tirx::SeqStmtNode>()) {
397 return op->seq.size() == 0;
398 }
399 return false;
400}
