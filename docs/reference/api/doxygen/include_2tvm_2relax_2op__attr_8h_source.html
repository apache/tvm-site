39enum OpPatternKind {
40 // Elementwise operation
41 kElemWise = 0,
42 // Broadcasting operator, can always map output axis to the input in order.
43 // for example :code:`out[i, ax1, j, ax2] = input[i, j]`.
44 // Note that the axis need to be in order so transpose is not a bcast operator.
45 kBroadcast = 1,
46 // Injective operator, can always injectively map output axis to a single input axis.
47 // All injective operator can still be safely fused to injective and reduction.
48 kInjective = 2,
49 // Communicative reduction operator.
50 kCommReduce = 3,
51 // Complex operation, can still fuse elemwise operations into its output.
52 // but cannot chain another complex op
53 kOutEWiseFusable = 4,
54 // The pattern for tuple nodes. Can fuse into subsequent injective ops,
55 // but treated specially
56 kTuple = 7,
57 // Opaque operation, cannot fuse anything.
58 kOpaque = 8
59};
