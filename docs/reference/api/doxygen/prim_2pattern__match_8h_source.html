147template <typename OpType, typename TA, typename TB, bool kFoldConstants = true>
148class PBinaryExpr : public Pattern<PBinaryExpr<OpType, TA, TB, kFoldConstants>> {
149 public:
150 PBinaryExpr(const TA& a, const TB& b) : a_(a), b_(b) {}
151
152 void InitMatch_() const {
153 a_.InitMatch_();
154 b_.InitMatch_();
155 }
156
157 bool Match_(const ffi::ObjectRef& node) const {
158 using NodeType = typename OpType::ContainerType;
159 if (const NodeType* ptr = node.as<NodeType>()) {
160 if (!a_.Match_(ptr->a)) return false;
161 if (!b_.Match_(ptr->b)) return false;
162 return true;
163 } else {
164 return false;
165 }
166 }
167
168 PrimExpr Eval() const {
169 PrimExpr lhs = a_.Eval();
170 PrimExpr rhs = b_.Eval();
171 if constexpr (kFoldConstants) {
172 if (auto ret = prim::TryConstFold<OpType>(lhs, rhs)) return ret.value();
173 }
174 return OpType(lhs, rhs);
175 }
176
177 private:
178 typename TA::Nested a_;
179 typename TB::Nested b_;
180};

611template <typename Op, typename... TArgs>
612class PCallExpr : public Pattern<PCallExpr<Op, TArgs...>> {
613 public:
614 explicit PCallExpr(const TArgs&... args) : args_(args...) {}
615
616 void InitMatch_() const {
617 detail::PCallExprInitMatchFunctor finit;
618 detail::tuple_for_each(finit, args_);
619 }
620
621 bool Match_(const ffi::ObjectRef& node) const {
622 if (const CallObj* ptr = node.as<CallObj>()) {
623 if (ptr->args.size() != sizeof...(TArgs)) return false;
624 if (!ptr->op.same_as(Op::GetOp())) return false;
625 detail::PCallExprMatchFunctor fmatch(ptr);
626 detail::tuple_for_each(fmatch, args_);
627 return fmatch.matched_;
628 } else {
629 return false;
630 }
631 }
632
633 PrimExpr Eval() const {
634 detail::PCallExprEvalArgsFunctor feval_args;
635 detail::tuple_for_each(feval_args, args_);
636 return Op::Eval(feval_args.args_);
637 }
638
639 private:
640 std::tuple<typename TArgs::Nested...> args_;
641};
