158inline ffi::Optional<ffi::Tuple<ffi::Array<PrimExpr>, ffi::Array<PrimExpr>, PrimExpr>>
159MatchShapeStridePattern(const TensorType& tensor) {
160 ffi::Array<PrimExpr> strides;
161 PrimExpr offset = IntImm(tensor->DefaultIndexType(), 0);
162 if (!tensor->layout.has_value()) {
163 PrimExpr stride = IntImm(tensor->DefaultIndexType(), 1);
164 strides = ffi::Array<PrimExpr>(tensor->shape.size(), stride);
165 for (size_t i = tensor->shape.size(); i > 0; --i) {
166 strides.Set(i - 1, stride);
167 stride = stride * tensor->shape[i - 1];
168 }
169 } else {
170 const auto* tile = tensor->layout.as<TileLayoutObj>();
171 if (!tile || !tile->replica.empty() || tile->shard.size() != tensor->shape.size()) {
172 return std::nullopt;
173 }
174 prim::ExprDeepEqual equal;
175 static const Axis memory_axis = Axis::Get("m");
176 for (size_t i = 0; i < tile->shard.size(); ++i) {
177 if (tile->shard[i]->axis != memory_axis || !equal(tile->shard[i]->extent, tensor->shape[i]))
178 return std::nullopt;
179 strides.push_back(tile->shard[i]->stride);
180 }
181 for (const auto& [axis, value] : tile->offset) {
182 if (axis != memory_axis) return std::nullopt;
183 offset = value;
184 }
185 }
186 return ffi::Tuple(tensor->shape, std::move(strides), std::move(offset));
187}
