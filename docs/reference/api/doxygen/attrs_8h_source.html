145 // Utils for accessing attributes
165 template <typename TObjectRef>
166 ffi::Optional<TObjectRef> GetAttr(
167 const std::string& attr_key,
168 ffi::Optional<TObjectRef> default_value = ffi::Optional<TObjectRef>(std::nullopt)) const {
169 const DictAttrsNode* node = get();
170 auto it = node->dict.find(attr_key);
171 if (it != node->dict.end()) {
172 return (*it).second.cast<TObjectRef>();
173 } else {
174 return default_value;
175 }
176 }
177 // variant that uses TObjectRef to enable implicit conversion to default value.
178 template <typename TObjectRef>
179 ffi::Optional<TObjectRef> GetAttr(const std::string& attr_key, TObjectRef default_value) const {
180 return GetAttr<TObjectRef>(attr_key, ffi::Optional<TObjectRef>(default_value));
181 }
201 bool HasNonzeroAttr(const std::string& attr_key) const {
202 return GetAttr<int64_t>(attr_key, 0).value_or(0) != 0;
203 }

246template <typename TFunc>
247inline TFunc WithAttr(TFunc input, const std::string& attr_key, Any attr_value) {
248 using TNode = typename TFunc::ContainerType;
249 if (!input.unique()) {
250 static ffi::reflection::TypeAttrColumn shallow_copy(ffi::reflection::type_attr::kShallowCopy);
251 ffi::AnyView copy_func = shallow_copy[input->type_index()];
252 TVM_FFI_CHECK(copy_func.type_index() == ffi::TypeIndex::kTVMFFIFunction, TypeError)
253 << "Type " << input->GetTypeKey()
254 << " must register an ffi.Function for __ffi_shallow_copy__";
255 Any copy = copy_func.cast<ffi::Function>()(input);
256 TVM_FFI_CHECK(copy.type_index() == input->type_index() && copy.as<ffi::Object>() != input.get(),
257 TypeError)
258 << "Shallow copy must return a distinct object of type " << input->GetTypeKey();
259 input = std::move(copy).cast<TFunc>();
260 }
261 TNode* node = const_cast<TNode*>(input.operator->());
262 // node->attrs is NOTNULLABLE by contract, but defend against a caller
263 // that left a moved-from DictAttrs in place by re-initializing here.
264 if (!node->attrs.defined()) node->attrs = DictAttrs();
265 node->attrs.CopyOnWrite()->dict.Set(attr_key, std::move(attr_value));
266 return input;
267}
