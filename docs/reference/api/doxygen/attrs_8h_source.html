270template <typename TFunc>
271inline TFunc WithAttr(TFunc input, const std::string& attr_key, Any attr_value) {
272 using TNode = typename TFunc::ContainerType;
273 if (!input.unique()) {
274 static ffi::reflection::TypeAttrColumn shallow_copy(ffi::type_attr::kShallowCopy);
275 ffi::AnyView copy_func = shallow_copy[input->type_index()];
276 TVM_FFI_CHECK(copy_func.type_index() == ffi::TypeIndex::kTVMFFIFunction, TypeError)
277 << "Type " << input->GetTypeKey()
278 << " must register an ffi.Function for __ffi_shallow_copy__";
279 Any copy = copy_func.cast<ffi::Function>()(input);
280 TVM_FFI_CHECK(copy.type_index() == input->type_index() && copy.as<ffi::Object>() != input.get(),
281 TypeError)
282 << "Shallow copy must return a distinct object of type " << input->GetTypeKey();
283 input = std::move(copy).cast<TFunc>();
284 }
285 TNode* node = const_cast<TNode*>(input.operator->());
286 // node->attrs is NOTNULLABLE by contract, but defend against a caller
287 // that left a moved-from DictAttrs in place by re-initializing here.
288 if (!node->attrs.defined()) node->attrs = DictAttrs();
289 node->attrs.CopyOnWrite()->dict.Set(attr_key, std::move(attr_value));
290 return input;
291}

319inline TFunc WithAttrs(TFunc input, ffi::Map<ffi::String, Any> attrs) {
320 using TNode = typename TFunc::ContainerType;
321 if (attrs.empty()) return input;
322 if (!input.unique()) {
323 static ffi::reflection::TypeAttrColumn shallow_copy(ffi::type_attr::kShallowCopy);
324 ffi::AnyView copy_func = shallow_copy[input->type_index()];
325 TVM_FFI_CHECK(copy_func.type_index() == ffi::TypeIndex::kTVMFFIFunction, TypeError)
326 << "Type " << input->GetTypeKey()
327 << " must register an ffi.Function for __ffi_shallow_copy__";
328 Any copy = copy_func.cast<ffi::Function>()(input);
329 TVM_FFI_CHECK(copy.type_index() == input->type_index() && copy.as<ffi::Object>() != input.get(),
330 TypeError)
331 << "Shallow copy must return a distinct object of type " << input->GetTypeKey();
332 input = std::move(copy).cast<TFunc>();
333 }
334 TNode* node = const_cast<TNode*>(input.operator->());
335 // node->attrs is NOTNULLABLE by contract, but defend against a caller
336 // that left a moved-from DictAttrs in place by re-initializing here.
337 if (!node->attrs.defined()) node->attrs = DictAttrs();
338 auto* dict_node = node->attrs.CopyOnWrite();
339 for (const auto& [k, v] : attrs) {
340 dict_node->dict.Set(k, v);
341 }
342 return input;
