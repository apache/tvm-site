165 void Finalize() {
166 TVM_FFI_ICHECK_EQ(begin_type_index_, 0) << "Can only call Finalize once";
167 while (begin_type_index_ < func_.size() && func_[begin_type_index_] == nullptr) {
168 ++begin_type_index_;
169 }
170 // shift up the function value
171 size_t new_ftable_size = func_.size() - begin_type_index_;
172 if (begin_type_index_ != 0) {
173 std::memmove(func_.data(), func_.data() + begin_type_index_,
174 new_ftable_size * sizeof(FPointer));
175 }
176 func_.resize(new_ftable_size);
177 func_.shrink_to_fit();
178 }
