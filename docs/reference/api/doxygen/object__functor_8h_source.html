102 template <typename T>
103 TVM_FFI_INLINE bool CanDispatch(const T& n) const {
104 uint32_t type_index = GetTypeIndex(n);
105 if (type_index < begin_type_index_) return false;
106 type_index -= begin_type_index_;
107 return type_index < func_.size() && func_[type_index] != nullptr;
108 }
115 TVM_FFI_INLINE R operator()(NodeArg n, Args... args) const {
116 uint32_t type_index = n->type_index();
117 if (type_index >= begin_type_index_) {
118 uint32_t index = type_index - begin_type_index_;
119 if (index < func_.size() && func_[index] != nullptr) {
120 return (*func_[index])(n, std::forward<Args>(args)...);
121 }
122 }
123
124 const TVMFFITypeInfo* type_info = TVMFFIGetTypeInfo(type_index);
125 for (int32_t i = type_info->type_depth - 1; i >= 0; --i) {
126 type_index = type_info->type_ancestors[i]->type_index;
127 if (type_index >= begin_type_index_) {
128 uint32_t index = type_index - begin_type_index_;
129 if (index < func_.size() && func_[index] != nullptr) {
130 return (*func_[index])(n, std::forward<Args>(args)...);
131 }
132 }
133 }
134 ThrowUnregistered(n);
135 }
142 template <typename TNode>
143 ObjectFunctor& SetDispatch(R (*f)(NodeArg n, Args...)) {
144 static_assert(std::is_base_of_v<ffi::Object, TNode>,
145 "ObjectFunctor dispatch requires an object node type");
146 uint32_t tindex = TNode::RuntimeTypeIndex();
147 if (func_.size() <= tindex) {
148 func_.resize(tindex + 1, nullptr);
149 }
150 TVM_FFI_ICHECK(func_[tindex] == nullptr)
151 << "Dispatch for " << TNode::_type_key << " is already set";
152 TVM_FFI_ICHECK_EQ(begin_type_index_, 0) << " Cannot call SetDispatch after calling Finalize";
153 func_[tindex] = f;
154 return *this;
155 }
162 template <typename TNode>
163 ObjectFunctor& ClearDispatch() {
164 uint32_t tindex = TNode::RuntimeTypeIndex();
165 TVM_FFI_ICHECK_LT(tindex, func_.size()) << "ClearDispatch: index out of range";
166 TVM_FFI_ICHECK_EQ(begin_type_index_, 0) << " Cannot call ClearDispatch after calling Finalize";
167 func_[tindex] = nullptr;
168 return *this;
169 }
175 void Finalize() {
176 TVM_FFI_ICHECK_EQ(begin_type_index_, 0) << "Can only call Finalize once";
177 while (begin_type_index_ < func_.size() && func_[begin_type_index_] == nullptr) {
178 ++begin_type_index_;
179 }
180 // shift up the function value
181 size_t new_ftable_size = func_.size() - begin_type_index_;
182 if (begin_type_index_ != 0) {
183 std::memmove(func_.data(), func_.data() + begin_type_index_,
184 new_ftable_size * sizeof(FPointer));
185 }
186 func_.resize(new_ftable_size);
187 func_.shrink_to_fit();
188 }
189
190 private:
191 // Map null pointers and undefined handles to None so CanDispatch rejects them without
192 // dereferencing.
193 TVM_FFI_INLINE static uint32_t GetTypeIndex(NodeArg n) {
194 if constexpr (std::is_pointer_v<NodeArg>) {
195 return n != nullptr ? n->type_index() : ffi::TypeIndex::kTVMFFINone;
196 } else {
197 return n.defined() ? n->type_index() : ffi::TypeIndex::kTVMFFINone;
198 }
199 }
200 TVM_FFI_INLINE static uint32_t GetTypeIndex(ffi::AnyView n) { return n.type_index(); }
201
202 [[noreturn]] TVM_FFI_COLD_CODE static void ThrowUnregistered(NodeArg n) {
203 TVM_FFI_THROW(InternalError) << "ObjectFunctor calls un-registered function on type "
204 << n->GetTypeKey();
205 TVM_FFI_UNREACHABLE();
206 }
207
209 using FPointer = R (*)(NodeArg n, Args...);
211 std::vector<FPointer> func_;
213 uint32_t begin_type_index_{0};
214};

242 virtual ffi::Optional<VisitInterrupt> Visit(ffi::AnyView value) {
243 if (native_vtable_->CanDispatch(value)) {
244 const auto* object =
245 ffi::details::AnyUnsafe::RawObjectPtrFromAnyViewAfterCheck<const ffi::Object>(value);
246 try {
247 return (*native_vtable_)(object, this);
248 } catch (ffi::Error& error) {
249 ffi::details::UpdateVisitErrorContext(error, ffi::GetRef<ffi::ObjectRef>(object));
250 throw;
251 }
252 }
253 return StructuralVisitDefault(value);
254 }

380 virtual UnchangedOr<ffi::Any> Mutate(ffi::AnyView value,
381 InplaceMode inplace_mode = InplaceMode::kDisallow) {
382 const auto* object = value.as<ffi::Object>();
383 if (object == nullptr) return ffi::Any(value);
384 if (!object->unique()) inplace_mode = InplaceMode::kDisallow;
385 if (native_vtable_->CanDispatch(object)) {
386 try {
387 return (*native_vtable_)(object, this, inplace_mode);
388 } catch (ffi::Error& error) {
389 ffi::details::UpdateVisitErrorContext(error, ffi::GetRef<ffi::ObjectRef>(object));
390 throw;
391 }
392 }
393 // Default descent already supplies the current node's error context.
394 return ffi::StructuralMutatorObj::DefaultMutateExpected(value, inplace_mode).value();
395 }
