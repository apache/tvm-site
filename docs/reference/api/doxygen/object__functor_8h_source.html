101 TVM_FFI_INLINE bool CanDispatch(NodeArg n) const {
102 uint32_t type_index = n->type_index();
103 if (type_index < begin_type_index_) return false;
104 type_index -= begin_type_index_;
105 return type_index < func_.size() && func_[type_index] != nullptr;
106 }
113 TVM_FFI_INLINE R operator()(NodeArg n, Args... args) const {
114 uint32_t type_index = n->type_index();
115 if (type_index >= begin_type_index_) {
116 uint32_t index = type_index - begin_type_index_;
117 if (index < func_.size() && func_[index] != nullptr) {
118 return (*func_[index])(n, std::forward<Args>(args)...);
119 }
120 }
121
122 const TVMFFITypeInfo* type_info = TVMFFIGetTypeInfo(type_index);
123 for (int32_t i = type_info->type_depth - 1; i >= 0; --i) {
124 type_index = type_info->type_ancestors[i]->type_index;
125 if (type_index >= begin_type_index_) {
126 uint32_t index = type_index - begin_type_index_;
127 if (index < func_.size() && func_[index] != nullptr) {
128 return (*func_[index])(n, std::forward<Args>(args)...);
129 }
130 }
131 }
132 ThrowUnregistered(n);
133 }
140 template <typename TNode>
141 ObjectFunctor& SetDispatch(R (*f)(NodeArg n, Args...)) {
142 uint32_t tindex = TNode::RuntimeTypeIndex();
143 if (func_.size() <= tindex) {
144 func_.resize(tindex + 1, nullptr);
145 }
146 TVM_FFI_ICHECK(func_[tindex] == nullptr)
147 << "Dispatch for " << TNode::_type_key << " is already set";
148 TVM_FFI_ICHECK_EQ(begin_type_index_, 0) << " Cannot call SetDispatch after calling Finalize";
149 func_[tindex] = f;
150 return *this;
151 }
158 template <typename TNode>
159 ObjectFunctor& ClearDispatch() {
160 uint32_t tindex = TNode::RuntimeTypeIndex();
161 TVM_FFI_ICHECK_LT(tindex, func_.size()) << "ClearDispatch: index out of range";
162 TVM_FFI_ICHECK_EQ(begin_type_index_, 0) << " Cannot call ClearDispatch after calling Finalize";
163 func_[tindex] = nullptr;
164 return *this;
165 }
171 void Finalize() {
172 TVM_FFI_ICHECK_EQ(begin_type_index_, 0) << "Can only call Finalize once";
173 while (begin_type_index_ < func_.size() && func_[begin_type_index_] == nullptr) {
174 ++begin_type_index_;
175 }
176 // shift up the function value
177 size_t new_ftable_size = func_.size() - begin_type_index_;
178 if (begin_type_index_ != 0) {
179 std::memmove(func_.data(), func_.data() + begin_type_index_,
180 new_ftable_size * sizeof(FPointer));
181 }
182 func_.resize(new_ftable_size);
183 func_.shrink_to_fit();
184 }
185
186 private:
187 [[noreturn]] TVM_FFI_COLD_CODE static void ThrowUnregistered(NodeArg n) {
188 TVM_FFI_THROW(InternalError) << "ObjectFunctor calls un-registered function on type "
189 << n->GetTypeKey();
190 throw;
191 }
192
194 using FPointer = R (*)(NodeArg n, Args...);
196 std::vector<FPointer> func_;
198 uint32_t begin_type_index_{0};
199};
