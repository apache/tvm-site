288inline const char* DLDeviceType2Str(int type) {
289 switch (type) {
290 case kDLCPU:
291 return "cpu";
292 case kDLCUDA:
293 return "cuda";
294 case kDLCUDAHost:
295 return "cuda_host";
296 case kDLCUDAManaged:
297 return "cuda_managed";
298 case kDLOpenCL:
299 return "opencl";
300 case kDLVulkan:
301 return "vulkan";
302 case kDLMetal:
303 return "metal";
304 case kDLVPI:
305 return "vpi";
306 case kDLROCM:
307 return "rocm";
308 case kDLROCMHost:
309 return "rocm_host";
310 case kDLExtDev:
311 return "ext_dev";
312 case kDLOneAPI:
313 return "oneapi";
314 case kDLWebGPU:
315 return "webgpu";
316 case kDLHexagon:
317 return "hexagon";
318 case kDLTrn:
319 return "trn";
320 default:
321 TVM_FFI_THROW(InternalError) << "unknown type = " << type;
322 }
323 throw;
324}
