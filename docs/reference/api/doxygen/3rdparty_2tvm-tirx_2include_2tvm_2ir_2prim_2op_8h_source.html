781template <typename ValueType>
782inline PrimExpr MakeConstScalar(PrimType dtype, ValueType value, Location loc = UnknownLoc()) {
783 if constexpr (std::is_enum_v<ValueType>) {
784 return MakeConstScalar(dtype, static_cast<std::underlying_type_t<ValueType>>(value), loc);
785 } else {
786 DLDataTypeCode code = dtype.code();
787 if (code == DLDataTypeCode::kDLInt || code == DLDataTypeCode::kDLBool) {
788 return IntImm(dtype, ffi::BigInt(value), loc);
789 }
790 if (code == DLDataTypeCode::kDLUInt) {
791 TVM_FFI_ICHECK(value >= 0) << "cannot make uint from negative value " << value;
792 return IntImm(dtype, ffi::BigInt(value), loc);
793 }
794 if (dtype.MatchesCode(DLDataTypeCode::kDLFloat, DLDataTypeCode::kDLFloat8_e3m4,
795 DLDataTypeCode::kDLFloat8_e4m3, DLDataTypeCode::kDLFloat8_e4m3b11fnuz,
796 DLDataTypeCode::kDLFloat8_e4m3fn, DLDataTypeCode::kDLFloat8_e4m3fnuz,
797 DLDataTypeCode::kDLFloat8_e5m2, DLDataTypeCode::kDLFloat8_e5m2fnuz,
798 DLDataTypeCode::kDLFloat8_e8m0fnu, DLDataTypeCode::kDLFloat6_e2m3fn,
799 DLDataTypeCode::kDLFloat6_e3m2fn, DLDataTypeCode::kDLFloat4_e2m1fn) ||
800 dtype.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) {
801 return FloatImm(dtype, static_cast<double>(value), loc);
802 }
803 TVM_FFI_THROW(InternalError) << "cannot make const for type " << dtype;
804 throw;
805 }
806}

809inline PrimExpr MakeConstScalar(PrimType dtype, bool value, Location loc) {
810 return MakeConstScalar(dtype, static_cast<int>(value), loc);

813template <typename ValueType, typename>
814inline PrimExpr MakeConst(PrimType dtype, ValueType value, Location loc) {
815 if (!dtype.IsScalableVector() && !dtype.IsFixedLengthVector()) {
816 return MakeConstScalar(dtype, value, loc);
817 }
818 PrimType elem_ty = dtype.WithLanes(1);
819 if (dtype.IsFixedLengthVector()) {
820 return prim::Broadcast(MakeConstScalar(elem_ty, value, loc), dtype.lanes(), loc);
821 }
822 PrimExpr lanes = prim::Mul(Call(PrimType::Int(32), prim::vscale_op(), {}).as_or_throw<PrimExpr>(),
823 dtype.VScaleFactor());
824 return prim::Broadcast(MakeConstScalar(elem_ty, value, loc), lanes, loc);
825}

927 constexpr bool div_ambiguity = !std::is_class<TA>::value;
928 static_assert(div_ambiguity,
929 "TVM supports multiple types of integer divisions, "
930 "please call div, indexdiv/indexmod, "
931 "floordiv/floormod or truncdiv/truncmod directly "
932 "to avoid ambiguity in the code. "
933 "Checkout these functions in ir/prim/op.h.");
934}

1229// Intrinsic operators
1230#define TVM_DECLARE_INTRIN_UNARY_WITH_CHECK(OpName, CheckInputDType) \
1231 inline PrimExpr OpName(PrimExpr x, Location loc = UnknownLoc()) { \
1232 static const Op op = Op::Get("prim." #OpName); \
1233 PrimType x_ty = x.ty(); \
1234 CheckInputDType(#OpName, x_ty); \
1235 if (x_ty.MatchesElementType(DLDataTypeCode::kDLBfloat, 16)) { \
1236 PrimType bf16_ty = x_ty; \
1237 PrimType f32_ty = \
1238 x_ty.IsScalableVector() \
1239 ? PrimType::ScalableVector(DLDataTypeCode::kDLFloat, 32, x_ty.VScaleFactor()) \
1240 : PrimType::Float(32, x_ty.lanes()); \
1241 PrimExpr x_fp32 = prim::Cast(f32_ty, x, loc); \
1242 PrimExpr result_fp32 = Call(f32_ty, op, {x_fp32}, {}, {}, loc).as_or_throw<PrimExpr>(); \
1243 return prim::Cast(bf16_ty, result_fp32, loc); \
1244 } else { \
1245 return Call(x_ty, op, {x}, {}, {}, loc).as_or_throw<PrimExpr>(); \
1246 } \
1247 }
